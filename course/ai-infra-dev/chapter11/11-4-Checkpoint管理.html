Checkpoint的重要性

Checkpoint是训练过程中模型状态的快照，它记录了模型参数、优化器状态、 学习率调度器状态等关键信息。完善的Checkpoint管理是大规模训练的基础， 能够有效应对硬件故障、系统崩溃等意外情况，避免训练成果丢失。

💾 Checkpoint基本概念

Checkpoint包含的内容

checkpoint = { # 模型状态 'model_state_dict': model.state_dict(), # 优化器状态 'optimizer_state_dict': optimizer.state_dict(), # 学习率调度器状态 'lr_scheduler_state_dict': lr_scheduler.state_dict(), # 训练进度 'epoch': epoch, 'global_step': global_step, 'best_metric': best_accuracy, # 随机数种子（确保可复现） 'random_states': { 'random': random.getstate(), 'numpy': np.random.get_state(), 'torch': torch.random.get_rng_state(), 'cuda': torch.cuda.get_rng_state_all() }, # 训练配置 'config': { 'learning_rate': learning_rate, 'batch_size': batch_size, 'architecture': config_dict }, # 元数据 'timestamp': time.time(), 'hostname': socket.gethostname() }

Checkpoint的类型

定期Checkpoint

按照固定间隔保存，如每个epoch或固定步数，用于常规进度保存

最佳Checkpoint

在指标达到最优时保存，用于保留最佳模型

紧急Checkpoint

在检测到异常或收到中断信号时保存

增量Checkpoint

只保存增量变化，减少存储空间

⚙️ Checkpoint保存策略

定期保存

class CheckpointSaver: def __init__(self, save_dir, save_every_n_epochs=1, save_every_n_steps=None, keep_n_best=3): self.save_dir = save_dir self.save_every_n_epochs = save_every_n_epochs self.save_every_n_steps = save_every_n_steps self.keep_n_best = keep_n_best self.best_checkpoints = [] def maybe_save(self, model, optimizer, lr_scheduler, epoch, step, metrics): """条件保存Checkpoint""" should_save = False # 按epoch保存 if self.save_every_n_epochs and epoch % self.save_every_n_epochs == 0: should_save = True # 按步数保存 if self.save_every_n_steps and step % self.save_every_n_steps == 0: should_save = True if should_save: self.save_checkpoint( model, optimizer, lr_scheduler, epoch, step, metrics ) def save_checkpoint(self, model, optimizer, lr_scheduler, epoch, step, metrics): """保存Checkpoint""" timestamp = datetime.now().strftime("%Y%m%d_%H%M%S") filename = f"checkpoint_epoch{epoch}_step{step}_{timestamp}.pt" filepath = os.path.join(self.save_dir, filename) checkpoint = { 'epoch': epoch, 'global_step': step, 'model_state_dict': model.state_dict(), 'optimizer_state_dict': optimizer.state_dict(), 'lr_scheduler_state_dict': lr_scheduler.state_dict(), 'metrics': metrics, 'timestamp': time.time() } torch.save(checkpoint, filepath) self._cleanup_old_checkpoints() def _cleanup_old_checkpoints(self): """清理旧的Checkpoint""" checkpoints = glob.glob(os.path.join(self.save_dir, "checkpoint_*.pt")) checkpoints.sort(key=os.path.getmtime, reverse=True) for old_checkpoint in checkpoints[self.keep_n_best:]: os.remove(old_checkpoint)

最佳模型保存

class BestModelSaver: def __init__(self, save_dir, metric_name='accuracy', mode='max'): self.save_dir = save_dir self.metric_name = metric_name self.mode = mode self.best_metric = float('-inf') if mode == 'max' else float('inf') self.best_filepath = None def update(self, model, optimizer, lr_scheduler, epoch, step, metrics): """更新最佳模型""" current_metric = metrics.get(self.metric_name) if current_metric is None: return # 检查是否是最佳指标 is_best = False if self.mode == 'max': is_best = current_metric > self.best_metric else: is_best = current_metric < self.best_metric if is_best: # 保存新最佳模型 filepath = os.path.join(self.save_dir, "best_model.pt") self._save(model, optimizer, lr_scheduler, epoch, step, metrics, filepath) # 删除旧的最佳模型 if self.best_filepath and self.best_filepath != filepath: os.remove(self.best_filepath) self.best_metric = current_metric self.best_filepath = filepath print(f"New best {self.metric_name}: {current_metric:.4f}")

🔄 断点续训

加载Checkpoint

def load_checkpoint(checkpoint_path, model, optimizer=None, lr_scheduler=None): """加载Checkpoint继续训练""" if not os.path.exists(checkpoint_path): print(f"Checkpoint not found: {checkpoint_path}") return None # 加载Checkpoint文件 checkpoint = torch.load(checkpoint_path) # 加载模型状态 model.load_state_dict(checkpoint['model_state_dict']) # 加载优化器状态 if optimizer is not None and 'optimizer_state_dict' in checkpoint: optimizer.load_state_dict(checkpoint['optimizer_state_dict']) # 加载学习率调度器状态 if lr_scheduler is not None and 'lr_scheduler_state_dict' in checkpoint: lr_scheduler.load_state_dict(checkpoint['lr_scheduler_state_dict']) # 恢复训练状态 start_epoch = checkpoint.get('epoch', 0) start_step = checkpoint.get('global_step', 0) best_metric = checkpoint.get('best_metric', None) print(f"Loaded checkpoint from epoch {start_epoch}, step {start_step}") return { 'epoch': start_epoch, 'step': start_step, 'best_metric': best_metric, 'metrics': checkpoint.get('metrics', {}) }

完整训练循环

class ResumeTrainer: def __init__(self, model, train_loader, config): self.model = model self.train_loader = train_loader self.config = config self.checkpoint_dir = config.checkpoint_dir # 初始化组件 self.optimizer = self._create_optimizer() self.lr_scheduler = self._create_scheduler() self.checkpoint_saver = CheckpointSaver(self.checkpoint_dir) self.best_saver = BestModelSaver(self.checkpoint_dir) def train(self): """训练循环""" # 尝试加载Checkpoint resume_state = load_checkpoint( self.config.resume_path, self.model, self.optimizer, self.lr_scheduler ) start_epoch = resume_state['epoch'] + 1 if resume_state else 0 global_step = resume_state['step'] if resume_state else 0 for epoch in range(start_epoch, self.config.num_epochs): self.model.train() for batch_idx, (data, target) in enumerate(self.train_loader): # 训练步骤 loss = self._train_step(data, target) global_step += 1 # 记录指标 metrics = self._compute_metrics() self._log_metrics(metrics, global_step) # 定期保存Checkpoint self.checkpoint_saver.maybe_save( self.model, self.optimizer, self.lr_scheduler, epoch, global_step, metrics ) # 验证 val_metrics = self._validate() self.best_saver.update( self.model, self.optimizer, self.lr_scheduler, epoch, global_step, val_metrics )

⚡ Checkpoint性能优化

异步保存

import threading import queue class AsyncCheckpointSaver: def __init__(self, save_dir, max_queue_size=5): self.save_dir = save_dir self.save_queue = queue.Queue(maxsize=max_queue_size) self.save_thread = threading.Thread(target=self._save_worker) self.save_thread.daemon = True self.save_thread.start() def _save_worker(self): """后台保存线程""" while True: checkpoint_info = self.save_queue.get() self._save_checkpoint(*checkpoint_info) self.save_queue.task_done() def save_async(self, model, optimizer, lr_scheduler, epoch, step, metrics): """异步保存Checkpoint""" # 创建状态字典的副本 checkpoint = { 'epoch': epoch, 'global_step': step, 'model_state_dict': {k: v.clone().cpu() for k, v in model.state_dict().items()}, 'optimizer_state_dict': optimizer.state_dict(), 'metrics': metrics } # 放入队列异步保存 try: self.save_queue.put_nowait(( checkpoint, f"checkpoint_epoch{epoch}_step{step}.pt" )) except queue.Full: print("Checkpoint save queue full, skipping...") def wait_for_completion(self): """等待所有保存完成""" self.save_queue.join()

内存优化

def save_checkpoint_optimized(model, optimizer, lr_scheduler, epoch, step, metrics, filepath): """优化的Checkpoint保存""" # 1. 清理GPU缓存 torch.cuda.empty_cache() # 2. 移动到CPU并创建副本 state_dict = {k: v.detach().cpu().clone() for k, v in model.state_dict().items()} # 3. 使用新ZIP格式（更快的保存速度） torch.save({ 'epoch': epoch, 'global_step': step, 'model_state_dict': state_dict, 'optimizer_state_dict': optimizer.state_dict(), 'lr_scheduler_state_dict': lr_scheduler.state_dict(), 'metrics': metrics }, filepath, _use_new_zipfile_serialization=True) # 4. 可选：压缩 # compress_checkpoint(filepath)

分布式训练Checkpoint

import torch.distributed as dist def save_distributed_checkpoint(model, optimizer, lr_scheduler, epoch, step, metrics, filepath): """分布式训练Checkpoint保存""" if dist.get_rank() == 0: # 只在主进程保存 checkpoint = { 'epoch': epoch, 'global_step': step, 'model_state_dict': model.module.state_dict(), 'optimizer_state_dict': optimizer.state_dict(), 'lr_scheduler_state_dict': lr_scheduler.state_dict(), 'metrics': metrics } torch.save(checkpoint, filepath) # 确保所有进程同步 dist.barrier() def load_distributed_checkpoint(model, optimizer, lr_scheduler, filepath): """分布式训练Checkpoint加载""" map_location = {'cuda:%d' % i: 'cuda:%d' % dist.get_rank() for i in range(torch.cuda.device_count())} checkpoint = torch.load(filepath, map_location=map_location) model.module.load_state_dict(checkpoint['model_state_dict']) optimizer.load_state_dict(checkpoint['optimizer_state_dict']) lr_scheduler.load_state_dict(checkpoint['lr_scheduler_state_dict']) return checkpoint['epoch'], checkpoint['global_step']

🛡️ 容错机制

异常处理

import signal import sys class GracefulCheckpointHandler: def __init__(self, trainer): self.trainer = trainer self.register_handlers() def register_handlers(self): """注册信号处理器""" signal.signal(signal.SIGINT, self._handle_signal) signal.signal(signal.SIGTERM, self._handle_signal) def _handle_signal(self, signum, frame): """处理中断信号""" print(f"\nReceived signal {signum}, saving emergency checkpoint...") try: # 保存紧急Checkpoint self.trainer.save_checkpoint(emergency=True) print("Emergency checkpoint saved successfully") except Exception as e: print(f"Failed to save emergency checkpoint: {e}") sys.exit(0) # 使用示例 trainer = Trainer(...) handler = GracefulCheckpointHandler(trainer) trainer.train() # 训练过程中可以安全地Ctrl+C

自动恢复

def auto_resume_training(checkpoint_dir): """自动恢复训练""" # 查找最新的Checkpoint checkpoints = glob.glob(os.path.join(checkpoint_dir, "checkpoint_*.pt")) if not checkpoints: print("No checkpoint found, starting from scratch") return None # 按修改时间排序 latest_checkpoint = max(checkpoints, key=os.path.getmtime) print(f"Auto-resuming from: {latest_checkpoint}") return latest_checkpoint # 使用 trainer = Trainer(...) # 自动查找并加载最新的Checkpoint resume_path = auto_resume_training(config.checkpoint_dir) if resume_path: trainer.resume_from(resume_path) trainer.train()
