20.2 对称加密

使用相同密钥加密解密

🎯 学习目标

  • 理解对称加密原理
  • 掌握 createCipheriv 方法
  • 学会加密和解密操作
  • 了解加密模式

📚 基本加密解密

const crypto = require('crypto'); // 对称加密：加密和解密使用相同密钥 // 生成密钥和 IV（初始化向量） const algorithm = 'aes-256-cbc'; const key = crypto.randomBytes(32); // 256 位密钥 const iv = crypto.randomBytes(16); // 128 位 IV // 加密 function encrypt(text, key, iv) { const cipher = crypto.createCipheriv(algorithm, key, iv); let encrypted = cipher.update(text, 'utf8', 'hex'); encrypted += cipher.final('hex'); return encrypted; } // 解密 function decrypt(encrypted, key, iv) { const decipher = crypto.createDecipheriv(algorithm, key, iv); let decrypted = decipher.update(encrypted, 'hex', 'utf8'); decrypted += decipher.final('utf8'); return decrypted; } // 使用 const plaintext = 'Hello World!'; const encrypted = encrypt(plaintext, key, iv); console.log('加密:', encrypted); const decrypted = decrypt(encrypted, key, iv); console.log('解密:', decrypted); // 注意：密钥和 IV 需要安全存储和传输 console.log('密钥:', key.toString('hex')); console.log('IV:', iv.toString('hex'));

🔧 使用密码派生密钥

const crypto = require('crypto'); // 从密码派生密钥 function deriveKey(password, salt) { return crypto.pbkdf2Sync(password, salt, 100000, 32, 'sha256'); } // 加密函数 function encryptWithPassword(plaintext, password) { const salt = crypto.randomBytes(16); const iv = crypto.randomBytes(16); const key = deriveKey(password, salt); const cipher = crypto.createCipheriv('aes-256-cbc', key, iv); let encrypted = cipher.update(plaintext, 'utf8', 'hex'); encrypted += cipher.final('hex'); // 返回 salt:iv:encrypted return `${salt.toString('hex')}:${iv.toString('hex')}:${encrypted}`; } // 解密函数 function decryptWithPassword(encryptedData, password) { const [saltHex, ivHex, encrypted] = encryptedData.split(':'); const salt = Buffer.from(saltHex, 'hex'); const iv = Buffer.from(ivHex, 'hex'); const key = deriveKey(password, salt); const decipher = crypto.createDecipheriv('aes-256-cbc', key, iv); let decrypted = decipher.update(encrypted, 'hex', 'utf8'); decrypted += decipher.final('utf8'); return decrypted; } // 使用 const secret = 'my-secret-password'; const encrypted = encryptWithPassword('机密数据', secret); console.log('加密结果:', encrypted); const decrypted = decryptWithPassword(encrypted, secret); console.log('解密结果:', decrypted);

📦 加密模式

const crypto = require('crypto'); // 常见加密模式 const modes = { 'CBC': 'aes-256-cbc', // 需要填充，需要 IV 'GCM': 'aes-256-gcm', // 认证加密，需要 IV 和 authTag 'CTR': 'aes-256-ctr', // 流密码模式，需要 IV 'ECB': 'aes-256-ecb' // 不推荐，相同明文产生相同密文 }; // GCM 模式（推荐） function encryptGCM(plaintext, key) { const iv = crypto.randomBytes(12); // GCM 推荐 12 字节 IV const cipher = crypto.createCipheriv('aes-256-gcm', key, iv); let encrypted = cipher.update(plaintext, 'utf8', 'hex'); encrypted += cipher.final('hex'); const authTag = cipher.getAuthTag(); return { iv: iv.toString('hex'), encrypted, authTag: authTag.toString('hex') }; } function decryptGCM(data, key) { const iv = Buffer.from(data.iv, 'hex'); const authTag = Buffer.from(data.authTag, 'hex'); const decipher = crypto.createDecipheriv('aes-256-gcm', key, iv); decipher.setAuthTag(authTag); let decrypted = decipher.update(data.encrypted, 'hex', 'utf8'); decrypted += decipher.final('utf8'); return decrypted; } // 使用 GCM const key = crypto.randomBytes(32); const gcmEncrypted = encryptGCM('Hello GCM', key); console.log('GCM 加密:', gcmEncrypted); const gcmDecrypted = decryptGCM(gcmEncrypted, key); console.log('GCM 解密:', gcmDecrypted); // 查看支持的加密算法 console.log(crypto.getCiphers());
💡
推荐模式

GCM 模式提供认证加密，可同时验证数据完整性和真实性，是现代应用的首选。

⚙️ 完整加密工具

const crypto = require('crypto'); class SymmetricCrypto { constructor(algorithm = 'aes-256-gcm') { this.algorithm = algorithm; } // 生成随机密钥 generateKey() { return crypto.randomBytes(32); } // 加密 encrypt(plaintext, key) { const iv = crypto.randomBytes(12); const cipher = crypto.createCipheriv(this.algorithm, key, iv); let encrypted = cipher.update(plaintext, 'utf8', 'hex'); encrypted += cipher.final('hex'); if (this.algorithm.includes('gcm')) { const authTag = cipher.getAuthTag(); return { iv: iv.toString('hex'), data: encrypted, tag: authTag.toString('hex') }; } return { iv: iv.toString('hex'), data: encrypted }; } // 解密 decrypt(encrypted, key) { const iv = Buffer.from(encrypted.iv, 'hex'); const decipher = crypto.createDecipheriv(this.algorithm, key, iv); if (encrypted.tag) { decipher.setAuthTag(Buffer.from(encrypted.tag, 'hex')); } let decrypted = decipher.update(encrypted.data, 'hex', 'utf8'); decrypted += decipher.final('utf8'); return decrypted; } // 加密对象 encryptObject(obj, key) { const json = JSON.stringify(obj); return this.encrypt(json, key); } // 解密对象 decryptObject(encrypted, key) { const json = this.decrypt(encrypted, key); return JSON.parse(json); } } // 使用 const cryptoUtil = new SymmetricCrypto(); const key = cryptoUtil.generateKey(); const encrypted = cryptoUtil.encrypt('敏感数据', key); console.log('加密:', encrypted); const decrypted = cryptoUtil.decrypt(encrypted, key); console.log('解密:', decrypted);
