📚 懒加载概述

# 什么是懒加载 延迟加载非关键资源，直到需要时才加载 # 懒加载的好处 ## 性能提升 - 减少初始加载时间 - 减少带宽消耗 - 降低服务器压力 ## 用户体验 - 页面更快可交互 - 按需加载内容 - 节省用户流量 # 懒加载类型 ┌─────────────────┬──────────────────────────────────┐ │ 类型 │ 场景 │ ├─────────────────┼──────────────────────────────────┤ │ 图片懒加载 │ 滚动到可视区域时加载 │ │ 组件懒加载 │ 需要显示时加载组件 │ │ 数据懒加载 │ 滚动到底部时加载更多 │ │ 路由懒加载 │ 访问路由时加载页面 │ │ 第三方脚本 │ 用户交互时加载 │ └─────────────────┴──────────────────────────────────┘ # 懒加载实现原理 1. 检测元素是否进入可视区域 2. 触发加载操作 3. 替换占位内容为真实内容

🔧 图片懒加载

# 原生懒加载 <!-- 最简单的方式 --> <img src="image.jpg" loading="lazy" alt="description"> <!-- 带占位图 --> <img src="placeholder.jpg" data-src="image.jpg" loading="lazy" alt="description" > # Intersection Observer 实现 class LazyLoader { constructor() { this.observer = new IntersectionObserver( this.handleIntersection.bind(this), { root: null, rootMargin: '50px', threshold: 0.1, } ); } handleIntersection(entries) { entries.forEach(entry => { if (entry.isIntersecting) { this.loadImage(entry.target); this.observer.unobserve(entry.target); } }); } loadImage(img) { const src = img.dataset.src; if (!src) return; img.src = src; img.classList.add('loaded'); img.removeAttribute('data-src'); } observe(elements) { elements.forEach(el => this.observer.observe(el)); } } // 使用 const loader = new LazyLoader(); loader.observe(document.querySelectorAll('img[data-src]')); # React 图片懒加载组件 function LazyImage({ src, alt, placeholder }) { const [imageSrc, setImageSrc] = useState(placeholder); const [isLoaded, setIsLoaded] = useState(false); const imgRef = useRef(null); useEffect(() => { const observer = new IntersectionObserver( ([entry]) => { if (entry.isIntersecting) { setImageSrc(src); observer.unobserve(entry.target); } }, { rootMargin: '50px' } ); if (imgRef.current) { observer.observe(imgRef.current); } return () => observer.disconnect(); }, [src]); return ( <img ref={imgRef} src={imageSrc} alt={alt} onLoad={() => setIsLoaded(true)} style={{ opacity: isLoaded ? 1 : 0.5, transition: 'opacity 0.3s' }} /> ); } # 使用第三方库 import LazyLoad from 'vanilla-lazyload'; const lazyLoad = new LazyLoad({ elements_selector: '.lazy', threshold: 50, });

🚀 数据懒加载

# 无限滚动加载 function useInfiniteScroll(fetchFn, options = {}) { const { threshold = 100 } = options; const [data, setData] = useState([]); const [loading, setLoading] = useState(false); const [hasMore, setHasMore] = useState(true); const pageRef = useRef(1); const loadMore = async () => { if (loading || !hasMore) return; setLoading(true); try { const result = await fetchFn(pageRef.current); setData(prev => [...prev, ...result.items]); setHasMore(result.hasMore); pageRef.current++; } finally { setLoading(false); } }; useEffect(() => { const handleScroll = () => { const { scrollTop, scrollHeight, clientHeight } = document.documentElement; if (scrollHeight - scrollTop - clientHeight < threshold) { loadMore(); } }; window.addEventListener('scroll', handleScroll); loadMore(); // 初始加载 return () => window.removeEventListener('scroll', handleScroll); }, []); return { data, loading, hasMore, loadMore }; } # 使用 Intersection Observer function InfiniteList({ fetchFn }) { const loaderRef = useRef(null); const [items, setItems] = useState([]); const [hasMore, setHasMore] = useState(true); useEffect(() => { const observer = new IntersectionObserver( async ([entry]) => { if (entry.isIntersecting && hasMore) { const result = await fetchFn(); setItems(prev => [...prev, ...result.items]); setHasMore(result.hasMore); } }, { rootMargin: '100px' } ); if (loaderRef.current) { observer.observe(loaderRef.current); } return () => observer.disconnect(); }, [hasMore]); return ( <div> {items.map(item => ( <Item key={item.id} data={item} /> ))} {hasMore && <div ref={loaderRef}>Loading...</div>} </div> ); }
