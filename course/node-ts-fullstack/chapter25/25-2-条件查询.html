25.2 条件查询

构建灵活的查询条件

🎯 学习目标

  • 掌握各种条件表达式
  • 学会动态条件构建
  • 了解 FindOptions 查询方式
  • 熟悉条件运算符的使用

📚 基础条件操作符

import { FindOperator, Equal, Not, In, Like, Between, IsNull } from 'typeorm'; // 使用 FindOptions 方式 const users = await userRepository.find({ where: { // 等于 status: Equal('active'), // 不等于 status: Not('banned'), // IN 查询 id: In([1, 2, 3]), // LIKE 查询 username: Like('%alice%'), // BETWEEN 范围 age: Between(18, 30), // IS NULL deletedAt: IsNull(), // 大于/小于 age: MoreThan(18), age: MoreThanOrEqual(18), age: LessThan(30), age: LessThanOrEqual(30), // 模糊匹配（不区分大小写） username: ILike('%alice%'), }, }); // 简写形式 const users = await userRepository.find({ where: { status: 'active', // 自动转为 Equal age: MoreThan(18), id: In([1, 2, 3]), }, });

🔧 QueryBuilder 条件

// 基础 WHERE const users = await userRepository .createQueryBuilder('user') .where('user.status = :status', { status: 'active' }) .getMany(); // AND 条件 const users = await userRepository .createQueryBuilder('user') .where('user.status = :status', { status: 'active' }) .andWhere('user.age >= :minAge', { minAge: 18 }) .getMany(); // OR 条件 const users = await userRepository .createQueryBuilder('user') .where('user.status = :status', { status: 'active' }) .orWhere('user.role = :role', { role: 'admin' }) .getMany(); // EXISTS 子查询 const users = await userRepository .createQueryBuilder('user') .where((qb) => { const subQuery = qb .subQuery() .select('order.user_id') .from(Order, 'order') .where('order.amount > :amount') .getQuery(); return 'user.id IN ' + subQuery; }) .setParameter('amount', 1000) .getMany(); // IN 条件 const users = await userRepository .createQueryBuilder('user') .where('user.id IN (:...ids)', { ids: [1, 2, 3] }) .getMany(); // BETWEEN 条件 const users = await userRepository .createQueryBuilder('user') .where('user.age BETWEEN :min AND :max', { min: 18, max: 30 }) .getMany(); // LIKE 条件 const users = await userRepository .createQueryBuilder('user') .where('user.username LIKE :keyword', { keyword: '%alice%' }) .getMany(); // IS NULL / IS NOT NULL const users = await userRepository .createQueryBuilder('user') .where('user.deletedAt IS NULL') .getMany(); const users = await userRepository .createQueryBuilder('user') .where('user.phone IS NOT NULL') .getMany();

📊 动态条件构建

// 根据参数动态添加条件 interface UserQuery { status?: string; minAge?: number; maxAge?: number; keyword?: string; } async function searchUsers(query: UserQuery) { const qb = userRepository.createQueryBuilder('user'); if (query.status) { qb.andWhere('user.status = :status', { status: query.status }); } if (query.minAge !== undefined) { qb.andWhere('user.age >= :minAge', { minAge: query.minAge }); } if (query.maxAge !== undefined) { qb.andWhere('user.age <= :maxAge', { maxAge: query.maxAge }); } if (query.keyword) { qb.andWhere( '(user.username LIKE :keyword OR user.email LIKE :keyword)', { keyword: `%${query.keyword}%` } ); } return qb.getMany(); } // 使用 Brackets 构建复杂动态条件 import { Brackets } from 'typeorm'; async function searchUsersAdvanced(filters: any) { const qb = userRepository.createQueryBuilder('user'); // 根据条件添加 OR 块 if (filters.searchBy === 'name') { qb.andWhere( new Brackets((subQb) => { subQb .where('user.firstName LIKE :keyword', { keyword: `%${filters.keyword}%` }) .orWhere('user.lastName LIKE :keyword', { keyword: `%${filters.keyword}%` }); }) ); } if (filters.searchBy === 'email') { qb.andWhere('user.email LIKE :keyword', { keyword: `%${filters.keyword}%` }); } return qb.getMany(); } // 使用条件对象 const where: any = {}; if (status) { where.status = status; } if (minAge || maxAge) { where.age = Between(minAge || 0, maxAge || 100); } const users = await userRepository.find({ where });
💡
动态条件最佳实践

对于简单条件，使用 FindOptions 更简洁；对于复杂动态条件，QueryBuilder 更灵活。使用参数绑定而非字符串拼接，防止 SQL 注入。

⚙️ 多条件组合

// AND 组合 - FindOptions const users = await userRepository.find({ where: { status: 'active', age: MoreThan(18), }, }); // OR 组合 - FindOptions（多个 where 对象） const users = await userRepository.find({ where: [ { status: 'active' }, { role: 'admin' }, ], }); // 生成的 SQL // SELECT * FROM users WHERE status = 'active' OR role = 'admin' // 复杂组合 - QueryBuilder const users = await userRepository .createQueryBuilder('user') .where( new Brackets((qb) => { qb.where('user.status = :status', { status: 'active' }) .andWhere('user.age >= :age', { age: 18 }); }) ) .orWhere( new Brackets((qb) => { qb.where('user.role = :role', { role: 'admin' }) .andWhere('user.verified = :verified', { verified: true }); }) ) .getMany(); // 生成的 SQL // WHERE (user.status = 'active' AND user.age >= 18) // OR (user.role = 'admin' AND user.verified = true) // NOT 条件 const users = await userRepository .createQueryBuilder('user') .where('user.status != :status', { status: 'banned' }) .andWhere('NOT user.email IS NULL') .getMany();
