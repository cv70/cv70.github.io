2. 应用配置

// src/app.ts import express from 'express'; import cors from 'cors'; import helmet from 'helmet'; import { json } from 'body-parser'; import userRouter from './routes/users'; import authRouter from './routes/auth'; import errorHandler from './middleware/errorHandler'; const app = express(); // 中间件 app.use(cors()); app.use(helmet()); app.use(json()); // 路由 app.use('/api/users', userRouter); app.use('/api/auth', authRouter); // 健康检查 app.get('/health', (req, res) => { res.json({ status: 'ok', timestamp: new Date().toISOString() }); }); // 错误处理 app.use(errorHandler); export default app;
// src/server.ts - 仅用于启动，不导出 import app from './app'; const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server running on port ${PORT}`); });
💡
分离 app 和 server

将 Express 应用配置与服务器启动分离，这样测试时可以只导入 app，而不需要监听端口。

// tests/integration/users.test.ts import request from 'supertest'; import app from '../../src/app'; import { setupDatabase, cleanupDatabase, getAuthToken } from '../helpers'; describe('User Routes', () => { beforeAll(async () => { await setupDatabase(); }); afterAll(async () => { await cleanupDatabase(); }); describe('GET /api/users', () => { test('returns all users', async () => { const response = await request(app) .get('/api/users') .expect('Content-Type', /json/) .expect(200); expect(response.body.data).toBeInstanceOf(Array); }); test('supports pagination', async () => { const response = await request(app) .get('/api/users') .query({ page: 1, limit: 10 }) .expect(200); expect(response.body.pagination).toBeDefined(); }); }); describe('GET /api/users/:id', () => { test('returns user by id', async () => { const response = await request(app) .get('/api/users/1') .expect(200); expect(response.body).toMatchObject({ id: expect.any(Number), name: expect.any(String) }); }); test('returns 404 for non-existent user', async () => { await request(app) .get('/api/users/99999') .expect(404); }); }); describe('POST /api/users', () => { test('creates a new user', async () => { const response = await request(app) .post('/api/users') .send({ name: 'John Doe', email: 'john@example.com', password: 'password123' }) .expect(201); expect(response.body).toMatchObject({ name: 'John Doe', email: 'john@example.com' }); }); test('validates required fields', async () => { const response = await request(app) .post('/api/users') .send({}) .expect(400); expect(response.body.errors).toBeDefined(); }); test('rejects duplicate email', async () => { await request(app) .post('/api/users') .send({ email: 'existing@example.com', name: 'User 1' }) .expect(201); await request(app) .post('/api/users') .send({ email: 'existing@example.com', name: 'User 2' }) .expect(409); }); }); });

7. 测试辅助函数

// tests/helpers/auth.ts import request from 'supertest'; import app from '../../src/app'; export async function getAuthToken( email: string = 'test@example.com', password: string = 'password123' ): Promise<string> { const response = await request(app) .post('/api/auth/login') .send({ email, password }); return response.body.token; } export async function createTestUser(userData: any = {}) { const defaultData = { name: 'Test User', email: `test${Date.now()}@example.com`, password: 'Password123!' }; const response = await request(app) .post('/api/auth/register') .send({ ...defaultData, ...userData }); return response.body; } export function authenticatedRequest(method: string, path: string) { return async (token: string) => { return request(app) [method](path) .set('Authorization', `Bearer ${token}`); }; }
// tests/helpers/database.ts import { Connection, createConnection } from 'typeorm'; import { User } from '../../src/models/User'; let connection: Connection; export async function setupDatabase(): Promise<Connection> { connection = await createConnection({ type: 'sqlite', database: ':memory:', entities: [User], synchronize: true }); return connection; } export async function cleanupDatabase(): Promise<void> { if (connection) { await connection.dropDatabase(); await connection.close(); } } export async function seedUsers(count: number = 3): Promise<User[]> { const users = []; for (let i = 0; i < count; i++) { const user = connection.manager.create(User, { name: `User ${i + 1}`, email: `user${i + 1}@example.com` }); users.push(await connection.manager.save(user)); } return users; }
