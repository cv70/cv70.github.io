什么是连续批处理

连续批处理（Continuous Batching），也称为Inflight Batching， 是大语言模型推理中的核心技术。与传统的静态批处理不同， 它可以在不等待整个批次完成的情况下，将新生成的请求加入正在处理的批次中。

🔄 传统批处理 vs 连续批处理

静态批处理

  • 等待策略：批次满或超时才开始处理
  • 批次大小：固定
  • 首token延迟：高（等待批次形成）
  • 吞吐量：高（大batch）
  • 适用：离线批处理

连续批处理

  • 等待策略：立即处理，动态加入
  • 批次大小：动态变化
  • 首token延迟：低（立即开始）
  • 吞吐量：高（持续填充）
  • 适用：在线推理服务

🎭 LLM推理的两个阶段

Prefill和Decode阶段

Prefill阶段
  • 处理输入prompt
  • 并行计算
  • 计算量与prompt长度成正比
  • 生成第一个token
Decode阶段
  • 逐个生成token
  • 串行计算
  • 每次生成一个token
  • 重复到完成

⚡ 连续批处理工作原理

动态调度机制

连续批处理的核心是动态调度：

请求A生成中
请求B生成中
请求C加入
请求D加入
请求A完成
请求E加入

💻 连续批处理实现示例

import queue
import threading
import time
from typing import List, Dict

class Request:
    """
    推理请求
    """
    def __init__(self, request_id, prompt, max_tokens):
        self.request_id = request_id
        self.prompt = prompt
        self.max_tokens = max_tokens
        self.generated_tokens = 0
        self.is_prefill = True
        self.completed = False
        self.created_time = time.time()

class ContinuousBatching:
    """
    连续批处理调度器
    """
    def __init__(self, max_batch_size=32, max_tokens_per_batch=4096):
        self.max_batch_size = max_batch_size
        self.max_tokens_per_batch = max_tokens_per_batch

        # 请求队列
        self.pending_queue = queue.Queue()

        # 当前批次的请求
        self.active_batch: Dict[str, Request] = {}

        # 请求结果
        self.results = {}

        # 调度锁
        self.lock = threading.Lock()

    def add_request(self, request: Request):
        """
        添加新请求
        """
        self.pending_queue.put(request)

    def get_next_batch(self) -> List[Request]:
        """
        获取下一批处理的请求
        """
        with self.lock:
            batch = []

            # 1. 首先从active_batch中选择未完成的请求
            for req_id, req in self.active_batch.items():
                if not req.completed:
                    batch.append(req)

            # 2. 从pending队列中添加新请求
            while len(batch) < self.max_batch_size and not self.pending_queue.empty():
                new_req = self.pending_queue.get()
                self.active_batch[new_req.request_id] = new_req
                batch.append(new_req)

            return batch

    def process_batch(self, batch: List[Request]):
        """
        处理一个批次
        """
        # Prefill阶段：并行处理所有新请求
        prefill_requests = [req for req in batch if req.is_prefill]

        if prefill_requests:
            self._prefill(prefill_requests)

        # Decode阶段：逐个生成token
        decode_requests = [req for req in batch if not req.is_prefill and not req.completed]

        if decode_requests:
            self._decode(decode_requests)

    def _prefill(self, requests: List[Request]):
        """
        Prefill阶段处理
        """
        # 并行处理所有prompt
        # 这里简化处理，实际应该并行计算
        for req in requests:
            # 模拟prefill计算
            time.sleep(0.1)

            # Prefill完成，标记为decode阶段
            req.is_prefill = False
            req.generated_tokens = 1

    def _decode(self, requests: List[Request]):
        """
        Decode阶段处理
        """
        # 逐个生成token
        for req in requests:
            if req.generated_tokens < req.max_tokens:
                # 生成一个token
                time.sleep(0.01)  # 模拟计算时间

                req.generated_tokens += 1

                # 检查是否完成
                if req.generated_tokens >= req.max_tokens:
                    req.completed = True
                    self.results[req.request_id] = "Generated output"

    def remove_completed(self):
        """
        移除已完成的请求
        """
        with self.lock:
            completed_ids = [
                req_id for req_id, req in self.active_batch.items()
                if req.completed
            ]

            for req_id in completed_ids:
                del self.active_batch[req_id]

    def run(self):
        """
        运行调度器
        """
        while True:
            # 获取下一批
            batch = self.get_next_batch()

            if not batch:
                time.sleep(0.01)
                continue

            # 处理批次
            self.process_batch(batch)

            # 移除完成的请求
            self.remove_completed()

# 使用示例
def example_usage():
    """
    使用连续批处理的示例
    """
    # 创建调度器
    scheduler = ContinuousBatching(
        max_batch_size=32,
        max_tokens_per_batch=4096
    )

    # 模拟请求到达
    requests = [
        Request("req_1", "你好", 100),
        Request("req_2", "解释量子计算", 200),
        Request("req_3", "写一个Python函数", 150),
    ]

    # 添加请求
    for req in requests:
        scheduler.add_request(req)

    # 运行调度器（简化版）
    # 在实际实现中，这应该是一个后台线程
    for _ in range(100):  # 模拟100次迭代
        scheduler.run()
        time.sleep(0.05)

    # 获取结果
    print("完成的结果:", scheduler.results)

# vLLM中的连续批处理
def vllm_continuous_batching():
    """
    vLLM中的连续批处理示例
    """
    from vllm import LLM, SamplingParams

    # 初始化LLM（自动启用连续批处理）
    llm = LLM(
        model="meta-llama/Llama-2-7b-hf",
        tensor_parallel_size=1,
        # vLLM自动启用continuous batching
    )

    # 采样参数
    sampling_params = SamplingParams(
        temperature=0.8,
        max_tokens=100
    )

    # 多个请求（会自动连续批处理）
    prompts = [
        "写一个Python函数",
        "解释什么是机器学习",
        "写一首诗",
        # ...更多请求会持续加入
    ]

    # 生成（自动连续批处理）
    outputs = llm.generate(prompts, sampling_params)

    for output in outputs:
        print(f"Prompt: {output.prompt}")
        print(f"Generated: {output.outputs[0].text}\n")
