📚 常用事件类型

import { ChangeEvent, FormEvent, MouseEvent, KeyboardEvent, FocusEvent, DragEvent, ClipboardEvent, } from 'react'; // ChangeEvent：表单输入变化 function Input() { const handleChange = (e: ChangeEvent<HTMLInputElement>) => { console.log(e.target.value); // string console.log(e.target.name); // string console.log(e.target.type); // string }; return <input onChange={handleChange} />; } // FormEvent：表单提交 function Form() { const handleSubmit = (e: FormEvent<HTMLFormElement>) => { e.preventDefault(); const formData = new FormData(e.currentTarget); console.log(formData.get('username')); }; return ( <form onSubmit={handleSubmit}> <input name="username" /> <button type="submit">Submit</button> </form> ); } // MouseEvent：鼠标事件 function Button() { const handleClick = (e: MouseEvent<HTMLButtonElement>) => { console.log(e.clientX, e.clientY); // 鼠标位置 console.log(e.currentTarget); // 事件绑定的元素 console.log(e.target); // 实际点击的元素 }; const handleContextMenu = (e: MouseEvent<HTMLDivElement>) => { e.preventDefault(); // 阻止右键菜单 }; return ( <> <button onClick={handleClick}>Click</button> <div onContextMenu={handleContextMenu}>Right click blocked</div> </> ); } // KeyboardEvent：键盘事件 function Input() { const handleKeyDown = (e: KeyboardEvent<HTMLInputElement>) => { if (e.key === 'Enter') { e.currentTarget.form?.requestSubmit(); } if (e.key === 'Escape') { e.currentTarget.blur(); } }; return <input onKeyDown={handleKeyDown} />; } // FocusEvent：焦点事件 function Input() { const handleFocus = (e: FocusEvent<HTMLInputElement>) => { console.log('Focused'); }; const handleBlur = (e: FocusEvent<HTMLInputElement>) => { console.log('Blurred'); }; return <input onFocus={handleFocus} onBlur={handleBlur} />; }

🔧 表单事件处理

// 完整表单示例 interface FormData { username: string; email: string; role: 'user' | 'admin'; subscribe: boolean; } function Form() { const [formData, setFormData] = useState<FormData>({ username: '', email: '', role: 'user', subscribe: false, }); // 通用 change 处理 const handleChange = (e: ChangeEvent<HTMLInputElement | HTMLSelectElement>) => { const { name, value, type } = e.target; const checked = (e.target as HTMLInputElement).checked; setFormData(prev => ({ ...prev, [name]: type === 'checkbox' ? checked : value, })); }; // 提交处理 const handleSubmit = async (e: FormEvent<HTMLFormElement>) => { e.preventDefault(); console.log('Submitting:', formData); try { await submitForm(formData); } catch (error) { console.error(error); } }; return ( <form onSubmit={handleSubmit}> <input name="username" value={formData.username} onChange={handleChange} /> <input name="email" type="email" value={formData.email} onChange={handleChange} /> <select name="role" value={formData.role} onChange={handleChange} > <option value="user">User</option> <option value="admin">Admin</option> </select> <input name="subscribe" type="checkbox" checked={formData.subscribe} onChange={handleChange} /> <button type="submit">Submit</button> </form> ); } // 文件上传 function FileUpload() { const handleFileChange = (e: ChangeEvent<HTMLInputElement>) => { const files = e.target.files; if (files) { Array.from(files).forEach(file => { console.log(file.name, file.size, file.type); }); } }; return <input type="file" multiple onChange={handleFileChange} />; }

📊 事件处理函数类型

// 作为 Props 的事件处理函数 interface ButtonProps { onClick?: (e: MouseEvent<HTMLButtonElement>) => void; onMouseEnter?: (e: MouseEvent<HTMLButtonElement>) => void; onMouseLeave?: (e: MouseEvent<HTMLButtonElement>) => void; } function Button({ onClick, onMouseEnter, onMouseLeave }: ButtonProps) { return ( <button onClick={onClick} onMouseEnter={onMouseEnter} onMouseLeave={onMouseLeave} > Click me </button> ); } // 或使用 React 提供的事件处理函数类型 import { MouseEventHandler, ChangeEventHandler, FormEventHandler } from 'react'; interface InputProps { onChange?: ChangeEventHandler<HTMLInputElement>; onClick?: MouseEventHandler<HTMLButtonElement>; onSubmit?: FormEventHandler<HTMLFormElement>; } // 等价于 interface InputProps { onChange?: (e: ChangeEvent<HTMLInputElement>) => void; onClick?: (e: MouseEvent<HTMLButtonElement>) => void; onSubmit?: (e: FormEvent<HTMLFormElement>) => void; } // 自定义事件 interface CustomEvent { type: string; payload: unknown; timestamp: number; } type CustomEventHandler = (event: CustomEvent) => void; interface EventEmitterProps { onEvent?: CustomEventHandler; } // 拖拽事件 function DropZone() { const handleDragOver = (e: DragEvent<HTMLDivElement>) => { e.preventDefault(); e.dataTransfer.dropEffect = 'copy'; }; const handleDrop = (e: DragEvent<HTMLDivElement>) => { e.preventDefault(); const files = e.dataTransfer.files; // 处理文件 }; return ( <div onDragOver={handleDragOver} onDrop={handleDrop} > Drop files here </div> ); }
