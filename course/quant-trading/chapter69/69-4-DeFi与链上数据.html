DeFi与链上数据

DeFi（去中心化金融）和链上数据是加密货币市场的重要组成部分。通过分析区块链上的交易、流动性、TVL等数据，投资者可以获得传统金融无法触及的信息优势，发现DeFi生态中的交易机会。

Aave, Compound, MakerDAO - 去中心化借贷平台

USDC, DAI, FRAX - 锚定法币的数字货币

# DeFi与链上数据分析示例
import requests
import pandas as pd
from web3 import Web3
import json

class OnChainDataAnalyzer:
    """链上数据分析器"""

    def __init__(self, rpc_url):
        self.w3 = Web3(Web3.HTTPProvider(rpc_url))

    def get_token_balance(self, token_address, wallet_address):
        """获取代币余额"""
        # ERC20 ABI (简化版)
        erc20_abi = json.loads('[{"constant":true,"inputs":[{"name":"_owner","type":"address"}],"name":"balanceOf","outputs":[{"name":"balance","type":"uint256"}],"type":"function"}]')

        contract = self.w3.eth.contract(
            address=token_address,
            abi=erc20_abi
        )

        balance = contract.functions.balanceOf(wallet_address).call()
        return balance

    def get_token_transfers(self, token_address, start_block, end_block):
        """获取代币转账记录"""
        # 调用链上API获取转账事件
        url = f"https://api.etherscan.io/api"
        params = {
            'module': 'account',
            'action': 'tokentx',
            'contractaddress': token_address,
            'startblock': start_block,
            'endblock': end_block,
            'sort': 'asc',
            'apikey': 'YOUR_API_KEY'
        }

        response = requests.get(url, params=params)
        data = response.json()

        if data['status'] == '1':
            df = pd.DataFrame(data['result'])
            df['timeStamp'] = pd.to_numeric(df['timeStamp'])
            df['date'] = pd.to_datetime(df['timeStamp'], unit='s')
            return df
        return None

class DefiProtocolAnalyzer:
    """DeFi协议分析器"""

    def __init__(self):
        self.endpoints = {
            'uniswap': 'https://api.thegraph.com/subgraphs/name/uniswap/uniswap-v3',
            'aave': 'https://api.thegraph.com/subgraphs/name/aave/aave-v3'
        }

    def query_uniswap_pool(self, pool_address):
        """查询Uniswap池子数据"""
        query = '''
        {
            pool(id: "%s") {
                token0 {
                    symbol
                    decimals
                }
                token1 {
                    symbol
                    decimals
                }
                liquidity
                tick
                sqrtPriceX96
                totalValueLockedUSD
                volumeUSD
                feeTier
            }
        }
        ''' % pool_address

        response = requests.post(
            self.endpoints['uniswap'],
            json={'query': query}
        )

        return response.json()

    def query_aave_rates(self):
        """查询Aave借贷利率"""
        query = '''
        {
            reserves(first: 10) {
                id
                name
                symbol
                liquidityIndex
                stableBorrowRate
                variableBorrowRate
                variableDepositRate
                utilizationRate
            }
        }
        '''

        response = requests.post(
            self.endpoints['aave'],
            json={'query': query}
        )

        return response.json()

class WhaleTracker:
    """大户追踪"""

    def __init__(self, rpc_url):
        self.w3 = Web3(Web3.HTTPProvider(rpc_url))

    def track_whale_transactions(self, whale_addresses):
        """追踪大户交易"""
        transactions = []

        for address in whale_addresses:
            # 获取最新区块
            latest_block = self.w3.eth.get_block('latest')['number']

            # 查询最近交易
            for i in range(latest_block - 100, latest_block + 1):
                block = self.w3.eth.get_block(i, full_transactions=True)

                for tx in block['transactions']:
                    if tx['from'].lower() == address.lower():
                        transactions.append({
                            'from': tx['from'],
                            'to': tx['to'],
                            'value': self.w3.from_wei(tx['value'], 'ether'),
                            'hash': tx['hash'].hex(),
                            'block': i
                        })

        return transactions

⚠️
风险提示

DeFi投资存在多重风险：1）智能合约漏洞风险，可能损失全部资金；2）流动性风险，极端行情下无法退出；3）预言机操纵风险；4）监管不确定性；5）技术门槛高，操作风险大。建议只投入能承受损失的资金，充分了解协议机制后再参与。
