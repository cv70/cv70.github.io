42.4 动态模块

创建可配置的灵活模块

🎯 学习目标

  • 理解动态模块的概念
  • 掌握动态模块的创建方式
  • 学会模块配置选项
  • 了解动态模块的实际应用

📚 什么是动态模块？

动态模块是可以根据配置动态改变其行为的模块。它允许在导入模块时传递配置参数，使模块更加灵活和可复用。

// 静态模块 - 固定配置 @Module({ providers: [DatabaseService], }) export class DatabaseModule {} // 动态模块 - 可配置 @Module({}) export class DatabaseModule { static forRoot(options: DatabaseOptions): DynamicModule { return { module: DatabaseModule, providers: [ { provide: 'DATABASE_OPTIONS', useValue: options, }, DatabaseService, ], exports: [DatabaseService], }; } }

🔧 创建动态模块

// database.module.ts import { Module, DynamicModule } from '@nestjs/common'; import { DatabaseService } from './database.service'; interface DatabaseModuleOptions { host: string; port: number; username: string; password: string; database: string; } @Module({}) export class DatabaseModule { static forRoot(options: DatabaseModuleOptions): DynamicModule { return { module: DatabaseModule, providers: [ { provide: 'DATABASE_OPTIONS', useValue: options, }, DatabaseService, ], exports: [DatabaseService], global: true, // 可选：设为全局模块 }; } static forRootAsync(options: { useFactory: (...args: any[]) => Promise; inject?: any[]; }): DynamicModule { return { module: DatabaseModule, providers: [ { provide: 'DATABASE_OPTIONS', useFactory: options.useFactory, inject: options.inject || [], }, DatabaseService, ], exports: [DatabaseService], }; } }

🚀 使用动态模块

// app.module.ts - 同步配置 @Module({ imports: [ DatabaseModule.forRoot({ host: 'localhost', port: 5432, username: 'postgres', password: 'password', database: 'mydb', }), ], }) export class AppModule {} // app.module.ts - 异步配置 @Module({ imports: [ DatabaseModule.forRootAsync({ useFactory: (configService: ConfigService) => ({ host: configService.get('DB_HOST'), port: configService.get('DB_PORT'), username: configService.get('DB_USERNAME'), password: configService.get('DB_PASSWORD'), database: configService.get('DB_DATABASE'), }), inject: [ConfigService], }), ], }) export class AppModule {} // 在服务中使用配置 @Injectable() export class DatabaseService { constructor(@Inject('DATABASE_OPTIONS') private options: DatabaseModuleOptions) { console.log('Database connected to:', options.host); } }

📦 实际应用示例

// TypeORM 动态模块配置 import { TypeOrmModule } from '@nestjs/typeorm'; @Module({ imports: [ TypeOrmModule.forRoot({ type: 'postgres', host: 'localhost', port: 5432, username: 'postgres', password: 'password', database: 'mydb', entities: [User, Product], synchronize: true, }), ], }) export class AppModule {} // ConfigModule 也是动态模块 import { ConfigModule } from '@nestjs/config'; @Module({ imports: [ ConfigModule.forRoot({ isGlobal: true, envFilePath: '.env.development', }), ], }) export class AppModule {}
💡
命名约定

动态模块方法通常命名为 forRoot、forRootAsync、forFeature 等。forRoot 用于全局配置，forFeature 用于特性模块配置。

📝 本节小结

✅
  • • 动态模块允许导入时传递配置
  • • 使用静态方法返回 DynamicModule
  • • forRoot 用于全局配置，forRootAsync 用于异步配置
  • • TypeOrmModule、ConfigModule 都是动态模块
