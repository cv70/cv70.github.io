🔧 限流与计费实现

// rate-limiter.service.ts @Injectable() export class RateLimiterService { constructor(private redis: Redis) {} // 检查限流 async checkRateLimit( appId: string, apiId: string, limit: number ): Promise<{ allowed: boolean; remaining: number }> { const key = `ratelimit:${appId}:${apiId}:${Math.floor(Date.now() / 1000)}`; const current = await this.redis.incr(key); if (current === 1) { await this.redis.expire(key, 1); // 1秒过期 } return { allowed: current <= limit, remaining: Math.max(0, limit - current), }; } // 检查日配额 async checkDailyQuota( appId: string, quota: number ): Promise<{ allowed: boolean; remaining: number }> { const today = format(new Date(), 'yyyyMMdd'); const key = `quota:${appId}:${today}`; const current = await this.redis.incr(key); if (current === 1) { await this.redis.expire(key, 86400); // 24小时过期 } return { allowed: current <= quota, remaining: Math.max(0, quota - current), }; } } // billing.service.ts @Injectable() export class BillingService { constructor( @InjectRepository(UsageRecord) private usageRepo: Repository, @InjectRepository(BillingRecord) private billingRepo: Repository, ) {} // 记录API调用 async recordUsage(appId: string, apiId: string, success: boolean) { const record = this.usageRepo.create({ appId, apiId, success, timestamp: new Date(), }); await this.usageRepo.save(record); } // 计算账单 async calculateBill(appId: string, month: string): Promise { const startDate = startOfMonth(parse(month, 'yyyyMM', new Date())); const endDate = endOfMonth(startDate); // 统计调用量 const usage = await this.usageRepo .createQueryBuilder('record') .select('apiId', 'api') .addSelect('COUNT(*)', 'count') .where('record.appId = :appId', { appId }) .andWhere('record.timestamp BETWEEN :start AND :end', { start: startDate, end: endDate, }) .groupBy('record.apiId') .getRawMany(); // 获取定价 const pricing = await this.getPricing(appId); // 计算费用 let totalAmount = 0; const details = usage.map(u => { const price = pricing[u.api] || 0; const amount = u.count * price; totalAmount += amount; return { api: u.api, count: parseInt(u.count), price, amount, }; }); // 创建账单 const bill = this.billingRepo.create({ appId, month, details, totalAmount, status: 'pending', }); return this.billingRepo.save(bill); } } // 计费模式 const billingModels = { // 按量计费 payAsYouGo: { description: '按实际调用次数计费', pricing: { '/api/v1/user/info': 0.001, // 每次调用0.001元 '/api/v1/order/list': 0.005, '/api/v1/payment/create': 0.01, } }, // 套餐计费 package: { description: '购买套餐，超出部分按量计费', packages: [ { name: '基础包', price: 99, calls: 10000 }, { name: '专业包', price: 299, calls: 50000 }, ] }, // 混合计费 hybrid: { description: '月费+按量', monthly: 99, includedCalls: 10000, overageRate: 0.001 } };
