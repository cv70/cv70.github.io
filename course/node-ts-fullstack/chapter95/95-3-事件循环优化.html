🔧 阻塞操作识别

# 常见阻塞操作 ## 1. CPU 密集计算 // 阻塞事件循环 function heavyComputation(n) { let result = 0; for (let i = 0; i < n; i++) { result += Math.sqrt(i); } return result; } // 调用时会阻塞整个应用 const result = heavyComputation(100000000); ## 2. 同步 I/O // 阻塞读取文件 const data = fs.readFileSync('large-file.txt'); // 阻塞加密 const hash = crypto.pbkdf2Sync(password, salt, 100000, 64, 'sha512'); ## 3. 大数组操作 // 大数组排序 largeArray.sort((a, b) => complexCompare(a, b)); // 正则匹配大字符串 const matches = largeString.match(/complex.*pattern/g); # 使用 blocked-at 检测阻塞 import blocked from 'blocked-at'; blocked((time, stack) => { console.log(`Blocked for ${time}ms`); console.log('Stack trace:', stack); }, { threshold: 100 }); // 阈值 100ms # 使用 clinic 检测 clinic doctor -- node app.js # 会显示事件循环延迟图
⚠️
阻塞危害

事件循环阻塞会影响所有请求，不仅当前请求变慢，其他所有请求也会被延迟，严重时导致请求超时。

⚙️ 任务分解技巧

# 使用 setImmediate 分解任务 // 不好的做法：一次性处理大量数据 function processLargeArray(array) { const results = []; for (const item of array) { results.push(processItem(item)); // 可能阻塞很久 } return results; } // 好的做法：分批处理 async function processLargeArrayChunked(array, chunkSize = 100) { const results = []; for (let i = 0; i < array.length; i += chunkSize) { const chunk = array.slice(i, i + chunkSize); // 处理当前批次 for (const item of chunk) { results.push(processItem(item)); } // 让出事件循环 await new Promise(resolve => setImmediate(resolve)); } return results; } # 使用 workerpool 处理 CPU 密集任务 import { WorkerPool } from 'workerpool'; const pool = WorkerPool.pool('./worker.js'); async function processHeavy(data) { const result = await pool.exec('heavyComputation', [data]); return result; } // worker.js import { workerpool as pool } from 'workerpool'; function heavyComputation(data) { // CPU 密集计算 let result = 0; for (let i = 0; i < data.iterations; i++) { result += Math.sqrt(i); } return result; } pool.worker({ heavyComputation, }); # 使用 async/await 配合 Promise async function processWithYield(array, processFn, yieldEvery = 10) { const results = []; for (let i = 0; i < array.length; i++) { results.push(processFn(array[i])); // 每处理 yieldEvery 个项目让出一次 if (i % yieldEvery === 0) { await new Promise(resolve => setTimeout(resolve, 0)); } } return results; }

🚀 异步优化实践

# 使用异步 API // 不好的做法：同步加密 const hash = crypto.pbkdf2Sync(password, salt, 100000, 64, 'sha512'); // 好的做法：异步加密 const hash = await new Promise((resolve, reject) => { crypto.pbkdf2(password, salt, 100000, 64, 'sha512', (err, derivedKey) => { if (err) reject(err); else resolve(derivedKey); }); }); // 或使用 util.promisify import { promisify } from 'util'; const pbkdf2Async = promisify(crypto.pbkdf2); const hash = await pbkdf2Async(password, salt, 100000, 64, 'sha512'); # 使用流处理大数据 import { createReadStream, createWriteStream } from 'fs'; import { Transform } from 'stream'; const transform = new Transform({ transform(chunk, encoding, callback) { // 处理数据块 const processed = processChunk(chunk); callback(null, processed); } }); createReadStream('input.txt') .pipe(transform) .pipe(createWriteStream('output.txt')); # 并行处理 // 串行处理（慢） for (const url of urls) { await fetch(url); } // 并行处理（快） await Promise.all(urls.map(url => fetch(url))); // 控制并发数 import pLimit from 'p-limit'; const limit = pLimit(10); // 最多 10 个并发 await Promise.all( urls.map(url => limit(() => fetch(url))) );
