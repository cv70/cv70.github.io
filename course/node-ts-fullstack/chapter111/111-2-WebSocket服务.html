📚 Socket.IO网关

// chat.gateway.ts import { WebSocketGateway, WebSocketServer, SubscribeMessage, OnGatewayConnection, OnGatewayDisconnect, ConnectedSocket, MessageBody, } from '@nestjs/websockets'; import { Server, Socket } from 'socket.io'; @WebSocketGateway({ cors: { origin: '*', }, namespace: '/chat', }) export class ChatGateway implements OnGatewayConnection, OnGatewayDisconnect { @WebSocketServer() server: Server; constructor( private authService: AuthService, private messageService: MessageService, private presenceService: PresenceService, ) {} // 客户端连接 async handleConnection(client: Socket) { try { // 从握手查询参数或头部获取token const token = client.handshake.auth.token || client.handshake.headers.authorization; if (!token) { client.disconnect(); return; } // 验证token const user = await this.authService.verifyToken(token); client.data.user = user; // 加入用户专属房间（用于私聊） client.join(`user:${user.id}`); // 更新在线状态 await this.presenceService.setOnline(user.id, client.id); // 通知好友用户上线 this.server.to(`friends:${user.id}`).emit('user:online', { userId: user.id, timestamp: Date.now(), }); console.log(`User ${user.id} connected`); } catch (error) { client.disconnect(); } } // 客户端断开 async handleDisconnect(client: Socket) { const user = client.data.user; if (user) { // 更新离线状态 await this.presenceService.setOffline(user.id); // 通知好友用户离线 this.server.to(`friends:${user.id}`).emit('user:offline', { userId: user.id, timestamp: Date.now(), }); console.log(`User ${user.id} disconnected`); } } // 发送消息 @SubscribeMessage('message:send') async handleSendMessage( @ConnectedSocket() client: Socket, @MessageBody() data: { to: string; content: string; type: MessageType } ) { const sender = client.data.user; // 保存消息 const message = await this.messageService.create({ from: sender.id, to: data.to, content: data.content, type: data.type, }); // 发送给接收者 this.server.to(`user:${data.to}`).emit('message:receive', message); // 发送确认给发送者 client.emit('message:sent', { messageId: message.id }); return message; } // 加入群聊 @SubscribeMessage('group:join') async handleJoinGroup( @ConnectedSocket() client: Socket, @MessageBody() data: { groupId: string } ) { const user = client.data.user; // 验证用户是否是群成员 const isMember = await this.groupService.isMember(data.groupId, user.id); if (!isMember) { return { error: 'Not a group member' }; } client.join(`group:${data.groupId}`); return { success: true }; } // 发送群消息 @SubscribeMessage('group:message') async handleGroupMessage( @ConnectedSocket() client: Socket, @MessageBody() data: { groupId: string; content: string } ) { const sender = client.data.user; const message = await this.messageService.create({ from: sender.id, to: data.groupId, content: data.content, type: MessageType.TEXT, isGroup: true, }); // 广播到群房间 this.server.to(`group:${data.groupId}`).emit('group:message', message); return message; } // 正在输入 @SubscribeMessage('typing:start') async handleTypingStart( @ConnectedSocket() client: Socket, @MessageBody() data: { to: string } ) { const user = client.data.user; this.server.to(`user:${data.to}`).emit('typing:start', { userId: user.id, }); } @SubscribeMessage('typing:stop') async handleTypingStop( @ConnectedSocket() client: Socket, @MessageBody() data: { to: string } ) { const user = client.data.user; this.server.to(`user:${data.to}`).emit('typing:stop', { userId: user.id, }); } }

🔧 客户端连接

// 前端客户端代码 import { io, Socket } from 'socket.io-client'; class ChatClient { private socket: Socket; private userId: string; constructor(userId: string, token: string) { this.userId = userId; this.socket = io('ws://localhost:3000/chat', { auth: { token }, transports: ['websocket'], }); this.setupListeners(); } private setupListeners() { // 连接事件 this.socket.on('connect', () => { console.log('Connected to server'); }); this.socket.on('disconnect', () => { console.log('Disconnected from server'); }); // 接收消息 this.socket.on('message:receive', (message: Message) => { this.onMessage?.(message); }); // 用户上线/离线 this.socket.on('user:online', (data) => { this.onUserOnline?.(data.userId, true); }); this.socket.on('user:offline', (data) => { this.onUserOnline?.(data.userId, false); }); // 正在输入 this.socket.on('typing:start', (data) => { this.onTyping?.(data.userId, true); }); this.socket.on('typing:stop', (data) => { this.onTyping?.(data.userId, false); }); } // 发送消息 sendMessage(to: string, content: string, type: MessageType = MessageType.TEXT) { return new Promise((resolve, reject) => { this.socket.emit('message:send', { to, content, type }, (response: any) => { if (response.error) { reject(response.error); } else { resolve(response); } }); }); } // 加入群聊 joinGroup(groupId: string) { return new Promise((resolve) => { this.socket.emit('group:join', { groupId }, resolve); }); } // 发送正在输入状态 startTyping(to: string) { this.socket.emit('typing:start', { to }); } stopTyping(to: string) { this.socket.emit('typing:stop', { to }); } // 回调函数 onMessage?: (message: Message) => void; onUserOnline?: (userId: string, online: boolean) => void; onTyping?: (userId: string, typing: boolean) => void; // 断开连接 disconnect() { this.socket.disconnect(); } }
