为什么需要混合检索

纯向量搜索擅长语义理解，但可能忽略精确匹配；纯关键词搜索精确但缺乏语义理解。 混合检索结合两者的优势，既能处理模糊语义，又能保持精确匹配， 是构建高质量搜索系统的关键策略。

向量搜索 语义相似度、高维空间 语义理解、跨语言 可能错过精确匹配 文档搜索、问答系统
混合搜索 融合多种检索方式 综合优势、高召回率 复杂度高 通用搜索引擎、推荐系统

Elasticsearch + 向量搜索

from elasticsearch import Elasticsearch from sentence_transformers import SentenceTransformer import numpy as np class HybridSearch: def __init__(self, es_host='localhost', es_port=9200): # 连接Elasticsearch self.es = Elasticsearch([f'http://{es_host}:{es_port}']) self.encoder = SentenceTransformer('all-MiniLM-L6-v2') def create_index(self): """创建混合检索索引""" index_mapping = { "mappings": { "properties": { "title": { "type": "text", "analyzer": "standard" }, "content": { "type": "text", "analyzer": "standard" }, "embedding": { "type": "dense_vector", "dims": 384, "index": True, "similarity": "cosine" }, "category": { "type": "keyword" }, "timestamp": { "type": "date" } } }, "settings": { "number_of_shards": 3, "number_of_replicas": 1 } } self.es.indices.create( index='documents', body=index_mapping, ignore=400 ) def index_document(self, doc): """索引文档""" # 生成向量 text = f"{doc['title']} {doc['content']}" embedding = self.encoder.encode(text).tolist() # 构建文档 es_doc = { "title": doc['title'], "content": doc['content'], "embedding": embedding, "category": doc.get('category'), "timestamp": doc.get('timestamp') } # 索引到Elasticsearch self.es.index( index='documents', id=doc['id'], body=es_doc ) def keyword_search(self, query, top_k=10): """关键词搜索""" query_body = { "query": { "multi_match": { "query": query, "fields": ["title^2", "content"], "type": "best_fields" } }, "size": top_k } response = self.es.search(index='documents', body=query_body) results = response['hits']['hits'] return [ { 'id': r['_id'], 'score': r['_score'], 'source': r['_source'], 'type': 'keyword' } for r in results ] def vector_search(self, query, top_k=10): """向量搜索""" # 生成查询向量 query_embedding = self.encoder.encode(query).tolist() query_body = { "query": { "script_score": { "query": {"match_all": {}}, "script": { "source": "cosineSimilarity(params.query_vector, 'embedding') + 1.0", "params": {"query_vector": query_embedding} } } }, "size": top_k } response = self.es.search(index='documents', body=query_body) results = response['hits']['hits'] return [ { 'id': r['_id'], 'score': r['_score'], 'source': r['_source'], 'type': 'vector' } for r in results ] def hybrid_search(self, query, alpha=0.5, top_k=10): """混合搜索""" # 分别执行关键词和向量搜索 keyword_results = self.keyword_search(query, top_k=top_k * 2) vector_results = self.vector_search(query, top_k=top_k * 2) # 合并结果 combined = {} # 处理关键词结果 for result in keyword_results: doc_id = result['id'] if doc_id not in combined: combined[doc_id] = { 'id': doc_id, 'source': result['source'], 'keyword_score': 0, 'vector_score': 0 } combined[doc_id]['keyword_score'] = result['score'] # 处理向量结果 for result in vector_results: doc_id = result['id'] if doc_id not in combined: combined[doc_id] = { 'id': doc_id, 'source': result['source'], 'keyword_score': 0, 'vector_score': 0 } combined[doc_id]['vector_score'] = result['score'] # 归一化分数并融合 max_keyword = max(r['keyword_score'] for r in combined.values()) or 1 max_vector = max(r['vector_score'] for r in combined.values()) or 1 for doc in combined.values(): normalized_keyword = doc['keyword_score'] / max_keyword normalized_vector = doc['vector_score'] / max_vector doc['combined_score'] = ( alpha * normalized_keyword + (1 - alpha) * normalized_vector ) # 排序并返回top_k结果 sorted_results = sorted( combined.values(), key=lambda x: x['combined_score'], reverse=True ) return sorted_results[:top_k] # 使用示例 hybrid_search = HybridSearch() hybrid_search.create_index() # 索引文档 for i in range(100): doc = { 'id': f"doc_{i}", 'title': f"Document {i}", 'content': f"This is the content of document {i}", 'category': f"category_{i % 5}" } hybrid_search.index_document(doc) # 执行混合搜索 query = "machine learning document" results = hybrid_search.hybrid_search(query, alpha=0.5) for i, result in enumerate(results[:5], 1): print(f"{i}. {result['source']['title']} (Score: {result['combined_score']:.4f})")

Reciprocal Rank Fusion

def reciprocal_rank_fusion(results_dict, k=60): """ Reciprocal Rank Fusion算法 参数: results_dict: {检索方法: [结果列表]} k: 平滑参数，通常设置为60 返回: 融合后的排序结果 """ fused_scores = {} for method, results in results_dict.items(): for rank, result in enumerate(results): doc_id = result['id'] # 计算RRF分数 rrf_score = 1.0 / (k + rank + 1) if doc_id in fused_scores: fused_scores[doc_id] += rrf_score else: fused_scores[doc_id] = rrf_score # 按分数排序 sorted_results = sorted( fused_scores.items(), key=lambda x: x[1], reverse=True ) # 构建结果列表 final_results = [] for doc_id, score in sorted_results: # 从原始结果中获取文档详情 for method, results in results_dict.items(): for result in results: if result['id'] == doc_id: final_results.append({ 'id': doc_id, 'score': score, 'source': result['source'] }) break break return final_results # 使用示例 keyword_results = hybrid_search.keyword_search("query") vector_results = hybrid_search.vector_search("query") # 使用RRF融合 fused_results = reciprocal_rank_fusion({ 'keyword': keyword_results, 'vector': vector_results }, k=60) for i, result in enumerate(fused_results[:10], 1): print(f"{i}. {result['source']['title']} (RRF Score: {result['score']:.4f})")

🔄 重排序（Rerank）技术

两阶段检索

Cross-Encoder重排序

from sentence_transformers import CrossEncoder class Reranker: def __init__(self, model_name='cross-encoder/ms-marco-MiniLM-L-6-v2'): self.model = CrossEncoder(model_name) def rerank(self, query, candidates, top_k=10): """ 使用Cross-Encoder重排序 参数: query: 查询文本 candidates: 候选文档列表 top_k: 返回的top_k数量 """ # 构建query-document对 pairs = [[query, doc['content']] for doc in candidates] # 计算相关性分数 scores = self.model.predict(pairs) # 为候选文档添加分数 for doc, score in zip(candidates, scores): doc['rerank_score'] = score # 按分数排序 reranked = sorted( candidates, key=lambda x: x['rerank_score'], reverse=True ) return reranked[:top_k] # 使用示例 reranker = Reranker() # 第一阶段检索（获取候选集） candidates = hybrid_search.hybrid_search(query, top_k=100) # 第二阶段重排序 final_results = reranker.rerank(query, candidates, top_k=10) for i, result in enumerate(final_results, 1): print(f"{i}. {result['title']} (Score: {result['rerank_score']:.4f})")

多特征融合

from sklearn.ensemble import RandomForestClassifier import numpy as np class FeatureBasedReranker: def __init__(self): self.model = RandomForestClassifier(n_estimators=100) def extract_features(self, query, document): """提取文档特征""" features = [] # 1. 关键词匹配特征 features.append(len(set(query.lower().split()) & set(document['title'].lower().split()))) # 2. 向量相似度特征 query_vec = self.encoder.encode(query) doc_vec = self.encoder.encode(document['content']) similarity = np.dot(query_vec, doc_vec) / ( np.linalg.norm(query_vec) * np.linalg.norm(doc_vec) ) features.append(similarity) # 3. 文档长度特征 features.append(len(document['content'].split())) # 4. 时间特征 if 'timestamp' in document: days_old = (time.time() - document['timestamp']) / 86400 features.append(days_old) return features def train(self, training_data): """ 训练重排序模型 training_data: [(query, positive_doc, negative_docs)] """ X = [] y = [] for query, pos_doc, neg_docs in training_data: # 正样本特征 pos_features = self.extract_features(query, pos_doc) X.append(pos_features) y.append(1) # 负样本特征 for neg_doc in neg_docs: neg_features = self.extract_features(query, neg_doc) X.append(neg_features) y.append(0) self.model.fit(X, y) def rerank(self, query, candidates, top_k=10): """使用训练好的模型重排序""" # 提取特征 features = [ self.extract_features(query, doc) for doc in candidates ] # 预测相关性分数 scores = self.model.predict_proba(features)[:, 1] # 添加分数并排序 for doc, score in zip(candidates, scores): doc['rerank_score'] = score reranked = sorted( candidates, key=lambda x: x['rerank_score'], reverse=True ) return reranked[:top_k]

A/B测试

# A/B测试框架 class ABTestFramework: def __init__(self, control_search, treatment_search): self.control = control_search self.treatment = treatment_search self.results = [] def run_test(self, query, ground_truth): """运行A/B测试""" # 对照组（baseline） control_results = self.control.search(query, top_k=10) control_metrics = self.evaluate(control_results, ground_truth) # 实验组（new） treatment_results = self.treatment.search(query, top_k=10) treatment_metrics = self.evaluate(treatment_results, ground_truth) # 记录结果 self.results.append({ 'query': query, 'control': control_metrics, 'treatment': treatment_metrics }) def evaluate(self, results, ground_truth): """评估搜索结果""" relevant_ids = set(ground_truth['relevant']) returned_ids = [r['id'] for r in results] # 计算指标 precision_at_10 = len(set(returned_ids[:10]) & relevant_ids) / 10 recall_at_10 = len(set(returned_ids[:10]) & relevant_ids) / len(relevant_ids) return { 'precision@10': precision_at_10, 'recall@10': recall_at_10 } def analyze(self): """分析测试结果""" control_precisions = [r['control']['precision@10'] for r in self.results] treatment_precisions = [r['treatment']['precision@10'] for r in self.results] import scipy.stats as stats t_stat, p_value = stats.ttest_ind(treatment_precisions, control_precisions) print(f"Control Precision@10: {np.mean(control_precisions):.4f}") print(f"Treatment Precision@10: {np.mean(treatment_precisions):.4f}") print(f"Improvement: {(np.mean(treatment_precisions) - np.mean(control_precisions)):.4f}") print(f"P-value: {p_value:.4f}") return p_value < 0.05 # 显著性检验
