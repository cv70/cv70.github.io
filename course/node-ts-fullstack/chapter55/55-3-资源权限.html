55.3 资源权限

细粒度的资源级权限控制

🎯 学习目标

  • 理解资源级权限控制的概念
  • 掌握所有权检查的实现方式
  • 学会实现数据行级权限控制
  • 实现资源访问授权管理

📚 资源权限概述

资源权限（Resource-Level Permission）是指对具体资源实例的访问控制。例如，用户只能编辑自己创建的文章，管理员可以编辑所有文章。这需要比角色权限更细粒度的控制。

// 权限控制层级 // 1. 功能权限（Function Level） // - 用户能否访问某个功能 // - 如：article:create, article:delete // 2. 资源权限（Resource Level） // - 用户能否操作某个具体资源 // - 如：能否编辑 ID=123 的文章 // 3. 字段权限（Field Level） // - 用户能否访问某些字段 // - 如：普通用户看不到用户手机号 // 示例场景 管理员 -> 可以删除任何文章 普通用户 -> 只能删除自己的文章 文章作者 -> 可以编辑自己的文章
💡
设计原则

资源权限检查应在功能权限检查之后进行。先判断用户是否有操作权限，再判断是否有资源访问权限。

🔧 所有权检查实现

// src/middleware/ownership.middleware.ts import { Request, Response, NextFunction } from 'express'; import { PrismaClient } from '@prisma/client'; const prisma = new PrismaClient(); /** * 检查文章所有权 */ export async function checkArticleOwnership( req: Request, res: Response, next: NextFunction ) { const userId = req.user?.id; const articleId = parseInt(req.params.id); if (!userId) { return res.status(401).json({ message: '未认证' }); } const article = await prisma.article.findUnique({ where: { id: articleId }, select: { authorId: true } }); if (!article) { return res.status(404).json({ message: '文章不存在' }); } // 检查是否是作者或管理员 const isOwner = article.authorId === userId; const isAdmin = req.user?.roles.includes('admin'); if (!isOwner && !isAdmin) { return res.status(403).json({ message: '无权访问此文章' }); } // 将资源信息附加到请求对象 req.resource = { articleId, isOwner }; next(); } /** * 通用所有权检查工厂函数 */ export function checkOwnership(options: { resourceModel: string; // Prisma 模型名 resourceIdParam: string; // 路由参数名 ownerField: string; // 所有者字段名 adminRoles?: string[]; // 管理员角色列表 }) { return async (req: Request, res: Response, next: NextFunction) => { const userId = req.user?.id; const resourceId = parseInt(req.params[options.resourceIdParam]); if (!userId) { return res.status(401).json({ message: '未认证' }); } // 动态查询资源 const resource = await (prisma as any)[options.resourceModel].findUnique({ where: { id: resourceId }, select: { [options.ownerField]: true } }); if (!resource) { return res.status(404).json({ message: '资源不存在' }); } const isOwner = resource[options.ownerField] === userId; const isAdmin = options.adminRoles?.some(role => req.user?.roles.includes(role) ); if (!isOwner && !isAdmin) { return res.status(403).json({ message: '无权访问此资源' }); } next(); }; }

📊 数据行级权限

// src/services/row-level-security.service.ts interface RowLevelFilter { userId: number; roles: string[]; model: string; } export class RowLevelSecurityService { /** * 根据用户权限生成查询过滤条件 */ static getArticleFilter(params: RowLevelFilter) { const { userId, roles } = params; // 管理员可以看到所有文章 if (roles.includes('admin')) { return {}; // 无过滤条件 } // 编辑可以看到已发布和自己草稿 if (roles.includes('editor')) { return { OR: [ { status: 'published' }, { authorId: userId } ] }; } // 普通用户只能看到已发布的文章 return { status: 'published' }; } /** * 用户可操作的文章过滤 */ static getEditableArticleFilter(params: RowLevelFilter) { const { userId, roles } = params; if (roles.includes('admin')) { return {}; // 管理员可编辑所有 } // 只能编辑自己的文章 return { authorId: userId }; } } // 在 Controller 中使用 export class ArticleController { async getArticles(req: Request, res: Response) { const filter = RowLevelSecurityService.getArticleFilter({ userId: req.user!.id, roles: req.user!.roles, model: 'article' }); const articles = await prisma.article.findMany({ where: filter, include: { author: { select: { id: true, name: true } } } }); res.json(articles); } }

🔄 资源授权管理

// Prisma Schema - 资源授权模型 model ResourceGrant { id Int @id @default(autoincrement()) resourceType String // 资源类型：article, document, project resourceId Int // 资源ID userId Int // 被授权用户 permission String // 授权权限：view, edit, manage grantedBy Int // 授权人 expiresAt DateTime? // 过期时间 createdAt DateTime @default(now()) user User @relation(fields: [userId], references: [id]) granter User @relation("GrantedBy", fields: [grantedBy], references: [id]) @@unique([resourceType, resourceId, userId, permission]) } // 授权服务 export class ResourceGrantService { /** * 检查用户对特定资源的权限 */ async checkResourcePermission( userId: number, resourceType: string, resourceId: number, permission: string ): Promise { // 检查是否有直接授权 const grant = await prisma.resourceGrant.findFirst({ where: { userId, resourceType, resourceId, permission, OR: [ { expiresAt: null }, { expiresAt: { gt: new Date() } } ] } }); if (grant) return true; // 检查是否是资源所有者 return this.checkOwnership(userId, resourceType, resourceId); } /** * 授权用户访问资源 */ async grantPermission(params: { resourceType: string; resourceId: number; userId: number; permission: string; grantedBy: number; expiresAt?: Date; }) { return prisma.resourceGrant.upsert({ where: { resourceType_resourceId_userId_permission: { resourceType: params.resourceType, resourceId: params.resourceId, userId: params.userId, permission: params.permission } }, update: { grantedBy: params.grantedBy, expiresAt: params.expiresAt }, create: params }); } /** * 撤销授权 */ async revokePermission( resourceType: string, resourceId: number, userId: number, permission: string ) { return prisma.resourceGrant.delete({ where: { resourceType_resourceId_userId_permission: { resourceType, resourceId, userId, permission } } }); } }
⚠️
安全提示

资源授权应有审计日志，记录授权人、被授权人、时间和权限内容。
