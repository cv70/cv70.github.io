81.4 组件基础

Vue 组件的定义与使用

📖 本节概述

组件是 Vue 应用的基本构建块。本节将介绍如何定义和使用组件、组件的 Props 和 Events、插槽的使用，以及组件注册方式。

🎯 学习目标

  • 掌握组件的定义方式
  • 学会使用 Props 传递数据
  • 掌握自定义事件的使用
  • 理解插槽的内容分发机制

📦 组件定义

单文件组件 (SFC)

<!-- Button.vue --> <template> <button :class="['btn', `btn-${type}`]" @click="handleClick"> <slot /> </button> </template> <script setup lang="ts"> // 定义 Props interface Props { type?: 'primary' | 'secondary' | 'danger'; } const props = withDefaults(defineProps<Props>(), { type: 'primary' }); // 定义 Events const emit = defineEmits<{ (e: 'click', event: MouseEvent): void; }>(); function handleClick(event: MouseEvent) { emit('click', event); } </script> <style scoped> .btn { padding: 8px 16px; border-radius: 4px; } .btn-primary { background: #42b883; color: white; } .btn-secondary { background: #666; color: white; } .btn-danger { background: #e74c3c; color: white; } </style>

📥 Props

定义 Props

// 使用 TypeScript 接口定义 interface Props { title: string; // 必需 count?: number; // 可选 items: string[]; // 数组类型 user: { // 对象类型 id: number; name: string; }; } const props = defineProps<Props>(); // 带默认值 const props = withDefaults(defineProps<Props>(), { count: 0, items: () => [] }); // 运行时声明（不推荐，TypeScript 支持较差） const props = defineProps({ title: { type: String, required: true }, count: { type: Number, default: 0 } });

使用 Props

<!-- 父组件 --> <template> <UserCard :title="userTitle" :count="10" :items="['a', 'b', 'c']" :user="{ id: 1, name: 'John' }" /> </template> <!-- 子组件 --> <template> <div> <h2>{{ title }}</h2> <p>Count: {{ count }}</p> <ul> <li v-for="item in items" :key="item">{{ item }}</li> </ul> </div> </template> <script setup lang="ts"> const props = defineProps<{ title: string; count: number; items: string[]; }>(); </script>

📤 自定义事件

定义和触发事件

// 定义事件 const emit = defineEmits<{ (e: 'update', value: string): void; (e: 'delete', id: number): void; (e: 'change'): void; }>(); // 触发事件 function handleUpdate() { emit('update', 'new value'); } // 运行时声明 const emit = defineEmits(['update', 'delete']);

监听事件

<!-- 父组件 --> <template> <ChildComponent @update="handleUpdate" @delete="handleDelete" /> </template> <script setup lang="ts"> function handleUpdate(value: string) { console.log('Updated:', value); } function handleDelete(id: number) { console.log('Deleted:', id); } </script>

v-model 双向绑定

<!-- 子组件 --> <template> <input :value="modelValue" @input="$emit('update:modelValue', $event.target.value)" /> </template> <script setup lang="ts"> const props = defineProps<{ modelValue: string }>(); const emit = defineEmits<{ (e: 'update:modelValue', value: string): void; }>(); </script> <!-- 父组件 --> <template> <CustomInput v-model="text" /> <!-- 等价于 --> <CustomInput :modelValue="text" @update:modelValue="text = $event" /> </template> <!-- 多个 v-model --> <template> <UserForm v-model:name="name" v-model:email="email" /> </template>

🔲 插槽

默认插槽

<!-- 子组件 Card.vue --> <template> <div class="card"> <slot>默认内容</slot> </div> </template> <!-- 父组件 --> <template> <Card> <p>自定义内容</p> </Card> </template>

具名插槽

<!-- 子组件 Layout.vue --> <template> <div class="layout"> <header> <slot name="header" /> </header> <main> <slot /> </main> <footer> <slot name="footer" /> </footer> </div> </template> <!-- 父组件 --> <template> <Layout> <template #header> <h1>页面标题</h1> </template> <p>主要内容</p> <template #footer> <p>页脚</p> </template> </Layout> </template>

作用域插槽

<!-- 子组件 List.vue --> <template> <ul> <li v-for="item in items" :key="item.id"> <slot :item="item" :index="index"> {{ item.name }} </slot> </li> </ul> </template> <script setup lang="ts"> defineProps<{ items: { id: number; name: string }[]; }>(); </script> <!-- 父组件 --> <template> <List :items="users"> <template #default="{ item, index }"> <span>{{ index + 1 }}. {{ item.name }}</span> <button @click="remove(item.id)">删除</button> </template> </List> </template>

📋 组件注册

局部注册

<!-- 使用 script setup 自动注册 --> <script setup lang="ts"> import MyButton from './MyButton.vue'; // 无需显式注册，直接可用 </script> <template> <MyButton /> </template> <!-- 不使用 script setup --> <script lang="ts"> import MyButton from './MyButton.vue'; export default { components: { MyButton } } </script>

全局注册

// main.ts import { createApp } from 'vue'; import App from './App.vue'; import MyButton from './components/MyButton.vue'; const app = createApp(App); // 全局注册 app.component('MyButton', MyButton); app.mount('#app'); // 批量全局注册 const components = import.meta.glob('./components/*.vue'); Object.entries(components).forEach(([path, component]) => { const name = path.match(/\/([^/]+)\.vue$/)?.[1]; if (name) { app.component(name, component); } });

📝 本节小结

  • 单文件组件是 Vue 推荐的组件定义方式
  • Props 用于父组件向子组件传递数据
  • 自定义事件用于子组件向父组件通信
  • v-model 实现双向绑定
  • 插槽用于内容分发，支持默认、具名、作用域插槽
