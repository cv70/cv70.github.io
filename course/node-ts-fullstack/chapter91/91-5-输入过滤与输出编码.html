91.5 输入过滤与输出编码

安全数据处理实践

🎯 学习目标

  • 掌握输入过滤的方法
  • 学会输出编码的最佳实践
  • 理解两者的区别和应用场景
  • 了解常用安全库的使用

📚 输入过滤与输出编码的区别

# 输入过滤 (Input Filtering) 在数据进入系统时进行清理和验证 ## 特点 - 在数据进入时处理 - 移除或转换危险字符 - 一次处理，多处使用 ## 问题 - 可能过滤掉合法内容 - 无法预见所有输出场景 - 可能被绕过 # 输出编码 (Output Encoding) 在数据输出时根据上下文进行编码 ## 特点 - 在数据输出时处理 - 根据输出上下文选择编码方式 - 不改变原始数据 ## 优势 - 更安全可靠 - 保持数据完整性 - 针对不同上下文正确处理 # 最佳实践 输入验证 + 输出编码 而非 输入过滤 # 处理流程 用户输入 → 输入验证 → 存储 → 输出编码 → 显示 (白名单) (根据上下文)
💡
优先输出编码

输出编码比输入过滤更可靠。因为输入过滤无法预见所有输出场景，而输出编码可以根据具体上下文正确处理。

🔧 输入验证

# 使用 Zod 进行输入验证 import { z } from 'zod'; // 定义验证模式 const userSchema = z.object({ username: z.string() .min(3, '用户名至少 3 个字符') .max(20, '用户名最多 20 个字符') .regex(/^[a-zA-Z0-9_]+$/, '用户名只能包含字母、数字和下划线'), email: z.string().email('请输入有效的邮箱地址'), age: z.number() .int('年龄必须是整数') .min(0, '年龄不能小于 0') .max(150, '年龄不能大于 150'), bio: z.string() .max(500, '个人简介最多 500 字符') .optional(), }); // 验证函数 function validateUser(input: unknown) { const result = userSchema.safeParse(input); if (!result.success) { return { success: false, errors: result.error.flatten().fieldErrors, }; } return { success: true, data: result.data, }; } // Express 中使用 app.post('/user', (req, res) => { const validation = validateUser(req.body); if (!validation.success) { return res.status(400).json({ errors: validation.errors }); } // 使用验证后的数据 createUser(validation.data); }); # 白名单验证 const ALLOWED_TAGS = ['p', 'b', 'i', 'em', 'strong']; const ALLOWED_PROTOCOLS = ['http', 'https', 'mailto']; function validateUrl(url: string): boolean { try { const parsed = new URL(url); return ALLOWED_PROTOCOLS.includes(parsed.protocol.slice(0, -1)); } catch { return false; } }

⚙️ 输出编码实践

# HTML 编码 import escapeHtml from 'escape-html'; function renderComment(comment: string): string { return `<div class="comment">${escapeHtml(comment)}</div>`; } # 属性编码 function renderLink(url: string, text: string): string { const safeUrl = escapeHtml(url); const safeText = escapeHtml(text); return `<a href="${safeUrl}">${safeText}</a>`; } # JavaScript 编码 function renderScript(data: object): string { return `<script>const data = ${JSON.stringify(data)};</script>`; } # URL 编码 function buildSearchUrl(query: string): string { return `/search?q=${encodeURIComponent(query)}`; } # CSS 编码（应避免动态 CSS） // 如必须，使用白名单 function safeColor(color: string): string { const allowed = ['red', 'blue', 'green', 'black', 'white']; return allowed.includes(color) ? color : 'black'; } # 使用 DOMPurify 清理 HTML import DOMPurify from 'dompurify'; function sanitizeHtml(dirty: string): string { return DOMPurify.sanitize(dirty, { ALLOWED_TAGS: ['p', 'b', 'i', 'em', 'strong', 'a', 'br'], ALLOWED_ATTR: ['href'], ALLOW_DATA_ATTR: false, }); } // 允许特定协议的链接 DOMPurify.addHook('uponSanitizeAttribute', (node, data) => { if (data.attrName === 'href') { const href = data.attrValue; if (!/^https?:\/\//i.test(href)) { data.attrValue = ''; } } });
