77.4 嵌套路由

构建多层路由结构

🎯 学习目标

  • 理解嵌套路由的概念
  • 掌握 Outlet 组件的使用
  • 学会配置多级嵌套路由
  • 了解嵌套布局的实现

📚 嵌套路由基础

import { Outlet } from 'react-router-dom'; // 父组件使用 Outlet 渲染子路由 function Dashboard() { return ( <div className="dashboard"> <Sidebar /> <div className="content"> <Outlet /> {/* 子路由在这里渲染 */} </div> </div> ); } // 路由配置 const router = createBrowserRouter([ { path: '/dashboard', element: <Dashboard />, // 父路由 children: [ { index: true, element: <DashboardHome /> }, // /dashboard { path: 'analytics', element: <Analytics /> }, // /dashboard/analytics { path: 'settings', element: <Settings /> }, // /dashboard/settings ], }, ]); // URL 映射： // /dashboard -> Dashboard + DashboardHome // /dashboard/analytics -> Dashboard + Analytics // /dashboard/settings -> Dashboard + Settings

🔧 多级嵌套路由

// 多级嵌套路由结构 function App() { return ( <div className="app"> <Header /> <Outlet /> <Footer /> </div> ); } function AdminLayout() { return ( <div className="admin"> <AdminSidebar /> <Outlet /> </div> ); } function UsersLayout() { return ( <div className="users"> <UserList /> <Outlet /> </div> ); } const router = createBrowserRouter([ { path: '/', element: <App />, // 第一层 children: [ { index: true, element: <Home /> }, { path: 'admin', element: <AdminLayout />, // 第二层 children: [ { index: true, element: <AdminHome /> }, { path: 'users', element: <UsersLayout />, // 第三层 children: [ { index: true, element: <UsersList /> }, { path: ':userId', element: <UserDetail /> }, { path: 'new', element: <NewUser /> }, ], }, { path: 'settings', element: <AdminSettings /> }, ], }, ], }, ]); // URL 映射： // / -> App + Home // /admin -> App + AdminLayout + AdminHome // /admin/users -> App + AdminLayout + UsersLayout + UsersList // /admin/users/123 -> App + AdminLayout + UsersLayout + UserDetail // /admin/users/new -> App + AdminLayout + UsersLayout + NewUser

📊 Outlet 上下文

import { Outlet, useOutletContext } from 'react-router-dom'; // 父组件传递上下文 function Dashboard() { const [user, setUser] = useState(currentUser); return ( <div className="dashboard"> <Sidebar /> <Outlet context={{ user, setUser }} /> </div> ); } // 子组件获取上下文 function Profile() { const { user, setUser } = useOutletContext<{ user: User; setUser: (user: User) => void; }>(); return ( <div> <h1>Profile</h1> <p>Name: {user.name}</p> <button onClick={() => setUser({ ...user, name: 'New Name' })}> Update Name </button> </div> ); } // 多层上下文 function AdminLayout() { const { user, setUser } = useOutletContext(); // 从父级获取 return ( <div> <AdminSidebar user={user} /> <Outlet context={{ user, setUser, adminMode: true }} /> {/* 传递给子级 */} </div> ); } // 自定义 Hook 封装 function useDashboardContext() { return useOutletContext<DashboardContext>(); } function Settings() { const { user, setUser } = useDashboardContext(); // ... }
