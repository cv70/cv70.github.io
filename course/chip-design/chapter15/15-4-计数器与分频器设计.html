🔢 二进制计数器

// 8位二进制计数器
module counter_8bit(
    input  clk,
    input  rst_n,
    input  enable,
    output [7:0] count,
    output       overflow
);
    reg [7:0] count_reg;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            count_reg <= 8'b0000_0000;
        else if (enable)
            count_reg <= count_reg + 1;
    end

    assign count = count_reg;
    assign overflow = (count_reg == 8'b1111_1111) && enable;
endmodule

🔟 BCD计数器

// 2位BCD计数器（0-99）
module bcd_counter_2digit(
    input  clk,
    input  rst_n,
    input  enable,
    output [7:0] bcd,      // [7:4]十位, [3:0]个位
    output       carry
);
    reg [3:0] ones, tens;
    wire ones_carry = (ones == 4'd9) && enable;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            ones <= 4'd0;
            tens <= 4'd0;
        end
        else if (enable) begin
            if (ones < 4'd9)
                ones <= ones + 1;
            else begin
                ones <= 4'd0;
                if (tens < 4'd9)
                    tens <= tens + 1;
                else
                    tens <= 4'd0;
            end
        end
    end

    assign bcd = {tens, ones};
    assign carry = (tens == 4'd9) && (ones == 4'd9) && enable;
endmodule

⏱️ 分频器

// 偶数分频器
module clock_divider_even #(parameter DIV = 10)(
    input  clk,
    input  rst_n,
    output clk_div
);
    reg [$clog2(DIV/2)-1:0] count;
    reg clk_div_reg;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            count <= 0;
            clk_div_reg <= 0;
        end
        else begin
            if (count == DIV/2 - 1) begin
                count <= 0;
                clk_div_reg <= ~clk_div_reg;
            end
            else
                count <= count + 1;
        end
    end

    assign clk_div = clk_div_reg;
endmodule
