📚 性能分析概述

# 为什么需要性能分析 ## 用户体验 - 响应速度直接影响用户满意度 - 慢请求导致用户流失 ## 资源效率 - 发现资源浪费 - 优化成本 ## 可扩展性 - 发现性能瓶颈 - 为扩展做准备 # 性能分析维度 ┌─────────────────┬──────────────────────────────────┐ │ 维度 │ 指标 │ ├─────────────────┼──────────────────────────────────┤ │ CPU │ 使用率、负载、函数执行时间 │ │ 内存 │ 使用量、泄漏、GC 频率 │ │ 事件循环 │ 延迟、阻塞、任务队列长度 │ │ I/O │ 磁盘读写、网络延迟 │ │ 应用 │ 响应时间、吞吐量、错误率 │ └─────────────────┴──────────────────────────────────┘ # Node.js 性能特点 - 单线程事件循环 - 非阻塞 I/O - V8 引擎优化 - 异步编程模型

🔧 内置诊断工具

# Node.js 内置性能标志 ## 启动参数 node --inspect app.js # 启用调试器 node --inspect-brk app.js # 启动时断点 node --prof app.js # 生成性能分析文件 node --trace-opt app.js # 追踪优化 node --trace-deopt app.js # 追踪反优化 ## 生成 CPU 分析 # 运行应用 node --prof app.js # 生成报告 node --prof-process isolate-*.log > processed.txt # 使用 Chrome DevTools node --inspect app.js # 打开 chrome://inspect # 点击 "Open dedicated DevTools for Node" # 选择 "Profiler" 标签 # V8 分析 API const v8 = require('v8'); const fs = require('fs'); // 获取堆统计 const stats = v8.getHeapStatistics(); console.log(stats); // 获取堆快照 const snapshotStream = v8.getHeapSnapshot(); const fileStream = fs.createWriteStream('heap.heapsnapshot'); snapshotStream.pipe(fileStream); // 代码覆盖 if (process.env.NODE_V8_COVERAGE) { // 自动输出覆盖率到指定目录 }

⚙️ 性能监控中间件

# Express 性能监控 import express from 'express'; const app = express(); // 请求时间监控 app.use((req, res, next) => { const start = process.hrtime.bigint(); res.on('finish', () => { const end = process.hrtime.bigint(); const duration = Number(end - start) / 1e6; // 毫秒 console.log({ method: req.method, url: req.url, status: res.statusCode, duration: `${duration.toFixed(2)}ms`, }); }); next(); }); // 事件循环延迟监控 let lastCheck = process.hrtime.bigint(); setInterval(() => { const now = process.hrtime.bigint(); const delay = Number(now - lastCheck) / 1e6 - 1000; // 预期 1 秒 if (delay > 10) { console.warn(`Event loop delay: ${delay.toFixed(2)}ms`); } lastCheck = now; }, 1000); # 使用 clinic.js # 安装 npm install -g clinic # CPU 分析 clinic doctor -- node app.js # 自动打开浏览器显示分析结果 # 内存分析 clinic heapprofiler -- node app.js # 事件循环分析 clinic bubbleprof -- node app.js # 综合分析 clinic flame -- node app.js

🚀 APM 工具集成

# New Relic 集成 # 安装 npm install newrelic # 配置 newrelic.js exports.config = { app_name: ['My Application'], license_key: 'YOUR_LICENSE_KEY', logging: { level: 'info', }, }; # 在应用入口最先引入 import 'newrelic'; # Elastic APM import apm from 'elastic-apm-node'; apm.start({ serviceName: 'my-service', serverUrl: 'http://localhost:8200', environment: 'production', }); # 自定义事务 const transaction = apm.startTransaction('CustomTransaction', 'custom'); const span = transaction.startSpan('operation'); // 执行操作... span.end(); transaction.end(); # Prometheus + Grafana import client from 'prom-client'; // 启用默认指标 client.collectDefaultMetrics(); // 自定义指标 const httpRequestDuration = new client.Histogram({ name: 'http_request_duration_seconds', help: 'Duration of HTTP requests in seconds', labelNames: ['method', 'route', 'status_code'], buckets: [0.1, 0.5, 1, 2, 5], }); // 在中间件中使用 app.use((req, res, next) => { const end = httpRequestDuration.startTimer(); res.on('finish', () => { end({ method: req.method, route: req.path, status_code: res.statusCode }); }); next(); }); // 暴露指标端点 app.get('/metrics', async (req, res) => { res.set('Content-Type', client.register.contentType); res.send(await client.register.metrics()); });
