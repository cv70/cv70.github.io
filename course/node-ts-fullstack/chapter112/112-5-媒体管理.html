🔧 文件上传服务

// media.entity.ts @Entity('media') export class Media { @PrimaryGeneratedColumn('uuid') id: string; @Column() filename: string; @Column() originalName: string; @Column() mimeType: string; @Column() size: number; @Column() path: string; @Column({ nullable: true }) url: string; @Column({ nullable: true }) thumbnailUrl: string; @Column({ nullable: true }) width: number; @Column({ nullable: true }) height: number; @Column() uploadedBy: string; @Column({ type: 'json', nullable: true }) metadata: Record; @CreateDateColumn() createdAt: Date; } // media.service.ts @Injectable() export class MediaService { constructor( @InjectRepository(Media) private mediaRepo: Repository, private s3Service: S3Service, ) {} async uploadImage( file: Express.Multer.File, userId: string, options?: { compress?: boolean; watermark?: boolean } ): Promise { // 生成文件名 const ext = path.extname(file.originalname); const filename = `${uuidv4()}${ext}`; // 图片处理 let buffer = file.buffer; let width: number, height: number; const metadata = await sharp(buffer).metadata(); width = metadata.width; height = metadata.height; // 压缩 if (options?.compress !== false) { buffer = await sharp(buffer) .resize(2048, null, { withoutEnlargement: true }) .jpeg({ quality: 80 }) .toBuffer(); } // 上传到存储 const url = await this.s3Service.upload(filename, buffer, file.mimetype); // 生成缩略图 const thumbnailBuffer = await sharp(file.buffer) .resize(200, 200, { fit: 'cover' }) .jpeg({ quality: 70 }) .toBuffer(); const thumbnailFilename = `thumb_${filename}`; const thumbnailUrl = await this.s3Service.upload( thumbnailFilename, thumbnailBuffer, 'image/jpeg' ); // 保存记录 const media = this.mediaRepo.create({ filename, originalName: file.originalname, mimeType: file.mimetype, size: buffer.length, path: filename, url, thumbnailUrl, width, height, uploadedBy: userId, }); return this.mediaRepo.save(media); } // 删除文件 async deleteMedia(id: string, userId: string): Promise { const media = await this.mediaRepo.findOne({ where: { id } }); if (!media) { throw new NotFoundException('文件不存在'); } // 删除存储文件 await this.s3Service.delete(media.path); if (media.thumbnailUrl) { await this.s3Service.delete(`thumb_${media.path}`); } // 删除记录 await this.mediaRepo.remove(media); } } // S3存储服务 @Injectable() export class S3Service { private s3: S3Client; private bucket: string; constructor() { this.s3 = new S3Client({ region: process.env.S3_REGION, endpoint: process.env.S3_ENDPOINT, credentials: { accessKeyId: process.env.S3_ACCESS_KEY, secretAccessKey: process.env.S3_SECRET_KEY, }, }); this.bucket = process.env.S3_BUCKET; } async upload(key: string, body: Buffer, contentType: string): Promise { await this.s3.send(new PutObjectCommand({ Bucket: this.bucket, Key: key, Body: body, ContentType: contentType, })); return `${process.env.CDN_URL}/${key}`; } async delete(key: string): Promise { await this.s3.send(new DeleteObjectCommand({ Bucket: this.bucket, Key: key, })); } // 生成签名URL（私有文件） async getSignedUrl(key: string, expiresIn: number = 3600): Promise { const command = new GetObjectCommand({ Bucket: this.bucket, Key: key, }); return getSignedUrl(this.s3, command, { expiresIn }); } }
