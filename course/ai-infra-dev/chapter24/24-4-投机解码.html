什么是投机解码

投机解码（Speculative Decoding）是一种大语言模型推理加速技术。 它使用一个快速的小模型（Draft模型）预测多个token， 然后用大模型（Target模型）验证这些预测，正确的被接受，错误的被丢弃。

💡 核心思想

Draft-Verify机制

投机解码的核心是Draft-Verify机制：

Draft阶段
  • 小模型快速预测N个token
  • 计算成本低，速度快
  • 生成候选序列
Verify阶段
  • 大模型验证预测结果
  • 接受正确的token
  • 拒绝错误的token
加速效果
  • 接受率高则加速明显
  • 通常加速2-3倍
  • 不改变输出结果

📊 加速效果

最佳情况

  • 接受率100%
  • 加速比 ≈ Draft速度
  • 理想情况
  • 可达10倍加速

  • 接受率60-80%
  • 加速比 ≈ 接受率 × Draft速度
  • 常见情况

💻 投机解码实现示例

import torch
import torch.nn as nn
from typing import Tuple

class SpeculativeDecoder:
    """
    投机解码器
    """
    def __init__(self, target_model, draft_model, n_draft_tokens=5):
        self.target_model = target_model  # 大模型
        self.draft_model = draft_model    # 小模型
        self.n_draft_tokens = n_draft_tokens

    def draft(self, input_ids: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Draft阶段：小模型预测N个token
        """
        draft_tokens = []
        draft_logits = []

        current_ids = input_ids.clone()

        for _ in range(self.n_draft_tokens):
            # 小模型预测下一个token
            with torch.no_grad():
                outputs = self.draft_model(current_ids)
                logits = outputs.logits[:, -1, :]  # (batch, vocab_size)
                probs = torch.softmax(logits, dim=-1)

            # 采样或greedy
            next_token = torch.argmax(probs, dim=-1, keepdim=True)

            draft_tokens.append(next_token)
            draft_logits.append(logits)

            # 更新输入
            current_ids = torch.cat([current_ids, next_token], dim=-1)

        draft_tokens = torch.cat(draft_tokens, dim=-1)  # (batch, n_draft)
        draft_logits = torch.stack(draft_logits, dim=1)   # (batch, n_draft, vocab_size)

        return draft_tokens, draft_logits

    def verify(self, input_ids: torch.Tensor, draft_tokens: torch.Tensor,
              draft_logits: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Verify阶段：大模型验证
        """
        # 拼接原始输入和draft tokens
        full_input = torch.cat([input_ids, draft_tokens], dim=-1)

        # 大模型前向传播
        with torch.no_grad():
            outputs = self.target_model(full_input)
            target_logits = outputs.logits[:, -(self.n_draft_tokens+1):, :]  # (batch, n_draft+1, vocab_size)

        # 计算接受情况
        accepted_tokens = []
        accept_mask = []

        for i in range(self.n_draft_tokens):
            # 比较draft和target的logits
            # 这里简化处理，实际应该计算概率比值
            draft_prob = torch.softmax(draft_logits[:, i, :], dim=-1)
            target_prob = torch.softmax(target_logits[:, i, :], dim=-1)

            draft_token = draft_tokens[:, i]

            # 接受条件（简化）
            accepted = (torch.argmax(target_prob, dim=-1) == draft_token) & \
                      (target_prob.gather(1, draft_token.unsqueeze(-1)).squeeze(-1) > 0.5)

            accept_mask.append(accepted)
            accepted_tokens.append(draft_token * accepted)

            # 如果不接受，停止接受后续token
            if not accepted.any():
                break

        accept_mask = torch.stack(accept_mask, dim=-1)  # (batch, n_draft)
        accepted_tokens = torch.stack(accepted_tokens, dim=-1)

        # 最后的target token总是接受
        final_token = torch.argmax(target_logits[:, -1, :], dim=-1, keepdim=True)

        return accepted_tokens, final_token, accept_mask

    def generate(self, input_ids: torch.Tensor, max_tokens: int) -> torch.Tensor:
        """
        使用投机解码生成
        """
        current_ids = input_ids.clone()
        generated_tokens = []

        for _ in range(max_tokens // self.n_draft_tokens):
            # Draft阶段
            draft_tokens, draft_logits = self.draft(current_ids)

            # Verify阶段
            accepted_tokens, final_token, accept_mask = self.verify(
                current_ids, draft_tokens, draft_logits
            )

            # 接受正确的token
            for i in range(self.n_draft_tokens):
                if accept_mask[:, i].any():
                    current_ids = torch.cat([current_ids, accepted_tokens[:, i:i+1]], dim=-1)
                    generated_tokens.append(accepted_tokens[:, i:i+1])
                else:
                    break

            # 接受最终的target token
            current_ids = torch.cat([current_ids, final_token], dim=-1)
            generated_tokens.append(final_token)

        return torch.cat(generated_tokens, dim=-1)

# Medusa方法（多分支投机）
class MedusaSpeculative:
    """
    Medusa：多分支投机解码
    """
    def __init__(self, target_model, num_heads=5):
        self.target_model = target_model
        self.num_heads = num_heads
        # Medusa heads（额外的小网络）
        self.medusa_heads = nn.ModuleList([
            nn.Linear(target_model.config.hidden_size, target_model.config.vocab_size)
            for _ in range(num_heads)
        ])

    def medusa_decode(self, input_ids: torch.Tensor, max_tokens: int) -> torch.Tensor:
        """
        Medusa投机解码
        """
        current_ids = input_ids.clone()

        for _ in range(max_tokens):
            # 获取target模型的隐藏状态
            with torch.no_grad():
                outputs = self.target_model(current_ids, output_hidden_states=True)
                hidden_states = outputs.hidden_states[-1]  # 最后一层隐藏状态

            # 每个head预测多个token
            medusa_predictions = []
            for head in self.medusa_heads:
                predictions = head(hidden_states)
                medusa_predictions.append(predictions)

            # 合并预测（简化处理）
            # 实际实现需要更复杂的验证逻辑
            next_token = torch.argmax(medusa_predictions[0], dim=-1, keepdim=True)
            current_ids = torch.cat([current_ids, next_token], dim=-1)

        return current_ids[:, input_ids.shape[-1]:]

# 使用示例
def example_usage():
    """
    使用投机解码的示例
    """
    from transformers import AutoModelForCausalLM, AutoTokenizer

    # 加载模型
    model_name = "meta-llama/Llama-2-7b-hf"
    target_model = AutoModelForCausalLM.from_pretrained(model_name)
    draft_model = AutoModelForCausalLM.from_pretrained("TinyLlama/TinyLlama-1.1B-Chat-v1.0")

    # 创建投机解码器
    spec_decoder = SpeculativeDecoder(
        target_model=target_model,
        draft_model=draft_model,
        n_draft_tokens=5
    )

    # Tokenize输入
    tokenizer = AutoTokenizer.from_pretrained(model_name)
    input_text = "写一个Python函数"
    input_ids = tokenizer.encode(input_text, return_tensors="pt")

    # 生成（使用投机解码）
    output_ids = spec_decoder.generate(input_ids, max_tokens=100)

    # 解码输出
    output_text = tokenizer.decode(output_ids[0], skip_special_tokens=True)
    print(output_text)

# vLLM中的投机解码
def vllm_speculative_decoding():
    """
    vLLM中的投机解码
    """
    from vllm import LLM, SamplingParams

    # 配置投机解码
    llm = LLM(
        model="meta-llama/Llama-2-7b-hf",
        speculative_model="TinyLlama/TinyLlama-1.1B-Chat-v1.0",
        num_speculative_tokens=5,  # Draft token数量
        tensor_parallel_size=1
    )

    sampling_params = SamplingParams(
        temperature=0.8,
        max_tokens=100
    )

    prompts = ["写一个Python函数"]

    outputs = llm.generate(prompts, sampling_params)

    for output in outputs:
        print(f"Output: {output.outputs[0].text}")
