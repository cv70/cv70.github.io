80.4 响应处理

正确处理 HTTP 响应和错误

📖 本节概述

正确处理 HTTP 响应是前端开发中的重要技能。本节将介绍如何处理成功响应、各种类型的错误，以及如何设计统一的响应处理机制。

🎯 学习目标

  • 理解 Axios 响应对象的结构
  • 掌握不同类型错误的处理方式
  • 学会设计统一的响应处理机制
  • 了解响应数据的类型定义

📦 响应对象结构

interface AxiosResponseany, D = any> { data: T; // 服务器响应的数据 status: number; // HTTP 状态码 (200, 404, 500...) statusText: string; // HTTP 状态文本 ('OK', 'Not Found'...) headers: RawAxiosResponseHeaders; // 响应头 config: InternalAxiosRequestConfig; // 请求配置 request?: any; // 原始请求对象 } // 使用示例 axios.get<User>('/api/users/1').then(response => { console.log('数据:', response.data); // User 对象 console.log('状态码:', response.status); // 200 console.log('状态文本:', response.statusText); // 'OK' console.log('响应头:', response.headers); });

✅ 成功响应处理

interface ApiResponse<T> { code: number; message: string; data: T; } interface User { id: number; name: string; email: string; } // 基本处理 async function getUser(id: number): Promise<User> { const response = await axios.get<ApiResponse<User>>( `/api/users/${id}` ); return response.data.data; } // 带状态检查的处理 async function fetchUser(id: number): Promise<User> { const response = await axios.get<ApiResponse<User>>( `/api/users/${id}` ); if (response.status !== 200) { throw new Error(`请求失败: ${response.status}`); } if (response.data.code !== 0) { throw new Error(response.data.message); } return response.data.data; }

❌ 错误处理

错误类型

interface AxiosError<T = any, D = any> extends Error { config: AxiosRequestConfig<D>; code?: string; request?: any; response?: AxiosResponse<T, D>; isAxiosError: boolean; status?: number; } // 完整的错误处理 async function handleRequest<T>( request: () => Promise<AxiosResponse<T>> ): Promise<{ data?: T; error?: string }> { try { const response = await request(); return { data: response.data }; } catch (error) { if (axios.isAxiosError(error)) { // Axios 错误 if (error.response) { // 服务器返回了响应，但状态码表示错误 switch (error.response.status) { case 400: return { error: '请求参数错误' }; case 401: return { error: '未授权，请重新登录' }; case 403: return { error: '拒绝访问' }; case 404: return { error: '请求的资源不存在' }; case 500: return { error: '服务器内部错误' }; default: return { error: `请求失败: ${error.response.status}` }; } } else if (error.request) { // 请求已发出但没有收到响应 return { error: '网络错误，请检查网络连接' }; } else { // 设置请求时出错 return { error: error.message || '请求失败' }; } } // 非 Axios 错误 return { error: '未知错误' }; } }

🔄 统一响应处理

// 定义通用响应类型 interface BaseResponse<T = any> { code: number; message: string; data: T; } // 创建带统一处理的实例 const api = axios.create({ baseURL: '/api', timeout: 10000 }); // 响应拦截器统一处理 api.interceptors.response.use( response => { const { data } = response; // 根据业务状态码判断 if (data.code === 0 || data.code === 200) { return data.data; // 直接返回数据 } // 业务错误 return Promise.reject(new Error(data.message || '请求失败')); }, error => { // 处理 HTTP 错误 if (error.response) { const { status } = error.response; switch (status) { case 401: // 跳转登录 window.location.href = '/login'; break; case 403: error.message = '没有权限访问'; break; case 404: error.message = '请求资源不存在'; break; case 500: error.message = '服务器错误'; break; default: error.message = `请求失败 (${status})`; } } else if (error.request) { error.message = '网络连接失败'; } return Promise.reject(error); } );

📁 特殊响应处理

文件下载

async function downloadFile(url: string, filename: string) { const response = await axios.get(url, { responseType: 'blob' }); // 创建下载链接 const blob = new Blob([response.data]); const downloadUrl = URL.createObjectURL(blob); const link = document.createElement('a'); link.href = downloadUrl; link.download = filename; link.click(); // 清理 URL.revokeObjectURL(downloadUrl); } // 带进度的文件下载 async function downloadWithProgress( url: string, onProgress: (percent: number) => void ) { const response = await axios.get(url, { responseType: 'blob', onDownloadProgress: (progressEvent) => { if (progressEvent.total) { const percent = Math.round( (progressEvent.loaded * 100) / progressEvent.total ); onProgress(percent); } } }); return response.data; }

文件上传

async function uploadFile( url: string, file: File, onProgress?: (percent: number) => void ) { const formData = new FormData(); formData.append('file', file); const response = await axios.post(url, formData, { headers: { 'Content-Type': 'multipart/form-data' }, onUploadProgress: (progressEvent) => { if (onProgress && progressEvent.total) { const percent = Math.round( (progressEvent.loaded * 100) / progressEvent.total ); onProgress(percent); } } }); return response.data; }

🔐 类型安全处理

// 定义 API 响应类型 interface ApiSuccess<T> { success: true; data: T; } interface ApiError { success: false; error: string; } type ApiResult<T> = ApiSuccess<T> | ApiError; // 类型安全的请求函数 async function safeRequest<T>( config: AxiosRequestConfig ): Promise<ApiResult<T>> { try { const response = await axios.request<T>(config); return { success: true, data: response.data }; } catch (error) { return { success: false, error: error instanceof Error ? error.message : '未知错误' }; } } // 使用示例 const result = await safeRequest<User[]>({ method: 'GET', url: '/api/users' }); if (result.success) { console.log('用户列表:', result.data); } else { console.error('错误:', result.error); }
