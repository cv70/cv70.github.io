📚 高阶函数定义

// 高阶函数：接受函数作为参数或返回函数的函数 // 函数作为参数 function executeWithLog(fn: () => T, message: string): T { console.log(`Before: ${message}`); const result = fn(); console.log(`After: ${message}`); return result; } // 函数作为返回值 function createMultiplier(factor: number): (n: number) => number { return (n: number) => n * factor; } const double = createMultiplier(2); double(5); // 10 // 内置高阶函数 const numbers = [1, 2, 3, 4, 5]; // map: 映射 const doubled = numbers.map(n => n * 2); // [2, 4, 6, 8, 10] // filter: 过滤 const evens = numbers.filter(n => n % 2 === 0); // [2, 4] // reduce: 归约 const sum = numbers.reduce((acc, n) => acc + n, 0); // 15 // find: 查找 const found = numbers.find(n => n > 3); // 4 // some: 存在判断 const hasEven = numbers.some(n => n % 2 === 0); // true // every: 全部判断 const allPositive = numbers.every(n => n > 0); // true

🔧 自定义高阶函数

// once: 确保函数只执行一次 function once any>(fn: T): T { let called = false; let result: any; return ((...args: any[]) => { if (!called) { called = true; result = fn(...args); } return result; }) as T; } const init = once(() => console.log('Initialized')); init(); // Initialized init(); // 无输出 // debounce: 防抖 function debounce any>( fn: T, delay: number ): (...args: Parameters) => void { let timer: NodeJS.Timeout; return (...args: Parameters) => { clearTimeout(timer); timer = setTimeout(() => fn(...args), delay); }; } const debouncedSearch = debounce((query: string) => { console.log('Searching:', query); }, 300); // throttle: 节流 function throttle any>( fn: T, interval: number ): (...args: Parameters) => void { let lastTime = 0; return (...args: Parameters) => { const now = Date.now(); if (now - lastTime >= interval) { lastTime = now; fn(...args); } }; } // pipe: 管道组合 function pipe(...fns: Array<(arg: T) => T>): (arg: T) => T { return (arg: T) => fns.reduce((acc, fn) => fn(acc), arg); } const processString = pipe( (s: string) => s.trim(), (s: string) => s.toLowerCase(), (s: string) => s.replace(/\s+/g, '-') ); processString(' Hello World '); // 'hello-world' // pick: 对象属性选择 function pick( obj: T, keys: K[] ): Pick { const result = {} as Pick; keys.forEach(key => { if (key in obj) { result[key] = obj[key]; } }); return result; } const user = { id: 1, name: 'Alice', email: 'alice@example.com', password: 'secret' }; const publicUser = pick(user, ['id', 'name', 'email']); // { id: 1, name: 'Alice', email: 'alice@example.com' } // groupBy: 分组 function groupBy(arr: T[], keyFn: (item: T) => string): Record { return arr.reduce((acc, item) => { const key = keyFn(item); if (!acc[key]) { acc[key] = []; } acc[key].push(item); return acc; }, {} as Record); } const people = [ { name: 'Alice', age: 25 }, { name: 'Bob', age: 30 }, { name: 'Charlie', age: 25 } ]; const grouped = groupBy(people, p => String(p.age)); // { '25': [{ name: 'Alice', age: 25 }, { name: 'Charlie', age: 25 }], '30': [{ name: 'Bob', age: 30 }] }
