8.4 条件类型约束

为条件类型添加类型约束

🎯 学习目标

  • 理解条件类型约束的概念
  • 掌握 extends 约束的用法
  • 学会约束 infer 推断的类型
  • 了解嵌套条件类型的约束

📚 基本约束

// 使用 extends 约束条件类型的输入 type MessageOf = T['message']; interface Email { message: string; } interface SMS { message: string; phone: string; } type EmailMessage = MessageOf; // string type SMSMessage = MessageOf; // string // 没有 message 属性会报错 // type Error = MessageOf; // Error: number 不满足约束 // 更灵活的约束 type MessageOf2 = T extends { message: unknown } ? T['message'] : never; type M1 = MessageOf2; // string type M2 = MessageOf2; // never

🔧 约束 infer 类型

// 约束推断的类型必须满足某些条件 type ExtractReturnType any> = T extends (...args: any[]) => infer R ? R : never; // 约束返回值必须是对象 type ExtractObjectReturn = T extends (...args: any[]) => infer R ? R extends object ? R : never : never; function returnObject() { return { x: 1, y: 2 }; } function returnNumber() { return 42; } type ObjReturn = ExtractObjectReturn; // { x: number; y: number } type NumReturn = ExtractObjectReturn; // never // 提取数组的非空元素类型 type NonNullableArrayElement = T extends (infer E)[] ? E extends null | undefined ? never : E : never; type Arr = NonNullableArrayElement<(string | null)[]>; // string

🔀 嵌套条件约束

// 深层类型提取 type DeepValue = K extends keyof T ? T[K] : K extends `${infer First}.${infer Rest}` ? First extends keyof T ? DeepValue : never : never; interface Config { database: { host: string; port: number; credentials: { username: string; password: string; }; }; } type Host = DeepValue; // string type Port = DeepValue; // number type Username = DeepValue; // string // 约束泛型参数必须可赋值给 infer 类型 type MatchPattern = T extends P ? T : never; type Matched = MatchPattern; // string
⚠️
注意事项

条件类型约束可以确保类型安全，但过于复杂的约束会降低代码可读性。保持约束简单明了。

  • • 约束提高类型安全性
