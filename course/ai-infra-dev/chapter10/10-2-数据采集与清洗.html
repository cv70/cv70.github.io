清洗策略

数据清洗流程：

import pandas as pd
import numpy as np

# 1. 处理缺失值
def handle_missing_data(df):
    # 数值型：填充均值或中位数
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    df[numeric_cols] = df[numeric_cols].fillna(
        df[numeric_cols].median())

    # 类别型：填充众数或特定值
    categorical_cols = df.select_dtypes(include=['object']).columns
    df[categorical_cols] = df[categorical_cols].fillna('Unknown')

    return df

# 2. 处理异常值
def handle_outliers(df, column, method='iqr'):
    if method == 'iqr':
        Q1 = df[column].quantile(0.25)
        Q3 = df[column].quantile(0.75)
        IQR = Q3 - Q1
        lower_bound = Q1 - 1.5 * IQR
        upper_bound = Q3 + 1.5 * IQR
        df[column] = df[column].clip(lower_bound, upper_bound)
    elif method == 'zscore':
        z_scores = np.abs((df[column] - df[column].mean()) /
                        df[column].std())
        df[column] = df[column][z_scores < 3]

    return df

# 3. 数据去重
def remove_duplicates(df, subset=None):
    return df.drop_duplicates(subset=subset, keep='first')

# 4. 数据类型转换
def convert_types(df):
    # 日期转换
    date_cols = ['created_at', 'updated_at']
    for col in date_cols:
        if col in df.columns:
            df[col] = pd.to_datetime(df[col])

    # 分类转换
    category_cols = ['category', 'status']
    for col in category_cols:
        if col in df.columns:
            df[col] = df[col].astype('category')

    return df

# 5. 数据验证
def validate_data(df, rules):
    for column, rule in rules.items():
        if 'range' in rule:
            df = df[(df[column] >= rule['range'][0]) &
                   (df[column] <= rule['range'][1])]
        if 'not_null' in rule and rule['not_null']:
            df = df[df[column].notna()]
    return df
