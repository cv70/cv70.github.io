18.1 创建 HTTP 服务器

使用 http 模块构建 Web 服务器

🎯 学习目标

  • 掌握 http.createServer 方法
  • 理解请求和响应对象
  • 学会监听端口
  • 了解服务器事件

📚 基本服务器

const http = require('http'); // 创建 HTTP 服务器 const server = http.createServer((req, res) => { // req: 请求对象 (IncomingMessage) // res: 响应对象 (ServerResponse) res.writeHead(200, { 'Content-Type': 'text/plain' }); res.end('Hello World!\n'); }); // 监听端口 server.listen(3000, () => { console.log('服务器运行在 http://localhost:3000'); }); // 指定主机 server.listen(3000, '127.0.0.1', () => { console.log('服务器运行在 http://127.0.0.1:3000'); }); // 使用选项对象 server.listen({ port: 3000, host: 'localhost', exclusive: true // 独占端口 }, () => { console.log('服务器启动'); }); // 查看服务器地址 const address = server.address(); console.log(address); // { address: '127.0.0.1', family: 'IPv4', port: 3000 }

🔧 请求对象

const server = http.createServer((req, res) => { // 请求方法 console.log('方法:', req.method); // GET, POST, PUT, DELETE... // 请求 URL console.log('URL:', req.url); // /path?query=value // HTTP 版本 console.log('HTTP 版本:', req.httpVersion); // 1.1 // 请求头 console.log('Headers:', req.headers); console.log('Content-Type:', req.headers['content-type']); console.log('User-Agent:', req.headers['user-agent']); // 获取请求体（流） let body = ''; req.on('data', (chunk) => { body += chunk; }); req.on('end', () => { console.log('请求体:', body); res.end('收到请求'); }); // 解析查询参数 const url = new URL(req.url, `http://${req.headers.host}`); console.log('路径:', url.pathname); console.log('查询参数:', url.searchParams); console.log('参数 name:', url.searchParams.get('name')); // 客户端信息 console.log('远程地址:', req.socket.remoteAddress); console.log('远程端口:', req.socket.remotePort); });

📦 响应对象

const server = http.createServer((req, res) => { // 设置状态码 res.statusCode = 200; // 设置响应头 res.setHeader('Content-Type', 'text/html; charset=utf-8'); res.setHeader('X-Custom-Header', 'value'); // 写入响应头 res.writeHead(200, { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' }); // 写入响应体 res.write('Hello '); res.write('World'); // 结束响应 res.end('!'); // 一步完成 res.end('Hello World'); // 发送 JSON res.writeHead(200, { 'Content-Type': 'application/json' }); res.end(JSON.stringify({ message: 'Hello' })); // 发送 HTML res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' }); res.end('

'); // 重定向 res.writeHead(302, { 'Location': '/new-path' }); res.end(); // 404 响应 res.writeHead(404, { 'Content-Type': 'text/plain' }); res.end('Not Found'); // 获取已设置的头 console.log(res.getHeader('Content-Type')); console.log(res.getHeaders()); // 移除头 res.removeHeader('X-Custom-Header'); });

⚙️ 服务器事件

const http = require('http'); const server = http.createServer(); // request 事件 server.on('request', (req, res) => { console.log('收到请求'); res.end('OK'); }); // connection 事件 server.on('connection', (socket) => { console.log('新连接:', socket.remoteAddress); }); // close 事件 server.on('close', () => { console.log('服务器关闭'); }); // error 事件 server.on('error', (err) => { if (err.code === 'EADDRINUSE') { console.error('端口已被占用'); } else { console.error('服务器错误:', err); } }); // listening 事件 server.on('listening', () => { console.log('服务器正在监听'); }); server.listen(3000); // 关闭服务器 setTimeout(() => { server.close(() => { console.log('服务器已关闭'); }); }, 10000); // 设置超时 server.timeout = 120000; // 2 分钟 server.keepAliveTimeout = 65000;

📝 本节小结

✅
  • • http.createServer() 创建服务器
  • • req 对象包含请求信息
  • • res 对象用于发送响应
  • • server.listen() 监听端口
