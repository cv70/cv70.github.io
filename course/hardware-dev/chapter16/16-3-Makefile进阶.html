Makefile进阶

Makefile进阶是"构建系统与工具链"这一章中的第3节内容。 Make是Unix/Linux系统中最流行的构建工具，通过编写Makefile可以自动化编译过程。 本节将深入讲解Makefile的高级特性，帮助你构建高效的嵌入式项目构建系统。

📐 Makefile基本结构

规则语法

目标: 依赖1 依赖2 ...
    命令1
    命令2
    ...

# 示例
main.o: main.c main.h
    arm-none-eabi-gcc -c -O2 -Wall $< -o $@

核心概念

概念 说明 示例
目标(Target) 要生成的文件 main.elf
依赖(Prerequisite) 生成目标所需的文件 main.o utils.o
命令(Command) 生成目标的shell命令 gcc -c main.c
伪目标 不对应文件的标签 clean、all

自动变量

变量 含义 示例
$@ 当前目标文件名 main.o
$< 第一个依赖文件名 main.c
$^ 所有依赖文件（去重） main.c utils.c
$+ 所有依赖文件（不去重） main.c main.c
$? 比目标新的依赖 main.c
$* 目标的主文件名 main

🔤 变量定义

变量类型

# 简单变量（:= 立即求值）
CC := arm-none-eabi-gcc
CFLAGS := -Wall -O2

# 递归变量（= 延迟求值）
SRC = $(wildcard *.c)

# 条件赋值（?= 未定义时赋值）
DEBUG_LEVEL ?= 0

# 追加赋值（+=）
CFLAGS += -g

# shell命令
TODAY := $(shell date +%Y-%m-%d)

内置变量

变量 默认值 说明
CC cc C编译器
CXX g++ C++编译器
LD ld 链接器
AR ar 归档工具
RANLIB ranlib 归档索引工具
CFLAGS - C编译选项
CXXFLAGS - C++编译选项
LDFLAGS - 链接选项
CPPFLAGS - C预处理器选项

🎨 文本处理函数

函数 语法 功能 示例
wildcard $(wildcard pattern) 匹配文件 $(wildcard *.c)
patsubst $(patsubst pattern,replacement) 模式替换 $(patsubst %.c,%.o,$(SRC))
subst $(subst from,to,text) 字符串替换 $(subst .c,.o,$(SRC))
addprefix $(addprefix prefix,names...) 添加前缀 $(addprefix src/,$(SRC))
addsuffix $(addsuffix suffix,names...) 添加后缀 $(addsuffix .o,$(OBJS))
basename $(basename names...) 取文件名 $(basename foo.c) → foo
notdir $(notdir names...) 去目录 $(notdir src/foo.c) → foo.c
dir $(dir names...) 取目录 $(dir src/foo.c) → src/
filter $(filter pattern...,text) 过滤匹配 $(filter %.c,$(FILES))
filter-out $(filter-out pattern...,text) 过滤不匹配 $(filter-out %.c,$(FILES))
foreach $(foreach var,list,text) 循环遍历 $(foreach f,$(SRC),$(f:.c=.o))

条件判断函数

# if语法
ifeq ($(DEBUG),1)
    CFLAGS += -g -DDEBUG
else
    CFLAGS += -O2 -DNDEBUG
endif

# ifneq语法
ifneq ($(V),)
    QUIET :=
else
    QUIET := @
endif

# ifdef语法
ifdef USE_HAL
    CFLAGS += -DUSE_HAL_DRIVER
endif

# ifndef语法
ifndef VERBOSE
    .SILENT:
endif

🏗️ 嵌入式Makefile模板

# ==================== 配置部分 ====================
TARGET := firmware.elf
CC := arm-none-eabi-gcc
LD := arm-none-eabi-gcc
OBJCOPY := arm-none-eabi-objcopy
SIZE := arm-none-eabi-size

# 编译选项
CFLAGS := -Wall -Wextra -Werror
CFLAGS += -mcpu=cortex-m4 -mthumb -mfloat-abi=hard -mfpu=fpv4-sp-d16
CFLAGS += -ffunction-sections -fdata-sections
CFLAGS += -DSTM32F407xx -DUSE_HAL_DRIVER

# 调试/发布配置
DEBUG ?= 0
ifeq ($(DEBUG),1)
    CFLAGS += -g -Og -DDEBUG
else
    CFLAGS += -Os -DNDEBUG
endif

# 链接选项
LDFLAGS := -mcpu=cortex-m4 -mthumb -mfloat-abi=hard -mfpu=fpv4-sp-d16
LDFLAGS += -specs=nano.specs -specs=nosys.specs
LDFLAGS += -Wl,--gc-sections
LDFLAGS += -Wl,-Map=$(TARGET:.elf=.map)

# 包含路径
INC_DIRS := ./Inc
INC_DIRS += ./Drivers/STM32F4xx_HAL_Driver/Inc
INC_DIRS += ./Drivers/CMSIS/Device/ST/STM32F4xx/Include
INC_DIRS += ./Drivers/CMSIS/Include

INCLUDES := $(addprefix -I,$(INC_DIRS))

# 源文件
SRC_DIRS := ./Src
SRC_DIRS += ./Drivers/STM32F4xx_HAL_Driver/Src

SRC := $(foreach dir,$(SRC_DIRS),$(wildcard $(dir)/*.c))

# 目标文件
OBJ := $(SRC:.c=.o)
OBJ += $(ASM:.s=.o)

# ==================== 规则部分 ====================
.PHONY: all clean flash

all: $(TARGET) $(TARGET:.elf=.hex) $(TARGET:.elf=.bin)
    @$(SIZE) $(TARGET)

$(TARGET): $(OBJ)
    @echo "  LD      $@"
    $(QUIET)$(LD) $(LDFLAGS) $(OBJ) -o $@

%.hex: %.elf
    @echo "  HEX     $@"
    $(QUIET)$(OBJCOPY) -O ihex $< $@

%.bin: %.elf
    @echo "  BIN     $@"
    $(QUIET)$(OBJCOPY) -O binary $< $@

%.o: %.c
    @echo "  CC      $<"
    $(QUIET)$(CC) $(CFLAGS) $(INCLUDES) -c $< -o $@

%.o: %.s
    @echo "  AS      $<"
    $(QUIET)$(CC) $(CFLAGS) $(INCLUDES) -c $< -o $@

clean:
    @echo "  CLEAN"
    $(QUIET)rm -f $(OBJ) $(TARGET) $(TARGET:.elf=.map)
    $(QUIET)rm -f $(TARGET:.elf=.hex) $(TARGET:.elf=.bin)

flash: all
    @echo "  FLASH   $(TARGET:.elf=.hex)"
    st-flash write $(TARGET:.elf=.hex) 0x8000000

# ==================== 依赖关系 ====================
-include $(OBJ:.o=.d)

📦 模块化组织

多文件结构

project/
├── Makefile          # 主Makefile
├── config.mk         # 配置文件
├── src/
│   └── Makefile      # 源文件Makefile
├── lib/
│   ├── lib1/
│   │   └── Makefile
│   └── lib2/
│       └── Makefile
└── build/
    └── Makefile      # 构建文件Makefile

包含其他Makefile

# 包含配置文件
include config.mk

# 包含子目录
include src/Makefile
include lib/lib1/Makefile
include lib/lib2/Makefile
