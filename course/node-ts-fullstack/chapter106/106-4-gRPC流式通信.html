106.4 gRPC 流式通信

高效的数据流传输模式

🎯 学习目标

  • 理解服务端流式RPC
  • 理解客户端流式RPC
  • 掌握双向流式RPC
  • 学会流式通信的实际应用

📚 服务端流式RPC

// Proto定义 service OrderService { // 服务端流：返回多个订单 rpc ListOrders(ListOrdersRequest) returns (stream Order); } // 服务端实现 import * as grpc from '@grpc/grpc-js'; class OrderServiceImpl { async listOrders( call: grpc.ServerWritableStream ) { const { page, pageSize } = call.request; // 模拟分批获取数据 for (let i = 0; i < pageSize; i++) { const order = await this.getOrder(page * pageSize + i); // 发送单个订单 call.write(order); // 模拟延迟 await new Promise(resolve => setTimeout(resolve, 100)); } // 结束流 call.end(); } } // 客户端调用 async function listOrders(client: OrderServiceClient) { const request = { page: 1, pageSize: 10 }; const stream = client.listOrders(request); stream.on('data', (order: Order) => { console.log('Received order:', order.id); }); stream.on('end', () => { console.log('All orders received'); }); stream.on('error', (error: Error) => { console.error('Error:', error); }); }
💡
服务端流适用场景

大量数据返回、实时数据推送、日志流、事件流。

🔧 客户端流式RPC

// Proto定义 service UploadService { // 客户端流：上传多个文件块 rpc UploadFile(stream FileChunk) returns (UploadResult); } // 服务端实现 class UploadServiceImpl { async uploadFile( call: grpc.ServerReadableStream, callback: grpc.sendUnaryData ) { const chunks: Buffer[] = []; call.on('data', (chunk: FileChunk) => { console.log('Received chunk:', chunk.sequence); chunks.push(Buffer.from(chunk.data)); }); call.on('end', async () => { // 合并所有块 const fileBuffer = Buffer.concat(chunks); // 保存文件 const filePath = await this.saveFile(fileBuffer); callback(null, { success: true, filePath, size: fileBuffer.length }); }); call.on('error', (error: Error) => { callback(error, null); }); } } // 客户端调用 async function uploadFile(client: UploadServiceClient, filePath: string) { return new Promise((resolve, reject) => { const stream = client.uploadFile((error, result) => { if (error) reject(error); else resolve(result); }); // 读取文件并分块发送 const fileStream = fs.createReadStream(filePath); let sequence = 0; fileStream.on('data', (chunk: Buffer) => { stream.write({ sequence: sequence++, data: chunk, filename: path.basename(filePath) }); }); fileStream.on('end', () => { stream.end(); }); }); }

🔀 双向流式RPC

// Proto定义 service ChatService { // 双向流：实时聊天 rpc Chat(stream ChatMessage) returns (stream ChatMessage); } // 服务端实现 class ChatServiceImpl { private clients: Map> = new Map(); async chat(call: grpc.ServerDuplexStream) { const clientId = uuidv4(); this.clients.set(clientId, call); call.on('data', (message: ChatMessage) => { console.log(`${message.sender}: ${message.content}`); // 广播给所有客户端 for (const [id, client] of this.clients) { if (id !== clientId) { client.write({ sender: message.sender, content: message.content, timestamp: Date.now() }); } } }); call.on('end', () => { this.clients.delete(clientId); call.end(); }); } } // 客户端实现 async function startChat(client: ChatServiceClient, username: string) { const stream = client.chat(); // 发送消息 const sendMessage = (content: string) => { stream.write({ sender: username, content, timestamp: Date.now() }); }; // 接收消息 stream.on('data', (message: ChatMessage) => { console.log(`${message.sender}: ${message.content}`); }); // 用户输入 readline.on('line', (line: string) => { sendMessage(line); }); // 结束 process.on('SIGINT', () => { stream.end(); process.exit(); }); }

📝 本节小结

✅
  • • 服务端流适合大量数据返回
  • • 客户端流适合批量数据上传
  • • 双向流适合实时通信场景
  • • 流式通信减少内存占用和延迟
