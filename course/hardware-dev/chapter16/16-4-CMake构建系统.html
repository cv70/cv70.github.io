CMake构建系统

CMake构建系统是"构建系统与工具链"这一章中的第4节内容。 CMake是一个跨平台的构建系统生成工具，能够生成Makefile、Visual Studio项目等多种构建文件。 本节将介绍CMake在嵌入式开发中的应用，帮助你建立现代化的构建流程。

💡 CMake基础概念

CMake工作流程

ℹ️
构建流程

CMakeLists.txt → CMake → Makefile/VS项目 → 编译器 → 可执行文件

  1. 编写CMakeLists.txt配置文件
  2. 运行cmake命令生成构建文件
  3. 运行构建命令（make、ninja等）编译项目

基本结构

# 最小示例
cmake_minimum_required(VERSION 3.10)
project(MyProject C CXX)

# 设置C标准
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# 添加可执行文件
add_executable(main main.c)

🔧 嵌入式CMake配置

主CMakeLists.txt

cmake_minimum_required(VERSION 3.15)
project(STM32_Project C CXX ASM)

# 设置C标准
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)

# 配置交叉编译工具链
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR ARM)

set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_CXX_COMPILER arm-none-eabi-g++)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)
set(CMAKE_AR arm-none-eabi-ar)
set(CMAKE_OBJCOPY arm-none-eabi-objcopy)
set(CMAKE_SIZE arm-none-eabi-size)

# 不链接系统库
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

# 编译选项
set(CMAKE_C_FLAGS_DEBUG "-g -Og -DDEBUG")
set(CMAKE_C_FLAGS_RELEASE "-Os -DNDEBUG")

set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -Werror")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -mcpu=cortex-m4 -mthumb")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -mfloat-abi=hard -mfpu=fpv4-sp-d16")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -ffunction-sections -fdata-sections")

# 链接选项
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -specs=nano.specs")
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -specs=nosys.specs")
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,--gc-sections")

# 定义宏
add_definitions(-DSTM32F407xx)
add_definitions(-DUSE_HAL_DRIVER)

# 包含目录
include_directories(
    ${CMAKE_SOURCE_DIR}/Inc
    ${CMAKE_SOURCE_DIR}/Drivers/STM32F4xx_HAL_Driver/Inc
    ${CMAKE_SOURCE_DIR}/Drivers/CMSIS/Device/ST/STM32F4xx/Include
    ${CMAKE_SOURCE_DIR}/Drivers/CMSIS/Include
)

# 收集源文件
file(GLOB_RECURSE SOURCES
    "Src/*.c"
    "Drivers/STM32F4xx_HAL_Driver/Src/*.c"
)

# 收集启动文件
file(GLOB ASM_SOURCES "startup/*.s")

# 添加可执行文件
add_executable(${PROJECT_NAME}.elf ${SOURCES} ${ASM_SOURCES})

# 设置链接脚本
target_link_options(${PROJECT_NAME}.elf PRIVATE
    -T${CMAKE_SOURCE_DIR}/linker/STM32F407VGTx_FLASH.ld
)

# 添加后处理命令
add_custom_command(TARGET ${PROJECT_NAME}.elf POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O ihex $ ${PROJECT_NAME}.hex
    COMMAND ${CMAKE_OBJCOPY} -O binary $ ${PROJECT_NAME}.bin
    COMMAND ${CMAKE_SIZE} $
    COMMENT "Building ${PROJECT_NAME}.hex and ${PROJECT_NAME}.bin"
)

# 添加flash目标
add_custom_target(flash
    DEPENDS ${PROJECT_NAME}.elf
    COMMAND st-flash write ${PROJECT_NAME}.hex 0x8000000
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    COMMENT "Flashing ${PROJECT_NAME}.hex"
)

🚀 构建流程

基本构建步骤

# 1. 创建构建目录
mkdir build
cd build

# 2. 配置项目（生成Makefile）
cmake ..

# 3. 编译项目
cmake --build .

# 4. 下载到目标板
cmake --build . --target flash

# 5. 清理
cmake --build . --target clean

多配置构建

# Debug构建
cmake -DCMAKE_BUILD_TYPE=Debug ..
cmake --build .

# Release构建
cmake -DCMAKE_BUILD_TYPE=Release ..
cmake --build .

# Ninja构建（更快）
cmake -G Ninja ..
ninja

常用CMake选项

选项 说明 示例
-DCMAKE_BUILD_TYPE 构建类型 Debug、Release
-G 生成器 "Unix Makefiles"、"Ninja"
-D 定义变量 -DOPTIMIZATION=3
-S 源码目录 -S /path/to/source
-B 构建目录 -B /path/to/build

⚙️ 高级特性

自定义命令

# 在编译前生成文件
add_custom_command(
    OUTPUT generated.c
    COMMAND python ${CMAKE_SOURCE_DIR}/tools/gen.py
    DEPENDS ${CMAKE_SOURCE_DIR}/tools/gen.py
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
)

# 使用生成的文件
add_executable(my_app main.c generated.c)

条件编译

# 检查编译器特性
include(CheckCSourceCompiles)

check_c_source_compiles("
    int main() { __builtin_clz(0); return 0; }
" HAS_BUILTIN_CLZ)

if(HAS_BUILTIN_CLZ)
    add_definitions(-DHAS_BUILTIN_CLZ)
endif()

# 检测平台
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_definitions(-DPLATFORM_LINUX)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Generic")
    add_definitions(-DPLATFORM_BAREMETAL)
endif()
