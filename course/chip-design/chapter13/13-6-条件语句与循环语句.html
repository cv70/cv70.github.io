13.6 条件语句与循环语句

Verilog中的控制流语句

🎯 学习目标

  • 掌握if-else条件语句的使用
  • 学会使用case语句进行多路选择
  • 理解for、while等循环语句
  • 能够根据需求选择合适的控制语句

🔀 if-else 条件语句

// 基本if-else结构
if (条件)
    语句1;
else
    语句2;

// 多重if-else
if (条件1)
    语句1;
else if (条件2)
    语句2;
else if (条件3)
    语句3;
else
    语句4;

// 示例：简单的多路选择器
always @(*) begin
    if (sel == 2'b00)
        y = a;
    else if (sel == 2'b01)
        y = b;
    else if (sel == 2'b10)
        y = c;
    else
        y = d;
end

// 示例：带优先级的逻辑
always @(*) begin
    if (req0)
        grant = 3'b001;
    else if (req1)
        grant = 3'b010;
    else if (req2)
        grant = 3'b100;
    else
        grant = 3'b000;
end

📌 case 语句

// 基本case结构
case (表达式)
    值1: 语句1;
    值2: 语句2;
    值3: 语句3;
    default: 语句4;
endcase

// 示例：解码器
module decoder3to8(
    input  [2:0] sel,
    output reg [7:0] y
);
    always @(*) begin
        case (sel)
            3'b000: y = 8'b0000_0001;
            3'b001: y = 8'b0000_0010;
            3'b010: y = 8'b0000_0100;
            3'b011: y = 8'b0000_1000;
            3'b100: y = 8'b0001_0000;
            3'b101: y = 8'b0010_0000;
            3'b110: y = 8'b0100_0000;
            3'b111: y = 8'b1000_0000;
        endcase
    end
endmodule

// 示例：状态机
typedef enum logic [1:0] {
    IDLE, RUN, DONE
} state_t;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n)
        state <= IDLE;
    else begin
        case (state)
            IDLE:  state <= (start) ? RUN : IDLE;
            RUN:   state <= (done) ? DONE : RUN;
            DONE:  state <= IDLE;
            default: state <= IDLE;
        endcase
    end
end

🔍 case 变体

// casez：不关心z的情况（常用于优先级编码）
casez (req)
    4'b1???: grant = 3'b100;  // 最高优先级
    4'b01??: grant = 3'b010;
    4'b001?: grant = 3'b001;
    default:  grant = 3'b000;
endcase

// casex：不关心x和z的情况
casex (code)
    4'b1xxx: type = 2'b11;
    4'b01xx: type = 2'b10;
    4'b001x: type = 2'b01;
    default: type = 2'b00;
endcase

// 注意：casez和casex不可综合，仅用于测试平台

🔄 for 循环

// 基本for循环
integer i;
always @(*) begin
    y = 0;
    for (i = 0; i < 8; i = i + 1) begin
        if (a[i])
            y = y + 1;
    end
end

// 示例：奇偶校验位生成
reg parity;
always @(*) begin
    parity = 1'b0;
    for (i = 0; i < 8; i = i + 1) begin
        parity = parity ^ data[i];
    end
end

// 示例：优先级编码器
integer i;
always @(*) begin
    result = 0;
    valid = 1'b0;
    for (i = 7; i >= 0; i = i - 1) begin
        if (req[i]) begin
            result = i;
            valid = 1'b1;
            break;  // Verilog-2001支持break
        end
    end
end

🔁 while 循环

// while循环
integer i;
always @(*) begin
    y = 0;
    i = 0;
    while (i < 8 && a[i] == 1'b0) begin
        i = i + 1;
    end
    result = i;
end

// 示例：找第一个1的位置
integer i;
always @(*) begin
    i = 0;
    while (i < 8 && data[i] != 1'b1) begin
        i = i + 1;
    end
    if (i == 8)
        found = 1'b0;
    else
        found = 1'b1;
end

🔃 repeat 循环

// repeat循环：固定次数重复
integer i;
always @(*) begin
    result = 0;
    repeat (8) begin
        result = result << 1;
        result = result | data[i];
        i = i + 1;
    end
end

// 示例：串并转换
always @(posedge clk) begin
    if (load) begin
        shift_reg <= data_in;
        count <= 0;
    end else if (count < 8) begin
        serial_out <= shift_reg[7];
        shift_reg <= {shift_reg[6:0], 1'b0};
        count <= count + 1;
    end
end

🔀 forever 循环

// forever循环：无限循环（用于测试平台）
initial begin
    clk = 0;
    forever #10 clk = ~clk;  // 生成时钟信号
end

// 示例：测试激励
initial begin
    rst_n = 0;
    #100;
    rst_n = 1;
    forever begin
        @(posedge clk);
        data_in = $random;
    end
end
⚠️
综合注意事项

forever循环不可综合。for、while、repeat循环在可综合代码中使用时，循环次数必须是确定的（常量）。
