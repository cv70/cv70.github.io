88.4 构建优化

Vite 生产构建优化技巧

📚 代码分割

// vite.config.ts export default defineConfig({ build: { rollupOptions: { output: { // 手动分割代码块 manualChunks: (id) => { // 第三方库 if (id.includes('node_modules')) { // Vue 生态 if (id.includes('vue') || id.includes('pinia') || id.includes('vue-router')) { return 'vue-vendor' } // UI 库 if (id.includes('element-plus') || id.includes('ant-design-vue')) { return 'ui-vendor' } // 工具库 if (id.includes('lodash') || id.includes('axios')) { return 'utils-vendor' } // 其他第三方 return 'vendor' } }, // 或者使用对象形式 manualChunks: { vue: ['vue', 'vue-router', 'pinia'], ui: ['element-plus'], utils: ['lodash', 'axios', 'dayjs'], }, }, }, // 分块大小警告阈值 chunkSizeWarningLimit: 1000, }, }) // 路由级代码分割 const routes = [ { path: '/home', component: () => import('@/views/Home.vue'), // 懒加载 }, { path: '/admin', component: () => import('@/views/Admin.vue'), }, ]
💡
代码分割原则

将不常变化的第三方库单独打包，业务代码分离，利用浏览器缓存提高加载效率。

🔧 资源优化

import viteCompression from 'vite-plugin-compression' import viteImagemin from 'vite-plugin-imagemin' export default defineConfig({ build: { // CSS 代码分割 cssCodeSplit: true, // 资源文件名 rollupOptions: { output: { chunkFileNames: 'js/[name]-[hash].js', entryFileNames: 'js/[name]-[hash].js', assetFileNames: 'assets/[name]-[hash][extname]', }, }, // 资源内联阈值 assetsInlineLimit: 4096, // 4KB 以下内联 // 清空输出目录 emptyOutDir: true, }, plugins: [ // Gzip 压缩 viteCompression({ algorithm: 'gzip', ext: '.gz', threshold: 1024, // 大于 1KB 才压缩 }), // Brotli 压缩 viteCompression({ algorithm: 'brotliCompress', ext: '.br', }), // 图片压缩 viteImagemin({ gifsicle: { optimizationLevel: 3 }, optipng: { optimizationLevel: 7 }, mozjpeg: { quality: 80 }, pngquant: { quality: [0.8, 0.9] }, svgo: { plugins: [ { name: 'removeViewBox', active: false }, ], }, }), ], })

⚙️ 构建性能优化

export default defineConfig({ // 依赖预构建 optimizeDeps: { // 预包含的依赖 include: ['vue', 'vue-router', 'pinia', 'axios'], // 排除预构建 exclude: ['your-linked-package'], // 强制重新构建 force: false, }, // esbuild 配置 esbuild: { // 移除 console drop: ['console', 'debugger'], // JSX 支持 jsxFactory: 'h', jsxFragment: 'Fragment', }, build: { // 使用 esbuild 压缩（更快） minify: 'esbuild', // 'terser' 压缩率更高但更慢 // Source Map sourcemap: false, // 生产环境关闭 // 报告压缩后大小 reportCompressedSize: true, // 禁用 brotli 大小报告 brotliSize: false, // 并行处理 maxParallelFileOps: 20, }, // CSS 优化 css: { devSourcemap: false, }, }) // Terser 配置（需要安装） export default defineConfig({ build: { minify: 'terser', terserOptions: { compress: { drop_console: true, drop_debugger: true, pure_funcs: ['console.log'], }, format: { comments: false, }, }, }, })

🚀 构建分析

import { visualizer } from 'rollup-plugin-visualizer' export default defineConfig({ plugins: [ visualizer({ open: true, // 自动打开分析页面 gzipSize: true, brotliSize: true, filename: 'stats.html', // 输出文件名 }), ], }) // 使用 rollup-plugin-visualizer 后 // 运行 npm run build // 自动生成 stats.html 文件 // 可视化展示各模块大小 # 分析构建产物 npm run build # 查看输出 dist/ ├── assets/ │ ├── index-abc123.js 150KB │ ├── vue-vendor-def456.js 80KB │ └── index-ghi789.css 20KB ├── index.html └── stats.html # 手动分析 npx vite-bundle-visualizer # 使用 rollup-plugin-visualizer import { visualizer } from 'rollup-plugin-visualizer' export default defineConfig({ plugins: [ visualizer({ template: 'treemap', // 'treemap' | 'sunburst' | 'network' }), ], })
