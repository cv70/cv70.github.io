43.4 响应格式化

掌握 NestJS 响应处理技巧

🎯 学习目标

  • 理解 NestJS 的响应机制
  • 掌握状态码设置
  • 学会自定义响应头
  • 了解响应序列化

📚 标准响应

NestJS 默认将返回值序列化为 JSON，状态码根据 HTTP 方法自动设置：

import { Controller, Get, Post } from '@nestjs/common'; @Controller('users') export class UsersController { @Get() findAll() { // 自动返回 JSON，状态码 200 return [ { id: 1, name: 'John' }, { id: 2, name: 'Jane' }, ]; } @Post() create() { // 自动返回 JSON，状态码 201 return { id: 1, name: 'John' }; } @Get('empty') getEmpty() { // 返回空响应，状态码 200 return; } }

🔧 自定义状态码

import { Controller, Get, Post, HttpCode, HttpStatus } from '@nestjs/common'; @Controller('users') export class UsersController { // 使用 @HttpCode 设置状态码 @Post() @HttpCode(HttpStatus.CREATED) // 201 create() { return { message: 'User created' }; } @Post('login') @HttpCode(HttpStatus.OK) // 200 login() { return { token: 'xxx' }; } @Delete(':id') @HttpCode(HttpStatus.NO_CONTENT) // 204 remove() { // 无返回内容 } // 使用 @Res 自定义响应 @Get('custom') customResponse(@Res() res: Response) { return res .status(201) .header('X-Custom-Header', 'value') .json({ message: 'Custom response' }); } }
💡
HttpStatus 枚举

使用 HttpStatus 枚举代替数字状态码，代码更清晰。常用值：OK(200), CREATED(201), NO_CONTENT(204), BAD_REQUEST(400), UNAUTHORIZED(401), NOT_FOUND(404)。

🌐 自定义响应头

import { Controller, Get, Header, Headers } from '@nestjs/common'; @Controller('api') export class ApiController { @Get() @Header('Cache-Control', 'none') @Header('X-Response-Time', '123ms') getData() { return { data: 'Hello' }; } @Get('download') @Header('Content-Type', 'text/csv') @Header('Content-Disposition', 'attachment; filename=data.csv') download() { return 'id,name\n1,John\n2,Jane'; } @Get('cors') @Header('Access-Control-Allow-Origin', '*') cors() { return { message: 'CORS enabled' }; } }

📦 响应 DTO 和序列化

// user.response.dto.ts import { Exclude, Expose, Type } from 'class-transformer'; export class UserResponseDto { @Expose() id: number; @Expose() name: string; @Expose() email: string; @Exclude() // 不会在响应中返回 password: string; @Expose() @Type(() => Date) createdAt: Date; } // controller import { ClassSerializerInterceptor, UseInterceptors } from '@nestjs/common'; @Controller('users') @UseInterceptors(ClassSerializerInterceptor) export class UsersController { @Get(':id') async findOne(@Param('id') id: string): Promise { const user = await this.usersService.findOne(+id); // password 字段会被自动排除 return new UserResponseDto(user); } }
⚠️
序列化注意

使用 ClassSerializerInterceptor 时，需要确保全局开启 class-transformer，或者在控制器上使用 @UseInterceptors 装饰器。
