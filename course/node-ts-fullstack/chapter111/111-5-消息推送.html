🔧 推送服务实现

// push.service.ts @Injectable() export class PushService { constructor( private presenceService: PresenceService, private messageService: MessageService, @InjectRepository(PushToken) private tokenRepo: Repository, ) {} // 推送消息 async pushMessage(message: Message) { const receiverId = message.receiverId; // 1. 获取用户所有在线连接 const onlineSockets = await this.getOnlineSockets(receiverId); if (onlineSockets.length > 0) { // 在线推送 await this.pushToOnlineUsers(message, onlineSockets); } else { // 离线推送 await this.pushToOfflineUser(message, receiverId); } } // 推送给在线用户 private async pushToOnlineUsers(message: Message, socketIds: string[]) { // 通过WebSocket推送 socketIds.forEach(socketId => { this.io.to(socketId).emit('message:receive', message); }); // 存储消息 await this.messageService.save(message); } // 推送给离线用户 private async pushToOfflineUser(message: Message, userId: string) { // 存储消息（用户上线后拉取） await this.messageService.save(message); // 获取用户的设备推送Token const tokens = await this.tokenRepo.find({ where: { userId } }); if (tokens.length === 0) return; // 分组推送 const iosTokens = tokens.filter(t => t.platform === 'ios'); const androidTokens = tokens.filter(t => t.platform === 'android'); // iOS APNs推送 if (iosTokens.length > 0) { await this.pushToAPNs( iosTokens.map(t => t.token), { title: '新消息', body: message.content, data: { messageId: message.id } } ); } // Android FCM推送 if (androidTokens.length > 0) { await this.pushToFCM( androidTokens.map(t => t.token), { title: '新消息', body: message.content, data: { messageId: message.id } } ); } } // APNs推送 private async pushToAPNs(tokens: string[], payload: PushPayload) { // 使用apn库 const apn = require('apn'); const notification = new apn.Notification(); notification.alert = { title: payload.title, body: payload.body, }; notification.payload = payload.data; notification.topic = process.env.APNS_BUNDLE_ID; // 发送推送 for (const token of tokens) { try { await this.apnProvider.send(notification, token); } catch (error) { // 处理无效Token if (error.status === 410) { await this.tokenRepo.delete({ token }); } } } } // FCM推送 private async pushToFCM(tokens: string[], payload: PushPayload) { const admin = require('firebase-admin'); const message = { notification: { title: payload.title, body: payload.body, }, data: payload.data, tokens: tokens, }; const response = await admin.messaging().sendMulticast(message); // 处理无效Token response.responses.forEach((resp, idx) => { if (!resp.success && resp.error?.code === 'messaging/registration-token-not-registered') { this.tokenRepo.delete({ token: tokens[idx] }); } }); } } // 离线消息拉取 @Injectable() export class OfflineMessageService { constructor( @InjectRepository(Message) private messageRepo: Repository, ) {} // 用户上线后拉取离线消息 async pullOfflineMessages(userId: string, lastMessageId?: string): Promise { const query = this.messageRepo.createQueryBuilder('message') .where('message.receiverId = :userId', { userId }) .andWhere('message.createdAt > :lastSync', { lastSync: new Date(lastMessageId ? parseInt(lastMessageId.split('-')[0]) : 0) }) .orderBy('message.createdAt', 'ASC') .limit(100); return query.getMany(); } }
