60.1 类型系统

GraphQL 强类型系统的深入理解

📖 类型系统概述

GraphQL 类型系统是 Schema 的核心，定义了 API 的数据结构。它提供了类型安全、自动验证和文档生成能力。

💡
类型系统的作用
  • 定义 API 数据结构
  • 提供编译时和运行时验证
  • 自动生成 API 文档
  • 支持开发工具（IDE 自动补全）

📝 标量类型

内置标量类型

类型 说明 示例值
Int 有符号 32 位整数 1, -5, 100
Float 有符号双精度浮点数 3.14, -0.5
String UTF-8 字符序列 "Hello", "张三"
Boolean 布尔值 true, false
ID 唯一标识符 "123", "abc-def"

自定义标量类型

// Schema 定义 scalar DateTime scalar Email scalar URL scalar JSON // 使用自定义标量 type User { id: ID! email: Email! avatar: URL createdAt: DateTime! metadata: JSON } // Resolver 实现 const resolvers = { DateTime: { // 序列化：发送给客户端 serialize: (value: Date) => value.toISOString(), // 解析字面量 parseLiteral: (ast: any) => new Date(ast.value), // 解析变量值 parseValue: (value: string) => new Date(value), }, Email: { serialize: (value: string) => value, parseValue: (value: string) => { if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) { throw new Error('Invalid email format'); } return value; }, parseLiteral: (ast: any) => { if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(ast.value)) { throw new Error('Invalid email format'); } return ast.value; }, }, };

📦 对象类型

// 定义对象类型 type User { id: ID! name: String! email: String! age: Int // 可为 null isActive: Boolean! posts: [Post!]! createdAt: DateTime! } type Post { id: ID! title: String! content: String author: User! comments(limit: Int = 10): [Comment!]! } // 字段参数 type Query { user(id: ID!): User users(limit: Int = 10, offset: Int = 0): [User!]! search(query: String!, type: SearchType): [SearchResult!]! }

非空修饰符

type Example { // 可为 null name: String // 非空（必填） name: String! // 列表变体 list1: [String] // 列表可为 null，元素可为 null list2: [String!] // 列表可为 null，元素非空 list3: [String]! // 列表非空，元素可为 null list4: [String!]! // 列表非空，元素非空（推荐） }

🔢 枚举类型

// 定义枚举 enum UserRole { USER ADMIN MODERATOR } enum PostStatus { DRAFT PUBLISHED ARCHIVED } enum SortOrder { ASC DESC } // 使用枚举 type User { id: ID! name: String! role: UserRole! } type Post { id: ID! title: String! status: PostStatus! } input PostFilter { status: PostStatus orderBy: SortOrder = DESC } // 查询示例 query { posts(filter: { status: PUBLISHED, orderBy: DESC }) { title } } // TypeScript 映射 enum UserRole { USER = 'USER', ADMIN = 'ADMIN', MODERATOR = 'MODERATOR', }

📋 列表类型

// 简单列表 type User { tags: [String!]! // 字符串列表 scores: [Int!]! // 整数列表 posts: [Post!]! // 对象列表 } // 嵌套列表 type Matrix { data: [[Int!]!]! // 二维数组 } // Resolver 示例 const resolvers = { User: { tags: (user) => user.tags || [], posts: async (user, _args, context) => { return context.dataSources.postApi.getByUserId(user.id); }, }, };

🔗 接口类型

接口定义了一组共享字段，其他类型可以实现这些接口。

// 定义接口 interface Node { id: ID! createdAt: DateTime! } interface Media { id: ID! url: String! type: MediaType! } // 实现接口 type User implements Node { id: ID! createdAt: DateTime! name: String! email: String! } type Post implements Node { id: ID! createdAt: DateTime! title: String! content: String } // 实现多个接口 type Image implements Node & Media { id: ID! createdAt: DateTime! url: String! type: MediaType! width: Int! height: Int! } // 查询返回接口类型 type Query { node(id: ID!): Node // 返回任何实现 Node 的类型 timeline: [Node!]! // 混合类型列表 } // 使用内联片段查询 query { node(id: "123") { id createdAt ... on User { name email } ... on Post { title content } } }

🔀 联合类型

联合类型表示可以是多种类型之一，但不需要共享字段。

// 定义联合类型 union SearchResult = User | Post | Comment union MediaContent = Image | Video | Audio // 使用联合类型 type Query { search(query: String!): [SearchResult!]! getMedia(id: ID!): MediaContent } // 查询联合类型 query Search { search(query: "test") { ... on User { id name email } ... on Post { id title } ... on Comment { id content } } } // Resolver 返回联合类型 const resolvers = { SearchResult: { __resolveType(obj: any) { if (obj.name && obj.email) return 'User'; if (obj.title) return 'Post'; if (obj.content && !obj.title) return 'Comment'; return null; }, }, };

📥 输入类型

Input 类型用于传递复杂参数，特别是 Mutation 的输入。

// 定义输入类型 input CreateUserInput { name: String! email: String! password: String! role: UserRole = USER profile: ProfileInput } input ProfileInput { bio: String website: String avatar: String } input UpdateUserInput { name: String email: String profile: ProfileInput } input UserFilter { role: UserRole isActive: Boolean search: String createdAt: DateRangeInput } input DateRangeInput { start: DateTime! end: DateTime! } // 使用输入类型 type Mutation { createUser(input: CreateUserInput!): User! updateUser(id: ID!, input: UpdateUserInput!): User } type Query { users(filter: UserFilter, limit: Int = 10): [User!]! } // Mutation 调用 mutation { createUser(input: { name: "张三" email: "zhangsan@example.com" password: "Password123!" profile: { bio: "前端开发者" website: "https://zhangsan.dev" } }) { id name } }

📊 类型对比

类型 用途 可包含字段
type 定义输出对象 可以是对象、标量、列表
input 定义输入参数 只能是标量、枚举、其他 input
interface 定义共享字段 输出字段
union 定义多选一类型 只能是对象类型
enum 定义枚举值 字符串值
scalar 定义标量类型 单个值

📝 小结

  • 标量类型是最基本的数据类型
  • 对象类型定义复杂的数据结构
  • 枚举类型定义有限的选项集合
  • 接口定义共享字段的抽象类型
  • 联合类型表示多种类型之一
  • Input 类型专门用于输入参数
  • 非空修饰符确保字段必填
