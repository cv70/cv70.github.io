🖥️ 服务端实现

通知服务

// services/notification.service.ts import { Server } from 'socket.io'; import { Notification, NotificationType, NotificationPriority, NotificationStatus } from '../types'; class NotificationService { private io: Server; private userSockets = new Map<string, Set<string>>(); constructor(io: Server) { this.io = io; } // 注册用户连接 registerUser(userId: string, socketId: string) { if (!this.userSockets.has(userId)) { this.userSockets.set(userId, new Set()); } this.userSockets.get(userId)!.add(socketId); } // 用户断开 unregisterUser(userId: string, socketId: string) { const sockets = this.userSockets.get(userId); if (sockets) { sockets.delete(socketId); if (sockets.size === 0) { this.userSockets.delete(userId); } } } // 用户是否在线 isUserOnline(userId: string): boolean { const sockets = this.userSockets.get(userId); return sockets !== undefined && sockets.size > 0; } // 发送通知 async send(notification: Omit<Notification, 'id' | 'status' | 'createdAt'>): Promise<Notification> { const fullNotification: Notification = { ...notification, id: this.generateId(), status: NotificationStatus.UNREAD, createdAt: new Date() }; // 保存到数据库 await this.saveNotification(fullNotification); // 如果用户在线，实时推送 if (this.isUserOnline(notification.userId)) { this.io.to(`user:${notification.userId}`).emit('notification:new', fullNotification); } // 如果是高优先级，发送邮件或短信 if (notification.priority === NotificationPriority.URGENT) { await this.sendUrgentNotification(fullNotification); } return fullNotification; } // 批量发送 async sendBulk(userIds: string[], notification: Omit<Notification, 'id' | 'userId' | 'status' | 'createdAt'>) { const notifications = await Promise.all( userIds.map(userId => this.send({ ...notification, userId })) ); return notifications; } // 发送给所有用户 async broadcast(notification: Omit<Notification, 'id' | 'userId' | 'status' | 'createdAt'>) { const fullNotification = { ...notification, id: this.generateId(), userId: 'all', status: NotificationStatus.UNREAD, createdAt: new Date() }; // 保存到数据库 await this.saveNotification(fullNotification); // 广播给所有在线用户 this.io.emit('notification:broadcast', fullNotification); return fullNotification; } // 获取未读通知 async getUnread(userId: string): Promise<Notification[]> { // 从数据库获取 return this.fetchUnreadFromDb(userId); } // 标记已读 async markAsRead(notificationId: string, userId: string): Promise<boolean> { const success = await this.updateStatus(notificationId, userId, NotificationStatus.READ); if (success) { this.io.to(`user:${userId}`).emit('notification:read', { notificationId }); } return success; } // 标记全部已读 async markAllAsRead(userId: string): Promise<number> { const count = await this.markAllReadInDb(userId); this.io.to(`user:${userId}`).emit('notification:all-read', { count }); return count; } // 用户上线时推送离线通知 async pushOfflineNotifications(userId: string) { const notifications = await this.getUnread(userId); this.io.to(`user:${userId}`).emit('notification:offline', notifications); } // 私有方法 private generateId(): string { return `notif_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`; } private async saveNotification(notification: Notification): Promise<void> { // 数据库保存逻辑 } private async fetchUnreadFromDb(userId: string): Promise<Notification[]> { // 数据库查询逻辑 return []; } private async updateStatus(id: string, userId: string, status: NotificationStatus): Promise<boolean> { // 数据库更新逻辑 return true; } private async markAllReadInDb(userId: string): Promise<number> { // 数据库更新逻辑 return 0; } private async sendUrgentNotification(notification: Notification): Promise<void> { // 发送邮件或短信 console.log('发送紧急通知:', notification.title); } } export { NotificationService };

Socket.IO 集成

// socket.ts import { Server } from 'socket.io'; import { NotificationService } from './services/notification.service'; export function setupNotificationSocket(io: Server) { const notificationService = new NotificationService(io); // 通知命名空间 const notificationNS = io.of('/notifications'); notificationNS.use((socket, next) => { const userId = socket.handshake.auth.userId; if (!userId) { return next(new Error('认证失败')); } socket.data.userId = userId; next(); }); notificationNS.on('connection', async (socket) => { const userId = socket.data.userId; // 加入用户专属房间 socket.join(`user:${userId}`); // 注册用户 notificationService.registerUser(userId, socket.id); console.log(`用户 ${userId} 连接通知服务`); // 推送离线通知 await notificationService.pushOfflineNotifications(userId); // 获取未读数量 socket.on('notification:unread-count', async (callback) => { const notifications = await notificationService.getUnread(userId); callback(notifications.length); }); // 获取通知列表 socket.on('notification:list', async (params: { page: number; limit: number }, callback) => { // 分页获取通知 callback({ notifications: [], total: 0 }); }); // 标记已读 socket.on('notification:read', async (notificationId: string, callback) => { const success = await notificationService.markAsRead(notificationId, userId); callback(success); }); // 标记全部已读 socket.on('notification:read-all', async (callback) => { const count = await notificationService.markAllAsRead(userId); callback(count); }); // 断开连接 socket.on('disconnect', () => { notificationService.unregisterUser(userId, socket.id); console.log(`用户 ${userId} 断开通知服务`); }); }); return notificationService; }

📱 客户端实现

通知 Hook

// hooks/useNotifications.ts import { useEffect, useState, useCallback } from 'react'; import { io, Socket } from 'socket.io-client'; import { Notification } from '../types'; export function useNotifications(userId: string) { const [socket, setSocket] = useState<Socket | null>(null); const [notifications, setNotifications] = useState<Notification[]>([]); const [unreadCount, setUnreadCount] = useState(0); useEffect(() => { const newSocket = io('http://localhost:3000/notifications', { auth: { userId } }); setSocket(newSocket); // 接收新通知 newSocket.on('notification:new', (notification: Notification) => { setNotifications(prev => [notification, ...prev]); setUnreadCount(prev => prev + 1); // 显示浏览器通知 showBrowserNotification(notification); }); // 接收离线通知 newSocket.on('notification:offline', (offlineNotifications: Notification[]) => { setNotifications(prev => [...offlineNotifications, ...prev]); setUnreadCount(offlineNotifications.filter(n => n.status === 'unread').length); }); // 广播通知 newSocket.on('notification:broadcast', (notification: Notification) => { setNotifications(prev => [notification, ...prev]); setUnreadCount(prev => prev + 1); showBrowserNotification(notification); }); // 已读状态更新 newSocket.on('notification:read', ({ notificationId }) => { setNotifications(prev => prev.map(n => n.id === notificationId ? { ...n, status: 'read' } : n) ); setUnreadCount(prev => Math.max(0, prev - 1)); }); // 全部已读 newSocket.on('notification:all-read', ({ count }) => { setNotifications(prev => prev.map(n => ({ ...n, status: 'read' })) ); setUnreadCount(0); }); // 获取未读数量 newSocket.emit('notification:unread-count', (count: number) => { setUnreadCount(count); }); return () => { newSocket.close(); }; }, [userId]); const markAsRead = useCallback((notificationId: string) => { socket?.emit('notification:read', notificationId, (success: boolean) => { if (success) { console.log('通知已标记已读'); } }); }, [socket]); const markAllAsRead = useCallback(() => { socket?.emit('notification:read-all', (count: number) => { console.log(`已标记 ${count} 条通知为已读`); }); }, [socket]); return { notifications, unreadCount, markAsRead, markAllAsRead }; } // 浏览器通知 function showBrowserNotification(notification: Notification) { if ('Notification' in window && Notification.permission === 'granted') { new Notification(notification.title, { body: notification.body, icon: notification.icon, tag: notification.id }); } } // 请求通知权限 export async function requestNotificationPermission(): Promise<boolean> { if ('Notification' in window) { const permission = await Notification.requestPermission(); return permission === 'granted'; } return false; }

通知组件

// components/NotificationBell.tsx import React, { useState } from 'react'; import { useNotifications } from '../hooks/useNotifications'; export function NotificationBell({ userId }: { userId: string }) { const { notifications, unreadCount, markAsRead, markAllAsRead } = useNotifications(userId); const [isOpen, setIsOpen] = useState(false); return ( <div className="notification-bell"> <button onClick={() => setIsOpen(!isOpen)} className="bell-button"> 🔔 {unreadCount > 0 && ( <span className="badge">{unreadCount > 99 ? '99+' : unreadCount}</span> )} </button> {isOpen && ( <div className="notification-dropdown"> <div className="dropdown-header"> <h3>通知</h3> {unreadCount > 0 && ( <button onClick={markAllAsRead}>全部已读</button> )} </div> <div className="notification-list"> {notifications.length === 0 ? ( <div className="empty">暂无通知</div> ) : ( notifications.slice(0, 10).map(notification => ( <div key={notification.id} className={`notification-item ${notification.status}`} onClick={() => markAsRead(notification.id)} > <div className="notification-icon"> {getNotificationIcon(notification.type)} </div> <div className="notification-content"> <div className="title">{notification.title}</div> <div className="body">{notification.body}</div> <div className="time">{formatTime(notification.createdAt)}</div> </div> </div> )) )} </div> </div> )} </div> ); } function getNotificationIcon(type: string): string { const icons: Record<string, string> = { system: '⚙️', message: '💬', alert: '⚠️', update: '🔄', promotion: '🎉', reminder: '⏰' }; return icons[type] || '📌'; } function formatTime(date: Date): string { const now = new Date(); const diff = now.getTime() - new Date(date).getTime(); const minutes = Math.floor(diff / 60000); const hours = Math.floor(diff / 3600000); const days = Math.floor(diff / 86400000); if (minutes < 1) return '刚刚'; if (minutes < 60) return `${minutes}分钟前`; if (hours < 24) return `${hours}小时前`; return `${days}天前`; }
