39.3 模板引擎

在 Koa 中使用模板渲染页面

🎯 学习目标

  • 学会使用 koa-views 配置模板引擎
  • 掌握 EJS 模板语法
  • 了解模板布局和继承
  • 学会模板数据传递

📦 安装配置

# 安装 koa-views 和 EJS npm install koa-views ejs
const Koa = require('koa'); const views = require('koa-views'); const path = require('path'); const app = new Koa(); // 配置模板引擎 app.use(views(path.join(__dirname, 'views'), { extension: 'ejs', // 模板扩展名 map: { html: 'ejs' } // 将 .html 文件也使用 ejs 解析 }));

📄 基本渲染

// views/index.ejs <%= title %>

<%= message %>

🔧 全局模板变量

// 中间件设置全局变量 app.use(async (ctx, next) => { ctx.state = ctx.state || {}; ctx.state.siteName = 'My Website'; ctx.state.currentYear = new Date().getFullYear(); ctx.state.user = ctx.session.user || null; await next(); }); // 模板中直接使用

⚡ 模板缓存

const views = require('koa-views'); app.use(views(path.join(__dirname, 'views'), { extension: 'ejs', options: { cache: process.env.NODE_ENV === 'production', // 生产环境启用缓存 debug: process.env.NODE_ENV !== 'production' // 开发环境调试 } }));
💡
缓存建议

开发环境禁用缓存便于调试，生产环境启用缓存提高性能。

🎨 使用其他模板引擎

// 使用 Pug (Jade) npm install pug app.use(views(path.join(__dirname, 'views'), { extension: 'pug' })); // 使用 Nunjucks npm install nunjucks app.use(views(path.join(__dirname, 'views'), { map: { html: 'nunjucks' } })); // 使用 Handlebars npm install handlebars app.use(views(path.join(__dirname, 'views'), { extension: 'hbs', map: { hbs: 'handlebars' } }));

💻 完整示例

// app.js const Koa = require('koa'); const Router = require('koa-router'); const views = require('koa-views'); const serve = require('koa-static'); const path = require('path'); const app = new Koa(); const router = new Router(); // 静态文件 app.use(serve(path.join(__dirname, 'public'))); // 模板引擎 app.use(views(path.join(__dirname, 'views'), { extension: 'ejs' })); // 全局变量 app.use(async (ctx, next) => { ctx.state = { siteName: 'My Blog', user: ctx.session?.user || null }; await next(); }); // 路由 router.get('/', async ctx => { const posts = await Post.findAll(); await ctx.render('index', { title: 'Home', posts }); }); router.get('/post/:id', async ctx => { const post = await Post.findById(ctx.params.id); await ctx.render('post', { title: post.title, post }); }); router.get('/about', async ctx => { await ctx.render('about', { title: 'About' }); }); app.use(router.routes()); app.listen(3000);
