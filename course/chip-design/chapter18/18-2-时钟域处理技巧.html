18.2 时钟域处理技巧

HDL 编码最佳实践

🎯 学习目标

  • 理解跨时钟域的问题
  • 掌握单bit信号同步
  • 学会多bit信号同步
  • 能够处理异步时钟域

🔄 双触发器同步

// 双触发器同步器（单bit信号）
module sync_2ff(
    input  clk_dst,    // 目标时钟
    input  rst_n,
    input  async_in,   // 异步输入
    output sync_out    // 同步输出
);
    reg meta, sync;

    always @(posedge clk_dst or negedge rst_n) begin
        if (!rst_n) begin
            meta <= 1'b0;
            sync <= 1'b0;
        end
        else begin
            meta <= async_in;
            sync <= meta;
        end
    end

    assign sync_out = sync;
endmodule

📊 握手协议

// 握手同步（多bit信号）
module handshake_sync(
    input  clk_src,
    input  clk_dst,
    input  rst_n,
    input  [7:0] data_in,
    input        req_in,
    output       ack_out,
    output [7:0] data_out
);
    // 源时钟域
    reg [7:0] data_reg;
    reg req_reg;
    
    always @(posedge clk_src or negedge rst_n) begin
        if (!rst_n) begin
            data_reg <= 0;
            req_reg <= 0;
        end
        else if (req_in && !req_reg) begin
            data_reg <= data_in;
            req_reg <= 1;
        end
        else if (ack_out)
            req_reg <= 0;
    end
    
    // 目标时钟域（同步req信号）
    reg req_meta, req_sync;
    always @(posedge clk_dst or negedge rst_n) begin
        if (!rst_n) begin
            req_meta <= 0;
            req_sync <= 0;
        end
        else begin
            req_meta <= req_reg;
            req_sync <= req_meta;
        end
    end
    
    // 数据捕获
    always @(posedge clk_dst or negedge rst_n) begin
        if (!rst_n)
            data_out <= 0;
        else if (req_sync && !ack_out)
            data_out <= data_reg;
    end
    
    // ACK生成
    reg ack_reg;
    always @(posedge clk_dst or negedge rst_n) begin
        if (!rst_n)
            ack_reg <= 0;
        else
            ack_reg <= req_sync;
    end
    
    assign ack_out = ack_reg;
endmodule

📝 本节小结

✅
  • • 跨时钟域需要同步
  • • 单bit用双触发器
  • • 多bit用握手或FIFO
  • • 避免亚稳态传播
