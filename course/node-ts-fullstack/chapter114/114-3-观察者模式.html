114.3 观察者模式

对象间的发布-订阅关系

📚 模式定义

// 观察者模式：定义对象间的一对多依赖，当一个对象状态改变时，所有依赖者都会收到通知 // 应用场景 const useCases = [ '事件处理系统', '消息订阅', '数据绑定', '状态管理', '日志系统' ]; // 基本实现 interface Observer { update(data: T): void; } interface Subject { attach(observer: Observer): void; detach(observer: Observer): void; notify(data: T): void; } class ConcreteSubject implements Subject { private observers: Set> = new Set(); attach(observer: Observer): void { this.observers.add(observer); } detach(observer: Observer): void { this.observers.delete(observer); } notify(data: T): void { this.observers.forEach(observer => observer.update(data)); } }
💡
观察者优点

解耦发布者和订阅者、支持广播通信、动态添加删除观察者。

🔧 TypeScript实现

// 事件发射器实现 type EventHandler = (data: T) => void; class EventEmitter> { private handlers: Map> = new Map(); on(event: K, handler: EventHandler) { if (!this.handlers.has(event)) { this.handlers.set(event, new Set()); } this.handlers.get(event)!.add(handler); return () => this.off(event, handler); } off(event: K, handler: EventHandler) { this.handlers.get(event)?.delete(handler); } emit(event: K, data: EventMap[K]) { this.handlers.get(event)?.forEach(handler => handler(data)); } once(event: K, handler: EventHandler) { const wrapper: EventHandler = (data) => { this.off(event, wrapper); handler(data); }; this.on(event, wrapper); } } // 使用示例 interface AppEvents { 'user:login': { userId: string; timestamp: Date }; 'user:logout': { userId: string }; 'order:created': { orderId: string; amount: number }; } const eventBus = new EventEmitter(); // 订阅事件 const unsubscribe = eventBus.on('user:login', (data) => { console.log(`User ${data.userId} logged in at ${data.timestamp}`); }); // 发射事件 eventBus.emit('user:login', { userId: 'user-123', timestamp: new Date() }); // 取消订阅 unsubscribe(); // 实际应用：订单状态通知 class OrderService { constructor(private eventBus: EventEmitter) {} async createOrder(orderData: any) { const order = await this.saveOrder(orderData); // 发布事件 this.eventBus.emit('order:created', { orderId: order.id, amount: order.amount }); return order; } } // 通知服务订阅订单事件 class NotificationService { constructor(eventBus: EventEmitter) { eventBus.on('order:created', async (data) => { await this.sendOrderNotification(data.orderId); }); } private async sendOrderNotification(orderId: string) { console.log(`Sending notification for order ${orderId}`); } }
