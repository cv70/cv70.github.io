什么是语音助手

语音助手是能够通过语音进行交互的智能系统，能够理解用户意图、执行相应任务并以语音形式反馈结果。从Siri、Alexa到ChatGPT Voice，语音助手正在改变人机交互方式，成为智能家居、车载系统、客服机器人的核心入口。

💻 代码示例：语音助手核心逻辑

import whisper
from transformers import pipeline
import edge_tts
import asyncio

class VoiceAssistant:
    def __init__(self):
        # 初始化各模块
        self.asr = whisper.load_model("base")
        self.nlu = pipeline("text-classification",
                           model="intent-model")
        self.skills = {
            "play_music": self.play_music,
            "set_alarm": self.set_alarm,
            "weather": self.get_weather
        }

    async def process_voice(self, audio_path):
        # 1. 语音识别
        text = self.asr.transcribe(audio_path)["text"]

        # 2. 意图识别
        intent_result = self.nlu(text)[0]
        intent = intent_result["label"]

        # 3. 槽位提取（简化示例）
        slots = self.extract_slots(text, intent)

        # 4. 执行技能
        response = await self.skills[intent](slots)

        # 5. 语音合成
        await self.speak(response)
        return response

    async def speak(self, text):
        communicate = edge_tts.Communicate(text, "zh-CN-XiaoxiaoNeural")
        await communicate.save("response.mp3")
        # 播放音频...

    def extract_slots(self, text, intent):
        # 基于规则或模型的槽位提取
        slots = {}
        if intent == "play_music":
            # 提取歌曲名
            slots["song"] = self.extract_song_name(text)
        return slots

# 使用示例
assistant = VoiceAssistant()
response = asyncio.run(assistant.process_voice("user_audio.wav"))
