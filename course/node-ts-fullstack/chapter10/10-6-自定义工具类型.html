10.6 自定义工具类型

创建项目专属的类型工具

🎯 学习目标

  • 掌握常用自定义工具类型的实现
  • 学会深层属性操作类型
  • 理解路径获取类型
  • 了解类型安全的实践

📚 深层属性类型

// DeepPartial - 深层可选 type DeepPartial = { [P in keyof T]?: T[P] extends object ? T[P] extends Function ? T[P] : DeepPartial : T[P]; }; interface Config { api: { url: string; headers: { auth: string; }; }; database: { host: string; port: number; }; } type PartialConfig = DeepPartial; // 所有层级属性都是可选的 const config: PartialConfig = { api: { headers: { auth: 'Bearer token' } } }; // DeepRequired - 深层必需 type DeepRequired = { [P in keyof T]-?: T[P] extends object ? T[P] extends Function ? T[P] : DeepRequired : T[P]; }; // DeepReadonly - 深层只读 type DeepReadonly = { readonly [P in keyof T]: T[P] extends object ? T[P] extends Function ? T[P] : DeepReadonly : T[P]; };

🔧 属性筛选类型

// 按类型筛选属性 type PickByType = { [K in keyof T as T[K] extends U ? K : never]: T[K]; }; type OmitByType = { [K in keyof T as T[K] extends U ? never : K]: T[K]; }; interface User { id: number; name: string; email: string; active: boolean; metadata: object; } type StringProps = PickByType; // { name: string; email: string; } type NonBooleanProps = OmitByType; // { id: number; name: string; email: string; metadata: object; } // 必需属性 type RequiredKeys = { [K in keyof T]-?: {} extends Pick ? never : K; }[keyof T]; type OptionalKeys = { [K in keyof T]-?: {} extends Pick ? K : never; }[keyof T]; interface Example { required: string; optional?: number; another: boolean; } type Req = RequiredKeys; // 'required' | 'another' type Opt = OptionalKeys; // 'optional'

🔀 路径访问类型

// 获取嵌套对象的路径类型 type Path = K extends keyof T ? T[K] : K extends `${infer First}.${infer Rest}` ? First extends keyof T ? Path : never : never; interface State { user: { profile: { name: string; age: number; }; settings: { theme: string; }; }; } type UserName = Path; // string type UserAge = Path; // number // 获取所有可能的路径 type AllPaths = T extends object ? { [K in keyof T]: | `${Prefix}${K & string}` | AllPaths }[keyof T] : never; type StatePaths = AllPaths; // 'user' | 'user.profile' | 'user.profile.name' | 'user.profile.age' | ... // 安全的路径访问 function getByPath( obj: T, path: P ): Path { return path.split('.').reduce((acc: any, key) => acc?.[key], obj); } const state: State = { user: { profile: { name: 'Alice', age: 30 }, settings: { theme: 'dark' } } }; const name = getByPath(state, 'user.profile.name'); // string

🎨 实用工具类型集合

// 合并两个类型，后者覆盖前者 type Merge = Omit & U; interface Defaults { timeout: number; retries: number; debug: boolean; } interface Overrides { timeout: number; logLevel: string; } type MergedConfig = Merge; // { retries: number; debug: boolean; timeout: number; logLevel: string; } // 互斥属性 type MutuallyExclusive = | (T & { [K in Exclude]?: never }) | (U & { [K in Exclude]?: never }); interface Cat { meow: () => void; lives: number; } interface Dog { bark: () => void; loyalty: number; } type Pet = MutuallyExclusive; // 只能是 Cat 或 Dog，不能混合 // 标记属性 type Brand = T & { __brand: B }; type UserId = Brand; type ProductId = Brand; const userId = 1 as UserId; const productId = 1 as ProductId; // userId === productId // 类型错误 // 不可为空的对象 type NonEmptyObject = keyof T extends never ? never : T;
💡
自定义类型工具建议

创建项目专属的工具类型可以显著提高代码复用性和类型安全性。但要注意保持简洁，避免过度复杂化。
