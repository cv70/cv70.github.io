70.3 参数化测试

第十部分：测试篇 - 使用参数化测试减少重复代码

1. 参数化测试简介

参数化测试允许使用不同的输入数据运行同一个测试用例，避免编写重复的测试代码。

💡
参数化测试的优势
  • 减少代码重复
  • 轻松添加新的测试用例
  • 测试数据与测试逻辑分离
  • 提高测试覆盖率

2. test.each 基础

// 基本语法 test.each(table)(name, fn, timeout) // table: 测试数据数组 // name: 测试名称，可以使用占位符 // fn: 测试函数，接收 table 中的数据

使用数组

describe('Calculator', () => { // 数组形式 test.each([ [1, 1, 2], [1, 2, 3], [2, 3, 5], [0, 0, 0], [-1, 1, 0], ])('.add(%i, %i) = %i', (a, b, expected) => { expect(add(a, b)).toBe(expected); }); test.each([ [10, 2, 5], [9, 3, 3], [100, 10, 10], ])('.divide(%i, %i) = %i', (a, b, expected) => { expect(divide(a, b)).toBe(expected); }); });

输出结果

PASS calculator.test.ts Calculator ✓ .add(1, 1) = 2 ✓ .add(1, 2) = 3 ✓ .add(2, 3) = 5 ✓ .add(0, 0) = 0 ✓ .add(-1, 1) = 0 ✓ .divide(10, 2) = 5 ✓ .divide(9, 3) = 3 ✓ .divide(100, 10) = 10

3. 使用对象数组

describe('Validation', () => { test.each([ { input: 'test@example.com', expected: true }, { input: 'invalid-email', expected: false }, { input: 'user@domain', expected: false }, { input: 'user.name@domain.com', expected: true }, { input: '', expected: false }, ])('validateEmail("$input") = $expected', ({ input, expected }) => { expect(validateEmail(input)).toBe(expected); }); test.each([ { password: 'Abc123!', valid: true, reason: 'meets all requirements' }, { password: 'abc123!', valid: false, reason: 'missing uppercase' }, { password: 'ABC123!', valid: false, reason: 'missing lowercase' }, { password: 'Abcdef!', valid: false, reason: 'missing number' }, { password: 'Abc1234', valid: false, reason: 'missing special char' }, { password: 'Ab1!', valid: false, reason: 'too short' }, ])('password "$password" is $valid because $reason', ({ password, valid }) => { expect(validatePassword(password)).toBe(valid); }); });

4. 模板字面量语法

// 使用模板字面量语法，更清晰 describe('Math Operations', () => { test.each` a | b | expected ${1} | ${1} | ${2} ${1} | ${2} | ${3} ${2} | ${3} | ${5} ${0} | ${0} | ${0} ${-1}| ${1} | ${0} `('returns $expected when adding $a and $b', ({ a, b, expected }) => { expect(add(a, b)).toBe(expected); }); test.each` value | percentage | expected ${50} | ${100} | ${50} ${25} | ${100} | ${25} ${1} | ${3} | ${33.33} ${0} | ${100} | ${0} ${100} | ${0} | ${0} `('percentage($value, $percentage) = $expected', ({ value, percentage, expected }) => { expect(percentage(value, percentage)).toBeCloseTo(expected, 1); }); });

5. describe.each

// 对测试套件进行参数化 describe.each([ { name: 'USD', symbol: '$', rate: 1 }, { name: 'EUR', symbol: '€', rate: 0.85 }, { name: 'GBP', symbol: '£', rate: 0.73 }, { name: 'JPY', symbol: '¥', rate: 110.0 }, ])('Currency: $name', ({ name, symbol, rate }) => { let currency: Currency; beforeEach(() => { currency = new Currency(name, symbol, rate); }); test('has correct symbol', () => { expect(currency.symbol).toBe(symbol); }); test('converts from USD correctly', () => { const amount = 100; expect(currency.fromUSD(amount)).toBeCloseTo(amount * rate); }); test('converts to USD correctly', () => { const amount = 100; expect(currency.toUSD(amount)).toBeCloseTo(amount / rate); }); });

模板字面量形式

describe.each` name | symbol | rate ${'USD'}| ${'$'} | ${1} ${'EUR'}| ${'€'} | ${0.85} ${'GBP'}| ${'£'} | ${0.73} `('Currency: $name', ({ name, symbol, rate }) => { test('properties are correct', () => { const currency = new Currency(name, symbol, rate); expect(currency.name).toBe(name); expect(currency.symbol).toBe(symbol); expect(currency.rate).toBe(rate); }); });

6. 异步参数化测试

describe('API Tests', () => { test.each([ { userId: '1', expectedName: 'John' }, { userId: '2', expectedName: 'Jane' }, { userId: '3', expectedName: 'Bob' }, ])('getUser($userId) returns $expectedName', async ({ userId, expectedName }) => { const user = await fetchUser(userId); expect(user.name).toBe(expectedName); }); // 使用模板字面量 test.each` endpoint | method | status ${'/users'} | ${'GET'} | ${200} ${'/users'} | ${'POST'} | ${201} ${'/users/1'} | ${'GET'} | ${200} ${'/users/999'} | ${'GET'} | ${404} `('$method $endpoint returns $status', async ({ endpoint, method, status }) => { const response = await request(app)[method.toLowerCase()](endpoint); expect(response.status).toBe(status); }); });

8. 复杂场景示例

// 测试用户权限系统 describe('Permission System', () => { const permissions = [ { role: 'admin', action: 'create', resource: 'user', allowed: true }, { role: 'admin', action: 'delete', resource: 'user', allowed: true }, { role: 'editor', action: 'create', resource: 'post', allowed: true }, { role: 'editor', action: 'delete', resource: 'post', allowed: false }, { role: 'viewer', action: 'read', resource: 'post', allowed: true }, { role: 'viewer', action: 'create', resource: 'post', allowed: false }, ]; test.each(permissions)( '$role can $action $resource: $allowed', ({ role, action, resource, allowed }) => { const hasPermission = checkPermission(role, action, resource); expect(hasPermission).toBe(allowed); } ); }); // 测试边界情况 describe('Edge Cases', () => { test.each([ { input: null, description: 'null' }, { input: undefined, description: 'undefined' }, { input: '', description: 'empty string' }, { input: [], description: 'empty array' }, { input: {}, description: 'empty object' }, { input: 0, description: 'zero' }, { input: NaN, description: 'NaN' }, ])('handles $description correctly', ({ input }) => { expect(() => processData(input)).not.toThrow(); }); });

9. 本节小结

📚
关键要点
  • test.each 用于参数化单个测试
  • describe.each 用于参数化测试套件
  • 支持数组、对象数组和模板字面量语法
  • 占位符用于生成有意义的测试名称
  • 参数化测试减少重复，提高可维护性
