10.5 Exclude 和 Extract

过滤联合类型

🎯 学习目标

  • 深入理解 Exclude 的实现原理
  • 掌握 Extract 的使用场景
  • 学会 NonNullable 的用法
  • 了解实际项目中的应用

📚 Exclude 深入

// Exclude 从联合类型中排除类型 // 内置实现： type MyExclude = T extends U ? never : T; // 基本用法 type AllTypes = 'a' | 'b' | 'c' | 'd'; type WithoutA = Exclude; // 'b' | 'c' | 'd' type WithoutAB = Exclude; // 'c' | 'd' // 排除多种类型 type Mixed = string | number | boolean | null | undefined; type OnlyStrings = Exclude; // string // 实际应用：从接口键中排除 interface User { id: number; name: string; email: string; password: string; createdAt: Date; } type UserKeys = keyof User; // 'id' | 'name' | 'email' | 'password' | 'createdAt' type EditableKeys = Exclude; // 'name' | 'email' | 'password' type EditableUser = Pick; // { name: string; email: string; password: string; }

🔧 Extract 深入

// Extract 从联合类型中提取类型 // 内置实现： type MyExtract = T extends U ? T : never; // 基本用法 type Values = 'click' | 'focus' | 'blur' | 'keydown' | 'keyup'; type KeyboardEvents = Extract; // 'keydown' | 'keyup' // 提取特定类型 type Primitive = string | number | boolean | object | Function; type OnlyFunctions = Extract; // Function // 提取接口实现 interface Animal { name: string; } interface Dog extends Animal { breed: string; } interface Cat extends Animal { color: string; } type Pet = Dog | Cat | string; type OnlyAnimals = Extract; // Dog | Cat // 提取有特定属性的类型 type HasId = { id: number }; type Entity1 = HasId & { name: string }; type Entity2 = { code: string }; type MixedEntities = Entity1 | Entity2; type WithId = Extract; // Entity1
💡
Exclude vs Extract

Exclude 排除匹配的类型，Extract 提取匹配的类型。它们互补，一个相当于"差集"，一个相当于"交集"。

⚙️ NonNullable

// NonNullable 排除 null 和 undefined // 内置实现： type MyNonNullable = Exclude; // 基本用法 type MaybeString = string | null | undefined; type DefiniteString = NonNullable; // string // 实际应用：处理 API 响应 interface ApiResponse { data: T | null; error: string | null; } function getData(response: ApiResponse): NonNullable { if (response.data === null) { throw new Error(response.error ?? 'Unknown error'); } return response.data; } // 数组过滤 type ArrayValue = T extends (infer U)[] ? NonNullable : never; type CleanArray = ArrayValue<(string | null)[]>; // string // 对象属性非空化 type NonNullableProps = { [K in keyof T]: NonNullable; }; interface OptionalUser { name: string | null; email: string | undefined; } type RequiredUser = NonNullableProps; // { name: string; email: string; }

🎨 实际应用

// 状态管理 type Status = 'idle' | 'loading' | 'success' | 'error'; type ActiveStatus = Exclude; // 'loading' | 'success' | 'error' // 事件处理 type AllEvents = 'click' | 'dblclick' | 'mousedown' | 'mouseup' | 'keydown' | 'keyup'; type MouseEvents = Exclude; // 'click' | 'dblclick' | 'mousedown' | 'mouseup' // 权限过滤 type Permission = 'read' | 'write' | 'delete' | 'admin'; type UserPermission = Exclude; // 'read' | 'write' | 'delete' // 路由守卫 type PublicRoute = '/' | '/login' | '/register'; type ProtectedRoute = '/dashboard' | '/profile' | '/settings'; type AllRoutes = PublicRoute | ProtectedRoute; type OnlyProtected = Extract; // '/dashboard' | '/profile' | '/settings' // API 方法过滤 type HttpMethod = 'GET' | 'POST' | 'PUT' | 'DELETE' | 'PATCH'; type ReadOnlyMethod = Extract; type WriteMethod = Exclude; // 'POST' | 'PUT' | 'DELETE' | 'PATCH'

📝 本节小结

✅
  • • Exclude 从联合类型中排除类型
  • • Extract 从联合类型中提取类型
  • • NonNullable 排除 null 和 undefined
  • • 常用于事件、权限、状态类型过滤
