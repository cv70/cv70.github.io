📚 领域事件

// 领域事件：领域中发生的、对业务有意义的事情 // 特点 const characteristics = { 过去式命名: 'OrderCreated, PaymentCompleted', 不可变: '事件一旦发生不能修改', 包含上下文: '事件携带发生的上下文信息', 时间戳: '记录事件发生时间' }; // 领域事件定义 interface DomainEvent { eventId: string; eventType: string; aggregateId: string; occurredAt: Date; payload: any; } class OrderCreatedEvent implements DomainEvent { readonly eventType = 'order.created'; readonly eventId: string; readonly occurredAt: Date; constructor( readonly aggregateId: string, readonly payload: { customerId: string; items: OrderItemData[]; total: number; } ) { this.eventId = uuidv4(); this.occurredAt = new Date(); } } class OrderConfirmedEvent implements DomainEvent { readonly eventType = 'order.confirmed'; readonly eventId: string; readonly occurredAt: Date; constructor( readonly aggregateId: string, readonly payload: { total: number; shippingAddress: Address; } ) { this.eventId = uuidv4(); this.occurredAt = new Date(); } }
💡
事件用途

解耦聚合、触发后续操作、审计日志、事件溯源。

🔧 事件发布与处理

// 事件发布器 interface EventPublisher { publish(event: DomainEvent): Promise; publishAll(events: DomainEvent[]): Promise; } @Injectable() class RabbitMQEventPublisher implements EventPublisher { constructor(private channel: Channel) {} async publish(event: DomainEvent): Promise { const message = JSON.stringify({ eventId: event.eventId, eventType: event.eventType, aggregateId: event.aggregateId, occurredAt: event.occurredAt.toISOString(), payload: event.payload }); this.channel.publish( 'domain-events', event.eventType, Buffer.from(message) ); } async publishAll(events: DomainEvent[]): Promise { for (const event of events) { await this.publish(event); } } } // 事件处理器 interface EventHandler { eventType: string; handle(event: T): Promise; } // 订单确认处理器：通知库存 @EventHandler('order.confirmed') class ReserveInventoryHandler implements EventHandler { eventType = 'order.confirmed'; constructor(private inventoryService: InventoryService) {} async handle(event: OrderConfirmedEvent): Promise { // 预留库存 await this.inventoryService.reserveStock( event.aggregateId, event.payload.items ); } } // 订单确认处理器：发送通知 @EventHandler('order.confirmed') class SendOrderNotificationHandler implements EventHandler { eventType = 'order.confirmed'; constructor(private notificationService: NotificationService) {} async handle(event: OrderConfirmedEvent): Promise { await this.notificationService.sendOrderConfirmation( event.payload.customerId, event.aggregateId ); } } // 事件分发器 @Injectable() class EventDispatcher { private handlers: Map = new Map(); register(handler: EventHandler): void { const handlers = this.handlers.get(handler.eventType) || []; handlers.push(handler); this.handlers.set(handler.eventType, handlers); } async dispatch(event: DomainEvent): Promise { const handlers = this.handlers.get(event.eventType) || []; await Promise.all( handlers.map(handler => handler.handle(event as any)) ); } } // 聚合中使用事件 abstract class AggregateRoot { private events: DomainEvent[] = []; protected addEvent(event: DomainEvent): void { this.events.push(event); } pullEvents(): DomainEvent[] { const events = [...this.events]; this.events = []; return events; } } class Order extends AggregateRoot { confirm(address: Address): void { // ... 验证和状态变更 ... this.status = OrderStatus.CONFIRMED; // 发布领域事件 this.addEvent(new OrderConfirmedEvent(this.id.value, { total: this.total.getAmount(), shippingAddress: address })); } }
