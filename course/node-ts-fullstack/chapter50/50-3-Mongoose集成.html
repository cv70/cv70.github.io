50.3 Mongoose 集成

在 NestJS 中使用 MongoDB

🎯 学习目标

  • 安装和配置 Mongoose
  • 定义 Schema 和模型
  • 使用 Mongoose 进行 CRUD
  • 理解 MongoDB 特性

📚 安装和配置

# 安装依赖 npm install @nestjs/mongoose mongoose // app.module.ts import { Module } from '@nestjs/common'; import { MongooseModule } from '@nestjs/mongoose'; @Module({ imports: [ MongooseModule.forRoot('mongodb://localhost:27017/mydb'), ], }) export class AppModule {} // 使用配置服务 import { ConfigModule, ConfigService } from '@nestjs/config'; @Module({ imports: [ ConfigModule.forRoot(), MongooseModule.forRootAsync({ imports: [ConfigModule], useFactory: (config: ConfigService) => ({ uri: config.get('MONGODB_URI'), }), inject: [ConfigService], }), ], }) export class AppModule {}

🔧 定义 Schema

// users/schemas/user.schema.ts import { Prop, Schema, SchemaFactory } from '@nestjs/mongoose'; import { Document, Types } from 'mongoose'; @Schema({ timestamps: true }) export class User extends Document { @Prop({ required: true }) name: string; @Prop({ required: true, unique: true }) email: string; @Prop({ required: true, select: false }) password: string; @Prop({ enum: ['user', 'admin'], default: 'user' }) role: string; @Prop({ default: true }) isActive: boolean; @Prop() createdAt: Date; @Prop() updatedAt: Date; } export const UserSchema = SchemaFactory.createForClass(User); // 添加索引 UserSchema.index({ email: 1 }); // 添加虚拟属性 UserSchema.virtual('profile').get(function() { return { name: this.name, email: this.email }; }); // orders/schemas/order.schema.ts @Schema({ timestamps: true }) export class Order extends Document { @Prop({ type: Types.ObjectId, ref: 'User', required: true }) userId: Types.ObjectId; @Prop({ type: Number, required: true }) total: number; @Prop({ enum: ['pending', 'paid', 'shipped'], default: 'pending' }) status: string; @Prop([{ productId: Types.ObjectId, quantity: Number, price: Number }]) items: OrderItem[]; } export const OrderSchema = SchemaFactory.createForClass(Order);
💡
Schema 装饰器

@Schema() 定义集合，@Prop() 定义字段。NestJS 提供的装饰器简化了 Mongoose Schema 的定义。

📦 使用 Model

// users.module.ts import { Module } from '@nestjs/common'; import { MongooseModule } from '@nestjs/mongoose'; import { User, UserSchema } from './schemas/user.schema'; import { UsersService } from './users.service'; import { UsersController } from './users.controller'; @Module({ imports: [ MongooseModule.forFeature([ { name: User.name, schema: UserSchema }, ]), ], controllers: [UsersController], providers: [UsersService], }) export class UsersModule {} // users.service.ts import { Injectable } from '@nestjs/common'; import { InjectModel } from '@nestjs/mongoose'; import { Model } from 'mongoose'; import { User } from './schemas/user.schema'; @Injectable() export class UsersService { constructor( @InjectModel(User.name) private userModel: Model, ) {} async findAll(): Promise { return this.userModel.find().exec(); } async findOne(id: string): Promise { return this.userModel.findById(id).exec(); } async findByEmail(email: string): Promise { return this.userModel.findOne({ email }).exec(); } async create(createUserDto: CreateUserDto): Promise { const createdUser = new this.userModel(createUserDto); return createdUser.save(); } async update(id: string, updateUserDto: UpdateUserDto): Promise { return this.userModel .findByIdAndUpdate(id, updateUserDto, { new: true }) .exec(); } async remove(id: string): Promise { return this.userModel.findByIdAndDelete(id).exec(); } // 复杂查询 async findActiveUsers() { return this.userModel .find({ isActive: true }) .select('-password') // 排除密码字段 .sort({ createdAt: -1 }) .limit(10) .exec(); } // 聚合查询 async getUserOrderStats(userId: string) { return this.userModel.aggregate([ { $match: { _id: Types.ObjectId(userId) } }, { $lookup: { from: 'orders', localField: '_id', foreignField: 'userId', as: 'orders', }, }, { $project: { name: 1, email: 1, orderCount: { $size: '$orders' }, totalSpent: { $sum: '$orders.total' }, }, }, ]); } }

📝 本节小结

✅
  • • MongooseModule 提供 MongoDB 集成
  • • @Schema() 和 @Prop() 定义数据模型
  • • @InjectModel 注入 Model 实例
  • • 支持 MongoDB 的所有特性
