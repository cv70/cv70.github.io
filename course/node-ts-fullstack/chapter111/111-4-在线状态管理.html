状态存储选择

Redis适合存储在线状态，支持TTL自动过期、高性能读写。

🔧 Redis实现在线状态

// presence.service.ts import { Redis } from 'ioredis'; @Injectable() export class PresenceService { private redis: Redis; constructor() { this.redis = new Redis(process.env.REDIS_URL); } private readonly PREFIX = 'presence:'; // 设置用户在线 async setOnline(userId: string, socketId: string, device: DeviceInfo) { const key = `${this.PREFIX}${userId}`; // 添加设备到集合 await this.redis.hset(key, socketId, JSON.stringify({ ...device, status: UserStatus.ONLINE, lastSeen: Date.now(), })); // 设置过期时间（30分钟无心跳自动离线） await this.redis.expire(key, 1800); // 加入在线用户集合 await this.redis.sadd('online_users', userId); } // 设置用户离线 async setOffline(userId: string, socketId: string) { const key = `${this.PREFIX}${userId}`; // 移除设备 await this.redis.hdel(key, socketId); // 检查是否还有其他设备在线 const devices = await this.redis.hlen(key); if (devices === 0) { // 没有设备在线，从在线集合移除 await this.redis.srem('online_users', userId); // 保存最后在线时间 await this.redis.set(`last_seen:${userId}`, Date.now()); } } // 获取用户状态 async getUserStatus(userId: string): Promise { const key = `${this.PREFIX}${userId}`; const devices = await this.redis.hgetall(key); if (!devices || Object.keys(devices).length === 0) { // 获取最后在线时间 const lastSeen = await this.redis.get(`last_seen:${userId}`); return { userId, status: UserStatus.OFFLINE, lastSeen: lastSeen ? new Date(parseInt(lastSeen)) : null, } as any; } // 返回最新的设备状态 const latestDevice = Object.values(devices) .map(d => JSON.parse(d)) .sort((a, b) => b.lastSeen - a.lastSeen)[0]; return { userId, status: latestDevice.status, lastSeen: new Date(latestDevice.lastSeen), }; } // 批量获取用户状态 async getBatchUserStatus(userIds: string[]): Promise> { const pipeline = this.redis.pipeline(); userIds.forEach(id => { pipeline.exists(`${this.PREFIX}${id}`); }); const results = await pipeline.exec(); const statusMap = new Map(); results.forEach((result, index) => { const exists = result[1] === 1; statusMap.set(userIds[index], exists ? UserStatus.ONLINE : UserStatus.OFFLINE); }); return statusMap; } // 心跳更新 async heartbeat(userId: string, socketId: string) { const key = `${this.PREFIX}${userId}`; const deviceData = await this.redis.hget(key, socketId); if (deviceData) { const device = JSON.parse(deviceData); device.lastSeen = Date.now(); await this.redis.hset(key, socketId, JSON.stringify(device)); await this.redis.expire(key, 1800); } } // 更新自定义状态 async updateStatus(userId: string, socketId: string, status: UserStatus) { const key = `${this.PREFIX}${userId}`; const deviceData = await this.redis.hget(key, socketId); if (deviceData) { const device = JSON.parse(deviceData); device.status = status; device.lastSeen = Date.now(); await this.redis.hset(key, socketId, JSON.stringify(device)); } } // 获取在线好友 async getOnlineFriends(userId: string): Promise { // 获取好友列表 const friendIds = await this.getFriendIds(userId); // 批量检查在线状态 const statusMap = await this.getBatchUserStatus(friendIds); return friendIds.filter(id => statusMap.get(id) === UserStatus.ONLINE); } }

🔀 心跳检测

// 在Gateway中实现心跳 @WebSocketGateway() export class ChatGateway { private heartbeatInterval = 30000; // 30秒 async handleConnection(client: Socket) { // ...认证逻辑... // 启动心跳检测 this.startHeartbeat(client); } private startHeartbeat(client: Socket) { const heartbeatTimer = setInterval(() => { if (client.disconnected) { clearInterval(heartbeatTimer); return; } // 发送心跳请求 client.emit('ping', { timestamp: Date.now() }); }, this.heartbeatInterval); // 监听心跳响应 client.on('pong', async () => { await this.presenceService.heartbeat( client.data.user.id, client.id ); }); // 客户端断开时清理 client.on('disconnect', () => { clearInterval(heartbeatTimer); }); } } // 客户端心跳响应 class ChatClient { constructor() { this.socket.on('ping', () => { this.socket.emit('pong', { timestamp: Date.now() }); }); } }
