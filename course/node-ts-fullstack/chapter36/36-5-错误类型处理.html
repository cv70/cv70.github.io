📝 自定义错误类

// src/utils/errors.ts // 基础应用错误 export class AppError extends Error { public readonly statusCode: number; public readonly status: string; public readonly isOperational: boolean; constructor(message: string, statusCode: number = 500) { super(message); this.statusCode = statusCode; this.status = `${statusCode}`.startsWith('4') ? 'fail' : 'error'; this.isOperational = true; Error.captureStackTrace(this, this.constructor); } } // 404 错误 export class NotFoundError extends AppError { constructor(resource: string = 'Resource') { super(`${resource} not found`, 404); } } // 验证错误 export class ValidationError extends AppError { public readonly errors: Record; constructor(message: string, errors: Record = {}) { super(message, 400); this.errors = errors; } } // 认证错误 export class AuthenticationError extends AppError { constructor(message: string = 'Authentication failed') { super(message, 401); } } // 权限错误 export class AuthorizationError extends AppError { constructor(message: string = 'Access denied') { super(message, 403); } } // 冲突错误 export class ConflictError extends AppError { constructor(message: string = 'Resource conflict') { super(message, 409); } } // 类型守卫 export const isAppError = (error: Error): error is AppError => { return error instanceof AppError; };

🛡️ 错误处理中间件

// src/middleware/errorHandler.ts import { Request, Response, NextFunction } from 'express'; import { AppError, isAppError, ValidationError } from '../utils/errors'; interface ErrorResponse { success: false; status: string; message: string; errors?: Record; stack?: string; } export const errorHandler = ( err: Error, req: Request, res: Response, next: NextFunction ) => { // 开发环境详细错误 const sendErrorDev = (err: Error, res: Response) => { const response: ErrorResponse = { success: false, status: 'error', message: err.message, stack: err.stack }; if (err instanceof ValidationError) { response.errors = err.errors; } res.status(isAppError(err) ? err.statusCode : 500).json(response); }; // 生产环境简化错误 const sendErrorProd = (err: Error, res: Response) => { if (isAppError(err)) { const response: ErrorResponse = { success: false, status: err.status, message: err.message }; if (err instanceof ValidationError) { response.errors = err.errors; } return res.status(err.statusCode).json(response); } // 未知错误，不暴露细节 res.status(500).json({ success: false, status: 'error', message: 'Something went wrong' }); }; if (process.env.NODE_ENV === 'development') { sendErrorDev(err, res); } else { sendErrorProd(err, res); } };

🔄 异步错误处理

// src/utils/asyncHandler.ts import { Request, Response, NextFunction, RequestHandler } from 'express'; // 类型安全的异步处理器 export const asyncHandler = ( fn: (req: Request, res: Response, next: NextFunction) => Promise ): RequestHandler => { return (req: Request, res: Response, next: NextFunction) => { Promise.resolve(fn(req, res, next)).catch(next); }; }; // 使用 import { asyncHandler } from '../utils/asyncHandler'; import { NotFoundError } from '../utils/errors'; router.get('/:id', asyncHandler(async (req, res) => { const user = await User.findById(req.params.id); if (!user) { throw new NotFoundError('User'); } res.json(user); }));

📊 完整错误处理示例

// src/controllers/userController.ts import { Request, Response, NextFunction } from 'express'; import { NotFoundError, ValidationError, ConflictError } from '../utils/errors'; export const getUser = async (req: Request, res: Response, next: NextFunction) => { try { const user = await User.findById(req.params.id); if (!user) { throw new NotFoundError('User'); } res.json(user); } catch (error) { next(error); } }; export const createUser = async (req: Request, res: Response, next: NextFunction) => { const { name, email, password } = req.body; // 验证 const errors: Record = {}; if (!name) errors.name = 'Name is required'; if (!email) errors.email = 'Email is required'; if (!password) errors.password = 'Password is required'; if (Object.keys(errors).length > 0) { throw new ValidationError('Validation failed', errors); } // 检查邮箱是否已存在 const existingUser = await User.findByEmail(email); if (existingUser) { throw new ConflictError('Email already exists'); } const user = await User.create({ name, email, password }); res.status(201).json(user); }; // app.ts - 注册错误处理中间件 import { errorHandler } from './middleware/errorHandler'; // ... 路由 ... // 404 处理 app.use((req, res, next) => { next(new NotFoundError('Route')); }); // 错误处理（必须在最后） app.use(errorHandler);
