Etcd特点

Kubernetes默认的存储后端、强一致性、简单API、支持租约和事务。

🔧 基本操作

import { Etcd3 } from 'etcd3'; // 创建客户端 const client = new Etcd3({ hosts: ['http://localhost:2379'] }); // ==================== KV操作 ==================== // 设置键值 await client.put('name', 'Alice'); // 设置带TTL的键值 const lease = client.lease(60); // 60秒过期 await lease.put('session', 'active'); // 获取键值 const value = await client.get('name').string(); console.log(value); // 'Alice' // 获取所有匹配前缀的键 const all = await client.getAll().prefix('config/').strings(); // { 'config/db.host': 'localhost', 'config/db.port': '3306' } // 删除键 await client.delete().key('name'); // 删除所有匹配前缀的键 await client.delete().prefix('cache/'); // ==================== 服务注册与发现 ==================== // 服务注册（使用租约） async function registerService(serviceName: string, instance: ServiceInstance) { const lease = client.lease(10); // 10秒TTL // 注册服务实例 const key = `/services/${serviceName}/${instance.id}`; await lease.put(key).value(JSON.stringify({ id: instance.id, host: instance.host, port: instance.port, metadata: instance.metadata })); // 定期续租 lease.on('lost', () => { console.log('Lease lost, re-registering...'); registerService(serviceName, instance); }); return lease; } // 服务发现 async function discoverService(serviceName: string) { const instances = await client.getAll() .prefix(`/services/${serviceName}/`) .strings(); return Object.entries(instances).map(([key, value]) => { const data = JSON.parse(value); return { id: data.id, host: data.host, port: data.port }; }); }

🔀 监听与事务

// ==================== Watch监听 ==================== // 监听单个键 const watcher = await client.watch().key('config').create(); watcher.on('put', (event) => { console.log(`Key ${event.key} set to ${event.value}`); }); watcher.on('delete', (event) => { console.log(`Key ${event.key} deleted`); }); // 监听前缀 const prefixWatcher = await client.watch() .prefix('/services/user-service/') .create(); prefixWatcher.on('put', (event) => { console.log('Service instance added:', event.key); // 更新本地服务列表 updateServiceList(event.key, JSON.parse(event.value)); }); prefixWatcher.on('delete', (event) => { console.log('Service instance removed:', event.key); // 从本地服务列表移除 removeFromServiceList(event.key); }); // ==================== 事务操作 ==================== // 原子性比较和设置 async function atomicUpdate(key: string, expectedValue: string, newValue: string) { const result = await client.if( // 条件：当前值等于期望值 client.key(key).value(expectedValue), 'value', '==', expectedValue ).then( // 成功则更新 client.put(key).value(newValue) ).commit(); return result.succeeded; } // 分布式锁实现 class EtcdLock { private client: Etcd3; private lockKey: string; private lease: any; constructor(client: Etcd3, lockName: string) { this.client = client; this.lockKey = `/locks/${lockName}`; } async acquire(timeout: number = 5000): Promise { const start = Date.now(); while (Date.now() - start < timeout) { try { // 创建租约 this.lease = this.client.lease(10); // 尝试获取锁 const succeeded = await this.client .if(this.client.key(this.lockKey).value(''), 'value', '==', '') .then(this.lease.put(this.lockKey).value('locked')) .commit(); if (succeeded.succeeded) { return true; } // 释放租约 await this.lease.revoke(); } catch (error) { // 继续重试 } await new Promise(resolve => setTimeout(resolve, 100)); } return false; } async release() { if (this.lease) { await this.lease.revoke(); } } }
