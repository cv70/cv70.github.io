图像处理流水线概述

图像处理流水线将图像数据从原始采集到可用特征的完整处理过程。流水线需要支持批量处理、并行计算、质量保证，为多模态AI提供高质量的图像输入。

🔧 Python图像处理

OpenCV与PIL

import cv2
import numpy as np
from PIL import Image, ImageEnhance

class ImagePreprocessor:
    def __init__(self, target_size=(224, 224)):
        self.target_size = target_size

    def load_image(self, image_path: str) -> np.ndarray:
        """加载图像"""
        image = cv2.imread(image_path)
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        return image

    def resize(self, image: np.ndarray) -> np.ndarray:
        """调整大小"""
        return cv2.resize(image, self.target_size)

    def normalize(self, image: np.ndarray) -> np.ndarray:
        """归一化"""
        return image.astype(np.float32) / 255.0

    def enhance(self, image: np.ndarray) -> np.ndarray:
        """图像增强"""
        # 转换为PIL Image
        pil_image = Image.fromarray((image * 255).astype(np.uint8))

        # 增强对比度
        enhancer = ImageEnhance.Contrast(pil_image)
        pil_image = enhancer.enhance(1.2)

        # 增强亮度
        enhancer = ImageEnhance.Brightness(pil_image)
        pil_image = enhancer.enhance(1.1)

        return np.array(pil_image) / 255.0

    def augment(self, image: np.ndarray) -> np.ndarray:
        """数据增强"""
        # 随机旋转
        angle = np.random.uniform(-15, 15)
        center = tuple(np.array(image.shape[:2][::-1]) / 2)
        rotation_matrix = cv2.getRotationMatrix2D(center, angle, 1.0)
        image = cv2.warpAffine(image, rotation_matrix,
                             image.shape[:2][::-1],
                             flags=cv2.INTER_LINEAR)

        # 随机翻转
        if np.random.random() > 0.5:
            image = cv2.flip(image, 1)

        return image

    def process(self, image_path: str) -> np.ndarray:
        """完整处理流程"""
        image = self.load_image(image_path)
        image = self.resize(image)
        image = self.enhance(image)
        image = self.augment(image)
        image = self.normalize(image)
        return image

# 使用示例
preprocessor = ImagePreprocessor()
processed_image = preprocessor.process("input.jpg")

🧠 特征提取

视觉特征提取

特征提取方法：

1. 传统特征
   ├── SIFT：尺度不变特征
   ├── SURF：加速SIFT
   ├── ORB：快速二进制特征
   └── HOG：方向梯度直方图

2. 深度学习特征
   ├── CNN特征：预训练模型
   ├── ResNet：残差网络特征
   ├── EfficientNet：高效网络特征
   └── ViT：Vision Transformer特征

3. 多模态特征
   ├── CLIP：文本-图像对齐
   ├── ALIGN：大规模对齐
   ├── FLAVA：多模态模型
   └── BLIP：图像-语言预训练

4. 注意力特征
   ├── Self-Attention：自注意力
   ├── Cross-Attention：交叉注意力
   ├── Spatial Attention：空间注意力
   └── Channel Attention：通道注意力

CLIP特征提取示例：
import torch
from transformers import CLIPProcessor, CLIPModel

model = CLIPModel.from_pretrained("openai/clip-vit-base-patch32")
processor = CLIPProcessor.from_pretrained("openai/clip-vit-base-patch32")

def extract_clip_features(image):
    """提取CLIP特征"""
    inputs = processor(images=image, return_tensors="pt")
    with torch.no_grad():
        image_features = model.get_image_features(**inputs)
        # 归一化
        image_features = image_features / image_features.norm(dim=-1,
                                                            keepdim=True)
    return image_features.numpy()[0]

⚡ 批量处理

高效批量处理

import concurrent.futures
import multiprocessing
from typing import List
import time

class BatchImageProcessor:
    def __init__(self, preprocessor, num_workers=4):
        self.preprocessor = preprocessor
        self.num_workers = num_workers

    def process_single(self, image_path: str):
        """处理单张图像"""
        try:
            result = self.preprocessor.process(image_path)
            return {"path": image_path, "result": result, "status": "success"}
        except Exception as e:
            return {"path": image_path, "error": str(e), "status": "error"}

    def process_batch(self, image_paths: List[str]):
        """批量处理图像"""
        # 使用线程池处理I/O密集型任务
        with concurrent.futures.ThreadPoolExecutor(
            max_workers=self.num_workers
        ) as executor:
            futures = [
                executor.submit(self.process_single, path)
                for path in image_paths
            ]
            results = []
            for future in concurrent.futures.as_completed(futures):
                results.append(future.result())

        return results

    def process_batch_multiprocess(self, image_paths: List[str]):
        """多进程批量处理"""
        with multiprocessing.Pool(
            processes=self.num_workers
        ) as pool:
            results = pool.map(self.process_single, image_paths)

        return results

# 使用示例
processor = BatchImageProcessor(preprocessor, num_workers=8)
results = processor.process_batch([
    "image1.jpg", "image2.jpg", "image3.jpg"
])

# 统计处理结果
success_count = sum(1 for r in results if r['status'] == 'success')
error_count = len(results) - success_count
print(f"成功: {success_count}, 失败: {error_count}")

🔄 流水线优化

优化策略

  • 并行处理：多线程/多进程并行
  • GPU加速：使用GPU加速处理
  • 批处理：批量处理提高效率
  • 缓存机制：缓存中间结果
  • 异步处理：异步I/O减少阻塞
  • 流水线并行：阶段并行处理
⚠️
处理注意事项

图像处理要考虑内存管理（避免OOM）、错误处理（失败重试）、质量检查（输出验证）。建立完善的监控和告警机制。
