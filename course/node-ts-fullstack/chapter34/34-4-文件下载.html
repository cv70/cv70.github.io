34.4 文件下载

掌握 Express 文件下载和发送的方法

🎯 学习目标

  • 掌握 res.download() 下载文件
  • 学会 res.sendFile() 发送文件
  • 理解文件下载的错误处理
  • 掌握动态生成文件下载

📥 res.download() 下载文件

const path = require('path'); // 基本下载 app.get('/download/:filename', (req, res) => { const filepath = path.join(__dirname, 'files', req.params.filename); res.download(filepath); }); // 指定下载文件名 app.get('/download/report', (req, res) => { const filepath = path.join(__dirname, 'reports', 'report-2024.pdf'); res.download(filepath, '年度报告.pdf'); }); // 带错误处理 app.get('/download/:filename', (req, res) => { const filepath = path.join(__dirname, 'files', req.params.filename); res.download(filepath, (err) => { if (err) { // 文件不存在或其他错误 if (!res.headersSent) { res.status(404).json({ error: 'File not found' }); } } }); }); // 带选项 app.get('/download/options', (req, res) => { res.download('/report.pdf', 'report.pdf', { root: path.join(__dirname, 'files'), dotfiles: 'ignore', headers: { 'X-Sent': 'true', 'X-Custom-Header': 'value' } }); });

📄 res.sendFile() 发送文件

const path = require('path'); // 基本发送 app.get('/file/:name', (req, res) => { const filepath = path.join(__dirname, 'public', req.params.name); res.sendFile(filepath); }); // 使用 root 选项 app.get('/document/:name', (req, res) => { res.sendFile(req.params.name, { root: path.join(__dirname, 'documents'), dotfiles: 'deny', headers: { 'X-Timestamp': Date.now() } }, (err) => { if (err) { res.status(404).json({ error: 'File not found' }); } }); }); // 发送 HTML 页面 app.get('/', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'index.html')); }); // 发送图片（浏览器内显示） app.get('/images/:name', (req, res) => { res.sendFile(req.params.name, { root: path.join(__dirname, 'images') }); });

res.download vs res.sendFile

方法 行为 适用场景
res.download() 触发浏览器下载 下载文件
res.sendFile() 在浏览器中显示/播放 图片、视频、HTML

🔄 动态生成文件下载

// 生成 CSV 下载 app.get('/export/csv', (req, res) => { const data = [ { name: 'Alice', age: 25 }, { name: 'Bob', age: 30 } ]; // 生成 CSV 内容 let csv = 'Name,Age\n'; data.forEach(row => { csv += `${row.name},${row.age}\n`; }); // 设置响应头 res.setHeader('Content-Type', 'text/csv'); res.setHeader('Content-Disposition', 'attachment; filename=users.csv'); res.send(csv); }); // 生成 JSON 下载 app.get('/export/json', (req, res) => { const data = { users: [...] }; res.setHeader('Content-Type', 'application/json'); res.setHeader('Content-Disposition', 'attachment; filename=data.json'); res.send(JSON.stringify(data, null, 2)); }); // 使用 ExcelJS 生成 Excel const ExcelJS = require('exceljs'); app.get('/export/excel', async (req, res) => { const workbook = new ExcelJS.Workbook(); const worksheet = workbook.addWorksheet('Users'); worksheet.columns = [ { header: 'Name', key: 'name' }, { header: 'Age', key: 'age' } ]; worksheet.addRows([ { name: 'Alice', age: 25 }, { name: 'Bob', age: 30 } ]); res.setHeader('Content-Type', 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'); res.setHeader('Content-Disposition', 'attachment; filename=users.xlsx'); await workbook.xlsx.write(res); res.end(); });

⚡ 大文件流式下载

const fs = require('fs'); // 流式下载大文件 app.get('/download/large/:filename', (req, res) => { const filepath = path.join(__dirname, 'large-files', req.params.filename); const stat = fs.statSync(filepath); res.setHeader('Content-Length', stat.size); res.setHeader('Content-Type', 'application/octet-stream'); res.setHeader('Content-Disposition', `attachment; filename="${req.params.filename}"`); const fileStream = fs.createReadStream(filepath); fileStream.pipe(res); fileStream.on('error', (err) => { res.status(500).json({ error: 'Download failed' }); }); }); // 支持断点续传 app.get('/download/resumable/:filename', (req, res) => { const filepath = path.join(__dirname, 'files', req.params.filename); const stat = fs.statSync(filepath); const range = req.headers.range; if (range) { const parts = range.replace(/bytes=/, '').split('-'); const start = parseInt(parts[0]); const end = parts[1] ? parseInt(parts[1]) : stat.size - 1; const chunkSize = end - start + 1; res.status(206); res.setHeader('Content-Range', `bytes ${start}-${end}/${stat.size}`); res.setHeader('Content-Length', chunkSize); res.setHeader('Content-Type', 'application/octet-stream'); fs.createReadStream(filepath, { start, end }).pipe(res); } else { res.setHeader('Content-Length', stat.size); res.setHeader('Content-Type', 'application/octet-stream'); fs.createReadStream(filepath).pipe(res); } });

✅ 小结

📌
本节要点
  • res.download() 触发浏览器下载
  • res.sendFile() 在浏览器中显示文件
  • 动态生成文件通过设置 Content-Disposition 头
  • 大文件使用流式传输，支持断点续传
