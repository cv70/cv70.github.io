10.4 Record 和 ReturnType

创建对象类型和获取返回类型

🎯 学习目标

  • 深入理解 Record 的实现原理
  • 掌握 ReturnType 的使用场景
  • 学会使用 Parameters 类型
  • 了解实际项目中的应用

📚 Record 深入

// Record 创建键值对类型 // 内置实现： type MyRecord = { [P in K]: V; }; // 简单示例 type UserRoles = Record; // { [x: string]: string } // 使用字符串字面量联合 type Role = 'admin' | 'user' | 'guest'; type RolePermissions = Record; const permissions: RolePermissions = { admin: ['read', 'write', 'delete'], user: ['read', 'write'], guest: ['read'] }; // 使用数字键 type Status = 200 | 404 | 500; type StatusMessages = Record; const messages: StatusMessages = { 200: 'OK', 404: 'Not Found', 500: 'Internal Server Error' }; // 嵌套对象 interface User { id: number; name: string; } type UserDirectory = Record; const users: UserDirectory = { 'alice': { id: 1, name: 'Alice' }, 'bob': { id: 2, name: 'Bob' } };
💡
Record 特点

Record 简化了创建对象类型的语法，特别是当键是已知的字符串字面量联合时非常有用。

🔧 ReturnType 深入

// ReturnType 获取函数返回值类型 // 内置实现： type MyReturnType any> = T extends (...args: any) => infer R ? R : any; // 从函数推断类型 function fetchUser() { return { id: 1, name: 'Alice', email: 'alice@example.com' }; } type User = ReturnType; // { id: number; name: string; email: string; } // 用于 API 响应 async function getProducts() { const response = await fetch('/api/products'); return response.json() as Promise<{ products: Array<{ id: number; name: string; price: number }>; total: number; }>; } type ProductsResponse = Awaited>; // { products: Array<{ id: number; name: string; price: number }>; total: number; } // 构造函数实例类型 class Person { constructor(public name: string, public age: number) {} greet() { return `Hello, ${this.name}`; } } type PersonInstance = InstanceType; // Person

⚙️ Parameters 类型

// Parameters 获取函数参数类型元组 // 内置实现： type MyParameters any> = T extends (...args: infer P) => any ? P : never; // 获取参数类型 function createUser(name: string, age: number, active: boolean = true) { return { name, age, active }; } type CreateUserParams = Parameters; // [name: string, age: number, active?: boolean] // 获取特定参数类型 type NameParam = CreateUserParams[0]; // string type AgeParam = CreateUserParams[1]; // number // 包装函数保持类型 function logCalls any>(fn: T): T { return ((...args: Parameters): ReturnType => { console.log('Called with:', args); const result = fn(...args); console.log('Returned:', result); return result; }) as T; } const loggedCreateUser = logCalls(createUser); // 构造函数参数 type ConstructorParams = ConstructorParameters; // [name: string, age: number]
