18.3 HTTP 客户端

发起 HTTP 请求

🎯 学习目标

  • 掌握 http.request 方法
  • 学会发送 GET 和 POST 请求
  • 理解请求选项配置
  • 掌握异步请求处理

📚 GET 请求

const http = require('http'); // 方式一：http.get 简化方法 http.get('http://localhost:3000/api/users', (res) => { let data = ''; res.on('data', (chunk) => { data += chunk; }); res.on('end', () => { console.log('响应:', data); const json = JSON.parse(data); console.log('用户列表:', json); }); }).on('error', (err) => { console.error('请求错误:', err.message); }); // 方式二：http.request 完整方法 const options = { hostname: 'localhost', port: 3000, path: '/api/users', method: 'GET', headers: { 'Accept': 'application/json' } }; const req = http.request(options, (res) => { console.log('状态码:', res.statusCode); console.log('响应头:', res.headers); let data = ''; res.on('data', (chunk) => data += chunk); res.on('end', () => console.log(data)); }); req.on('error', (err) => { console.error('请求错误:', err); }); req.end(); // 必须调用

🔧 POST 请求

const http = require('http'); // 发送 JSON 数据 const postData = JSON.stringify({ name: 'Alice', age: 25 }); const options = { hostname: 'localhost', port: 3000, path: '/api/users', method: 'POST', headers: { 'Content-Type': 'application/json', 'Content-Length': Buffer.byteLength(postData) } }; const req = http.request(options, (res) => { let data = ''; res.on('data', (chunk) => data += chunk); res.on('end', () => { console.log('响应:', data); }); }); req.on('error', (err) => { console.error('请求错误:', err); }); // 写入请求体 req.write(postData); req.end(); // 发送表单数据 const querystring = require('querystring'); const formData = querystring.stringify({ username: 'alice', password: 'secret' }); const formOptions = { hostname: 'localhost', port: 3000, path: '/login', method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded', 'Content-Length': Buffer.byteLength(formData) } }; const formReq = http.request(formOptions, (res) => { // 处理响应 }); formReq.write(formData); formReq.end();

📦 请求选项详解

const options = { // 基本选项 protocol: 'http:', // 协议 hostname: 'example.com', // 主机名 port: 80, // 端口 path: '/api/data?foo=bar', // 路径（含查询参数） method: 'GET', // 方法 // 请求头 headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer token123', 'User-Agent': 'MyApp/1.0', 'Accept': 'application/json' }, // 认证 auth: 'username:password', // Basic 认证 // 超时 timeout: 5000, // 5 秒超时 // 代理 // agent: false, // 禁用连接池 // 其他 family: 4, // IPv4 localAddress: '127.0.0.1', // 本地地址 }; // 使用 URL 对象 const url = new URL('http://example.com:3000/api/data'); const req = http.request({ hostname: url.hostname, port: url.port, path: url.pathname + url.search, method: 'GET' }, (res) => { // ... });

⚙️ 封装请求函数

const http = require('http'); const https = require('https'); // Promise 封装 function request(url, options = {}) { return new Promise((resolve, reject) => { const parsedUrl = new URL(url); const isHttps = parsedUrl.protocol === 'https:'; const client = isHttps ? https : http; const reqOptions = { hostname: parsedUrl.hostname, port: parsedUrl.port || (isHttps ? 443 : 80), path: parsedUrl.pathname + parsedUrl.search, method: options.method || 'GET', headers: options.headers || {}, timeout: options.timeout || 10000 }; const req = client.request(reqOptions, (res) => { let data = ''; res.on('data', (chunk) => data += chunk); res.on('end', () => { resolve({ statusCode: res.statusCode, headers: res.headers, body: data }); }); }); req.on('error', reject); req.on('timeout', () => { req.destroy(); reject(new Error('请求超时')); }); if (options.body) { req.write(options.body); } req.end(); }); } // 使用 async function main() { try { // GET 请求 const getRes = await request('http://localhost:3000/api/users'); console.log('用户:', JSON.parse(getRes.body)); // POST 请求 const postRes = await request('http://localhost:3000/api/users', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ name: 'Bob' }) }); console.log('创建用户:', postRes); } catch (err) { console.error('请求失败:', err); } }
