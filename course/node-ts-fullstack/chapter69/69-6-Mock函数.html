69.6 Mock 函数

第十部分：测试篇 - 掌握 Mock 函数的使用

1. Mock 函数简介

Mock 函数（也称为"间谍"）是测试中常用的技术，用于：

  • 捕获函数调用（包括参数和调用次数）
  • 控制函数的返回值
  • 替换真实实现
  • 验证函数是否被正确调用
💡
为什么需要 Mock？

隔离测试单元，避免依赖外部系统（数据库、API、文件系统等），使测试更快、更稳定、更独立。

2. 创建 Mock 函数

// 使用 jest.fn() 创建 Mock 函数 const mockFn = jest.fn(); // 基本使用 test('mock function', () => { mockFn('hello', 'world'); // 检查是否被调用 expect(mockFn).toHaveBeenCalled(); expect(mockFn).toHaveBeenCalledTimes(1); expect(mockFn).toHaveBeenCalledWith('hello', 'world'); });

设置返回值

describe('Mock Return Values', () => { test('mockReturnValue', () => { const mock = jest.fn(); mock.mockReturnValue(42); expect(mock()).toBe(42); expect(mock()).toBe(42); // 每次都返回相同值 }); test('mockReturnValueOnce', () => { const mock = jest.fn(); mock .mockReturnValueOnce(1) .mockReturnValueOnce(2) .mockReturnValue(3); // 默认返回值 expect(mock()).toBe(1); expect(mock()).toBe(2); expect(mock()).toBe(3); expect(mock()).toBe(3); }); test('mockImplementation', () => { const mock = jest.fn(); mock.mockImplementation((a: number, b: number) => a + b); expect(mock(1, 2)).toBe(3); expect(mock(5, 3)).toBe(8); }); test('mockImplementationOnce', () => { const mock = jest.fn(); mock .mockImplementationOnce(() => 'first') .mockImplementationOnce(() => 'second') .mockImplementation(() => 'default'); expect(mock()).toBe('first'); expect(mock()).toBe('second'); expect(mock()).toBe('default'); }); });

3. Mock 属性和方法

describe('Mock Properties', () => { test('mock.calls - 调用参数', () => { const mock = jest.fn(); mock('first', 'call'); mock('second', 'call'); // mock.calls 存储所有调用的参数 expect(mock.mock.calls).toHaveLength(2); expect(mock.mock.calls[0]).toEqual(['first', 'call']); expect(mock.mock.calls[1]).toEqual(['second', 'call']); }); test('mock.results - 返回值', () => { const mock = jest.fn(); mock.mockReturnValue('result'); mock(); mock(); // mock.results 存储所有返回值 expect(mock.mock.results).toHaveLength(2); expect(mock.mock.results[0].value).toBe('result'); }); test('mock.instances - this 值', () => { const mock = jest.fn(); const obj = { method: mock }; obj.method(); // mock.instances 存储 this 值 expect(mock.mock.instances[0]).toBe(obj); }); test('mock.lastCall - 最后一次调用', () => { const mock = jest.fn(); mock('first'); mock('last'); expect(mock.mock.lastCall).toEqual(['last']); }); });

4. Mock 断言方法

describe('Mock Assertions', () => { test('toHaveBeenCalled', () => { const mock = jest.fn(); expect(mock).not.toHaveBeenCalled(); mock(); expect(mock).toHaveBeenCalled(); }); test('toHaveBeenCalledTimes', () => { const mock = jest.fn(); mock(); mock(); mock(); expect(mock).toHaveBeenCalledTimes(3); }); test('toHaveBeenCalledWith', () => { const mock = jest.fn(); mock('hello', 'world'); expect(mock).toHaveBeenCalledWith('hello', 'world'); expect(mock).not.toHaveBeenCalledWith('other'); }); test('toHaveBeenLastCalledWith', () => { const mock = jest.fn(); mock('first'); mock('last'); expect(mock).toHaveBeenLastCalledWith('last'); }); test('toHaveBeenNthCalledWith', () => { const mock = jest.fn(); mock('first'); mock('second'); mock('third'); expect(mock).toHaveBeenNthCalledWith(1, 'first'); expect(mock).toHaveBeenNthCalledWith(2, 'second'); expect(mock).toHaveBeenNthCalledWith(3, 'third'); }); test('toHaveReturned', () => { const mock = jest.fn(() => 'result'); mock(); expect(mock).toHaveReturned(); expect(mock).toHaveReturnedTimes(1); expect(mock).toHaveReturnedWith('result'); }); test('toHaveLastReturnedWith', () => { const mock = jest.fn(); mock.mockReturnValueOnce('first'); mock.mockReturnValueOnce('last'); mock(); mock(); expect(mock).toHaveLastReturnedWith('last'); }); });

5. Mock 实现

describe('Mock Implementations', () => { test('mockImplementation', () => { const mock = jest.fn((x: number) => x * 2); expect(mock(5)).toBe(10); expect(mock(3)).toBe(6); }); test('mockReturnThis', () => { const mock = jest.fn().mockReturnThis(); const context = { method: mock }; expect(context.method()).toBe(context); }); test('mockName', () => { const mock = jest.fn().mockName('myMock'); // 在错误消息中会显示名称 // expect(mock).toHaveBeenCalled(); // Error: expect(myMock).toHaveBeenCalled() }); test('mockClear, mockReset, mockRestore', () => { const mock = jest.fn().mockReturnValue('value'); mock('arg'); expect(mock).toHaveBeenCalled(); // mockClear: 清除调用记录 mock.mockClear(); expect(mock).not.toHaveBeenCalled(); expect(mock()).toBe('value'); // 返回值还在 // mockReset: 清除所有设置 mock.mockReset(); expect(mock()).toBeUndefined(); // 返回默认 undefined // mockRestore: 恢复原始实现（用于 spyOn） }); });

6. jest.spyOn

// spyOn 用于监视对象的方法 const video = { play: () => 'playing', pause: () => 'paused' }; describe('spyOn', () => { test('spyOn method', () => { const spy = jest.spyOn(video, 'play'); // 调用原方法 const result = video.play(); expect(result).toBe('playing'); expect(spy).toHaveBeenCalled(); }); test('spyOn with mock implementation', () => { const spy = jest.spyOn(video, 'play'); spy.mockReturnValue('mocked'); expect(video.play()).toBe('mocked'); // 恢复原始实现 spy.mockRestore(); expect(video.play()).toBe('playing'); }); test('spyOn with mockImplementation', () => { const spy = jest.spyOn(video, 'pause'); spy.mockImplementation(() => 'mocked pause'); expect(video.pause()).toBe('mocked pause'); spy.mockRestore(); }); });
💡
spyOn vs jest.fn
  • jest.fn(): 创建全新的 Mock 函数
  • jest.spyOn(): 监视现有对象的方法，可以调用原始实现

7. TypeScript 中的 Mock

// 定义 Mock 类型 interface UserRepository { findById(id: string): Promise<User | null>; create(data: CreateUserData): Promise<User>; update(id: string, data: UpdateData): Promise<User>; } // 创建类型安全的 Mock function createMockRepository(): jest.Mocked<UserRepository> { return { findById: jest.fn(), create: jest.fn(), update: jest.fn() }; } describe('UserService', () => { let service: UserService; let mockRepository: jest.Mocked<UserRepository>; beforeEach(() => { mockRepository = createMockRepository(); service = new UserService(mockRepository); }); test('should find user', async () => { const mockUser = { id: '1', name: 'John' }; mockRepository.findById.mockResolvedValue(mockUser); const result = await service.findUser('1'); expect(result).toEqual(mockUser); expect(mockRepository.findById).toHaveBeenCalledWith('1'); }); test('should create user', async () => { mockRepository.create.mockResolvedValue({ id: '1', name: 'John', email: 'john@example.com' }); const result = await service.createUser({ name: 'John', email: 'john@example.com' }); expect(result.id).toBe('1'); }); });

8. Mock 时间

describe('Date Mocking', () => { beforeEach(() => { jest.useFakeTimers(); jest.setSystemTime(new Date('2024-01-15T10:00:00Z')); }); afterEach(() => { jest.useRealTimers(); }); test('current date is mocked', () => { expect(new Date()).toEqual(new Date('2024-01-15T10:00:00Z')); }); test('Date.now returns mocked time', () => { expect(Date.now()).toBe(1705312800000); }); });

9. Mock 方法速查

方法 说明
jest.fn() 创建 Mock 函数
jest.fn(impl) 创建带实现的 Mock
jest.spyOn(obj, method) 监视对象方法
mockReturnValue(val) 设置返回值
mockReturnValueOnce(val) 设置一次性返回值
mockImplementation(fn) 设置实现函数
mockResolvedValue(val) 设置 Promise resolve 值
mockRejectedValue(val) 设置 Promise reject 值
mockClear() 清除调用记录
mockReset() 重置所有设置
mockRestore() 恢复原始实现

10. 本节小结

📚
关键要点
  • jest.fn() 创建全新的 Mock 函数
  • jest.spyOn() 监视现有方法
  • mockReturnValue/mockImplementation 设置行为
  • 使用 toHaveBeenCalled* 断言调用情况
  • TypeScript 中使用 jest.Mocked<T> 获得类型安全
