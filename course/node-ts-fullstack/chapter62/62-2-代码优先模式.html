62.2 代码优先模式

使用 TypeScript 装饰器定义 GraphQL 类型

📖 代码优先概述

代码优先模式使用 TypeScript 装饰器定义 GraphQL Schema，自动生成 GraphQL 类型定义。这种方式保持 TypeScript 类型和 GraphQL Schema 同步。

💡
代码优先优势
  • TypeScript 类型安全
  • 减少重复代码
  • 自动生成 Schema
  • 开发效率高

📦 定义对象类型

import { ObjectType, Field, ID, Int } from '@nestjs/graphql'; import { Post } from './post.entity'; @ObjectType() export class User { @Field(() => ID) id: number; @Field({ description: '用户名' }) name: string; @Field() email: string; @Field({ nullable: true }) bio?: string; @Field(() => Int, { nullable: true }) age?: number; @Field(() => [Post], { description: '用户文章列表' }) posts: Post[]; @Field() createdAt: Date; } // 生成的 GraphQL Schema type User { id: ID! name: String! email: String! bio: String age: Int posts: [Post!]! createdAt: DateTime! }

📋 定义输入类型

import { InputType, Field, PartialType } from '@nestjs/graphql'; import { IsString, IsEmail, MinLength } from 'class-validator'; @InputType() export class CreateUserInput { @Field() @IsString() @MinLength(2) name: string; @Field() @IsEmail() email: string; @Field() @MinLength(6) password: string; } @InputType() export class UpdateUserInput extends PartialType(CreateUserInput) { // 所有字段变为可选 }

🔍 定义 Query 和 Mutation

import { Resolver, Query, Mutation, Args, ID } from '@nestjs/graphql'; import { UserService } from './user.service'; import { User } from './user.entity'; import { CreateUserInput, UpdateUserInput } from './user.input'; @Resolver(() => User) export class UserResolver { constructor(private readonly userService: UserService) {} @Query(() => [User], { name: 'users' }) async findAll(): Promise { return this.userService.findAll(); } @Query(() => User, { name: 'user', nullable: true }) async findOne( @Args('id', { type: () => ID }) id: string, ): Promisenull> { return this.userService.findOne(+id); } @Mutation(() => User) async createUser( @Args('input') input: CreateUserInput, ): Promise { return this.userService.create(input); } @Mutation(() => User, { nullable: true }) async updateUser( @Args('id', { type: () => ID }) id: string, @Args('input') input: UpdateUserInput, ): Promisenull> { return this.userService.update(+id, input); } @Mutation(() => Boolean) async deleteUser( @Args('id', { type: () => ID }) id: string, ): Promise<boolean> { await this.userService.remove(+id); return true; } }

📝 小结

  • 使用 @ObjectType 定义 GraphQL 对象类型
  • 使用 @InputType 定义输入类型
  • 使用 @Query 和 @Mutation 定义操作
  • 使用 @Args 定义参数
  • 可以与 class-validator 配合验证
  • 自动生成 Schema 文件
