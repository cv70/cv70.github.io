🔔 事件与同步

// 事件
event data_ready;

// 触发事件
-> data_ready;

// 等待事件
@(data_ready);

// 信号量
semaphore sem = new(1);

// 获取信号量
sem.get();

// 释放信号量
sem.put();

// 邮箱
mailbox #(int) mb = new();

// 放入邮箱
mb.put(10);

// 从邮箱取出
int val;
mb.get(val);
