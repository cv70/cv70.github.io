📚 useState 基础

import { useState } from 'react'; // 基本语法 // const [state, setState] = useState(initialValue); // 基本类型 function Counter() { const [count, setCount] = useState(0); return ( <div> <p>Count: {count}</p> <button onClick={() => setCount(count + 1)}>+1</button> <button onClick={() => setCount(prev => prev - 1)}>-1</button> </div> ); } // TypeScript 类型 function Counter() { // 类型推断 const [count, setCount] = useState(0); // number // 显式类型 const [name, setName] = useState<string>(''); // 联合类型 const [status, setStatus] = useState<'idle' | 'loading' | 'success' | 'error'>('idle'); return ( <div> <p>Count: {count}</p> <p>Name: {name}</p> <p>Status: {status}</p> </div> ); } // 惰性初始化 // 初始值只在首次渲染时计算 function ExpensiveComponent() { // 每次渲染都会执行 // const [state, setState] = useState(computeExpensiveValue()); // 只在首次渲染时执行 const [state, setState] = useState(() => { console.log('Initial computation'); return computeExpensiveValue(); }); return <div>{state}</div>; }

🔧 对象和数组状态

// 对象状态 function UserForm() { const [user, setUser] = useState({ name: '', email: '', age: 0, }); // 更新单个字段 const updateField = (field, value) => { setUser(prev => ({ ...prev, [field]: value, })); }; // 或使用单独的 useState const [name, setName] = useState(''); const [email, setEmail] = useState(''); const [age, setAge] = useState(0); return ( <form> <input value={user.name} onChange={e => updateField('name', e.target.value)} /> </form> ); } // 数组状态 function TodoList() { const [todos, setTodos] = useState<Todo[]>([]); // 添加元素 const addTodo = (text: string) => { setTodos(prev => [...prev, { id: Date.now(), text, done: false }]); }; // 删除元素 const removeTodo = (id: number) => { setTodos(prev => prev.filter(todo => todo.id !== id)); }; // 更新元素 const toggleTodo = (id: number) => { setTodos(prev => prev.map(todo => todo.id === id ? { ...todo, done: !todo.done } : todo )); }; // 清空数组 const clearTodos = () => setTodos([]); return ( <div> <button onClick={() => addTodo('New Task')}>Add</button> <ul> {todos.map(todo => ( <li key={todo.id}> <input type="checkbox" checked={todo.done} onChange={() => toggleTodo(todo.id)} /> {todo.text} <button onClick={() => removeTodo(todo.id)}>Delete</button> </li> ))} </ul> </div> ); } interface Todo { id: number; text: string; done: boolean; }

📊 函数式更新

💡
何时使用函数式更新

1. 新状态依赖于旧状态
2. 在异步操作中更新状态
3. 批量更新时确保正确性

// 直接更新
setCount(count + 1); // 可能导致问题

// 函数式更新
setCount(prev => prev + 1); // 始终正确
