📚 限流概述

// 限流（Rate Limiting）控制请求速率，保护后端服务 const reasons = { protectBackend: '保护后端服务不被压垮', preventAbuse: '防止恶意请求和爬虫', fairUsage: '保证资源公平分配', costControl: '控制API调用成本' }; // 常见限流算法 // 1. 固定窗口算法 class FixedWindowLimiter { private windows: Map = new Map(); private limit: number; private windowSize: number; // 毫秒 constructor(limit: number, windowSize: number) { this.limit = limit; this.windowSize = windowSize; } isAllowed(key: string): boolean { const now = Date.now(); let window = this.windows.get(key); if (!window || now - window.startTime >= this.windowSize) { window = { count: 0, startTime: now }; this.windows.set(key, window); } window.count++; return window.count <= this.limit; } } // 2. 滑动窗口算法 class SlidingWindowLimiter { private requests: Map = new Map(); private limit: number; private windowSize: number; isAllowed(key: string): boolean { const now = Date.now(); const requests = this.requests.get(key) || []; // 移除窗口外的请求 const validRequests = requests.filter(t => now - t < this.windowSize); if (validRequests.length >= this.limit) { return false; } validRequests.push(now); this.requests.set(key, validRequests); return true; } } // 3. 令牌桶算法 class TokenBucket { private tokens: number; private lastRefill: number; constructor( private capacity: number, private refillRate: number // 每秒补充数量 ) { this.tokens = capacity; this.lastRefill = Date.now(); } isAllowed(cost: number = 1): boolean { this.refill(); if (this.tokens >= cost) { this.tokens -= cost; return true; } return false; } private refill() { const now = Date.now(); const elapsed = (now - this.lastRefill) / 1000; const tokensToAdd = elapsed * this.refillRate; this.tokens = Math.min(this.capacity, this.tokens + tokensToAdd); this.lastRefill = now; } } // 4. 漏桶算法 class LeakyBucket { private queue: any[] = []; private processing = false; constructor( private capacity: number, private leakRate: number // 每秒处理数量 ) {} async add(item: any): Promise { if (this.queue.length >= this.capacity) { return false; } this.queue.push(item); this.process(); return true; } private async process() { if (this.processing) return; this.processing = true; while (this.queue.length > 0) { const item = this.queue.shift(); await this.handleItem(item); await new Promise(r => setTimeout(r, 1000 / this.leakRate)); } this.processing = false; } }
💡
算法选择

固定窗口简单但有边界问题；滑动窗口精确但内存消耗大；令牌桶允许突发流量；漏桶恒定速率输出。

🔧 熔断器模式

// 熔断器防止故障蔓延 enum CircuitState { CLOSED, // 正常状态 OPEN, // 熔断状态 HALF_OPEN // 半开状态 } class CircuitBreaker { private state = CircuitState.CLOSED; private failureCount = 0; private successCount = 0; private lastFailureTime = 0; constructor( private threshold: number = 5, // 失败阈值 private timeout: number = 60000, // 熔断超时时间 private halfOpenSuccesses: number = 3 // 半开状态成功阈值 ) {} async execute(fn: () => Promise): Promise { if (this.state === CircuitState.OPEN) { if (Date.now() - this.lastFailureTime >= this.timeout) { this.state = CircuitState.HALF_OPEN; this.successCount = 0; } else { throw new Error('Circuit breaker is OPEN'); } } try { const result = await fn(); if (this.state === CircuitState.HALF_OPEN) { this.successCount++; if (this.successCount >= this.halfOpenSuccesses) { this.reset(); } } else { this.failureCount = 0; } return result; } catch (error) { this.recordFailure(); throw error; } } private recordFailure() { this.failureCount++; this.lastFailureTime = Date.now(); if (this.state === CircuitState.HALF_OPEN) { this.state = CircuitState.OPEN; } else if (this.failureCount >= this.threshold) { this.state = CircuitState.OPEN; } } private reset() { this.state = CircuitState.CLOSED; this.failureCount = 0; this.successCount = 0; } getState(): CircuitState { return this.state; } } // 使用熔断器 const userCircuit = new CircuitBreaker(5, 60000); async function callUserService(userId: string) { return userCircuit.execute(async () => { const response = await fetch(`http://user-service/users/${userId}`); if (!response.ok) throw new Error('Service error'); return response.json(); }); }

🔀 降级策略

// 服务降级：在系统压力大或故障时，返回备选响应 class DegradableService { private circuit: CircuitBreaker; private fallbackData: any; constructor() { this.circuit = new CircuitBreaker(); } // 带降级的服务调用 async callWithFallback( fn: () => Promise, fallback: () => T | Promise ): Promise { try { return await this.circuit.execute(fn); } catch (error) { console.warn('Service degraded, using fallback'); return fallback(); } } } // 使用示例 const service = new DegradableService(); // 商品推荐服务 async function getRecommendations(userId: string) { return service.callWithFallback( // 正常调用 async () => { const response = await fetch(`/api/recommendations/${userId}`); return response.json(); }, // 降级策略：返回默认推荐 () => [ { id: 1, name: '热门商品1' }, { id: 2, name: '热门商品2' }, { id: 3, name: '热门商品3' } ] ); } // 评论服务降级 async function getComments(productId: string) { return service.callWithFallback( async () => { const response = await fetch(`/api/comments/${productId}`); return response.json(); }, // 降级：显示占位信息 () => ({ items: [], message: '评论服务暂时不可用，请稍后再试' }) ); } // Express中间件实现限流 import rateLimit from 'express-rate-limit'; const apiLimiter = rateLimit({ windowMs: 15 * 60 * 1000, // 15分钟 max: 100, // 最多100个请求 message: { error: 'Too many requests, please try again later.' }, standardHeaders: true, legacyHeaders: false }); app.use('/api/', apiLimiter); // 更细粒度的限流 const strictLimiter = rateLimit({ windowMs: 60 * 1000, // 1分钟 max: 5, // 最多5个请求 skipSuccessfulRequests: false, keyGenerator: (req) => { return req.user?.id || req.ip; // 按用户或IP限流 } }); app.post('/api/password/reset', strictLimiter, handlePasswordReset);
