📥 客户端安装

方式一：CDN 引入

<!-- 从服务端自动提供（推荐） --> <script src="/socket.io/socket.io.js"></script> <!-- 使用 CDN --> <script src="https://cdn.socket.io/4.7.2/socket.io.min.js"></script> <!-- 使用后 --> <script> const socket = io(); </script>

方式二：NPM 安装

# 安装 npm install socket.io-client # 使用 import { io } from 'socket.io-client'; const socket = io('http://localhost:3000');

方式三：TypeScript 项目

// 安装 npm install socket.io-client // 类型定义（通常已包含在包中） import { io, Socket } from 'socket.io-client'; interface ServerToClientEvents { message: (data: { content: string }) => void; } interface ClientToServerEvents { send: (content: string) => void; } const socket: Socket = io();

🔌 连接配置

import { io } from 'socket.io-client'; const socket = io('http://localhost:3000', { // ==================== 连接配置 ==================== autoConnect: true, // 自动连接，默认 true reconnection: true, // 自动重连，默认 true reconnectionAttempts: Infinity, // 重连次数，默认无限 reconnectionDelay: 1000, // 重连延迟，默认 1000ms reconnectionDelayMax: 5000, // 最大延迟，默认 5000ms timeout: 20000, // 连接超时，默认 20000ms // ==================== 传输配置 ==================== transports: ['polling', 'websocket'], // 传输方式 upgrade: true, // 允许升级传输，默认 true // ==================== 认证配置 ==================== auth: { token: 'user-jwt-token', userId: '123' }, // ==================== 查询参数 ==================== query: { roomId: 'room-123', type: 'chat' }, // ==================== 请求头 ==================== extraHeaders: { 'X-Custom-Header': 'value' }, // ==================== 路径配置 ==================== path: '/socket.io/', // 服务端路径，默认 '/socket.io/' forceNew: false, // 强制新连接，默认 false withCredentials: true // 跨域携带 Cookie });

⚛️ React 集成

自定义 Hook

import { useEffect, useState, useRef } from 'react'; import { io, Socket } from 'socket.io-client'; interface UseSocketOptions { url: string; auth?: { token?: string }; onConnect?: () => void; onDisconnect?: () => void; } export function useSocket(options: UseSocketOptions) { const { url, auth, onConnect, onDisconnect } = options; const socketRef = useRef<Socket | null>(null); const [isConnected, setIsConnected] = useState(false); useEffect(() => { // 创建连接 socketRef.current = io(url, { auth, autoConnect: true }); const socket = socketRef.current; socket.on('connect', () => { setIsConnected(true); onConnect?.(); }); socket.on('disconnect', () => { setIsConnected(false); onDisconnect?.(); }); // 清理 return () => { socket.disconnect(); }; }, [url]); return { socket: socketRef.current, isConnected }; } // 使用示例 function ChatComponent() { const { socket, isConnected } = useSocket({ url: 'http://localhost:3000', auth: { token: localStorage.getItem('token') || '' } }); const [messages, setMessages] = useState<string[]>([]); useEffect(() => { if (!socket) return; socket.on('message', (content: string) => { setMessages(prev => [...prev, content]); }); return () => { socket.off('message'); }; }, [socket]); const sendMessage = (content: string) => { socket?.emit('send-message', content); }; return ( <div> <div>状态: {isConnected ? '已连接' : '未连接'}</div> <ul> {messages.map((msg, i) => <li key={i}>{msg}</li>)} </ul> <button onClick={() => sendMessage('Hello')}>发送</button> </div> ); }

Context Provider

import { createContext, useContext, useEffect, useState, ReactNode } from 'react'; import { io, Socket } from 'socket.io-client'; interface SocketContextType { socket: Socket | null; isConnected: boolean; } const SocketContext = createContext<SocketContextType>({ socket: null, isConnected: false }); export function SocketProvider({ children }: { children: ReactNode }) { const [socket, setSocket] = useState<Socket | null>(null); const [isConnected, setIsConnected] = useState(false); useEffect(() => { const newSocket = io('http://localhost:3000', { auth: { token: localStorage.getItem('token') } }); newSocket.on('connect', () => setIsConnected(true)); newSocket.on('disconnect', () => setIsConnected(false)); setSocket(newSocket); return () => { newSocket.close(); }; }, []); return ( <SocketContext.Provider value={{ socket, isConnected }}> {children} </SocketContext.Provider> ); } export function useSocketContext() { return useContext(SocketContext); } // App.tsx function App() { return ( <SocketProvider> <ChatApp /> </SocketProvider> ); }

💚 Vue 集成

Composition API

<script setup lang="ts"> import { ref, onMounted, onUnmounted } from 'vue'; import { io, Socket } from 'socket.io-client'; const socket = ref<Socket | null>(null); const isConnected = ref(false); const messages = ref<string[]>([]); onMounted(() => { socket.value = io('http://localhost:3000', { auth: { token: localStorage.getItem('token') } }); socket.value.on('connect', () => { isConnected.value = true; }); socket.value.on('disconnect', () => { isConnected.value = false; }); socket.value.on('message', (content: string) => { messages.value.push(content); }); }); onUnmounted(() => { socket.value?.disconnect(); }); function sendMessage(content: string) { socket.value?.emit('send-message', content); } </script> <template> <div> <div>状态: {{ isConnected ? '已连接' : '未连接' }}</div> <ul> <li v-for="(msg, i) in messages" :key="i">{{ msg }}</li> </ul> <button @click="sendMessage('Hello')">发送</button> </div> </template>

全局插件

// socket.ts import { io, Socket } from 'socket.io-client'; import type { App } from 'vue'; export const SocketPlugin = { install(app: App, options: { url: string }) { const socket = io(options.url); // 提供全局属性 app.config.globalProperties.$socket = socket; // 或使用 provide/inject app.provide('socket', socket); } }; // main.ts import { createApp } from 'vue'; import App from './App.vue'; import { SocketPlugin } from './socket'; const app = createApp(App); app.use(SocketPlugin, { url: 'http://localhost:3000' }); app.mount('#app'); // 组件中使用 import { inject } from 'vue'; const socket = inject<Socket>('socket');

常用方法

// 连接 socket.connect(); // 断开 socket.disconnect(); // 发送事件 socket.emit('event', data); // 发送并等待确认 socket.emit('event', data, (response) => { console.log('确认:', response); }); // 监听事件 socket.on('event', handler); // 监听一次 socket.once('event', handler); // 移除监听 socket.off('event', handler); // 移除所有监听 socket.off('event'); // 发送 volatile 消息 socket.volatile.emit('event', data);
