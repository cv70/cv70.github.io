🛡️ 安全响应头

// 安装 koa-helmet npm install koa-helmet
const helmet = require('koa-helmet'); // 使用默认安全头 app.use(helmet()); // 自定义配置 app.use(helmet({ contentSecurityPolicy: { directives: { defaultSrc: ["'self'"], scriptSrc: ["'self'", 'https://cdn.example.com'], styleSrc: ["'self'", "'unsafe-inline'"], imgSrc: ["'self'", 'data:', 'https:'], connectSrc: ["'self'", 'https://api.example.com'], fontSrc: ["'self'", 'https://fonts.gstatic.com'], objectSrc: ["'none'"], frameAncestors: ["'none'"] } }, hsts: { maxAge: 31536000, includeSubDomains: true, preload: true } }));

📊 安全头详解

// 手动设置安全响应头 app.use(async (ctx, next) => { // 防止点击劫持 ctx.set('X-Frame-Options', 'DENY'); // 防止 MIME 类型嗅探 ctx.set('X-Content-Type-Options', 'nosniff'); // XSS 保护 ctx.set('X-XSS-Protection', '1; mode=block'); // HSTS (强制 HTTPS) ctx.set('Strict-Transport-Security', 'max-age=31536000; includeSubDomains'); // 内容安全策略 ctx.set('Content-Security-Policy', "default-src 'self'"); // 权限策略 ctx.set('Permissions-Policy', 'geolocation=(), microphone=()'); // 引用策略 ctx.set('Referrer-Policy', 'strict-origin-when-cross-origin'); await next(); });

🔒 XSS 防护

// 1. 输入验证 function sanitizeInput(input) { if (typeof input !== 'string') return input; return input .replace(/&/g, '&') .replace(//g, '>') .replace(/"/g, '"') .replace(/'/g, '''); } // 2. 使用验证库 const validator = require('validator'); router.post('/comment', async ctx => { const { content } = ctx.request.body; // 清理 HTML 标签 const cleanContent = validator.stripLow(content); const escapedContent = validator.escape(cleanContent); await Comment.create({ content: escapedContent }); ctx.body = { message: 'Comment created' }; }); // 3. EJS 模板自动转义 // <%= %> 自动转义 // <%- %> 不转义（谨慎使用）

🔄 CSRF 防护

# 安装 koa-csrf npm install koa-csrf
const CSRF = require('koa-csrf'); // 配置 CSRF app.use(new CSRF({ invalidTokenMessage: 'Invalid CSRF token', invalidTokenStatusCode: 403, excludedMethods: ['GET', 'HEAD', 'OPTIONS'], disableQuery: true })); // 获取 CSRF Token router.get('/csrf-token', async ctx => { ctx.body = { csrf: ctx.csrf }; }); // 表单提交（需要携带 CSRF token） router.post('/submit', async ctx => { // CSRF 中间件会自动验证 ctx.body = { message: 'Success' }; }); // 前端请求 /* fetch('/submit', { method: 'POST', headers: { 'Content-Type': 'application/json', 'X-CSRF-Token': csrfToken }, body: JSON.stringify(data) }); */

💉 SQL 注入防护

// 危险！直接拼接 SQL router.get('/user', async ctx => { const name = ctx.query.name; // 可被注入: name = "admin' OR '1'='1" const sql = `SELECT * FROM users WHERE name = '${name}'`; const users = await sequelize.query(sql); ctx.body = users; }); // 安全：使用参数化查询 router.get('/user', async ctx => { const name = ctx.query.name; const users = await sequelize.query( 'SELECT * FROM users WHERE name = ?', { replacements: [name] } ); ctx.body = users; }); // 安全：使用 ORM router.get('/user', async ctx => { const name = ctx.query.name; const users = await User.findAll({ where: { name } }); ctx.body = users; });

⏱️ 请求限流

const rateLimit = require('koa-ratelimit'); // 使用 Redis 存储 const Redis = require('ioredis'); const redis = new Redis(); app.use(rateLimit({ db: redis, duration: 60000, // 时间窗口：1 分钟 max: 100, // 最大请求数 id: ctx => ctx.ip, // 限流标识 errorMessage: 'Too many requests', disableHeader: false })); // 响应头 // X-RateLimit-Limit: 100 // X-RateLimit-Remaining: 99 // X-RateLimit-Reset: 60

🔐 数据验证

const Joi = require('joi'); // 验证中间件 function validate(schema) { return async (ctx, next) => { const data = ctx.method === 'GET' ? ctx.query : ctx.request.body; try { const value = await schema.validateAsync(data, { abortEarly: false }); // 替换验证后的数据 if (ctx.method === 'GET') { ctx.query = value; } else { ctx.request.body = value; } await next(); } catch (err) { ctx.status = 400; ctx.body = { error: 'Validation failed', details: err.details }; } }; } // 使用 const userSchema = Joi.object({ username: Joi.string().alphanum().min(3).max(30).required(), email: Joi.string().email().required(), password: Joi.string().pattern(new RegExp('^[a-zA-Z0-9]{3,30}$')).required() }); router.post('/register', validate(userSchema), async ctx => { const { username, email, password } = ctx.request.body; // 创建用户... });

💻 完整安全配置

const Koa = require('koa'); const helmet = require('koa-helmet'); const rateLimit = require('koa-ratelimit'); const cors = require('@koa/cors'); const Redis = require('ioredis'); const app = new Koa(); // 1. 安全响应头 app.use(helmet()); // 2. CORS 配置 app.use(cors({ origin: 'https://example.com', credentials: true })); // 3. 请求限流 app.use(rateLimit({ db: new Redis(), duration: 60000, max: 100, id: ctx => ctx.ip })); // 4. 错误处理 app.use(async (ctx, next) => { try { await next(); } catch (err) { ctx.status = err.status || 500; ctx.body = { error: process.env.NODE_ENV === 'production' ? 'Internal Server Error' : err.message }; ctx.app.emit('error', err, ctx); } }); // 5. 业务路由 // ... app.listen(3000);
