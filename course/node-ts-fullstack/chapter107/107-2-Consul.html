🔧 服务注册与发现

import Consul from 'consul'; // 创建Consul客户端 const consul = new Consul({ host: 'localhost', port: 8500, promisify: true }); // ==================== 服务注册 ==================== // 注册服务 async function registerService() { await consul.agent.service.register({ name: 'user-service', id: 'user-service-1', address: '192.168.1.100', port: 3001, tags: ['v1', 'primary'], check: { http: 'http://192.168.1.100:3001/health', interval: '10s', timeout: '5s', deregistercriticalserviceafter: '1m' } }); console.log('Service registered'); } // 注销服务 async function deregisterService() { await consul.agent.service.deregister('user-service-1'); console.log('Service deregistered'); } // ==================== 服务发现 ==================== // 获取服务列表 async function discoverService() { // 获取健康的服务实例 const services = await consul.health.service({ service: 'user-service', passing: true // 只返回健康的实例 }); console.log('Available instances:', services.length); // 选择一个实例（负载均衡） const instance = services[Math.floor(Math.random() * services.length)]; const address = `http://${instance.Service.Address}:${instance.Service.Port}`; return address; } // 使用服务发现调用服务 async function callUserService(userId: string) { const serviceUrl = await discoverService(); const response = await fetch(`${serviceUrl}/users/${userId}`); return response.json(); } // ==================== 监听服务变化 ==================== // Watch服务变化 function watchService() { const watch = consul.watch({ method: consul.health.service, options: { service: 'user-service', passing: true } }); watch.on('change', (data) => { console.log('Service instances changed:', data.length); // 更新本地缓存 updateLocalCache(data); }); watch.on('error', (err) => { console.error('Watch error:', err); }); return watch; }

🔀 KV存储

// ==================== KV存储操作 ==================== // 设置键值 async function setConfig(key: string, value: any) { await consul.kv.set(`config/${key}`, JSON.stringify(value)); console.log(`Config ${key} saved`); } // 获取键值 async function getConfig(key: string) { const result = await consul.kv.get(`config/${key}`); if (result) { return JSON.parse(result.Value); } return null; } // 获取所有键 async function listKeys(prefix: string) { const keys = await consul.kv.keys({ key: prefix, recurse: true }); return keys; } // 删除键 async function deleteConfig(key: string) { await consul.kv.del(`config/${key}`); } // 使用KV存储实现配置中心 class ConsulConfigCenter { private consul: Consul; private cache: Map = new Map(); private watchers: Map = new Map(); constructor(consul: Consul) { this.consul = consul; } // 获取配置 async get(key: string): Promise { if (this.cache.has(key)) { return this.cache.get(key); } const config = await getConfig(key); this.cache.set(key, config); return config; } // 监听配置变化 async watch(key: string, callback: (value: any) => void) { const watch = this.consul.watch({ method: this.consul.kv.get, options: { key: `config/${key}` } }); watch.on('change', (data) => { const value = data ? JSON.parse(data.Value) : null; this.cache.set(key, value); callback(value); }); this.watchers.set(key, watch); } } // 使用示例 const configCenter = new ConsulConfigCenter(consul); // 监听数据库配置变化 await configCenter.watch('database', (config) => { console.log('Database config changed:', config); // 重新连接数据库 reconnectDatabase(config); });
