25.1 QueryBuilder

构建复杂的数据库查询

🎯 学习目标

  • 理解 QueryBuilder 的作用
  • 掌握基本查询构建方法
  • 学会链式调用和参数绑定
  • 了解获取结果的不同方式

📚 QueryBuilder 简介

QueryBuilder 是 TypeORM 提供的查询构建器，允许使用链式 API 构建复杂 SQL 查询，同时保持类型安全。

// 创建 QueryBuilder import { AppDataSource } from './data-source'; import { User } from './entity/User'; const userRepository = AppDataSource.getRepository(User); // 方式一：从 Repository 创建 const qb = userRepository.createQueryBuilder('user'); // 方式二：从 DataSource 创建 const qb = AppDataSource.createQueryBuilder(User, 'user'); // 方式三：从 EntityManager 创建 const qb = AppDataSource.manager.createQueryBuilder(User, 'user'); // 基本查询 const users = await userRepository .createQueryBuilder('user') .where('user.status = :status', { status: 'active' }) .orderBy('user.createdAt', 'DESC') .getMany(); // 生成的 SQL // SELECT user.id, user.username, user.status, user.createdAt // FROM users user // WHERE user.status = 'active' // ORDER BY user.createdAt DESC
💡
别名的重要性

createBuider 的第一个参数是实体类，第二个参数是别名（如 'user'）。别名在后续的 WHERE、ORDER BY 等子句中引用实体时使用。

🔧 基本查询方法

// SELECT 查询 const users = await userRepository .createQueryBuilder('user') .select(['user.id', 'user.username', 'user.email']) .getMany(); // SELECT DISTINCT const uniqueEmails = await userRepository .createQueryBuilder('user') .select('DISTINCT user.email', 'email') .getRawMany(); // 添加选择列 const usersWithCount = await userRepository .createQueryBuilder('user') .leftJoin('user.posts', 'post') .addSelect('COUNT(post.id)', 'postCount') .groupBy('user.id') .getRawMany(); // WHERE 条件 const users = await userRepository .createQueryBuilder('user') .where('user.status = :status', { status: 'active' }) .andWhere('user.age >= :minAge', { minAge: 18 }) .getMany(); // OR 条件 const users = await userRepository .createQueryBuilder('user') .where('user.status = :status', { status: 'active' }) .orWhere('user.role = :role', { role: 'admin' }) .getMany(); // 复杂条件组合 const users = await userRepository .createQueryBuilder('user') .where( new Brackets((qb) => { qb.where('user.status = :status', { status: 'active' }) .andWhere('user.age >= :minAge', { minAge: 18 }); }) ) .orWhere( new Brackets((qb) => { qb.where('user.role = :role', { role: 'admin' }); }) ) .getMany(); // 生成的 SQL // SELECT * FROM users user // WHERE (user.status = 'active' AND user.age >= 18) // OR (user.role = 'admin')

📊 参数绑定

// 命名参数（推荐） const users = await userRepository .createQueryBuilder('user') .where('user.status = :status', { status: 'active' }) .andWhere('user.age > :age', { age: 18 }) .getMany(); // 链式参数 const users = await userRepository .createQueryBuilder('user') .where('user.status = :status') .setParameter('status', 'active') .andWhere('user.age > :age') .setParameter('age', 18) .getMany(); // 批量参数 - IN 查询 const users = await userRepository .createQueryBuilder('user') .where('user.id IN (:...ids)', { ids: [1, 2, 3] }) .getMany(); // 生成的 SQL // SELECT * FROM users user WHERE user.id IN (1, 2, 3) // LIKE 查询 const users = await userRepository .createQueryBuilder('user') .where('user.username LIKE :keyword', { keyword: '%alice%' }) .getMany(); // 安全防注入 // 参数化查询自动处理 SQL 注入防护 const userInput = "alice'; DROP TABLE users;--"; const users = await userRepository .createQueryBuilder('user') .where('user.username = :username', { username: userInput }) .getMany(); // 安全：username 会被转义，不会执行恶意 SQL

⚙️ 获取结果

// 获取多条记录 const users = await userRepository .createQueryBuilder('user') .getMany(); // 返回 User[] // 获取单条记录 const user = await userRepository .createQueryBuilder('user') .where('user.id = :id', { id: 1 }) .getOne(); // 返回 User | null // 获取原始数据 const rawUsers = await userRepository .createQueryBuilder('user') .select(['user.id AS id', 'user.username AS name']) .getRawMany(); // 返回 { id: number, name: string }[] // 获取单条原始数据 const rawUser = await userRepository .createQueryBuilder('user') .where('user.id = :id', { id: 1 }) .getRawOne(); // 获取数量 const count = await userRepository .createQueryBuilder('user') .where('user.status = :status', { status: 'active' }) .getCount(); // 返回 number // 获取结果和总数 const [users, total] = await userRepository .createQueryBuilder('user') .where('user.status = :status', { status: 'active' }) .getManyAndCount(); // 返回 [User[], number] // 检查是否存在 const exists = await userRepository .createQueryBuilder('user') .where('user.email = :email', { email: 'test@mail.com' }) .getExists(); // 返回 boolean // 获取 SQL（不执行） const sql = userRepository .createQueryBuilder('user') .where('user.status = :status', { status: 'active' }) .getSql(); // 返回 SQL 字符串 console.log(sql); // SELECT ... FROM users user WHERE user.status = ?
