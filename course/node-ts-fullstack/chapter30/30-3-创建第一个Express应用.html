30.3 创建第一个 Express 应用

从零开始手动搭建一个 Express Web 应用

🎯 学习目标

  • 掌握手动创建 Express 项目的完整流程
  • 理解 Express 应用的基本结构
  • 学会配置和使用常用中间件
  • 能够编写简单的路由处理

🛠️ 项目初始化

让我们从零开始创建一个 Express 应用，深入理解每个步骤：

# 1. 创建项目目录 mkdir my-first-express cd my-first-express # 2. 初始化 package.json npm init -y # 3. 安装 Express npm install express # 4. 安装开发依赖（可选） npm install -D nodemon
💡
关于 nodemon

nodemon 是一个开发工具，监听文件变化自动重启服务器，大大提高开发效率。生产环境不需要安装。

📝 创建基本服务器

创建 index.js 文件，编写最简单的 Express 应用：

// index.js const express = require('express'); // 创建 Express 应用实例 const app = express(); const PORT = 3000; // 定义根路由 app.get('/', (req, res) => { res.send('Hello, Express!'); }); // 启动服务器 app.listen(PORT, () => { console.log(`服务器已启动: http://localhost:${PORT}`); });
# 运行应用 node index.js # 或使用 nodemon（开发模式） npx nodemon index.js

访问 http://localhost:3000，你将看到 "Hello, Express!" 的响应。

📁 完善项目结构

为了更好的代码组织，我们创建完整的项目结构：

my-first-express/ ├── src/ │ ├── routes/ │ │ ├── index.js # 路由汇总 │ │ ├── home.js # 首页路由 │ │ └── api.js # API 路由 │ ├── middleware/ │ │ └── logger.js # 日志中间件 │ ├── config/ │ │ └── index.js # 配置文件 │ └── app.js # Express 配置 ├── public/ # 静态文件 │ ├── css/ │ ├── js/ │ └── images/ ├── views/ # 模板文件 ├── index.js # 入口文件 └── package.json

⚙️ 配置 Express 应用

1. 创建配置文件 (src/config/index.js)

// src/config/index.js module.exports = { port: process.env.PORT || 3000, env: process.env.NODE_ENV || 'development', cors: { origin: '*', methods: ['GET', 'POST', 'PUT', 'DELETE'] } };

2. 创建 Express 配置 (src/app.js)

// src/app.js const express = require('express'); const path = require('path'); // 引入路由 const routes = require('./routes'); // 创建应用 const app = express(); // ========== 中间件配置 ========== // JSON 请求体解析 app.use(express.json()); // URL 编码请求体解析 app.use(express.urlencoded({ extended: true })); // 静态文件服务 app.use(express.static(path.join(__dirname, '../public'))); // ========== 路由配置 ========== // 注册路由 app.use('/', routes); // 404 处理 app.use((req, res) => { res.status(404).json({ success: false, message: 'Not Found' }); }); // 全局错误处理 app.use((err, req, res, next) => { console.error(err.stack); res.status(500).json({ success: false, message: 'Internal Server Error', error: process.env.NODE_ENV === 'development' ? err.message : {} }); }); module.exports = app;

3. 创建路由文件

// src/routes/index.js const express = require('express'); const router = express.Router(); const homeRouter = require('./home'); const apiRouter = require('./api'); // 首页路由 router.use('/', homeRouter); // API 路由 router.use('/api', apiRouter); module.exports = router;
// src/routes/home.js const express = require('express'); const router = express.Router(); // GET / - 首页 router.get('/', (req, res) => { res.json({ message: 'Welcome to Express!', timestamp: new Date().toISOString() }); }); // GET /health - 健康检查 router.get('/health', (req, res) => { res.json({ status: 'ok', uptime: process.uptime() }); }); module.exports = router;
// src/routes/api.js const express = require('express'); const router = express.Router(); // 模拟数据 let users = [ { id: 1, name: 'Alice', email: 'alice@example.com' }, { id: 2, name: 'Bob', email: 'bob@example.com' } ]; // GET /api/users - 获取用户列表 router.get('/users', (req, res) => { res.json({ success: true, data: users }); }); // GET /api/users/:id - 获取单个用户 router.get('/users/:id', (req, res) => { const id = parseInt(req.params.id); const user = users.find(u => u.id === id); if (!user) { return res.status(404).json({ success: false, message: 'User not found' }); } res.json({ success: true, data: user }); }); // POST /api/users - 创建用户 router.post('/users', (req, res) => { const { name, email } = req.body; if (!name || !email) { return res.status(400).json({ success: false, message: 'Name and email are required' }); } const newUser = { id: users.length + 1, name, email }; users.push(newUser); res.status(201).json({ success: true, data: newUser }); }); // PUT /api/users/:id - 更新用户 router.put('/users/:id', (req, res) => { const id = parseInt(req.params.id); const userIndex = users.findIndex(u => u.id === id); if (userIndex === -1) { return res.status(404).json({ success: false, message: 'User not found' }); } const { name, email } = req.body; users[userIndex] = { ...users[userIndex], name, email }; res.json({ success: true, data: users[userIndex] }); }); // DELETE /api/users/:id - 删除用户 router.delete('/users/:id', (req, res) => { const id = parseInt(req.params.id); const userIndex = users.findIndex(u => u.id === id); if (userIndex === -1) { return res.status(404).json({ success: false, message: 'User not found' }); } users.splice(userIndex, 1); res.json({ success: true, message: 'User deleted' }); }); module.exports = router;

4. 更新入口文件 (index.js)

// index.js const app = require('./src/app'); const config = require('./src/config'); app.listen(config.port, () => { console.log(`服务器已启动: http://localhost:${config.port}`); console.log(`运行环境: ${config.env}`); });

🔌 添加自定义中间件

创建日志中间件 (src/middleware/logger.js)

// src/middleware/logger.js const logger = (req, res, next) => { const start = Date.now(); // 响应完成后记录日志 res.on('finish', () => { const duration = Date.now() - start; const timestamp = new Date().toISOString(); console.log(`[${timestamp}] ${req.method} ${req.originalUrl} ${res.statusCode} - ${duration}ms`); }); next(); }; module.exports = logger;

在 app.js 中使用

// 在路由之前添加 const logger = require('./middleware/logger'); app.use(logger);

📦 更新 package.json

{ "name": "my-first-express", "version": "1.0.0", "main": "index.js", "scripts": { "start": "node index.js", "dev": "nodemon index.js" }, "dependencies": { "express": "^4.18.2" }, "devDependencies": { "nodemon": "^3.0.0" } }

🧪 测试 API

启动服务器后，可以使用浏览器、Postman 或 curl 测试 API：

# 测试首页 curl http://localhost:3000/ # 获取用户列表 curl http://localhost:3000/api/users # 获取单个用户 curl http://localhost:3000/api/users/1 # 创建用户 curl -X POST http://localhost:3000/api/users \ -H "Content-Type: application/json" \ -d '{"name":"Charlie","email":"charlie@example.com"}' # 更新用户 curl -X PUT http://localhost:3000/api/users/1 \ -H "Content-Type: application/json" \ -d '{"name":"Alice Updated"}' # 删除用户 curl -X DELETE http://localhost:3000/api/users/1

✅ 小结

📌
本节要点
  • 手动创建 Express 项目需要：初始化 npm、安装 express、创建入口文件
  • 良好的项目结构有助于代码维护和扩展
  • app.js 负责应用配置，routes 负责路由处理
  • 中间件是 Express 的核心概念，按顺序执行
  • 路由参数、请求体解析、错误处理是常用功能
