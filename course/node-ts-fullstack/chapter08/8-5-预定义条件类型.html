8.5 预定义条件类型

TypeScript 内置的条件类型工具

🎯 学习目标

  • 掌握 Exclude 和 Extract 类型
  • 理解 NonNullable 的作用
  • 学会使用 ReturnType 和 Parameters
  • 了解 InstanceType 的用法

📚 Exclude 和 Extract

// Exclude - 从 T 中排除可赋值给 U 的类型 // type Exclude = T extends U ? never : T; type T1 = Exclude<'a' | 'b' | 'c', 'a'>; // 'b' | 'c' type T2 = Exclude; // string | boolean type T3 = Exclude; // never // Extract - 从 T 中提取可赋值给 U 的类型 // type Extract = T extends U ? T : never; type T4 = Extract<'a' | 'b' | 'c', 'a' | 'f'>; // 'a' type T5 = Extract; // number type T6 = Extract; // never // 实际应用：过滤事件类型 type AllEvents = 'click' | 'focus' | 'blur' | 'keydown' | 'keyup'; type KeyboardEvents = Extract; // 'keydown' | 'keyup' type MouseEvents = Exclude; // 'click' | 'focus' | 'blur'

🔧 NonNullable

// NonNullable - 从 T 中排除 null 和 undefined // type NonNullable = T extends null | undefined ? never : T; type T7 = NonNullable; // string type T8 = NonNullable; // number type T9 = NonNullable; // never // 实际应用：确保值不为空 function processValue(value: NonNullable): void { console.log(value); } processValue('hello'); // OK processValue(42); // OK // processValue(null); // Error // 对象属性非空化 interface User { name: string | null; email: string | undefined; } type NonNullUser = { [K in keyof User]: NonNullable; }; // { name: string; email: string; }

⚙️ ReturnType 和 Parameters

// ReturnType - 获取函数返回值类型 // type ReturnType any> = // T extends (...args: any) => infer R ? R : any; function getUser() { return { id: 1, name: 'Alice', email: 'alice@example.com' }; } type User = ReturnType; // { id: number; name: string; email: string; } // Parameters - 获取函数参数类型元组 // type Parameters any> = // T extends (...args: infer P) => any ? P : never; function greet(name: string, age: number): string { return `Hello, ${name}!`; } type GreetParams = Parameters; // [name: string, age: number] // 实际应用：包装函数保持类型 function logResult any>(fn: T) { return (...args: Parameters): ReturnType => { console.log('Calling with:', args); const result = fn(...args); console.log('Result:', result); return result; }; } const loggedGreet = logResult(greet);

🏗️ InstanceType

// InstanceType - 获取构造函数的实例类型 // type InstanceType any> = // T extends new (...args: any) => infer R ? R : any; class Person { constructor(public name: string, public age: number) {} } type PersonInstance = InstanceType; // Person function createInstance any>( ctor: T, ...args: ConstructorParameters ): InstanceType { return new ctor(...args); } const person = createInstance(Person, 'Alice', 30); // person: Person // ConstructorParameters - 获取构造函数参数类型 type PersonParams = ConstructorParameters; // [name: string, age: number]
💡
内置条件类型一览

Exclude、Extract、NonNullable、ReturnType、Parameters、InstanceType、ConstructorParameters 都是 TypeScript 内置的条件类型工具，在日常开发中非常实用。
