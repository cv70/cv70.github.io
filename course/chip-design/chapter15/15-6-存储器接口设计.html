💾 SRAM接口

// SRAM控制器
module sram_controller(
    input  clk,
    input  rst_n,
    input        read_req,
    input        write_req,
    input  [15:0] addr,
    input  [7:0]  wdata,
    output reg [7:0]  rdata,
    output       rdata_valid,
    // SRAM接口
    output        sram_ce_n,
    output        sram_oe_n,
    output        sram_we_n,
    output [15:0] sram_addr,
    inout  [7:0]  sram_data
);
    reg [1:0] state;
    reg [7:0] sram_data_out;
    
    assign sram_ce_n = 1'b0;
    assign sram_oe_n = (state != 2'b10);
    assign sram_we_n = (state != 2'b01);
    assign sram_addr = addr;
    assign sram_data = (state == 2'b01) ? sram_data_out : 8'hzz;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= 2'b00;
            rdata <= 0;
            rdata_valid <= 0;
        end
        else begin
            case (state)
                2'b00: begin
                    rdata_valid <= 0;
                    if (write_req) begin
                        state <= 2'b01;
                        sram_data_out <= wdata;
                    end
                    else if (read_req)
                        state <= 2'b10;
                end
                2'b01: state <= 2'b00;
                2'b10: begin
                    rdata <= sram_data;
                    rdata_valid <= 1;
                    state <= 2'b00;
                end
            endcase
        end
    end
endmodule
