🔌 手动连接控制

延迟连接

// 禁用自动连接 const socket = io('http://localhost:3000', { autoConnect: false }); // 在需要时手动连接 document.getElementById('connectBtn').addEventListener('click', () => { socket.connect(); }); // 断开连接 document.getElementById('disconnectBtn').addEventListener('click', () => { socket.disconnect(); });

条件连接

// 等待用户登录后再连接 const socket = io('http://localhost:3000', { autoConnect: false }); async function login(username: string, password: string) { const response = await fetch('/api/login', { method: 'POST', body: JSON.stringify({ username, password }) }); const { token } = await response.json(); // 设置认证信息后连接 socket.auth = { token }; socket.connect(); localStorage.setItem('token', token); } // 页面刷新时自动连接 const savedToken = localStorage.getItem('token'); if (savedToken) { socket.auth = { token: savedToken }; socket.connect(); }

🔁 重连机制

重连配置

const socket = io('http://localhost:3000', { // 重连配置 reconnection: true, // 启用自动重连 reconnectionAttempts: 10, // 最大重连次数 reconnectionDelay: 1000, // 初始重连延迟 reconnectionDelayMax: 5000, // 最大重连延迟 randomizationFactor: 0.5, // 随机化因子（避免同时重连） // 指数退避重连 // 第1次: 1000ms // 第2次: 1000 * (1 + 0.5) = 1500ms // 第3次: 1500 * (1 + 0.5) = 2250ms // ...直到达到 reconnectionDelayMax });

自定义重连逻辑

const socket = io('http://localhost:3000', { reconnection: false // 禁用默认重连 }); let reconnectAttempts = 0; const maxAttempts = 5; const baseDelay = 1000; function attemptReconnect() { if (reconnectAttempts >= maxAttempts) { console.error('重连失败'); showOfflineMessage(); return; } reconnectAttempts++; const delay = Math.min(baseDelay * Math.pow(2, reconnectAttempts - 1), 30000); console.log(`将在 ${delay}ms 后重连 (第 ${reconnectAttempts} 次)`); setTimeout(() => { socket.connect(); }, delay); } socket.on('disconnect', (reason) => { if (reason === 'io server disconnect') { // 服务端主动断开，不重连 console.log('服务端断开连接'); } else { // 其他原因断开，尝试重连 attemptReconnect(); } }); socket.on('connect', () => { reconnectAttempts = 0; // 重置计数 });

🌐 多命名空间连接

连接多个命名空间

// 连接多个命名空间（共享底层连接） const mainSocket = io('http://localhost:3000'); const chatSocket = io('http://localhost:3000/chat'); const notificationSocket = io('http://localhost:3000/notifications'); // 各命名空间独立管理 mainSocket.on('connect', () => { console.log('主命名空间已连接'); }); chatSocket.on('connect', () => { console.log('聊天命名空间已连接'); }); notificationSocket.on('connect', () => { console.log('通知命名空间已连接'); }); // 断开特定命名空间 chatSocket.disconnect(); // 断开所有连接 mainSocket.disconnect();

动态命名空间

// 动态创建命名空间连接 function createNamespaceConnection(namespace: string) { const socket = io(`http://localhost:3000${namespace}`, { autoConnect: false }); socket.on('connect', () => { console.log(`已连接到 ${namespace}`); }); return socket; } // 按需连接 let gameSocket: Socket | null = null; function joinGame(roomId: string) { gameSocket = createNamespaceConnection('/game'); gameSocket.auth = { roomId }; gameSocket.connect(); } function leaveGame() { gameSocket?.disconnect(); gameSocket = null; }

📊 连接状态监控

状态检查

// 检查连接状态 if (socket.connected) { console.log('已连接'); socket.emit('event', data); } else { console.log('未连接，消息已缓存'); messageQueue.push(data); } // 连接后发送缓存消息 socket.on('connect', () => { while (messageQueue.length > 0) { const msg = messageQueue.shift(); socket.emit('event', msg); } });

连接状态 Hook（React）

import { useState, useEffect, useCallback } from 'react'; import { io, Socket } from 'socket.io-client'; interface ConnectionState { socket: Socket | null; isConnected: boolean; isReconnecting: boolean; reconnectAttempts: number; error: Error | null; } export function useConnection(url: string) { const [state, setState] = useState<ConnectionState>({ socket: null, isConnected: false, isReconnecting: false, reconnectAttempts: 0, error: null }); useEffect(() => { const socket = io(url, { autoConnect: false }); socket.on('connect', () => { setState(prev => ({ ...prev, socket, isConnected: true, isReconnecting: false, reconnectAttempts: 0, error: null })); }); socket.on('disconnect', () => { setState(prev => ({ ...prev, isConnected: false })); }); socket.on('connect_error', (error) => { setState(prev => ({ ...prev, error, isReconnecting: true, reconnectAttempts: prev.reconnectAttempts + 1 })); }); socket.on('reconnect_failed', () => { setState(prev => ({ ...prev, isReconnecting: false, error: new Error('重连失败') })); }); socket.connect(); return () => { socket.disconnect(); }; }, [url]); const reconnect = useCallback(() => { state.socket?.connect(); }, [state.socket]); const disconnect = useCallback(() => { state.socket?.disconnect(); }, [state.socket]); return { ...state, reconnect, disconnect }; }

⚠️ 断开原因详解

原因 说明 处理建议
io server disconnect 服务端主动断开 不自动重连，需用户操作
io client disconnect 客户端主动断开 正常行为，无需处理
ping timeout 心跳超时 自动重连
transport close 传输层关闭 自动重连
transport error 传输层错误 自动重连

📝 本节小结

✅

  • • 不同断开原因需要不同处理策略
