79.2 Props 类型

定义组件属性类型

🎯 学习目标

  • 掌握基本 Props 类型定义
  • 学会可选属性和默认值
  • 理解 children 类型
  • 掌握高级 Props 模式

📚 基本 Props 类型

import { ReactNode, MouseEventHandler, ChangeEventHandler } from 'react'; // 基本类型 interface BasicProps { string: string; number: number; boolean: boolean; callback: () => void; nullable: string | null; } // 可选属性 interface OptionalProps { required: string; // 必需 optional?: string; // 可选（undefined） optionalWithDefault?: string; // 有默认值 } function Component({ required, optional, optionalWithDefault = 'default' }: OptionalProps) { return <div>{required} {optional} {optionalWithDefault}</div>; } // 函数类型 interface FunctionProps { onClick: () => void; // 无参数 onChange: (value: string) => void; // 有参数 onSubmit: (event: React.FormEvent) => void; // 事件参数 // 或使用 React 提供的类型 onClick2: MouseEventHandler<HTMLButtonElement>; onChange2: ChangeEventHandler<HTMLInputElement>; } // 对象和数组 interface DataProps { user: { id: string; name: string; email: string; }; items: string[]; users: User[]; // 使用已定义的类型 } interface User { id: string; name: string; } // 联合类型 interface UnionProps { status: 'loading' | 'success' | 'error'; size: 'sm' | 'md' | 'lg'; value: string | number; }

🔧 children 类型

import { ReactNode, ReactElement, ComponentType } from 'react'; // ReactNode：最宽泛，包含所有可渲染内容 interface CardProps { children: ReactNode; // string | number | JSX | null | undefined } function Card({ children }: CardProps) { return <div className="card">{children}</div>; } // 使用 <Card> <h1>Title</h1> <p>Content</p> </Card> <Card>Just a string</Card> <Card>{condition && <span>Conditional</span>}</Card> // ReactElement：只接受 JSX 元素 interface StrictProps { children: ReactElement; // 只接受单个 JSX 元素 } function Strict({ children }: StrictProps) { return <div>{children}</div>; } // 使用 <Strict><span>OK</span></Strict> // 正确 <Strict>String</Strict> // 错误：不能是字符串 // 多个 children slots interface LayoutProps { header: ReactNode; children: ReactNode; footer?: ReactNode; } function Layout({ header, children, footer }: LayoutProps) { return ( <div> <header>{header}</header> <main>{children}</main> {footer && <footer>{footer}</footer>} </div> ); } // 使用 <Layout header={<Header />} footer={<Footer />} > <Main /> </Layout> // 组件作为 Props interface WrapperProps { component: ComponentType<{ data: string }>; data: string; } function Wrapper({ component: Component, data }: WrapperProps) { return <Component data={data} />; }

📊 高级 Props 模式

// 条件 Props：根据一个属性决定另一个属性是否必需 interface BaseButtonProps { variant?: 'primary' | 'secondary'; } // 如果 variant 是 'icon'，则 icon 必需 type ButtonProps = BaseButtonProps & ( | { variant?: 'primary' | 'secondary'; icon?: never } | { variant: 'icon'; icon: ReactNode } ); function Button({ variant = 'primary', icon }: ButtonProps) { if (variant === 'icon') { return <button className="icon-btn">{icon}</button>; } return <button className={`btn-${variant}`} />; } // 使用 <Button variant="primary" /> // 正确 <Button variant="icon" icon={<Icon />} /> // 正确 <Button variant="icon" /> // 错误：缺少 icon // 工具类型 // Partial<T>：所有属性可选 type PartialProps = Partial<RequiredProps>; // Required<T>：所有属性必需 type RequiredProps = Required<OptionalProps>; // Pick<T, K>：选取部分属性 type OnlyIdAndName = Pick<User, 'id' | 'name'>; // Omit<T, K>：排除部分属性 type WithoutEmail = Omit<User, 'email'>; // 扩展 HTML 元素属性 interface InputProps extends React.InputHTMLAttributes<HTMLInputElement> { label?: string; } function Input({ label, ...props }: InputProps) { return ( <div> {label && <label>{label}</label>} <input {...props} /> </div> ); } // 使用 <Input label="Username" type="text" placeholder="Enter username" onChange={(e) => console.log(e.target.value)} />
