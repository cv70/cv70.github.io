88.2 Vite 配置

Vite 配置文件详解

🎯 学习目标

  • 掌握 Vite 配置文件语法
  • 学会常用配置选项
  • 了解环境变量配置
  • 掌握代理和别名配置

📚 配置文件基础

// vite.config.ts import { defineConfig } from 'vite' import vue from '@vitejs/plugin-vue' export default defineConfig({ // 插件配置 plugins: [vue()], // 开发服务器配置 server: { port: 3000, open: true, }, // 构建配置 build: { outDir: 'dist', }, }) // 条件配置 export default defineConfig(({ mode }) => { if (mode === 'development') { return { // 开发模式配置 } } else { return { // 生产模式配置 } } }) // 异步配置 export default defineConfig(async () => { const data = await someAsyncFunction() return { // 使用 data } })

🔧 常用配置选项

import { defineConfig } from 'vite' import vue from '@vitejs/plugin-vue' import path from 'path' export default defineConfig({ // 项目根目录 root: process.cwd(), // 公共基础路径 base: '/', // 模式 // development | production mode: 'development', // 插件 plugins: [vue()], // 路径解析 resolve: { alias: { '@': path.resolve(__dirname, 'src'), '@components': path.resolve(__dirname, 'src/components'), }, extensions: ['.mjs', '.js', '.ts', '.jsx', '.tsx', '.json'], }, // CSS 配置 css: { modules: { localsConvention: 'camelCase', }, preprocessorOptions: { scss: { additionalData: `@import "@/styles/variables.scss";`, }, }, }, // JSON 配置 json: { namedExports: true, stringify: false, }, })

⚙️ 开发服务器配置

export default defineConfig({ server: { // 主机 host: '0.0.0.0', // 端口 port: 3000, // 自动打开浏览器 open: true, // CORS cors: true, // HMR 配置 hmr: { overlay: true, }, // 代理配置 proxy: { '/api': { target: 'http://localhost:8080', changeOrigin: true, rewrite: (path) => path.replace(/^\/api/, ''), }, '/ws': { target: 'ws://localhost:8080', ws: true, }, }, // HTTPS https: { key: fs.readFileSync('key.pem'), cert: fs.readFileSync('cert.pem'), }, // 监听文件变化 watch: { ignored: ['**/node_modules/**', '**/dist/**'], }, }, // 预构建依赖 optimizeDeps: { include: ['axios', 'lodash'], exclude: ['your-linked-package'], }, })
💡
代理配置

开发环境下通过代理解决跨域问题，请求 /api/* 会被代理到 target 指定的服务器。

🚀 构建配置

export default defineConfig({ build: { // 输出目录 outDir: 'dist', // 静态资源目录 assetsDir: 'assets', // 文件名格式 rollupOptions: { output: { entryFileNames: 'js/[name]-[hash].js', chunkFileNames: 'js/[name]-[hash].js', assetFileNames: 'assets/[name]-[hash][extname]', }, }, // 代码分割策略 rollupOptions: { output: { manualChunks: { vendor: ['vue', 'vue-router', 'pinia'], utils: ['lodash', 'axios'], }, }, }, // 压缩配置 minify: 'terser', // 'terser' | 'esbuild' terserOptions: { compress: { drop_console: true, drop_debugger: true, }, }, // Source Map sourcemap: true, // 分块大小警告阈值 chunkSizeWarningLimit: 500, // CSS 代码分割 cssCodeSplit: true, // 清理输出目录 emptyOutDir: true, }, })

📊 环境变量

# .env 文件命名规则 .env # 所有环境 .env.local # 所有环境，不提交 git .env.development # 开发环境 .env.production # 生产环境 .env.staging # 预发布环境 # .env.development VITE_API_URL=http://localhost:8080/api VITE_APP_TITLE=My App (Dev) # .env.production VITE_API_URL=https://api.example.com VITE_APP_TITLE=My App # 使用环境变量 const apiUrl = import.meta.env.VITE_API_URL const mode = import.meta.env.MODE // 'development' | 'production' const isDev = import.meta.env.DEV const isProd = import.meta.env.PROD // 类型定义 (env.d.ts) /// <reference types="vite/client" /> interface ImportMetaEnv { readonly VITE_API_URL: string readonly VITE_APP_TITLE: string } interface ImportMeta { readonly env: ImportMetaEnv }
