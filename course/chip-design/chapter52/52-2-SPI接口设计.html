52.2 SPI接口设计

常用外设接口设计

🎯 学习目标

  • 掌握SPI通信协议的基本原理
  • 理解SPI的四种工作模式
  • 学会SPI主从设备的硬件设计
  • 了解SPI多设备通信方法

📚 SPI概述

SPI（Serial Peripheral Interface）串行外设接口是一种高速同步串行通信总线，由Motorola公司开发。它采用主从架构，支持全双工通信，具有高速、简单、灵活的特点。

💡
SPI特点

• 同步通信：使用时钟信号同步
• 全双工：可同时收发数据
• 高速：支持MHz级数据速率
• 主从架构：一个主设备，多个从设备

🔌 SPI信号接口

信号方向描述主设备从设备
SCLK输出→输入串行时钟输出输入
MOSI输出→输入主出从入数据输出输入
MISO输入←输出主入从出数据输入输出
CS/SS输出→输入片选/从机选择输出输入

信号说明

SCLK（Serial Clock）：由主设备输出的时钟信号，控制数据传输的时序。

MOSI（Master Out Slave In）：主设备向从设备发送数据的信号线。

MISO（Master In Slave Out）：从设备向主设备发送数据的信号线。

CS/SS（Chip Select/Slave Select）：片选信号，低电平有效，用于选择要通信的从设备。

📊 SPI四种工作模式

SPI有四种工作模式，由时钟极性（CPOL）和时钟相位（CPHA）决定：

模式CPOLCPHA时钟空闲状态数据采样边沿数据变化边沿
Mode 000低电平上升沿下降沿
Mode 101低电平下降沿上升沿
Mode 210高电平下降沿上升沿
Mode 311高电平上升沿下降沿
💡
模式选择建议

• Mode 0最常用，大多数SPI设备都支持
• 不同模式的设备无法直接通信
• 设计时应支持所有四种模式以提高兼容性

⚙️ SPI主机模块设计

// SPI主机模块
module spi_master #(
    parameter DATA_WIDTH = 8,
    parameter CLK_DIV = 4  // 时钟分频系数
) (
    input  wire                  clk,
    input  wire                  rst_n,
    input  wire                  start,          // 启动传输
    input  wire [2:0]            cpol_cpha,      // CPOL和CPHA
    input  wire [DATA_WIDTH-1:0] tx_data,        // 发送数据
    output wire [DATA_WIDTH-1:0] rx_data,        // 接收数据
    output reg                   sclk,           // 串行时钟
    output reg                   mosi,           // 主出从入
    input  wire                  miso,           // 主入从出
    output reg                   cs_n,           // 片选（低有效）
    output reg                   busy            // 忙标志
);

    // 状态定义
    localparam IDLE = 0, TRANSFER = 1, DONE = 2;
    reg [1:0] state;

    // 内部寄存器
    reg [DATA_WIDTH-1:0] tx_shift;
    reg [DATA_WIDTH-1:0] rx_shift;
    reg [3:0] bit_cnt;
    reg [CLK_DIV-1:0] clk_div;
    wire sclk_edge = (clk_div == 0);

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            sclk <= cpol_cpha[2];
            mosi <= 1;
            cs_n <= 1;
            busy <= 0;
            bit_cnt <= 0;
            clk_div <= 0;
        end else begin
            case (state)
                IDLE: begin
                    cs_n <= 1;
                    sclk <= cpol_cpha[2];
                    busy <= 0;
                    if (start) begin
                        state <= TRANSFER;
                        tx_shift <= tx_data;
                        rx_shift <= 0;
                        bit_cnt <= DATA_WIDTH - 1;
                        cs_n <= 0;
                        busy <= 1;
                        clk_div <= 0;
                    end
                end

                TRANSFER: begin
                    if (sclk_edge) begin
                        // 时钟翻转
                        sclk <= ~sclk;

                        // 根据CPHA决定采样和更新时机
                        if (cpol_cpha[0] == 0) begin
                            // CPHA=0: 第一个边沿采样
                            if (sclk == cpol_cpha[2]) begin
                                // 上升沿/下降沿采样
                                rx_shift[bit_cnt] <= miso;
                            end else begin
                                // 更新MOSI
                                mosi <= tx_shift[bit_cnt];
                                if (bit_cnt == 0) begin
                                    state <= DONE;
                                end else begin
                                    bit_cnt <= bit_cnt - 1;
                                end
                            end
                        end else begin
                            // CPHA=1: 第二个边沿采样
                            if (sclk == cpol_cpha[2]) begin
                                // 更新MOSI
                                mosi <= tx_shift[bit_cnt];
                                if (bit_cnt == 0) begin
                                    state <= DONE;
                                end else begin
                                    bit_cnt <= bit_cnt - 1;
                                end
                            end else begin
                                // 采样MISO
                                rx_shift[bit_cnt] <= miso;
                            end
                        end
                    end
                    clk_div <= (clk_div >= CLK_DIV-1) ? 0 : clk_div + 1;
                end

                DONE: begin
                    cs_n <= 1;
                    busy <= 0;
                    state <= IDLE;
                end
            endcase
        end
    end

    // 输出接收数据
    assign rx_data = rx_shift;

endmodule

🔌 SPI从机模块设计

// SPI从机模块
module spi_slave #(
    parameter DATA_WIDTH = 8
) (
    input  wire                  clk,
    input  wire                  rst_n,
    input  wire                  sclk,           // 串行时钟
    input  wire                  mosi,           // 主出从入
    output reg                   miso,           // 主入从出
    input  wire                  cs_n,           // 片选（低有效）
    input  wire [DATA_WIDTH-1:0] tx_data,        // 发送数据
    output reg  [DATA_WIDTH-1:0] rx_data,        // 接收数据
    output reg                   data_valid      // 数据有效
);

    // 边沿检测
    reg sclk_prev;
    wire sclk_rising = sclk & ~sclk_prev;
    wire sclk_falling = ~sclk & sclk_prev;

    // 同步CS信号
    reg cs_prev;
    wire cs_falling = cs_prev & ~cs_n;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            sclk_prev <= 0;
            cs_prev <= 1;
            miso <= 1;
            data_valid <= 0;
        end else begin
            sclk_prev <= sclk;
            cs_prev <= cs_n;
            data_valid <= 0;
        end
    end

    // 移位寄存器
    reg [DATA_WIDTH-1:0] tx_shift;
    reg [DATA_WIDTH-1:0] rx_shift;
    reg [3:0] bit_cnt;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            tx_shift <= 0;
            rx_shift <= 0;
            bit_cnt <= 0;
        end else begin
            if (cs_falling) begin
                // CS下降沿，初始化传输
                tx_shift <= tx_data;
                rx_shift <= 0;
                bit_cnt <= DATA_WIDTH - 1;
            end else if (cs_n == 0) begin
                // CS有效，进行数据传输
                if (sclk_rising) begin
                    // 上升沿：采样MOSI
                    rx_shift <= {rx_shift[DATA_WIDTH-2:0], mosi};
                    bit_cnt <= bit_cnt - 1;
                    if (bit_cnt == 0) begin
                        rx_data <= {rx_shift[DATA_WIDTH-2:0], mosi};
                        data_valid <= 1;
                    end
                end
                if (sclk_falling) begin
                    // 下降沿：更新MISO
                    miso <= tx_shift[bit_cnt];
                end
            end
        end
    end

endmodule

🌐 多从设备通信

硬件连接方式

SPI支持一个主设备连接多个从设备，有两种连接方式：

1. 独立片选线：每个从设备有独立的CS线，硬件简单，需要较多IO引脚。

2. 级联方式：从设备串联，使用一条CS线，节省IO但通信延迟增加。

💡
多设备控制

• 独立CS方式可同时与多个设备通信
• CS信号应确保不通信时为高电平
• 通信时应先拉低CS，传输结束后拉高

📊 SPI配置寄存器

控制寄存器（SPI_CTRL）

| 位 | 名称 | 描述 |

| [31:16] | CLK_DIV | 时钟分频值 |

| [3] | CPOL | 时钟极性 |

| [2] | CPHA | 时钟相位 |

| [1] | MSB_FIRST | 1=MSB先传, 0=LSB先传 |

| [0] | ENABLE | SPI使能 |

状态寄存器（SPI_STATUS）

| 位 | 名称 | 描述 |

| [3] | TX_EMPTY | 发送FIFO空 |

| [2] | TX_FULL | 发送FIFO满 |

| [1] | RX_AVAIL | 接收数据可用 |

| [0] | BUSY | 传输忙 |

🐛 常见问题与调试

⚠️
常见问题

1. 工作模式不匹配
症状：数据错误或无法通信
解决：检查CPOL和CPHA设置，确保主从一致

2. 时钟频率过高
症状：数据不稳定或出错
解决：降低SCLK频率，增加建立保持时间

3. CS时序错误
症状：通信异常
解决：确保CS在SCLK变化前稳定，传输结束后释放

4. 多设备冲突
症状：多个设备同时响应
解决：确保同一时间只有一个CS有效
