48.5 缓存拦截

使用拦截器实现响应缓存

🎯 学习目标

  • 理解缓存拦截器原理
  • 实现简单的内存缓存
  • 学会使用 Redis 缓存
  • 掌握缓存装饰器

📚 缓存拦截器原理

缓存拦截器在请求到达控制器前检查缓存，如果命中则直接返回缓存数据，否则执行控制器方法并缓存结果。

缓存流程： 请求 ↓ 检查缓存 ├─ 命中 → 返回缓存数据 └─ 未命中 ↓ 执行控制器 ↓ 缓存结果 ↓ 返回响应

🔧 内存缓存拦截器

import { Injectable, NestInterceptor, ExecutionContext, CallHandler } from '@nestjs/common'; import { Observable, of } from 'rxjs'; import { tap } from 'rxjs/operators'; import { Reflector } from '@nestjs/core'; export const CACHE_KEY = 'cache'; export const Cache = (ttl: number = 60) => SetMetadata(CACHE_KEY, { ttl }); @Injectable() export class CacheInterceptor implements NestInterceptor { private cache = new Map(); constructor(private reflector: Reflector) {} intercept(context: ExecutionContext, next: CallHandler): Observable { const cacheConfig = this.reflector.get(CACHE_KEY, context.getHandler()); if (!cacheConfig) { return next.handle(); } const request = context.switchToHttp().getRequest(); const cacheKey = this.generateCacheKey(request); const cached = this.cache.get(cacheKey); // 检查缓存是否存在且未过期 if (cached && cached.expireAt > Date.now()) { console.log(`Cache hit: ${cacheKey}`); return of(cached.data); } console.log(`Cache miss: ${cacheKey}`); return next.handle().pipe( tap((data) => { this.cache.set(cacheKey, { data, expireAt: Date.now() + cacheConfig.ttl * 1000, }); }), ); } private generateCacheKey(request: Request): string { const { method, url, body } = request; return `${method}:${url}:${JSON.stringify(body)}`; } } // 使用装饰器 @Controller('users') export class UsersController { @Cache(60) // 缓存 60 秒 @Get() findAll() { return this.usersService.findAll(); } @Cache(300) // 缓存 5 分钟 @Get(':id') findOne(@Param('id') id: string) { return this.usersService.findOne(+id); } }
💡
of 操作符

of() 创建一个立即发射指定值的 Observable。缓存命中时使用 of(cachedData) 直接返回缓存数据，跳过控制器执行。

📦 Redis 缓存拦截器

import { Injectable, NestInterceptor, ExecutionContext, CallHandler } from '@nestjs/common'; import { Observable, of, from } from 'rxjs'; import { switchMap, tap } from 'rxjs/operators'; import { RedisService } from './redis.service'; @Injectable() export class RedisCacheInterceptor implements NestInterceptor { constructor( private redisService: RedisService, private reflector: Reflector, ) {} intercept(context: ExecutionContext, next: CallHandler): Observable { const cacheConfig = this.reflector.get(CACHE_KEY, context.getHandler()); if (!cacheConfig) { return next.handle(); } const request = context.switchToHttp().getRequest(); const cacheKey = this.generateCacheKey(request); return from(this.redisService.get(cacheKey)).pipe( switchMap((cached) => { if (cached) { console.log(`Cache hit: ${cacheKey}`); return of(JSON.parse(cached)); } console.log(`Cache miss: ${cacheKey}`); return next.handle().pipe( tap((data) => { this.redisService.set( cacheKey, JSON.stringify(data), cacheConfig.ttl, ); }), ); }), ); } private generateCacheKey(request: Request): string { return `cache:${request.method}:${request.url}`; } } // Redis 服务 @Injectable() export class RedisService { private client: Redis; constructor() { this.client = new Redis({ host: process.env.REDIS_HOST, port: parseInt(process.env.REDIS_PORT, 10), }); } async get(key: string): Promise { return this.client.get(key); } async set(key: string, value: string, ttl: number): Promise { await this.client.setex(key, ttl, value); } async del(key: string): Promise { await this.client.del(key); } async delPattern(pattern: string): Promise { const keys = await this.client.keys(pattern); if (keys.length > 0) { await this.client.del(...keys); } } }

✨ 缓存清除装饰器

export const CACHE_CLEAR_KEY = 'cacheClear'; export const CacheClear = (...patterns: string[]) => SetMetadata(CACHE_CLEAR_KEY, patterns); @Injectable() export class CacheClearInterceptor implements NestInterceptor { constructor(private redisService: RedisService) {} intercept(context: ExecutionContext, next: CallHandler): Observable { const patterns = this.reflector.get(CACHE_CLEAR_KEY, context.getHandler()); return next.handle().pipe( tap(async () => { if (patterns) { for (const pattern of patterns) { await this.redisService.delPattern(`cache:${pattern}*`); } } }), ); } } // 使用 @Controller('users') export class UsersController { @Cache(60) @Get() findAll() {} // 缓存结果 @CacheClear('GET:/users') // 清除用户列表缓存 @Post() create(@Body() createUserDto: CreateUserDto) { return this.usersService.create(createUserDto); } @CacheClear('GET:/users', 'GET:/users/:id') @Put(':id') update(@Param('id') id: string, @Body() updateUserDto: UpdateUserDto) { return this.usersService.update(+id, updateUserDto); } }
⚠️
缓存注意

缓存拦截器应该只用于 GET 请求。POST、PUT、DELETE 等修改操作不应该缓存。确保缓存键能唯一标识请求。
