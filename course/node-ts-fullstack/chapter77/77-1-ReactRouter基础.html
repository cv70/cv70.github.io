📚 前端路由概念

// 什么是前端路由？ // - 在单页应用中，通过 JavaScript 控制页面显示 // - URL 变化时，不刷新页面，只更新视图 // - 提供类似多页应用的导航体验 // 两种路由模式 // 1. Hash 路由：URL 中有 #，如 /#/home // - 兼容性好，不需要服务器配置 // 2. History 路由：正常的 URL，如 /home // - 需要 server 端配置支持 // 安装 React Router // npm install react-router-dom // 基本结构 import { createBrowserRouter, RouterProvider } from 'react-router-dom'; // 创建路由配置 const router = createBrowserRouter([ { path: '/', element: <Home />, }, { path: '/about', element: <About />, }, { path: '/users', element: <Users />, }, ]); // 在应用中使用 function App() { return <RouterProvider router={router} />; }

🔧 路由导航

import { Link, useNavigate } from 'react-router-dom'; // Link 组件：声明式导航 function Navigation() { return ( <nav> <Link to="/">Home</Link> <Link to="/about">About</Link> <Link to="/users">Users</Link> </nav> ); } // useNavigate：编程式导航 function LoginForm() { const navigate = useNavigate(); const handleSubmit = async (e) => { e.preventDefault(); const success = await login(credentials); if (success) { navigate('/dashboard'); // 跳转到仪表盘 // navigate(-1); // 返回上一页 // navigate('/users', { replace: true }); // 替换当前历史记录 } }; return <form onSubmit={handleSubmit}>...</form>; } // NavLink：带激活状态的链接 function Navigation() { return ( <nav> <NavLink to="/" className={({ isActive }) => isActive ? 'active' : ''} > Home </NavLink> <NavLink to="/about" style={({ isActive }) => ({ color: isActive ? 'red' : 'black', })} > About </NavLink> </nav> ); } // Navigate 组件：重定向 function ProtectedRoute({ children }) { const { isAuthenticated } = useAuth(); if (!isAuthenticated) { return <Navigate to="/login" replace />; } return children; }

📊 获取路由信息

import { useParams, useLocation, useSearchParams } from 'react-router-dom'; // useParams：获取 URL 参数 function UserProfile() { const { userId } = useParams(); // 从 /users/:userId 获取 return <div>User ID: {userId}</div>; } // useLocation：获取当前位置信息 function Analytics() { const location = useLocation(); useEffect(() => { // 记录页面访问 logPageView(location.pathname); }, [location]); return null; } // location 对象 // { // pathname: '/users/123', // search: '?tab=posts', // hash: '#comments', // state: { from: 'home' }, // } // useSearchParams：获取查询参数 function SearchPage() { const [searchParams, setSearchParams] = useSearchParams(); const query = searchParams.get('q'); // 获取 ?q=value const page = searchParams.get('page') || '1'; const handleSearch = (newQuery) => { setSearchParams({ q: newQuery, page: '1' }); }; return ( <div> <p>Search: {query}</p> <p>Page: {page}</p> </div> ); } // 使用 state 传递数据 function UserList() { const navigate = useNavigate(); const handleView = (userId) => { navigate(`/users/${userId}`, { state: { from: 'user-list', timestamp: Date.now() } }); }; return <button onClick={() => handleView(1)}>View</button>; } function UserDetail() { const location = useLocation(); const from = location.state?.from; // 'user-list' }
