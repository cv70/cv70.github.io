34.2 设置响应头

掌握 Express 响应头的设置方法

🎯 学习目标

  • 掌握 res.set() 和 res.setHeader() 的用法
  • 了解常用响应头的作用
  • 学会设置 CORS 相关响应头
  • 理解缓存相关响应头

📝 设置响应头方法

// res.set() 或 res.setHeader() // 设置单个响应头 app.get('/single', (req, res) => { res.set('Content-Type', 'application/json'); res.set('X-Custom-Header', 'Custom Value'); res.json({ message: 'OK' }); }); // 设置多个响应头 app.get('/multiple', (req, res) => { res.set({ 'Content-Type': 'application/json', 'X-API-Version': '1.0', 'X-Request-ID': 'abc123' }); res.json({ message: 'OK' }); }); // res.append() - 追加响应头 app.get('/append', (req, res) => { res.append('Set-Cookie', 'name=value'); res.append('Set-Cookie', 'token=abc123'); res.send('Cookies set'); }); // res.removeHeader() - 移除响应头 app.get('/remove', (req, res) => { res.removeHeader('X-Powered-By'); res.send('Header removed'); });

📋 常用响应头

app.get('/headers', (req, res) => { res.set({ // 内容类型 'Content-Type': 'application/json; charset=utf-8', 'Content-Encoding': 'gzip', // 缓存控制 'Cache-Control': 'public, max-age=3600', 'ETag': 'abc123', 'Last-Modified': new Date().toUTCString(), 'Expires': new Date(Date.now() + 3600000).toUTCString(), // 安全相关 'X-Content-Type-Options': 'nosniff', 'X-Frame-Options': 'DENY', 'X-XSS-Protection': '1; mode=block', 'Strict-Transport-Security': 'max-age=31536000; includeSubDomains', // 自定义头 'X-API-Version': '1.0', 'X-Request-ID': 'abc123' }); res.json({ message: 'Headers set' }); });

🌐 CORS 响应头

// 手动设置 CORS app.get('/api', (req, res) => { res.set({ 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type, Authorization', 'Access-Control-Allow-Credentials': 'true', 'Access-Control-Max-Age': '86400' }); res.json({ message: 'CORS enabled' }); }); // 预检请求处理 app.options('/api', (req, res) => { res.set({ 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE', 'Access-Control-Allow-Headers': 'Content-Type, Authorization' }); res.sendStatus(204); }); // 或使用 cors 中间件 const cors = require('cors'); app.use(cors());

📦 缓存响应头

// 禁用缓存 app.get('/no-cache', (req, res) => { res.set({ 'Cache-Control': 'no-store, no-cache, must-revalidate', 'Pragma': 'no-cache', 'Expires': '0' }); res.json({ data: 'fresh' }); }); // 启用缓存 app.get('/cached', (req, res) => { res.set({ 'Cache-Control': 'public, max-age=3600', 'ETag': 'v1', 'Last-Modified': new Date().toUTCString() }); res.json({ data: 'cached' }); }); // 条件请求 app.get('/conditional', (req, res) => { const etag = 'v1'; const lastModified = new Date('2024-01-01'); // 检查客户端缓存 if (req.headers['if-none-match'] === etag) { return res.sendStatus(304); // Not Modified } if (req.headers['if-modified-since']) { const clientDate = new Date(req.headers['if-modified-since']); if (clientDate >= lastModified) { return res.sendStatus(304); } } res.set('ETag', etag); res.set('Last-Modified', lastModified.toUTCString()); res.json({ data: 'fresh' }); });

✅ 小结

📌
本节要点
  • res.set() 设置响应头，支持单个和批量设置
  • res.append() 追加响应头，不覆盖已有值
  • CORS 响应头用于跨域访问控制
  • 缓存响应头用于控制客户端缓存行为
