📚 gRPC概述

// gRPC是Google开源的高性能RPC框架 const features = { protocol: 'HTTP/2作为传输协议', serialization: 'Protocol Buffers序列化', streaming: '支持双向流式通信', multiplexing: '支持多路复用', headerCompression: '头部压缩减少开销', flowControl: '内置流量控制' }; // gRPC优势 const advantages = [ '高性能：二进制协议，序列化快', '强类型：IDL定义接口，编译检查', '跨语言：支持多种编程语言', '双向流：支持客户端和服务端流', '代码生成：自动生成客户端和服务端代码' ]; // gRPC vs REST性能对比 const performance = { gRPC: { latency: '~20-30ms', throughput: '~10000 req/s', payload: '二进制，体积小' }, REST: { latency: '~50-100ms', throughput: '~2000 req/s', payload: 'JSON文本，体积大' } };

gRPC适用场景

微服务间通信、实时通信、高性能API、跨语言服务调用。

🔧 四种通信模式

// 1. 一元RPC（Unary RPC） // 客户端发送一个请求，服务端返回一个响应 service UserService { rpc GetUser(GetUserRequest) returns (User); } // 2. 服务端流RPC（Server Streaming RPC） // 客户端发送一个请求，服务端返回流式响应 service OrderService { rpc ListOrders(ListOrdersRequest) returns (stream Order); } // 3. 客户端流RPC（Client Streaming RPC） // 客户端发送流式请求，服务端返回一个响应 service UploadService { rpc UploadFile(stream FileChunk) returns (UploadResult); } // 4. 双向流RPC（Bidirectional Streaming RPC） // 双方都可以发送流 service ChatService { rpc Chat(stream Message) returns (stream Message); } // 使用场景 const useCases = { unary: ['简单查询', '创建资源', '更新资源'], serverStream: ['大量数据返回', '实时推送', '日志流'], clientStream: ['文件上传', '批量数据提交'], bidirectionalStream: ['聊天应用', '实时协作', '游戏同步'] };

🔀 Proto文件定义

// user.proto syntax = "proto3"; package user; // 导入其他proto文件 import "google/protobuf/timestamp.proto"; // 服务定义 service UserService { // 一元RPC rpc GetUser(GetUserRequest) returns (User); rpc CreateUser(CreateUserRequest) returns (User); rpc UpdateUser(UpdateUserRequest) returns (User); rpc DeleteUser(DeleteUserRequest) returns (Empty); // 服务端流 rpc ListUsers(ListUsersRequest) returns (stream User); // 双向流 rpc WatchUsers(WatchRequest) returns (stream UserEvent); } // 消息定义 message User { int32 id = 1; string name = 2; string email = 3; repeated string roles = 4; google.protobuf.Timestamp created_at = 5; } message GetUserRequest { int32 id = 1; } message CreateUserRequest { string name = 1; string email = 2; repeated string roles = 3; } message UpdateUserRequest { int32 id = 1; string name = 2; string email = 3; } message DeleteUserRequest { int32 id = 1; } message ListUsersRequest { int32 page = 1; int32 page_size = 2; string filter = 3; } message Empty {} message WatchRequest { repeated int32 user_ids = 1; } message UserEvent { string event_type = 1; // created, updated, deleted User user = 2; }

📝 本节小结

✅

  • • 自动生成代码，支持多语言
