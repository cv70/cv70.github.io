83.3 State 与 Getter

Pinia 状态管理核心

📖 本节概述

State 是 Store 中存储数据的地方，Getter 类似于计算属性，用于派生状态。本节将详细介绍如何定义和使用 State 与 Getter。

🎯 学习目标

  • 掌握 State 的定义和访问方式
  • 学会修改 State 的多种方法
  • 掌握 Getter 的定义和使用
  • 了解 State 的订阅和重置

📦 State 定义

Options Store

interface UserState { name: string; age: number; email: string; preferences: { theme: 'light' | 'dark'; language: string; }; } export const useUserStore = defineStore('user', { // state 必须是返回初始状态的函数 state: (): UserState => ({ name: '', age: 0, email: '', preferences: { theme: 'light', language: 'zh-CN' } }) });

Setup Store

export const useUserStore = defineStore('user', () => { // 使用 ref 或 reactive const name = ref(''); const age = ref(0); const email = ref(''); const preferences = reactive({ theme: 'light' as 'light' | 'dark', language: 'zh-CN' }); return { name, age, email, preferences }; });

📖 访问 State

<script setup lang="ts"> import { useUserStore } from '@/stores/user'; import { storeToRefs } from 'pinia'; const userStore = useUserStore(); // 直接访问 console.log(userStore.name); console.log(userStore.age); // 解构（需要使用 storeToRefs 保持响应性） const { name, age } = storeToRefs(userStore); // 错误：直接解构会失去响应性 const { name, age } = userStore; // name, age 不是响应式的！ </script> <template> <div> <p>Name: {{ userStore.name }}</p> <p>Age: {{ age }}</p> </div> </template>

✏️ 修改 State

直接修改

const userStore = useUserStore(); // 直接修改 userStore.name = 'John'; userStore.age = 25; // 修改嵌套对象 userStore.preferences.theme = 'dark';

$patch 批量修改

// 使用对象批量修改 userStore.$patch({ name: 'John', age: 25, email: 'john@example.com' }); // 使用函数批量修改（适合复杂逻辑） userStore.$patch((state) => { state.name = 'John'; state.age = 25; state.preferences.theme = 'dark'; state.preferences.language = 'en-US'; });

重置 State

// 重置到初始状态（仅 Options Store） userStore.$reset(); // Setup Store 需要自己实现重置 export const useUserStore = defineStore('user', () => { const name = ref(''); const age = ref(0); function $reset() { name.value = ''; age.value = 0; } return { name, age, $reset }; });

替换整个 State

userStore.$state = { name: 'Jane', age: 30, email: 'jane@example.com', preferences: { theme: 'dark', language: 'zh-CN' } };

👂 订阅 State 变化

// 订阅 state 变化 userStore.$subscribe((mutation, state) => { // mutation.type: 'direct' | 'patch object' | 'patch function' // mutation.storeId: store 的 id // mutation.payload: $patch 的参数 console.log('Type:', mutation.type); console.log('Store ID:', mutation.storeId); // 持久化到 localStorage localStorage.setItem('user', JSON.stringify(state)); }, { detached: true }); // detached: 组件卸载后继续监听

🔧 Getter 定义

Options Store

export const useUserStore = defineStore('user', { state: () => ({ firstName: '', lastName: '', age: 0 }), getters: { // 简单 getter fullName(state): string { return `${state.firstName} ${state.lastName}`; }, // 使用 this 访问其他 getter greeting(): string { return `Hello, ${this.fullName}!`; }, // 判断类 getter isAdult(state): boolean { return state.age >= 18; } } });

Setup Store

export const useUserStore = defineStore('user', () => { const firstName = ref(''); const lastName = ref(''); const age = ref(0); // 使用 computed const fullName = computed(() => `${firstName.value} ${lastName.value}`); const greeting = computed(() => `Hello, ${fullName.value}!`); const isAdult = computed(() => age.value >= 18); return { firstName, lastName, age, fullName, greeting, isAdult }; });

📖 访问 Getter

const userStore = useUserStore(); // 直接访问 console.log(userStore.fullName); console.log(userStore.greeting); console.log(userStore.isAdult); // 解构 const { fullName, greeting } = storeToRefs(userStore); <!-- 模板中使用 --> <template> <div> <p>{{ userStore.fullName }}</p> <p>{{ userStore.greeting }}</p> <p v-if="userStore.isAdult">成人</p> </div> </template>

📊 Getter 传参

// Options Store export const useProductsStore = defineStore('products', { state: () => ({ products: [] as Product[] }), getters: { // 返回函数来实现传参 getById(state) { return (id: number) => { return state.products.find(p => p.id === id); }; }, filterByCategory(state) { return (category: string) => { return state.products.filter(p => p.category === category); }; } } }); // Setup Store export const useProductsStore = defineStore('products', () => { const products = ref<Product[]>([]); // 普通函数即可 function getById(id: number) { return products.value.find(p => p.id === id); } function filterByCategory(category: string) { return products.value.filter(p => p.category === category); } return { products, getById, filterByCategory }; }); // 使用 const product = productsStore.getById(1); const electronics = productsStore.filterByCategory('electronics');

📝 本节小结

  • State 使用 state() 函数或 ref/reactive 定义
  • 解构 State 需要使用 storeToRefs 保持响应性
  • 修改 State 可以直接修改或使用 $patch
  • $reset 可以重置到初始状态
  • Getter 类似计算属性，支持传参
