🔧 express.static 配置

const express = require('express'); const app = express(); // 基本配置 app.use(express.static('public')); // 带路径前缀 app.use('/static', express.static('public')); // 访问: /static/css/main.css // 多个静态目录 app.use(express.static('public')); app.use(express.static('uploads')); // 完整配置 app.use('/assets', express.static('public', { dotfiles: 'ignore', // 隐藏文件处理 etag: true, // 启用 ETag extensions: ['html'], // 默认扩展名 fallthrough: true, // 404 时继续 immutable: false, // immutable 缓存 index: 'index.html', // 默认索引文件 lastModified: true, // Last-Modified 头 maxAge: 0, // 缓存时间 redirect: true, // 目录重定向 setHeaders: (res, path) => { // 自定义响应头 if (path.endsWith('.js')) { res.set('X-Content-Type-Options', 'nosniff'); } } }));

⚡ 缓存策略

// 开发环境 - 禁用缓存 if (process.env.NODE_ENV === 'development') { app.use(express.static('public', { maxAge: 0, etag: false, lastModified: false })); } // 生产环境 - 启用缓存 if (process.env.NODE_ENV === 'production') { app.use(express.static('public', { maxAge: '1y', // 缓存一年 immutable: true, // 文件内容不变 etag: true })); } // 按文件类型设置不同缓存 app.use(express.static('public', { setHeaders: (res, path) => { // HTML 不缓存 if (path.endsWith('.html')) { res.set('Cache-Control', 'no-cache'); } // CSS/JS 长期缓存 else if (path.match(/\.(css|js)$/)) { res.set('Cache-Control', 'public, max-age=31536000, immutable'); } // 图片中期缓存 else if (path.match(/\.(png|jpg|jpeg|gif|ico|svg|webp)$/)) { res.set('Cache-Control', 'public, max-age=2592000'); // 30天 } // 字体长期缓存 else if (path.match(/\.(woff|woff2|ttf|eot)$/)) { res.set('Cache-Control', 'public, max-age=31536000, immutable'); } } }));

🔗 CDN 集成

// 模板中使用 CDN app.locals.assetUrl = (path) => { if (process.env.NODE_ENV === 'production') { return `https://cdn.example.com${path}`; } return path; }; // EJS 模板 <link rel="stylesheet" href="<%= assetUrl('/css/main.css') %>"> <script src="<%= assetUrl('/js/main.js') %>"></script> // 使用版本号防止缓存 app.locals.assetVersion = Date.now(); // 或使用文件 hash const getAssetHash = () => { // 构建时生成的 hash return process.env.ASSET_HASH || ''; }; // 模板 <link rel="stylesheet" href="/css/main.css?v=<%= assetVersion %>">

📦 构建工具集成

// 使用 Webpack/Vite 生成的 manifest const manifest = require('./public/manifest.json'); app.locals.getAssetPath = (name) => { return manifest[name] || name; }; // manifest.json 示例 { "main.css": "/css/main.a1b2c3.css", "main.js": "/js/main.x1y2z3.js" } // 模板中使用 <link rel="stylesheet" href="<%= getAssetPath('main.css') %>"> <script src="<%= getAssetPath('main.js') %>"></script>
