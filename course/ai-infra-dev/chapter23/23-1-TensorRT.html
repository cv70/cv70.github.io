什么是TensorRT

TensorRT是NVIDIA推出的高性能深度学习推理优化SDK。 它可以将训练好的模型转换为优化后的推理引擎，通过层融合、精度校准、内核自动调整等技术， 在NVIDIA GPU上实现极致的推理性能。

💻 使用示例

import tensorrt as trt
import pycuda.driver as cuda
import pycuda.autoinit
import numpy as np
import torch

# 将PyTorch模型转换为TensorRT引擎
def build_engine(onnx_model_path, precision='fp16'):
    """
    构建TensorRT引擎
    """
    # 创建logger
    logger = trt.Logger(trt.Logger.WARNING)

    # 创建builder
    builder = trt.Builder(logger)
    network = builder.create_network(
        1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH)
    )
    parser = trt.OnnxParser(network, logger)

    # 解析ONNX模型
    with open(onnx_model_path, 'rb') as model:
        parser.parse(model.read())

    # 创建配置
    config = builder.create_builder_config()

    # 设置精度
    if precision == 'fp16':
        config.set_flag(trt.BuilderFlag.FP16)
    elif precision == 'int8':
        config.set_flag(trt.BuilderFlag.INT8)
        # 设置INT8校准器
        # calibrator = MyCalibrator(...)
        # config.int8_calibrator = calibrator

    # 设置最大工作空间
    config.set_memory_pool_limit(
        trt.MemoryPoolType.WORKSPACE, 1 << 30  # 1GB
    )

    # 构建引擎
    engine = builder.build_serialized_network(network, config)

    return engine

# 使用TensorRT推理
class TensorRTInference:
    def __init__(self, engine_path):
        self.logger = trt.Logger(trt.Logger.WARNING)

        # 加载引擎
        with open(engine_path, 'rb') as f:
            self.engine = trt.Runtime(self.logger).deserialize_cuda_engine(f.read())

        # 创建执行上下文
        self.context = self.engine.create_execution_context()

        # 分配GPU内存
        self.bindings = []
        self.outputs = []
        for i in range(self.engine.num_io_tensors):
            tensor_name = self.engine.get_tensor_name(i)
            size = trt.volume(self.engine.get_tensor_shape(tensor_name))
            dtype = trt.nptype(self.engine.get_tensor_dtype(tensor_name))

            # 分配内存
            host_mem = cuda.pagelocked_empty(size, dtype)
            device_mem = cuda.mem_alloc(host_mem.nbytes)

            self.bindings.append(int(device_mem))
            if self.engine.get_tensor_mode(tensor_name) == trt.TensorIOMode.OUTPUT:
                self.outputs.append((host_mem, device_mem))

    def infer(self, input_data):
        """
        执行推理
        """
        # 将输入数据传输到GPU
        cuda.memcpy_htod(self.bindings[0], input_data)

        # 执行推理
        self.context.execute_v2(self.bindings)

        # 将输出数据传输到CPU
        outputs = []
        for host_mem, device_mem in self.outputs:
            cuda.memcpy_dtoh(host_mem, device_mem)
            outputs.append(host_mem)

        return outputs

# PyTorch中使用
# 1. 导出ONNX
dummy_input = torch.randn(1, 3, 224, 224).cuda()
torch.onnx.export(
    model,
    dummy_input,
    "model.onnx",
    input_names=['input'],
    output_names=['output']
)

# 2. 构建TensorRT引擎
engine = build_engine("model.onnx", precision='fp16')

# 3. 保存引擎
with open("model.trt", 'wb') as f:
    f.write(engine)

# 4. 加载并使用
inferencer = TensorRTInference("model.trt")
output = inferencer.infer(input_data)

🤖 TensorRT-LLM

大语言模型推理优化

  • INT4/INT8量化：支持GPTQ/AWQ等量化方法
  • Flash Attention：加速注意力计算
  • Inflight Batching：连续批处理
  • Paged KV Cache：动态KV缓存管理
  • Fused Operators：融合算子优化
  • 多GPU支持：张量并行和流水线并行
💡
TensorRT使用建议
  • 优先使用FP16精度，在精度和性能间取得平衡
  • INT8需要校准数据，建议使用有代表性的数据集
  • 构建引擎时设置足够的工作空间（1-4GB）
  • 使用动态形状时需要指定优化配置文件
