87.3 Loader 配置

Webpack Loader 详解

🎯 学习目标

  • 理解 Loader 的作用和工作原理
  • 掌握常用 Loader 配置
  • 学会处理各种资源类型
  • 了解 Loader 执行顺序

📚 Loader 概述

Loader 用于处理非 JavaScript 文件，将各种资源转换为 Webpack 能处理的模块。

# Loader 工作原理 源文件 → Loader 转换 → JavaScript 模块 # 配置语法 module.exports = { module: { rules: [ { test: /\.css$/, // 匹配文件 use: ['style-loader', 'css-loader'], // 使用的 loader exclude: /node_modules/, // 排除目录 include: /src/, // 包含目录 }, ], }, }; # Loader 执行顺序 从右到左，从下到上 ['style-loader', 'css-loader', 'sass-loader'] 执行顺序: sass-loader → css-loader → style-loader
💡
链式调用

多个 Loader 可以链式调用，前一个 Loader 的输出是后一个 Loader 的输入，最终输出 JavaScript 模块。

🔧 CSS 处理

# 安装依赖 npm install --save-dev \ style-loader \ css-loader \ sass \ sass-loader \ postcss \ postcss-loader \ autoprefixer // webpack.config.js module.exports = { module: { rules: [ // CSS 文件 { test: /\.css$/, use: ['style-loader', 'css-loader'], }, // Sass/SCSS 文件 { test: /\.scss$/, use: [ 'style-loader', 'css-loader', { loader: 'sass-loader', options: { sourceMap: true, }, }, ], }, // PostCSS 处理 { test: /\.css$/, use: [ 'style-loader', { loader: 'css-loader', options: { modules: true, // CSS Modules }, }, { loader: 'postcss-loader', options: { postcssOptions: { plugins: [ require('autoprefixer'), require('cssnano')({ preset: 'default' }), ], }, }, }, ], }, ], }, }; // postcss.config.js module.exports = { plugins: [ require('autoprefixer')({ overrideBrowserslist: ['> 1%', 'last 2 versions'], }), ], };

⚙️ TypeScript 和 JavaScript

# 安装依赖 npm install --save-dev \ typescript \ ts-loader \ @babel/core \ @babel/preset-env \ @babel/preset-typescript \ babel-loader // webpack.config.js module.exports = { module: { rules: [ // TypeScript (ts-loader) { test: /\.tsx?$/, use: 'ts-loader', exclude: /node_modules/, }, // TypeScript (babel-loader) { test: /\.tsx?$/, exclude: /node_modules/, use: { loader: 'babel-loader', options: { presets: [ ['@babel/preset-env', { targets: 'defaults' }], '@babel/preset-typescript', ], }, }, }, // JavaScript (Babel) { test: /\.m?js$/, exclude: /node_modules/, use: { loader: 'babel-loader', options: { presets: ['@babel/preset-env'], plugins: ['@babel/plugin-transform-runtime'], }, }, }, ], }, resolve: { extensions: ['.tsx', '.ts', '.js'], }, }; // babel.config.js module.exports = { presets: [ ['@babel/preset-env', { modules: false }], '@babel/preset-typescript', ], };

🚀 资源文件处理

// Webpack 5 资源模块 (Asset Modules) module.exports = { module: { rules: [ // 图片资源 { test: /\.(png|jpg|jpeg|gif|svg|webp)$/i, type: 'asset/resource', generator: { filename: 'images/[hash][ext][query]', }, // 小于 8KB 转为 base64 parser: { dataUrlCondition: { maxSize: 8 * 1024, // 8KB }, }, }, // 字体文件 { test: /\.(woff|woff2|eot|ttf|otf)$/i, type: 'asset/resource', generator: { filename: 'fonts/[hash][ext][query]', }, }, // 音视频文件 { test: /\.(mp3|mp4|wav|ogg)$/i, type: 'asset/resource', generator: { filename: 'media/[hash][ext][query]', }, }, // 数据文件 (JSON, CSV, XML) { test: /\.json$/i, type: 'json', }, { test: /\.(csv|tsv)$/i, use: ['csv-loader'], }, { test: /\.xml$/i, use: ['xml-loader'], }, ], }, }; // 在代码中使用 import logo from './assets/logo.png'; import './styles/main.scss'; const img = document.createElement('img'); img.src = logo; // 输出路径

📝 本节小结

✅
  • • Loader 用于转换非 JavaScript 模块
  • • 执行顺序从右到左、从下到上
  • • CSS 需要 style-loader 和 css-loader 配合
  • • Webpack 5 内置资源模块处理文件
