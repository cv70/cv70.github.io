📦 模块定义

聊天模块

// chat/chat.module.ts import { Module } from '@nestjs/common'; import { ChatGateway } from './chat.gateway'; import { ChatService } from './chat.service'; import { AuthModule } from '../auth/auth.module'; @Module({ imports: [AuthModule], // 导入认证模块 providers: [ChatGateway, ChatService], exports: [ChatService] }) export class ChatModule {} // chat/chat.gateway.ts import { WebSocketGateway, WebSocketServer } from '@nestjs/websockets'; import { Server } from 'socket.io'; import { ChatService } from './chat.service'; import { AuthService } from '../auth/auth.service'; @WebSocketGateway({ namespace: 'chat', cors: { origin: '*' } }) export class ChatGateway { @WebSocketServer() server: Server; constructor( private readonly chatService: ChatService, private readonly authService: AuthService ) {} } // chat/chat.service.ts import { Injectable } from '@nestjs/common'; @Injectable() export class ChatService { private rooms = new Map<string, any>(); async createRoom(name: string) { // 创建房间逻辑 } async getMessages(roomId: string) { // 获取消息逻辑 } }

通知模块

// notification/notification.module.ts import { Module } from '@nestjs/common'; import { NotificationGateway } from './notification.gateway'; import { NotificationService } from './notification.service'; @Module({ providers: [NotificationGateway, NotificationService], exports: [NotificationService] }) export class NotificationModule {} // notification/notification.gateway.ts import { WebSocketGateway, WebSocketServer } from '@nestjs/websockets'; import { Server } from 'socket.io'; @WebSocketGateway({ namespace: 'notifications', cors: { origin: '*' } }) export class NotificationGateway { @WebSocketServer() server: Server; // 通知相关事件处理 } // notification/notification.service.ts import { Injectable } from '@nestjs/common'; import { NotificationGateway } from './notification.gateway'; @Injectable() export class NotificationService { constructor(private readonly gateway: NotificationGateway) {} // 发送通知给特定用户 sendToUser(userId: string, notification: any) { this.gateway.server.to(`user:${userId}`).emit('notification', notification); } // 广播通知 broadcast(notification: any) { this.gateway.server.emit('notification', notification); } }

🔗 跨模块通信

使用 Event Emitter

// app.module.ts import { Module } from '@nestjs/common'; import { EventEmitterModule } from '@nestjs/event-emitter'; @Module({ imports: [ EventEmitterModule.forRoot(), ChatModule, NotificationModule ] }) export class AppModule {} // chat/chat.gateway.ts import { WebSocketGateway, SubscribeMessage } from '@nestjs/websockets'; import { EventEmitter2 } from '@nestjs/event-emitter'; @WebSocketGateway({ namespace: 'chat' }) export class ChatGateway { constructor(private eventEmitter: EventEmitter2) {} @SubscribeMessage('message') handleMessage(client: any, payload: any) { // 发送事件 this.eventEmitter.emit('message.created', { userId: client.data.userId, message: payload }); } } // notification/notification.service.ts import { Injectable, OnModuleInit } from '@nestjs/common'; import { OnEvent } from '@nestjs/event-emitter'; import { NotificationGateway } from './notification.gateway'; @Injectable() export class NotificationService { constructor(private readonly gateway: NotificationGateway) {} @OnEvent('message.created') handleMessageCreated(payload: any) { // 发送通知 this.gateway.server.emit('notification', { type: 'new_message', data: payload }); } }

🌐 全局网关配置

全局守卫和过滤器

// main.ts import { NestFactory } from '@nestjs/core'; import { AppModule } from './app.module'; import { IoAdapter } from '@nestjs/platform-socket.io'; import { WsJwtGuard } from './common/guards/ws-jwt.guard'; import { WsExceptionFilter } from './common/filters/ws-exception.filter'; async function bootstrap() { const app = await NestFactory.create(AppModule); // WebSocket 适配器 app.useWebSocketAdapter(new IoAdapter(app)); // 全局 WebSocket 守卫 app.useGlobalGuards(new WsJwtGuard()); // 全局 WebSocket 过滤器 app.useGlobalFilters(new WsExceptionFilter()); await app.listen(3000); } bootstrap(); // 或者使用 Provider // app.module.ts import { Module } from '@nestjs/common'; import { APP_GUARD, APP_FILTER } from '@nestjs/core'; import { WsJwtGuard } from './common/guards/ws-jwt.guard'; import { WsExceptionFilter } from './common/filters/ws-exception.filter'; @Module({ providers: [ { provide: APP_GUARD, useClass: WsJwtGuard }, { provide: APP_FILTER, useClass: WsExceptionFilter } ] }) export class AppModule {}

自定义 WebSocket 适配器

// common/adapters/socket.adapter.ts import { IoAdapter } from '@nestjs/platform-socket.io'; import { ServerOptions } from 'socket.io'; import { createAdapter } from '@socket.io/redis-adapter'; import { createClient } from 'redis'; export class RedisIoAdapter extends IoAdapter { private adapterConstructor: any; async connectToRedis(): Promise<void> { const pubClient = createClient({ url: 'redis://localhost:6379' }); const subClient = pubClient.duplicate(); await Promise.all([ pubClient.connect(), subClient.connect() ]); this.adapterConstructor = createAdapter(pubClient, subClient); } createIOServer(port: number, options?: ServerOptions): any { const server = super.createIOServer(port, options); server.adapter(this.adapterConstructor); return server; } } // main.ts async function bootstrap() { const app = await NestFactory.create(AppModule); const redisIoAdapter = new RedisIoAdapter(app); await redisIoAdapter.connectToRedis(); app.useWebSocketAdapter(redisIoAdapter); await app.listen(3000); }

🔄 完整示例

// app.module.ts import { Module } from '@nestjs/common'; import { EventEmitterModule } from '@nestjs/event-emitter'; import { AuthModule } from './auth/auth.module'; import { ChatModule } from './chat/chat.module'; import { NotificationModule } from './notification/notification.module'; import { GameModule } from './game/game.module'; @Module({ imports: [ EventEmitterModule.forRoot(), AuthModule, ChatModule, NotificationModule, GameModule ] }) export class AppModule {} // main.ts import { NestFactory } from '@nestjs/core'; import { AppModule } from './app.module'; import { IoAdapter } from '@nestjs/platform-socket.io'; import { Logger } from '@nestjs/common'; async function bootstrap() { const app = await NestFactory.create(AppModule); const logger = new Logger('Bootstrap'); // 启用 CORS app.enableCors({ origin: ['http://localhost:3000', 'http://localhost:5173'], credentials: true }); // WebSocket 适配器 app.useWebSocketAdapter(new IoAdapter(app)); const port = process.env.PORT || 3000; await app.listen(port); logger.log(`Application running on port ${port}`); } bootstrap();
