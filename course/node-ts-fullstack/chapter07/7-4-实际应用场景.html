7.4 实际应用场景

交叉类型在真实项目中的应用

🎯 学习目标

  • 掌握交叉类型的实际应用模式
  • 学会组件 props 组合
  • 理解状态管理中的类型组合
  • 了解 API 响应类型增强

🎨 组件 Props 组合

// 基础 Props interface BaseProps { className?: string; style?: React.CSSProperties; } // 可点击 interface ClickableProps { onClick: () => void; disabled?: boolean; } // 可禁用 interface DisableableProps { disabled: boolean; disabledReason?: string; } // 组合不同能力 type ButtonProps = BaseProps & ClickableProps & { variant: 'primary' | 'secondary' | 'danger'; size?: 'sm' | 'md' | 'lg'; }; type IconButtonProps = BaseProps & ClickableProps & DisableableProps & { icon: string; label: string; }; // 条件性 Props 组合 type InputProps = BaseProps & ( | { type: 'text'; value: string; onChange: (v: string) => void } | { type: 'number'; value: number; onChange: (v: number) => void } | { type: 'checkbox'; checked: boolean; onChange: (v: boolean) => void } );

📊 状态管理

// 基础状态类型 interface EntityState { items: T[]; loading: boolean; error: string | null; } // 分页状态 interface PaginationState { page: number; pageSize: number; total: number; } // 筛选状态 interface FilterState { filters: Partial; sortBy: keyof T | null; sortOrder: 'asc' | 'desc'; } // 组合完整状态 type ListViewState = EntityState & PaginationState & FilterState; // 用户列表状态 interface User { id: number; name: string; email: string; role: 'admin' | 'user'; } type UserListState = ListViewState; const userState: UserListState = { items: [], loading: false, error: null, page: 1, pageSize: 10, total: 100, filters: { role: 'admin' }, sortBy: 'name', sortOrder: 'asc' };

🌐 API 响应增强

// 标准 API 响应包装 interface ApiResponse { data: T; status: number; message: string; } // 分页响应 interface PaginatedResponse { data: T[]; pagination: { page: number; pageSize: number; total: number; totalPages: number; }; } // 带元数据的响应 interface WithMetadata { data: T; metadata: { timestamp: string; requestId: string; cached: boolean; }; } // 组合响应类型 type FullApiResponse = ApiResponse & PaginatedResponse & WithMetadata; // 使用示例 async function fetchUsers(): Promise> { const response = await fetch('/api/users'); return response.json(); } // 请求配置 interface RequestConfig { timeout?: number; retries?: number; } interface AuthConfig { token: string; userId: string; } type ApiConfig = RequestConfig & AuthConfig; const config: ApiConfig = { token: 'abc123', userId: 'user1', timeout: 5000, retries: 3 };
