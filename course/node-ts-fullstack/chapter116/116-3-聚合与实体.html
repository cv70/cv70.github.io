116.3 聚合与实体

管理领域对象的一致性边界

🎯 学习目标

  • 理解聚合的设计原则
  • 掌握聚合根的作用
  • 学会识别聚合边界
  • 了解聚合间的关系

📚 聚合概念

// 聚合（Aggregate）：一组相关对象的集合，作为数据修改的单元 // 聚合设计原则 const aggregatePrinciples = { 一致性边界: '聚合内部保证强一致性', 聚合根: '聚合中唯一的外部入口', 边界规则: '外部只能引用聚合根', 最小化: '聚合应该尽可能小' }; // 聚合根职责 const rootResponsibilities = { 控制访问: '外部只能通过聚合根访问内部对象', 维护不变性: '保证聚合内部业务规则', 发布事件: '对外通知状态变化' }; // 聚合示例：订单聚合 /* ┌─────────────────────────────────────┐ │ Order Aggregate │ │ ┌─────────────────────────────┐ │ │ │ Order (Aggregate Root) │ │ │ │ - id: OrderId │ │ │ │ - status: OrderStatus │ │ │ │ - shippingAddress: Address │ │ │ └─────────────────────────────┘ │ │ │ │ │ ▼ │ │ ┌─────────────────────────────┐ │ │ │ OrderItem[] │ │ │ │ - productId │ │ │ │ - quantity │ │ │ │ - unitPrice │ │ │ └─────────────────────────────┘ │ └─────────────────────────────────────┘ */
💡
聚合设计要点

聚合要小，外部只引用聚合根ID，使用最终一致性跨聚合。

🔧 聚合实现

// 订单聚合完整实现 // 领域事件 interface DomainEvent { occurredAt: Date; eventType: string; } class OrderCreatedEvent implements DomainEvent { readonly occurredAt: Date; readonly eventType = 'OrderCreated'; constructor( readonly orderId: string, readonly userId: string ) { this.occurredAt = new Date(); } } class OrderPaidEvent implements DomainEvent { readonly occurredAt: Date; readonly eventType = 'OrderPaid'; constructor( readonly orderId: string, readonly amount: number ) { this.occurredAt = new Date(); } } // 聚合基类 abstract class AggregateRoot<ID> { protected _id: ID; private _domainEvents: DomainEvent[] = []; get id(): ID { return this._id; } get domainEvents(): ReadonlyArray<DomainEvent> { return this._domainEvents; } protected addDomainEvent(event: DomainEvent): void { this._domainEvents.push(event); } clearDomainEvents(): void { this._domainEvents = []; } } // 订单聚合根 class Order extends AggregateRoot<OrderId> { private _items: OrderItem[] = []; private _status: OrderStatus; private _shippingAddress: Address; private _createdAt: Date; private constructor() { super(); } // 工厂方法：创建新订单 static create( id: OrderId, userId: string, shippingAddress: Address ): Order { const order = new Order(); order._id = id; order._status = OrderStatus.PENDING; order._shippingAddress = shippingAddress; order._createdAt = new Date(); // 发布领域事件 order.addDomainEvent(new OrderCreatedEvent(id.value, userId)); return order; } // 从持久化重建 static reconstitute( id: OrderId, items: OrderItem[], status: OrderStatus, shippingAddress: Address, createdAt: Date ): Order { const order = new Order(); order._id = id; order._items = items; order._status = status; order._shippingAddress = shippingAddress; order._createdAt = createdAt; return order; } // 业务方法：添加商品 addItem(item: OrderItem): void { this.ensureStatus(OrderStatus.PENDING, 'Cannot add items'); const existing = this._items.find(i => i.productId === item.productId); if (existing) { existing.increaseQuantity(item.Quantity); } else { this._items.push(item); } } // 业务方法：支付 pay(): void { this.ensureStatus(OrderStatus.PENDING, 'Order is not pending'); this.ensureNonEmpty('Cannot pay for empty order'); this._status = OrderStatus.PAID; this.addDomainEvent(new OrderPaidEvent( this._id.value, this.getTotalAmount().amount )); } // 业务方法：发货 ship(): void { this.ensureStatus(OrderStatus.PAID, 'Order is not paid'); this._status = OrderStatus.SHIPPED; } // 业务方法：取消 cancel(reason: string): void { if (this._status === OrderStatus.SHIPPED || this._status === OrderStatus.DELIVERED) { throw new Error('Cannot cancel shipped or delivered order'); } this._status = OrderStatus.CANCELLED; } // 查询方法 get items(): ReadonlyArray<OrderItem> { return this._items; } get status(): OrderStatus { return this._status; } getTotalAmount(): Money { return this._items.reduce( (total, item) => total.add(item.getTotalPrice()), Money.create(0) ); } // 私有方法 private ensureStatus(expected: OrderStatus, message: string): void { if (this._status !== expected) { throw new Error(`${message}: expected ${expected}, got ${this._status}`); } } private ensureNonEmpty(message: string): void { if (this._items.length === 0) { throw new Error(message); } } } // 使用示例 const orderId = OrderId.generate(); const address = new Address('北京', '北京', '朝阳区', 'xxx街道'); const order = Order.create(orderId, 'user-123', address); order.addItem(new OrderItem( 'product-1', '商品A', Money.create(100), 2 )); order.pay(); // 获取领域事件并发布 console.log(order.domainEvents); // [OrderCreatedEvent, OrderPaidEvent]

✅
