13.3 console 对象

调试输出与日志记录

🎯 学习目标

  • 掌握 console 基本输出方法
  • 学会格式化输出
  • 了解性能计时方法
  • 掌握日志输出到文件

📚 基本输出

// 基本日志 console.log('Hello, World!'); console.log('Hello', 'World', '!'); // 空格分隔 console.log('Hello %s', 'World'); // 格式化 // 信息日志 console.info('这是信息日志'); // 警告日志（带警告图标） console.warn('这是警告日志'); // 错误日志（带错误图标） console.error('这是错误日志'); // 调试日志 console.debug('这是调试日志'); // 清空控制台 console.clear(); // 对象输出 const obj = { name: 'Alice', age: 25, city: 'Beijing' }; console.log(obj); console.dir(obj, { depth: null, colors: true });

🔧 格式化输出

// %s - 字符串 console.log('姓名: %s', 'Alice'); // 姓名: Alice // %d 或 %i - 整数 console.log('年龄: %d', 25); // 年龄: 25 // %f - 浮点数 console.log('价格: %f', 99.99); // 价格: 99.99 // %j - JSON const user = { name: 'Bob', age: 30 }; console.log('用户: %j', user); // %o 或 %O - 对象 console.log('对象: %o', user); // %% - 百分号 console.log('进度: %d%%', 80); // 进度: 80% // 模板字符串 const name = 'Alice'; const age = 25; console.log(`姓名: ${name}, 年龄: ${age}`); // 样式输出（终端可能不支持） console.log('\x1b[31m%s\x1b[0m', '红色文字'); console.log('\x1b[32m%s\x1b[0m', '绿色文字'); console.log('\x1b[33m%s\x1b[0m', '黄色文字');

📊 表格输出

// 表格形式输出数组 const users = [ { id: 1, name: 'Alice', age: 25 }, { id: 2, name: 'Bob', age: 30 }, { id: 3, name: 'Charlie', age: 35 } ]; console.table(users); // 输出： // ┌─────────┬────┬─────────┬─────┐ // │ (index) │ id │ name │ age │ // ├─────────┼────┼─────────┼─────┤ // │ 0 │ 1 │ 'Alice' │ 25 │ // │ 1 │ 2 │ 'Bob' │ 30 │ // │ 2 │ 3 │'Charlie'│ 35 │ // └─────────┴────┴─────────┴─────┘ // 只显示指定列 console.table(users, ['name', 'age']); // 输出对象 const config = { host: 'localhost', port: 3000, debug: true }; console.table(config);

⏱️ 性能计时

// 计时器 console.time('循环耗时'); let sum = 0; for (let i = 0; i < 1000000; i++) { sum += i; } console.timeEnd('循环耗时'); // 循环耗时: 2.345ms // 多个计时器 console.time('操作A'); console.time('操作B'); // ... 一些操作 console.timeEnd('操作A'); // 操作A: 1.234ms console.timeEnd('操作B'); // 操作B: 2.345ms // 断言 console.assert(1 === 1, '1 等于 1'); // 不输出 console.assert(1 === 2, '1 不等于 2'); // 输出错误信息 // 计数器 function test() { console.count('test 调用次数'); } test(); // test 调用次数: 1 test(); // test 调用次数: 2 test(); // test 调用次数: 3 console.countReset('test 调用次数'); test(); // test 调用次数: 1

📁 输出到文件

const fs = require('fs'); const { Console } = require('console'); // 创建新的 Console 实例 const output = fs.createWriteStream('./stdout.log'); const errorOutput = fs.createWriteStream('./stderr.log'); const logger = new Console({ stdout: output, stderr: errorOutput }); logger.log('普通日志'); logger.error('错误日志'); logger.table([{ a: 1, b: 2 }]); // 使用文件系统追加日志 function logToFile(message) { const timestamp = new Date().toISOString(); const logMessage = `[${timestamp}] ${message}\n`; fs.appendFileSync('app.log', logMessage); } logToFile('应用启动'); logToFile('用户登录'); // 生产环境建议使用专业日志库：winston, pino, bunyan

📝 本节小结

✅
  • • console.log/info/warn/error 用于不同级别日志
  • • console.table 表格形式输出数据
  • • console.time/timeEnd 测量执行时间
  • • 生产环境使用专业日志库
