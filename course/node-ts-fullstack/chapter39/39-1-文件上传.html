🔐 文件类型验证

// 允许的文件类型 const ALLOWED_TYPES = ['image/jpeg', 'image/png', 'image/gif']; const ALLOWED_EXTENSIONS = ['.jpg', '.jpeg', '.png', '.gif']; function validateFile(file) { const errors = []; // 检查 MIME 类型 if (!ALLOWED_TYPES.includes(file.mimetype)) { errors.push(`不支持的文件类型: ${file.mimetype}`); } // 检查扩展名 const ext = path.extname(file.originalFilename).toLowerCase(); if (!ALLOWED_EXTENSIONS.includes(ext)) { errors.push(`不支持的扩展名: ${ext}`); } // 检查文件大小 const maxSize = 5 * 1024 * 1024; // 5MB if (file.size > maxSize) { errors.push(`文件过大: ${(file.size / 1024 / 1024).toFixed(2)}MB`); } return errors; } router.post('/upload/avatar', async ctx => { const file = ctx.request.files.avatar; if (!file) { ctx.status = 400; ctx.body = { error: 'No file uploaded' }; return; } const errors = validateFile(file); if (errors.length > 0) { // 删除无效文件 fs.unlinkSync(file.filepath); ctx.status = 400; ctx.body = { errors }; return; } ctx.body = { success: true, file: file.newFilename }; });

📁 文件存储管理

const fs = require('fs'); const path = require('path'); // 按日期创建目录 function getUploadDir() { const date = new Date(); const year = date.getFullYear(); const month = String(date.getMonth() + 1).padStart(2, '0'); const day = String(date.getDate()).padStart(2, '0'); const dir = path.join(__dirname, 'uploads', `${year}`, `${month}`, `${day}`); if (!fs.existsSync(dir)) { fs.mkdirSync(dir, { recursive: true }); } return dir; } // 重命名文件 async function saveFile(file, customName) { const uploadDir = getUploadDir(); const ext = path.extname(file.originalFilename); const filename = customName || `${Date.now()}${ext}`; const filepath = path.join(uploadDir, filename); // 移动文件 await fs.promises.rename(file.filepath, filepath); return { filename, filepath, relativePath: path.relative(path.join(__dirname, 'uploads'), filepath) }; } router.post('/upload', async ctx => { const file = ctx.request.files.avatar; const result = await saveFile(file); ctx.body = { success: true, path: result.relativePath }; });

☁️ 上传到云存储

const AWS = require('aws-sdk'); const s3 = new AWS.S3({ accessKeyId: process.env.AWS_ACCESS_KEY, secretAccessKey: process.env.AWS_SECRET_KEY }); async function uploadToS3(file) { const stream = fs.createReadStream(file.filepath); const params = { Bucket: 'my-bucket', Key: `uploads/${Date.now()}_${file.originalFilename}`, Body: stream, ContentType: file.mimetype }; const result = await s3.upload(params).promise(); // 删除临时文件 fs.unlinkSync(file.filepath); return result.Location; } router.post('/upload/cloud', async ctx => { const file = ctx.request.files.file; const url = await uploadToS3(file); ctx.body = { success: true, url }; });

🖼️ 图片处理

const sharp = require('sharp'); async function processImage(file) { const outputPath = path.join( path.dirname(file.filepath), `thumb_${file.newFilename}` ); // 缩略图 await sharp(file.filepath) .resize(200, 200, { fit: 'cover' }) .toFile(outputPath); // 获取图片信息 const metadata = await sharp(file.filepath).metadata(); return { width: metadata.width, height: metadata.height, format: metadata.format, thumbnail: `thumb_${file.newFilename}` }; } router.post('/upload/image', async ctx => { const file = ctx.request.files.image; // 验证图片类型 if (!file.mimetype.startsWith('image/')) { ctx.status = 400; ctx.body = { error: 'Please upload an image' }; return; } const info = await processImage(file); ctx.body = { success: true, original: file.newFilename, ...info }; });

💻 完整上传服务

// services/uploadService.js const fs = require('fs').promises; const path = require('path'); const sharp = require('sharp'); class UploadService { constructor(options = {}) { this.uploadDir = options.uploadDir || './uploads'; this.maxSize = options.maxSize || 10 * 1024 * 1024; this.allowedTypes = options.allowedTypes || ['image/jpeg', 'image/png']; } async save(file, options = {}) { // 验证 this.validate(file); // 生成文件名 const filename = options.filename || this.generateFilename(file); const dir = this.getDailyDir(); const filepath = path.join(dir, filename); // 移动文件 await fs.rename(file.filepath, filepath); return { filename, filepath, url: `/uploads/${filename}` }; } validate(file) { if (file.size > this.maxSize) { throw new Error(`File too large: ${file.size} bytes`); } if (!this.allowedTypes.includes(file.mimetype)) { throw new Error(`Type not allowed: ${file.mimetype}`); } } generateFilename(file) { const ext = path.extname(file.originalFilename); return `${Date.now()}_${Math.random().toString(36).slice(2)}${ext}`; } getDailyDir() { const date = new Date(); const dir = path.join( this.uploadDir, `${date.getFullYear()}`, `${date.getMonth() + 1}`.padStart(2, '0') ); fs.mkdir(dir, { recursive: true }).catch(() => {}); return dir; } async createThumbnail(filepath, size = 200) { const dir = path.dirname(filepath); const name = path.basename(filepath); const thumbPath = path.join(dir, `thumb_${name}`); await sharp(filepath) .resize(size, size, { fit: 'cover' }) .toFile(thumbPath); return thumbPath; } } module.exports = UploadService;
