52.8 DMA设计

常用外设接口设计

🎯 学习目标

  • 掌握DMA的基本原理与优势
  • 理解DMA的传输模式与配置
  • 学会DMA控制器的硬件设计
  • 了解DMA与外设的协同工作

📚 DMA概述

DMA（Direct Memory Access）直接内存访问是一种无需CPU干预的数据传输机制。DMA控制器可以在外设和内存之间、或内存和内存之间直接传输数据，大幅减轻CPU负担，提高系统效率。

💡
DMA优势

• 减轻CPU负担：CPU可执行其他任务
• 提高传输效率：高速数据传输
• 降低功耗：减少CPU唤醒次数
• 实时性好：满足实时数据传输需求

📊 DMA工作原理

DMA传输流程

1. 请求：外设向DMA控制器发起传输请求

2. 仲裁：DMA控制器仲裁多个请求，选择优先级最高的

3. 响应：DMA控制器响应请求，获得总线控制权

4. 传输：DMA控制器执行数据传输

5. 完成：传输完成，触发中断通知CPU

🔄 DMA传输类型

类型源地址目的地址应用场景
内存到内存内存内存内存复制、数据搬移
外设到内存外设内存ADC数据采集、UART接收
内存到外设内存外设DAC输出、UART发送
外设到外设外设外设外设间数据转发

⚙️ DMA控制器设计

// DMA控制器模块
module dma_controller #(
    parameter CHANNEL_NUM = 8,        // DMA通道数
    parameter ADDR_WIDTH = 32,        // 地址宽度
    parameter DATA_WIDTH = 32,        // 数据宽度
    parameter BURST_MAX = 16          // 最大突发传输长度
) (
    // 系统接口
    input  wire                          clk,
    input  wire                          rst_n,

    // 寄存器接口
    input  wire                          reg_wr,
    input  wire                          reg_rd,
    input  wire [4:0]                    reg_addr,
    input  wire [31:0]                   reg_wdata,
    output wire [31:0]                   reg_rdata,

    // DMA请求（来自外设）
    input  wire [CHANNEL_NUM-1:0]        dma_req,       // DMA请求
    output wire [CHANNEL_NUM-1:0]        dma_ack,       // DMA应答

    // 总线接口（主端口）
    output wire                          m_cmd_valid,
    input  wire                          m_cmd_ready,
    output wire [ADDR_WIDTH-1:0]         m_addr,
    output wire                          m_cmd_write,
    output wire [DATA_WIDTH/8-1:0]       m_cmd_strb,
    output wire [DATA_WIDTH-1:0]         m_cmd_data,
    output wire [7:0]                    m_cmd_len,     // 突发长度

    input  wire                          m_rsp_valid,
    output wire                          m_rsp_ready,
    input  wire [DATA_WIDTH-1:0]         m_rsp_data,
    input  wire                          m_rsp_last,

    // 中断输出
    output wire [CHANNEL_NUM-1:0]        transfer_done,  // 传输完成中断
    output wire [CHANNEL_NUM-1:0]        transfer_error   // 传输错误中断
);

    // 通道寄存器
    reg [ADDR_WIDTH-1:0] src_addr [CHANNEL_NUM-1:0];    // 源地址
    reg [ADDR_WIDTH-1:0] dst_addr [CHANNEL_NUM-1:0];    // 目的地址
    reg [15:0]           transfer_cnt [CHANNEL_NUM-1:0]; // 传输计数
    reg [7:0]            burst_size [CHANNEL_NUM-1:0];   // 突发大小
    reg [CHANNEL_NUM-1:0] channel_enable;               // 通道使能
    reg [CHANNEL_NUM-1:0] channel_priority;             // 通道优先级
    reg [2:0]            data_size [CHANNEL_NUM-1:0];   // 数据大小
    reg [1:0]            addr_mode [CHANNEL_NUM-1:0];   // 地址模式

    // 通道状态
    reg [CHANNEL_NUM-1:0] channel_active;
    reg [CHANNEL_NUM-1:0] channel_done;
    reg [CHANNEL_NUM-1:0] channel_error;

    // 仲裁器
    reg [3:0] arbiter_ptr;
    wire [CHANNEL_NUM-1:0] arbiter_grant;
    reg [3:0] current_channel;

    // 当前通道控制信号
    reg [ADDR_WIDTH-1:0] curr_src_addr;
    reg [ADDR_WIDTH-1:0] curr_dst_addr;
    reg [15:0]           curr_transfer_cnt;
    reg [7:0]            curr_burst_cnt;
    reg                  curr_dir;  // 0=读(源->目的), 1=写
    reg                  curr_active;

    // 通道优先级编码器
    integer k;
    always @(*) begin
        arbiter_grant = {CHANNEL_NUM{1'b0}};
        for (k = 0; k < CHANNEL_NUM; k = k + 1) begin
            if (dma_req[k] && channel_enable[k] && !channel_active[k]) begin
                arbiter_grant[k] = 1;
            end
        end
    end

    // 优先级仲裁器
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            arbiter_ptr <= 0;
            current_channel <= 0;
            curr_active <= 0;
        end else begin
            if (curr_active && (curr_burst_cnt == 0 || curr_transfer_cnt == 0)) begin
                // 当前通道传输完成
                curr_active <= 0;
                channel_active[current_channel] <= 0;
            end else if (!curr_active && |arbiter_grant) begin
                // 选择新通道
                // 简单轮询仲裁
                for (k = 0; k < CHANNEL_NUM; k = k + 1) begin
                    if (arbiter_grant[(arbiter_ptr + k) % CHANNEL_NUM]) begin
                        current_channel <= (arbiter_ptr + k) % CHANNEL_NUM;
                        curr_src_addr <= src_addr[(arbiter_ptr + k) % CHANNEL_NUM];
                        curr_dst_addr <= dst_addr[(arbiter_ptr + k) % CHANNEL_NUM];
                        curr_transfer_cnt <= transfer_cnt[(arbiter_ptr + k) % CHANNEL_NUM];
                        curr_burst_cnt <= burst_size[(arbiter_ptr + k) % CHANNEL_NUM];
                        curr_active <= 1;
                        channel_active[(arbiter_ptr + k) % CHANNEL_NUM] <= 1;
                        arbiter_ptr <= (arbiter_ptr + k + 1) % CHANNEL_NUM;
                    end
                end
            end
        end
    end

    // DMA应答
    assign dma_ack = channel_active;

    // 总线主端口控制
    assign m_cmd_valid = curr_active && (curr_burst_cnt > 0);
    assign m_addr = curr_dir ? curr_dst_addr : curr_src_addr;
    assign m_cmd_write = curr_dir;  // 1=写, 0=读
    assign m_cmd_strb = {DATA_WIDTH/8{1'b1}};
    assign m_cmd_len = curr_burst_cnt - 1;

    // 总线响应处理
    assign m_rsp_ready = 1;

    // 传输状态更新
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            channel_done <= 0;
            channel_error <= 0;
        end else begin
            channel_done <= 0;
            channel_error <= 0;

            if (curr_active && m_cmd_valid && m_cmd_ready) begin
                // 命令已发送
                if (!curr_dir) begin
                    // 读操作，等待响应
                    if (m_rsp_valid) begin
                        // 数据读取完成，写入目的地址
                        // （此处简化处理，实际需要写入FIFO或缓存）
                        curr_src_addr <= curr_src_addr + (DATA_WIDTH/8);
                        curr_dst_addr <= curr_dst_addr + (DATA_WIDTH/8);
                        curr_transfer_cnt <= curr_transfer_cnt - 1;
                        curr_burst_cnt <= curr_burst_cnt - 1;

                        if (m_rsp_last || curr_transfer_cnt == 1) begin
                            channel_done[current_channel] <= 1;
                        end
                    end
                end else begin
                    // 写操作
                    curr_src_addr <= curr_src_addr + (DATA_WIDTH/8);
                    curr_dst_addr <= curr_dst_addr + (DATA_WIDTH/8);
                    curr_transfer_cnt <= curr_transfer_cnt - 1;
                    curr_burst_cnt <= curr_burst_cnt - 1;

                    if (curr_transfer_cnt == 1) begin
                        channel_done[current_channel] <= 1;
                    end
                end
            end
        end
    end

    // 中断输出
    assign transfer_done = channel_done;
    assign transfer_error = channel_error;

    // 寄存器读写（简化）
    reg [31:0] rdata;
    always @(*) begin
        rdata = 32'h0;
        // 寄存器读逻辑
    end

    assign reg_rdata = rdata;

    // 寄存器写（简化）
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            channel_enable <= 0;
            channel_priority <= 0;
        end else if (reg_wr) begin
            // 寄存器写逻辑
        end
    end

endmodule

📐 DMA地址模式

固定地址

传输过程中地址保持不变：

• 适用于外设寄存器访问
• 如UART数据寄存器
• 每次传输到相同地址

递增地址

传输后地址自动递增：

• 适用于内存块传输
• 连续的数据缓冲区
• 每次传输地址+数据宽度

递减地址

传输后地址自动递减：

• 特殊场景使用
• 堆栈操作
• 每次传输地址-数据宽度

重装载地址

传输完成后地址重置：

• 循环缓冲区
• 音频/视频数据流
• 循环传输模式

🌊 DMA数据流

DMA数据流架构

DMA控制器通常包含多个数据流（Stream），每个数据流包含：

1. 源通道：配置源地址、源数据宽度

2. 目的通道：配置目的地址、目的数据宽度

3. 传输计数器：记录剩余传输量

4. FIFO：缓冲数据，处理宽度转换

5. 控制器：管理传输流程和状态

🔗 DMA与外设协同

外设DMA请求

外设通过DMA请求信号与DMA控制器协同：

• UART：接收数据就绪或发送缓冲区空

• ADC：转换完成

• SPI/I2C：数据接收/发送就绪

• 定时器：比较匹配/捕获事件

💡
DMA请求映射

• 每个DMA通道可映射到多个外设请求
• 通过多路选择器选择请求源
• 支持多路复用和优先级配置

⚡ DMA性能优化

突发传输

一次传输多个数据：

• 减少总线开销
• 提高传输效率
• 支持可配置突发长度

双缓冲

两个缓冲区交替使用：

• 实现无间断数据流
• 防止数据丢失
• 适用于实时数据采集

链式传输

自动启动下一个传输：

• 减少CPU干预
• 复杂的数据搬移
• 提高自动化程度

FIFO优化

使用FIFO缓冲数据：

• 处理数据宽度转换
• 平衡传输速率
• 减少总线等待

🐛 常见问题与调试

⚠️
常见问题

1. 数据丢失
症状：传输数据不完整
解决：增加FIFO深度，优化缓冲区管理

2. 总线冲突
症状：DMA传输效率低
解决：优化优先级，使用多通道并行

3. 地址越界
症状：访问错误地址
解决：检查地址配置，添加边界检查

4. 中断丢失
症状：无法收到完成中断
解决：检查中断使能，确保中断服务及时

📝 本节小结

✅
  • • DMA是直接内存访问机制，无需CPU干预即可传输数据
  • • DMA支持内存到内存、外设到内存等多种传输类型
  • • DMA控制器包含通道管理、仲裁器、总线接口等模块
  • • 支持固定、递增、递减、重装载等多种地址模式
  • • 突发传输、双缓冲、链式传输可优化DMA性能
