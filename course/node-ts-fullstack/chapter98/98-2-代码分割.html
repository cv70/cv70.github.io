📚 代码分割概述

# 为什么需要代码分割 ## 问题：大体积打包文件 - 单个 JS 文件过大（几 MB） - 首次加载耗时长 - 用户下载不需要的代码 ## 解决：代码分割 - 将代码拆分成多个小块 - 按需加载 - 减小初始加载体积 - 更好的缓存利用 # 代码分割策略 ┌─────────────────┬──────────────────────────────────┐ │ 策略 │ 说明 │ ├─────────────────┼──────────────────────────────────┤ │ 入口分割 │ 多入口，独立打包 │ │ 动态导入 │ import() 语法 │ │ 路由分割 │ 按路由懒加载组件 │ │ 组件分割 │ 大组件延迟加载 │ │ 第三方库分割 │ vendor chunk 单独打包 │ └─────────────────┴──────────────────────────────────┘ # 打包结果示例 Before: main.js (2MB) - 包含所有代码 After: main.js (200KB) - 核心代码 vendor.js (500KB) - 第三方库 route-home.js (50KB) route-about.js (30KB) component-chart.js (100KB)

🔧 动态导入

# import() 语法 // 静态导入（打包时加载） import { add } from './math'; add(1, 2); // 动态导入（运行时加载） async function calculate() { const math = await import('./math'); return math.add(1, 2); } # 按需加载模块 // 条件加载 async function loadModule(condition) { if (condition) { const module = await import('./module-a'); module.doSomething(); } else { const module = await import('./module-b'); module.doSomethingElse(); } } // 事件触发加载 button.addEventListener('click', async () => { const { showModal } = await import('./modal'); showModal(); }); # Webpack 魔法注释 // 命名 chunk const module = await import( /* webpackChunkName: "my-module" */ './module' ); // 预取 const module = await import( /* webpackPrefetch: true */ './module' ); // 预加载 const module = await import( /* webpackPreload: true */ './module' ); # Vite 动态导入 // Vite 自动处理代码分割 const module = await import('./module'); // 命名 chunk const module = await import( /* @vite-ignore */ './module?chunkName=my-module' );
💡
预取 vs 预加载

prefetch: 空闲时加载，用于下一页面可能需要的代码。preload: 立即并行加载，用于当前页面需要的代码。

⚙️ 路由级代码分割

# React Router 懒加载 import { lazy, Suspense } from 'react'; import { BrowserRouter, Routes, Route } from 'react-router-dom'; // 懒加载路由组件 const Home = lazy(() => import('./pages/Home')); const About = lazy(() => import('./pages/About')); const Dashboard = lazy(() => import('./pages/Dashboard')); function App() { return ( <BrowserRouter> <Suspense fallback={<Loading />}> <Routes> <Route path="/" element={<Home />} /> <Route path="/about" element={<About />} /> <Route path="/dashboard" element={<Dashboard />} /> </Routes> </Suspense> </BrowserRouter> ); } # Vue Router 懒加载 import { createRouter, createWebHistory } from 'vue-router'; const routes = [ { path: '/', component: () => import('./views/Home.vue'), }, { path: '/about', component: () => import('./views/About.vue'), }, { path: '/dashboard', component: () => import('./views/Dashboard.vue'), }, ]; const router = createRouter({ history: createWebHistory(), routes, }); # Next.js 动态导入 import dynamic from 'next/dynamic'; const DynamicComponent = dynamic(() => import('./components/Chart'), { loading: () => <p>Loading...</p>, ssr: false, // 禁用服务端渲染 }); # 组件级懒加载 Hook function useLazyComponent(importFn) { const [component, setComponent] = useState(null); useEffect(() => { importFn().then((module) => { setComponent(() => module.default); }); }, [importFn]); return component; } // 使用 const Chart = useLazyComponent(() => import('./Chart')); if (!Chart) return <Loading />; return <Chart data={data} />;

🚀 第三方库分割

# Vite 自动分割 // vite.config.ts export default defineConfig({ build: { rollupOptions: { output: { manualChunks: { // 单独打包 'react-vendor': ['react', 'react-dom'], 'ui-vendor': ['antd', '@ant-design/icons'], 'chart': ['echarts', 'echarts-for-react'], }, }, }, }, }); # 动态分割函数 manualChunks(id) { if (id.includes('node_modules')) { // React 生态 if (id.includes('react') || id.includes('react-dom')) { return 'react'; } // UI 库 if (id.includes('antd') || id.includes('@ant-design')) { return 'ui'; } // 工具库 if (id.includes('lodash') || id.includes('dayjs')) { return 'utils'; } // 其他第三方 return 'vendor'; } } # Webpack SplitChunks // webpack.config.js optimization: { splitChunks: { chunks: 'all', cacheGroups: { vendor: { test: /[\\/]node_modules[\\/]/, name: 'vendor', chunks: 'all', }, react: { test: /[\\/]node_modules[\\/](react|react-dom)[\\/]/, name: 'react', chunks: 'all', priority: 10, }, }, }, }
