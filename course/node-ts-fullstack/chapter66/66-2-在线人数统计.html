服务端实现

import { Server, Socket } from 'socket.io'; interface OnlineUser { id: string; username: string; socketId: string; connectedAt: Date; lastActivity: Date; ip?: string; userAgent?: string; } class OnlineUserManager { private users = new Map<string, OnlineUser>(); // userId -> User private socketToUser = new Map<string, string>(); // socketId -> userId add(socket: Socket, userId: string, username: string): OnlineUser { const user: OnlineUser = { id: userId, username, socketId: socket.id, connectedAt: new Date(), lastActivity: new Date(), ip: socket.handshake.address, userAgent: socket.handshake.headers['user-agent'] }; this.users.set(userId, user); this.socketToUser.set(socket.id, userId); return user; } remove(socketId: string): OnlineUser | null { const userId = this.socketToUser.get(socketId); if (!userId) return null; const user = this.users.get(userId); this.users.delete(userId); this.socketToUser.delete(socketId); return user || null; } updateActivity(userId: string): void { const user = this.users.get(userId); if (user) { user.lastActivity = new Date(); } } getCount(): number { return this.users.size; } getAll(): OnlineUser[] { return Array.from(this.users.values()); } getByRoom(roomId: string, io: Server): OnlineUser[] { const roomSockets = io.sockets.adapter.rooms.get(roomId); if (!roomSockets) return []; const users: OnlineUser[] = []; roomSockets.forEach(socketId => { const userId = this.socketToUser.get(socketId); if (userId) { const user = this.users.get(userId); if (user) users.push(user); } }); return users; } } const onlineManager = new OnlineUserManager(); export function setupOnlineTracking(io: Server) { // 定时广播在线人数 setInterval(() => { const stats = { total: onlineManager.getCount(), timestamp: Date.now() }; io.emit('online:stats', stats); }, 5000); io.on('connection', (socket) => { const userId = socket.handshake.auth.userId; const username = socket.handshake.auth.username; // 添加在线用户 const user = onlineManager.add(socket, userId, username); // 广播用户上线 io.emit('user:online', { user, onlineCount: onlineManager.getCount() }); // 发送当前在线用户列表给新用户 socket.emit('online:list', onlineManager.getAll()); // 更新活动状态 socket.onAny(() => { onlineManager.updateActivity(userId); }); // 断开连接 socket.on('disconnect', () => { const leftUser = onlineManager.remove(socket.id); if (leftUser) { // 广播用户下线 io.emit('user:offline', { user: leftUser, onlineCount: onlineManager.getCount() }); } }); }); }

🏠 房间级别统计

房间在线人数

// 房间在线统计 class RoomStatsManager { private roomStats = new Map<string, { onlineCount: number; peakCount: number; totalVisits: number; }>(); joinRoom(roomId: string): void { if (!this.roomStats.has(roomId)) { this.roomStats.set(roomId, { onlineCount: 0, peakCount: 0, totalVisits: 0 }); } const stats = this.roomStats.get(roomId)!; stats.onlineCount++; stats.totalVisits++; if (stats.onlineCount > stats.peakCount) { stats.peakCount = stats.onlineCount; } } leaveRoom(roomId: string): void { const stats = this.roomStats.get(roomId); if (stats && stats.onlineCount > 0) { stats.onlineCount--; } } getStats(roomId: string) { return this.roomStats.get(roomId); } getAllStats() { const result: Record<string, any> = {}; this.roomStats.forEach((value, key) => { result[key] = value; }); return result; } } const roomStats = new RoomStatsManager(); // 在 Socket.IO 中使用 io.on('connection', (socket) => { socket.on('room:join', (roomId: string) => { socket.join(roomId); roomStats.joinRoom(roomId); // 广播房间统计更新 const stats = roomStats.getStats(roomId); io.to(roomId).emit('room:stats', { roomId, ...stats }); }); socket.on('room:leave', (roomId: string) => { socket.leave(roomId); roomStats.leaveRoom(roomId); // 广播房间统计更新 const stats = roomStats.getStats(roomId); if (stats) { io.to(roomId).emit('room:stats', { roomId, ...stats }); } }); socket.on('disconnect', () => { // 用户断开时，从所有房间统计中移除 socket.rooms.forEach(roomId => { if (roomId !== socket.id) { roomStats.leaveRoom(roomId); } }); }); });

React 组件

import { useState, useEffect } from 'react'; import { useSocket } from '../hooks/useSocket'; interface OnlineStats { total: number; timestamp: number; } interface OnlineUser { id: string; username: string; connectedAt: Date; } export function OnlineIndicator() { const { socket, isConnected } = useSocket(); const [onlineStats, setOnlineStats] = useState<OnlineStats>({ total: 0, timestamp: 0 }); const [onlineUsers, setOnlineUsers] = useState<OnlineUser[]>([]); useEffect(() => { if (!socket) return; // 接收在线统计 socket.on('online:stats', (stats) => { setOnlineStats(stats); }); // 接收在线用户列表 socket.on('online:list', (users) => { setOnlineUsers(users); }); // 用户上线 socket.on('user:online', ({ user, onlineCount }) => { setOnlineUsers(prev => { if (!prev.find(u => u.id === user.id)) { return [...prev, user]; } return prev; }); setOnlineStats(prev => ({ ...prev, total: onlineCount })); }); // 用户下线 socket.on('user:offline', ({ user, onlineCount }) => { setOnlineUsers(prev => prev.filter(u => u.id !== user.id)); setOnlineStats(prev => ({ ...prev, total: onlineCount })); }); return () => { socket.off('online:stats'); socket.off('online:list'); socket.off('user:online'); socket.off('user:offline'); }; }, [socket]); return ( <div className="online-indicator"> <div className="status-bar"> <span className={`status-dot ${isConnected ? 'connected' : 'disconnected'}`} /> <span className="count">在线: {onlineStats.total}</span> </div> <div className="user-list"> <h4>在线用户</h4> <ul> {onlineUsers.map(user => ( <li key={user.id}> <span className="avatar">{user.username[0]}</span> <span className="name">{user.username}</span> </li> ))} </ul> </div> </div> ); }

管理后台统计

// 服务端统计 API import express from 'express'; const app = express(); interface SystemStats { totalConnections: number; activeConnections: number; uniqueUsers: number; peakToday: number; avgSessionDuration: number; topRooms: Array<{ roomId: string; users: number }>; } // 统计中间件 let totalConnections = 0; const dailyStats = new Map<string, number>(); io.on('connection', (socket) => { totalConnections++; const today = new Date().toDateString(); const currentDaily = dailyStats.get(today) || 0; dailyStats.set(today, currentDaily + 1); socket.on('disconnect', () => { // 记录会话时长等 }); }); // 获取统计信息 app.get('/api/stats', (req, res) => { const stats: SystemStats = { totalConnections, activeConnections: onlineManager.getCount(), uniqueUsers: onlineManager.getAll().length, peakToday: Math.max(...Array.from(dailyStats.values())), avgSessionDuration: 0, // 计算平均会话时长 topRooms: getTopRooms() }; res.json(stats); }); // 获取在线用户详情 app.get('/api/online', (req, res) => { res.json(onlineManager.getAll()); }); // 获取房间统计 app.get('/api/rooms/stats', (req, res) => { res.json(roomStats.getAllStats()); }); function getTopRooms() { return Object.entries(roomStats.getAllStats()) .map(([roomId, stats]) => ({ roomId, users: stats.onlineCount })) .sort((a, b) => b.users - a.users) .slice(0, 10); }
