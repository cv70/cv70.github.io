65.3 事件监听与触发

客户端事件处理详解

🎯 学习目标

  • 掌握客户端事件监听和发送方法
  • 理解消息确认机制的使用
  • 学会管理事件监听器
  • 了解事件的类型定义

👂 事件监听

基本监听

// 监听事件 socket.on('message', (data) => { console.log('收到消息:', data); }); // 监听多个参数 socket.on('notification', (title, body, timestamp) => { console.log(`${title}: ${body} at ${timestamp}`); }); // 监听对象数据 socket.on('user-joined', (user) => { console.log(`${user.name} 加入了聊天室`); });

一次性监听

// 只监听一次 socket.once('welcome', (message) => { console.log('欢迎消息:', message); // 此后不再触发 });

TypeScript 类型定义

// 定义服务端到客户端的事件 interface ServerToClientEvents { message: (data: { id: string; content: string; userId: string }) => void; notification: (title: string, body: string) => void; userStatus: (userId: string, status: 'online' | 'offline') => void; } // 定义客户端到服务端的事件 interface ClientToServerEvents { sendMessage: (content: string, callback: (success: boolean) => void) => void; joinRoom: (roomId: string) => void; typing: (isTyping: boolean) => void; } // 应用类型 const socket: Socket = io(); // 类型安全的事件监听 socket.on('message', (data) => { // data 自动推断为 { id: string; content: string; userId: string } console.log(data.content); }); socket.on('userStatus', (userId, status) => { // userId: string, status: 'online' | 'offline' console.log(`User ${userId} is ${status}`); });

📤 发送事件

基本发送

// 发送简单数据 socket.emit('ping'); // 发送字符串 socket.emit('message', 'Hello World'); // 发送对象 socket.emit('join-room', { roomId: 'room-123', password: 'secret' }); // 发送多个参数 socket.emit('chat', 'room-123', 'Hello everyone', Date.now());

带回调确认

// 发送并等待确认 socket.emit('join-room', 'room-123', (response) => { if (response.success) { console.log('加入成功'); console.log('房间信息:', response.room); } else { console.log('加入失败:', response.error); } }); // 服务端 socket.on('join-room', (roomId, callback) => { try { // 处理逻辑 const room = getRoom(roomId); callback({ success: true, room }); } catch (error) { callback({ success: false, error: error.message }); } });

Promise 封装

// 封装为 Promise function emitAsync<T>(event: string, ...args: any[]): Promise<T> { return new Promise((resolve, reject) => { socket.emit(event, ...args, (response: T) => { resolve(response); }); // 设置超时 setTimeout(() => { reject(new Error('请求超时')); }, 5000); }); } // 使用 async function joinRoom(roomId: string) { try { const response = await emitAsync<JoinResponse>('join-room', roomId); console.log('加入成功:', response); } catch (error) { console.error('加入失败:', error); } }

🔧 监听器管理

移除监听器

// 定义命名函数 function onMessage(data) { console.log('消息:', data); } // 添加监听 socket.on('message', onMessage); // 移除特定监听 socket.off('message', onMessage); // 移除某事件的所有监听 socket.off('message'); // 移除所有监听 socket.removeAllListeners();

React 清理监听器

import { useEffect, useState } from 'react'; function ChatRoom({ roomId }: { roomId: string }) { const [messages, setMessages] = useState<Message[]>([]); useEffect(() => { // 定义处理函数 const handleNewMessage = (message: Message) => { setMessages(prev => [...prev, message]); }; const handleUserJoined = (user: User) => { console.log(`${user.name} 加入了房间`); }; // 添加监听 socket.on('new-message', handleNewMessage); socket.on('user-joined', handleUserJoined); // 加入房间 socket.emit('join-room', roomId); // 清理函数 - 组件卸载时执行 return () => { socket.off('new-message', handleNewMessage); socket.off('user-joined', handleUserJoined); socket.emit('leave-room', roomId); }; }, [roomId]); // roomId 变化时重新执行 return ( <div> {messages.map(msg => ( <div key={msg.id}>{msg.content}</div> ))} </div> ); }

⚡ 高级技巧

事件拦截器

// 创建事件拦截器 const originalOn = socket.on.bind(socket); socket.on = (event, handler) => { const wrappedHandler = (...args) => { console.log(`[Event] ${event}`, args); handler(...args); }; return originalOn(event, wrappedHandler); }; // 所有事件都会被记录 socket.on('message', (data) => { // ... }); // 控制台: [Event] message [{ content: '...' }]

事件队列（离线时缓存）

class SocketQueue { private queue: Array<{ event: string; args: any[] }> = []; emit(event: string, ...args: any[]) { if (socket.connected) { socket.emit(event, ...args); } else { // 离线时缓存 this.queue.push({ event, args }); console.log(`消息已缓存: ${event}`); } } flush() { while (this.queue.length > 0 && socket.connected) { const { event, args } = this.queue.shift()!; socket.emit(event, ...args); } } } const socketQueue = new SocketQueue(); // 连接成功后发送缓存 socket.on('connect', () => { socketQueue.flush(); }); // 使用 socketQueue.emit('send-message', 'Hello');

事件类型映射

// 统一的事件处理 type EventHandler<T = any> = (data: T) => void; class EventEmitter<Events extends Record<string, any>> { private handlers = new Map<keyof Events, Set<EventHandler>>(); on<K extends keyof Events>(event: K, handler: EventHandler<Events[K]>) { if (!this.handlers.has(event)) { this.handlers.set(event, new Set()); } this.handlers.get(event)!.add(handler); } off<K extends keyof Events>(event: K, handler: EventHandler<Events[K]>) { this.handlers.get(event)?.delete(handler); } emit<K extends keyof Events>(event: K, data: Events[K]) { this.handlers.get(event)?.forEach(handler => handler(data)); } } // 使用 interface AppEvents { 'user-joined': { userId: string; username: string }; 'user-left': { userId: string }; 'message': { content: string; timestamp: number }; } const events = new EventEmitter<AppEvents>(); events.on('user-joined', (data) => { console.log(`${data.username} 加入了`); });

📊 内置事件

事件名 说明 参数
connect 连接成功 无
connect_error 连接错误 Error 对象
disconnect 断开连接 断开原因字符串
reconnect 重连成功 重连次数
reconnect_attempt 重连尝试 尝试次数
reconnect_error 重连错误 Error 对象
reconnect_failed 重连失败 无
