向量数据库生态

随着AI应用的普及，向量数据库生态日益丰富。从开源的Milvus到托管的Pinecone， 从云原生的Weaviate到轻量级的Qdrant，不同的向量数据库各有特色， 选择合适的工具对于构建高效的AI应用至关重要。

🔍 向量数据库对比

主流产品对比

数据库 类型 索引算法 扩展性 元数据过滤 优势 适用场景
Milvus 开源 HNSW/IVF/PQ 分布式 强 性能高、功能全 企业级大规模
Pinecone 托管 HNSW/PQ 云端扩展 中 简单易用 快速原型
Weaviate 开源/托管 HNSW 单机/集群 强 语义搜索、GraphQL RAG、知识图谱
Qdrant 开源/托管 HNSW 分布式 强 Rust高性能 实时应用
Chroma 开源 HNSW 单机 中 简单易用 原型开发
Pgvector 扩展 IVF/HNSW PostgreSQL 强 集成PostgreSQL 已有PostgreSQL

使用Milvus

from pymilvus import connections, Collection, FieldSchema, CollectionSchema, DataType import numpy as np # 连接Milvus connections.connect(host='localhost', port='19530') # 定义Collection Schema fields = [ FieldSchema(name="id", dtype=DataType.INT64, is_primary=True, auto_id=True), FieldSchema(name="embedding", dtype=DataType.FLOAT_VECTOR, dim=768), FieldSchema(name="category", dtype=DataType.VARCHAR, max_length=100), FieldSchema(name="timestamp", dtype=DataType.INT64) ] schema = CollectionSchema( fields=fields, description="Document embeddings" ) # 创建Collection collection = Collection(name="documents", schema=schema) # 插入数据 data = [ np.random.random((1000, 768)).astype('float32'), # embeddings ["category_" + str(i % 10) for i in range(1000)], # categories [int(time.time()) for _ in range(1000)] # timestamps ] collection.insert(data) collection.flush() # 创建索引 index_params = { "metric_type": "IP", "index_type": "HNSW", "params": { "M": 16, "efConstruction": 200 } } collection.create_index(field_name="embedding", index_params=index_params) # 加载Collection到内存 collection.load() # 搜索 search_params = { "metric_type": "IP", "params": {"ef": 64} } query_vector = np.random.random((1, 768)).astype('float32') results = collection.search( data=query_vector, anns_field="embedding", param=search_params, limit=10, expr="category == 'category_1'", # 元数据过滤 output_fields=["category", "timestamp"] ) # 处理结果 for result in results[0]: print(f"ID: {result.id}, Distance: {result.distance}") print(f"Category: {result.entity.get('category')}")

Milvus高级特性

# 1. 分区 collection.create_partition(name="partition_a") # 向特定分区插入数据 partition = Partition(collection, name="partition_a") partition.insert(data) # 只在特定分区搜索 results = collection.search( data=query_vector, anns_field="embedding", param=search_params, limit=10, expr="category == 'category_1'", partition_names=["partition_a"] ) # 2. 删除向量 collection.delete(expr="id in [1, 2, 3]") # 3. 导入/导出数据 from pymilvus import utility # 导出为JSON utility.export_data('documents', 'export_data.json') # 从JSON导入 utility.import_data('documents', 'import_data.json')

使用Pinecone

import pinecone import numpy as np # 初始化Pinecone pinecone.init( api_key="your-api-key", environment="us-east-1-aws" ) # 创建Index index_name = "document-search" # 如果不存在则创建 if index_name not in pinecone.list_indexes(): pinecone.create_index( name=index_name, dimension=768, metric="cosine", pods=1, pod_type="p1.x1" ) # 连接到Index index = pinecone.Index(index_name) # 插入向量 vectors = [ (str(i), np.random.random(768).tolist(), { "category": f"cat_{i % 5}", "timestamp": int(time.time()) }) for i in range(1000) ] index.upsert(vectors=vectors) # 搜索 query_vector = np.random.random(768).tolist() results = index.query( vector=query_vector, top_k=10, filter={"category": {"$eq": "cat_1"}}, # 元数据过滤 include_metadata=True ) # 处理结果 for match in results['matches']: print(f"ID: {match['id']}, Score: {match['score']}") print(f"Metadata: {match['metadata']}") # 删除向量 index.delete(ids=["1", "2", "3"]) # 批量操作 index.delete(filter={"category": {"$eq": "cat_1"}})

🔮 Weaviate详解

核心特性

Weaviate的独特之处

  • 语义搜索: 内置向量模型，自动向量化
  • GraphQL API: 灵活的查询语言
  • 模块化: 可扩展的向量模块
  • 知识图谱: 支持关系型数据
  • 多模态: 支持文本、图像、音频
  • 实时更新: 无需重建索引

使用Weaviate

import weaviate import json # 连接Weaviate client = weaviate.Client("http://localhost:8080") # 创建Schema schema = { "classes": [ { "class": "Document", "description": "A document with text content", "vectorizer": "text2vec-transformers", # 自动向量化 "moduleConfig": { "text2vec-transformers": { "vectorizeClassName": True } }, "properties": [ { "name": "title", "dataType": ["text"], "description": "The title of the document" }, { "name": "content", "dataType": ["text"], "description": "The content of the document" }, { "name": "category", "dataType": ["string"], "description": "The category of the document" } ] } ] } client.schema.create(schema) # 添加数据 with client.batch as batch: for i in range(100): properties = { "title": f"Document {i}", "content": f"This is the content of document {i}", "category": f"category_{i % 5}" } client.batch.add_data_object( properties, "Document" ) # 语义搜索 near_text = { "concepts": ["machine learning"], "distance": 0.7 } response = ( client.query .get("Document", ["title", "content", "category"]) .with_near_text(near_text) .do() ) # 处理结果 for result in response["data"]["Get"]["Document"]: print(f"Title: {result['title']}") print(f"Category: {result['category']}") # 过滤 + 语义搜索 response = ( client.query .get("Document", ["title", "content", "category"]) .with_where({ "path": ["category"], "operator": "Equal", "valueString": "category_1" }) .with_near_text(near_text) .with_limit(5) .do() )

Qdrant

from qdrant_client import QdrantClient from qdrant_client.models import Distance, VectorParams, PointStruct # 连接Qdrant client = QdrantClient(host="localhost", port=6333) # 创建Collection client.recreate_collection( collection_name="documents", vectors_config=VectorParams(size=768, distance=Distance.COSINE) ) # 插入数据 client.upsert( collection_name="documents", points=[ PointStruct( id=i, vector=np.random.random(768).tolist(), payload={"category": f"cat_{i % 5}"} ) for i in range(1000) ] ) # 搜索 results = client.search( collection_name="documents", query_vector=np.random.random(768).tolist(), query_filter={ "must": [ { "key": "category", "match": {"value": "cat_1"} } ] }, limit=10 )

Pgvector

-- 安装扩展 CREATE EXTENSION vector; -- 创建表 CREATE TABLE documents ( id SERIAL PRIMARY KEY, content TEXT, category VARCHAR(100), embedding vector(768) ); -- 创建索引 CREATE INDEX ON documents USING hnsw (embedding vector_cosine_ops); -- 插入数据 INSERT INTO documents (content, category, embedding) VALUES ('Document 1', 'cat_1', '[0.1, 0.2, ...]'), ('Document 2', 'cat_2', '[0.3, 0.4, ...]'); -- 搜索 SELECT id, content, category, 1 - (embedding <=> '[0.1, 0.2, ...]') as similarity FROM documents WHERE category = 'cat_1' ORDER BY embedding <=> '[0.1, 0.2, ...]' LIMIT 10;

Chroma

import chromadb from chromadb.config import Settings # 创建客户端 client = chromadb.Client(Settings(chroma_db_impl="duckdb+parquet", persist_directory="./db")) # 创建Collection collection = client.create_collection(name="documents") # 添加数据 collection.add( documents=["Document 1", "Document 2"], metadatas=[{"category": "cat_1"}, {"category": "cat_2"}], ids=["doc1", "doc2"] ) # 搜索 results = collection.query( query_texts=["machine learning"], n_results=10, where={"category": "cat_1"} )
💡
选型建议

Milvus适合企业级大规模部署；Pinecone适合快速开发和云原生应用；Weaviate适合语义搜索和知识图谱应用；Qdrant适合需要高性能和简单部署的场景；Pgvector适合已有PostgreSQL基础架构。
