🧹 数据清洗

# 数据清洗脚本
import re
from langdetect import detect

class DataCleaner:
    def __init__(self):
        # 噪声模式
        self.noise_patterns = [
            r'http\S+',  # URL
            r'www\.\S+',  # 网址
            r'[<>{}]',    # 特殊字符
            r'[^a-zA-Z0-9\u4e00-\u9fff\s.,!?;:]',  # 非中英文
        ]

    def clean_text(self, text):
        """清洗文本"""
        # 移除噪声
        for pattern in self.noise_patterns:
            text = re.sub(pattern, ' ', text)

        # 标准化空格
        text = re.sub(r'\s+', ' ', text).strip()

        # 移除过短文本
        if len(text) < 10:
            return None

        # 语言检测
        try:
            lang = detect(text)
            if lang not in ['en', 'zh', 'zh-cn']:
                return None
        except:
            return None

        return text

    def remove_duplicates(self, dataset):
        """去重"""
        seen = set()
        unique_data = []
        for item in dataset:
            text_hash = hash(item['text'])
            if text_hash not in seen:
                seen.add(text_hash)
                unique_data.append(item)
        return unique_data

    def filter_by_length(self, dataset, min_len=10, max_len=2048):
        """按长度过滤"""
        filtered = []
        for item in dataset:
            text_len = len(item['text'])
            if min_len <= text_len <= max_len:
                filtered.append(item)
        return filtered

🚀 数据分发优化

# 高效数据加载
import torch
from torch.utils.data import Dataset, DataLoader
import numpy as np

class EfficientDataset(Dataset):
    def __init__(self, data_path, cache_size=10000):
        self.data_path = data_path
        self.cache_size = cache_size
        self.cache = {}

        # 预加载索引
        self.indexes = np.load(f"{data_path}/indexes.npy")

    def __len__(self):
        return len(self.indexes)

    def __getitem__(self, idx):
        # 检查缓存
        if idx in self.cache:
            return self.cache[idx]

        # 加载数据
        data = np.load(f"{self.data_path}/{idx}.npy")

        # 更新缓存
        if len(self.cache) < self.cache_size:
            self.cache[idx] = data
        elif len(self.cache) >= self.cache_size:
            # 移除最旧的缓存
            oldest_key = list(self.cache.keys())[0]
            del self.cache[oldest_key]
            self.cache[idx] = data

        return data

# 创建DataLoader
dataset = EfficientDataset("training_data")

# 使用多进程加载
dataloader = DataLoader(
    dataset,
    batch_size=64,
    shuffle=True,
    num_workers=8,
    pin_memory=True,
    prefetch_factor=2
