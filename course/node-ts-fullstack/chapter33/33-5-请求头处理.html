33.5 请求头处理

掌握 Express 中请求头的读取和处理方法

🎯 学习目标

  • 掌握请求头的读取方法
  • 理解常用请求头的含义
  • 学会使用请求头进行认证和内容协商
  • 了解请求头安全注意事项

📖 常用请求头

请求头 说明 示例
Content-Type 请求体类型 application/json
Accept 接受的内容类型 application/json
Authorization 认证凭证 Bearer token
User-Agent 客户端信息 Mozilla/5.0...
Host 目标主机 example.com
Cookie Cookie 数据 sessionId=abc123
Referer 来源页面 https://example.com
X-Forwarded-For 客户端真实 IP 192.168.1.1

📖 读取请求头

基本读取

app.get('/headers', (req, res) => { // 所有请求头 console.log(req.headers); // 获取单个请求头（推荐方式） const contentType = req.get('Content-Type'); const auth = req.get('Authorization'); // 直接访问（小写） const userAgent = req.headers['user-agent']; const host = req.headers.host; res.json({ contentType, auth, userAgent, host }); });

req.get() 方法

// req.get(name) 不区分大小写 req.get('Content-Type'); req.get('content-type'); req.get('CONTENT-TYPE'); // 以上三种写法等效 // 返回值 // 存在 -> 返回字符串 // 不存在 -> 返回 undefined // 处理可能不存在的头 const token = req.get('Authorization') || ''; if (!token) { return res.status(401).json({ error: 'No token' }); }

🔐 Authorization 头处理

// 解析 Bearer Token function extractBearerToken(req) { const auth = req.get('Authorization'); if (!auth) return null; const parts = auth.split(' '); if (parts.length !== 2 || parts[0] !== 'Bearer') { return null; } return parts[1]; } // 认证中间件 function authMiddleware(req, res, next) { const token = extractBearerToken(req); if (!token) { return res.status(401).json({ error: 'Authorization header missing or invalid' }); } try { const decoded = jwt.verify(token, process.env.JWT_SECRET); req.user = decoded; next(); } catch (error) { return res.status(401).json({ error: 'Invalid token' }); } } // Basic Auth function parseBasicAuth(req) { const auth = req.get('Authorization'); if (!auth || !auth.startsWith('Basic ')) return null; const encoded = auth.slice(6); const decoded = Buffer.from(encoded, 'base64').toString('utf-8'); const [username, password] = decoded.split(':'); return { username, password }; }

📋 Content-Type 处理

// 检查请求内容类型 app.post('/api/data', (req, res) => { const contentType = req.get('Content-Type'); if (!contentType || !contentType.includes('application/json')) { return res.status(415).json({ error: 'Unsupported Media Type', message: 'Content-Type must be application/json' }); } // 处理 JSON 请求 res.json(req.body); }); // 使用 req.is() 检查类型 app.post('/upload', (req, res) => { if (req.is('multipart/form-data')) { // 文件上传处理 } else if (req.is('application/json')) { // JSON 数据处理 } else { res.status(415).json({ error: 'Unsupported content type' }); } }); // 检查多种类型 app.post('/submit', (req, res) => { const type = req.is(['json', 'urlencoded']); if (type === false) { return res.status(415).json({ error: 'Unsupported type' }); } // type 为 'json' 或 'urlencoded' res.json(req.body); });

🌐 Accept 头和内容协商

// 根据客户端接受类型返回不同格式 app.get('/api/data', (req, res) => { const data = { message: 'Hello' }; // 检查客户端接受什么类型 const accepted = req.accepts(['json', 'xml', 'html']); switch (accepted) { case 'json': res.json(data); break; case 'xml': res.set('Content-Type', 'application/xml'); res.send(`${data.message}`); break; case 'html': res.send(`

${data.message}

`); break; default: res.status(406).send('Not Acceptable'); } }); // Accept-Encoding - 压缩协商 app.get('/large-data', (req, res) => { const encoding = req.acceptsEncodings(['gzip', 'deflate', 'br']); if (encoding) { // 使用对应的压缩方式 console.log(`Client accepts ${encoding} encoding`); } // 通常由 compression 中间件自动处理 }); // Accept-Language - 语言协商 app.get('/', (req, res) => { const language = req.acceptsLanguages(['zh', 'en', 'ja']); switch (language) { case 'zh': res.send('你好'); break; case 'en': res.send('Hello'); break; case 'ja': res.send('こんにちは'); break; default: res.send('Hello'); } });

📱 User-Agent 处理

// 简单的设备检测 function detectDevice(userAgent) { const ua = userAgent.toLowerCase(); if (/mobile|android|iphone|ipad|ipod|blackberry|iemobile|opera mini/i.test(ua)) { return 'mobile'; } if (/tablet|ipad/i.test(ua)) { return 'tablet'; } return 'desktop'; } app.get('/', (req, res) => { const userAgent = req.get('User-Agent'); const device = detectDevice(userAgent); res.json({ userAgent, device }); }); // 使用 ua-parser-js 进行详细解析 const UAParser = require('ua-parser-js'); app.get('/ua-info', (req, res) => { const parser = new UAParser(); parser.setUA(req.get('User-Agent')); res.json({ browser: parser.getBrowser(), os: parser.getOS(), device: parser.getDevice(), engine: parser.getEngine(), cpu: parser.getCPU() }); });

🔄 X-Forwarded-For 和真实 IP

// 获取客户端真实 IP（反向代理后） function getClientIP(req) { return req.headers['x-forwarded-for']?.split(',')[0].trim() || req.headers['x-real-ip'] || req.connection?.remoteAddress || req.socket?.remoteAddress; } // Express 设置信任代理 app.set('trust proxy', true); // 信任所有代理 // 或指定特定代理 app.set('trust proxy', 'loopback, 123.123.123.123'); // 或使用函数 app.set('trust proxy', (ip) => { return ip === '127.0.0.1' || ip.startsWith('10.'); }); // 然后可以直接使用 req.ip app.get('/ip', (req, res) => { res.json({ ip: req.ip, ips: req.ips // X-Forwarded-For 数组 }); });

🛡️ 自定义请求头

// 读取自定义请求头 app.post('/webhook', (req, res) => { const signature = req.get('X-Webhook-Signature'); const timestamp = req.get('X-Webhook-Timestamp'); const eventType = req.get('X-Event-Type'); // 验证签名 if (!verifySignature(req.body, signature)) { return res.status(401).json({ error: 'Invalid signature' }); } res.json({ received: true }); }); // CORS 相关自定义头 // 客户端需要设置 Access-Control-Request-Headers // 服务端需要设置 Access-Control-Allow-Headers app.options('/api', (req, res) => { res.set('Access-Control-Allow-Headers', 'X-API-Key, X-Request-ID'); res.sendStatus(204); }); app.get('/api', (req, res) => { const apiKey = req.get('X-API-Key'); const requestId = req.get('X-Request-ID'); res.json({ apiKey, requestId }); });

✅ 小结

📌
本节要点
  • 使用 req.get() 读取请求头，不区分大小写
  • Authorization 头用于认证，常用 Bearer Token
  • Accept 头用于内容协商
  • 使用 trust proxy 配置获取真实 IP
  • 自定义请求头常用于 API 认证和追踪
