33.3 文件上传 multer

使用 multer 中间件处理文件上传

🎯 学习目标

  • 掌握 multer 的基本配置和使用
  • 学会单文件和多文件上传
  • 理解文件存储策略
  • 掌握文件类型和大小验证

📦 multer 简介

multer 是 Express 官方推荐的文件上传中间件，用于处理 multipart/form-data 类型的表单数据。

# 安装 multer npm install multer

基本概念

方法 说明
.single(fieldname) 单个文件上传
.array(fieldname[, maxCount]) 多个文件上传（同名字段）
.fields(fields) 多个文件上传（不同字段）
.none() 只接受文本字段
.any() 接受所有文件

📤 基本使用

单文件上传

const express = require('express'); const multer = require('multer'); const path = require('path'); const app = express(); // 基本配置 - 保存到临时目录 const upload = multer({ dest: 'uploads/' }); // 单文件上传 app.post('/upload/avatar', upload.single('avatar'), (req, res) => { // req.file - 文件信息 // req.body - 其他表单字段 console.log(req.file); /* { fieldname: 'avatar', originalname: 'photo.jpg', encoding: '7bit', mimetype: 'image/jpeg', destination: 'uploads/', filename: 'abc123', path: 'uploads/abc123', size: 102400 } */ res.json({ message: 'File uploaded', file: req.file }); });

多文件上传（同名字段）

// 最多上传 5 张图片 app.post('/upload/photos', upload.array('photos', 5), (req, res) => { // req.files - 文件数组 console.log(req.files); /* [ { fieldname: 'photos', originalname: '1.jpg', ... }, { fieldname: 'photos', originalname: '2.jpg', ... }, ... ] */ res.json({ message: `${req.files.length} files uploaded`, files: req.files }); });

多字段文件上传

// 不同字段上传不同文件 const uploadFields = upload.fields([ { name: 'avatar', maxCount: 1 }, { name: 'gallery', maxCount: 8 } ]); app.post('/upload/profile', uploadFields, (req, res) => { // req.files - 对象，键为字段名 /* { avatar: [{ ... }], gallery: [{ ... }, { ... }] } */ console.log(req.files); console.log(req.body); // 其他表单字段 res.json({ message: 'Files uploaded', avatar: req.files['avatar']?.[0], gallery: req.files['gallery'] }); });

💾 自定义存储配置

const multer = require('multer'); const path = require('path'); // 自定义存储配置 const storage = multer.diskStorage({ // 目标目录 destination: (req, file, cb) => { // 可以根据文件类型分目录 const dir = file.mimetype.startsWith('image/') ? 'uploads/images' : 'uploads/files'; // 确保目录存在 fs.mkdirSync(dir, { recursive: true }); cb(null, dir); }, // 文件名 filename: (req, file, cb) => { // 保留原始扩展名 const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1E9); const ext = path.extname(file.originalname); cb(null, file.fieldname + '-' + uniqueSuffix + ext); } }); const upload = multer({ storage: storage }); // 内存存储（用于处理后直接上传到云存储） const memoryStorage = multer.memoryStorage(); const uploadToMemory = multer({ storage: memoryStorage }); app.post('/upload/memory', uploadToMemory.single('file'), (req, res) => { // req.file.buffer - 文件 Buffer // 适合直接上传到 S3、OSS 等 console.log(req.file.buffer); res.send('File in memory'); });

🛡️ 文件过滤和限制

const upload = multer({ storage: storage, // 文件过滤器 fileFilter: (req, file, cb) => { // 允许的文件类型 const allowedTypes = /jpeg|jpg|png|gif|webp/; const extname = allowedTypes.test( path.extname(file.originalname).toLowerCase() ); const mimetype = allowedTypes.test(file.mimetype); if (extname && mimetype) { cb(null, true); } else { cb(new Error('Only image files are allowed')); } }, // 大小限制 limits: { fileSize: 5 * 1024 * 1024, // 5MB files: 10, // 最多 10 个文件 fields: 20, // 最多 20 个文本字段 fieldNameSize: 100, // 字段名最大长度 fieldSize: 1024 * 1024 // 字段值最大 1MB } });

不同类型文件的过滤

// 图片文件过滤器 const imageFilter = (req, file, cb) => { const allowedMimes = ['image/jpeg', 'image/png', 'image/gif', 'image/webp']; if (allowedMimes.includes(file.mimetype)) { cb(null, true); } else { cb(new Error('Only image files are allowed')); } }; // 文档文件过滤器 const documentFilter = (req, file, cb) => { const allowedMimes = [ 'application/pdf', 'application/msword', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document', 'text/plain' ]; if (allowedMimes.includes(file.mimetype)) { cb(null, true); } else { cb(new Error('Only document files are allowed')); } }; // 使用不同过滤器 const uploadImage = multer({ storage, fileFilter: imageFilter }); const uploadDoc = multer({ storage, fileFilter: documentFilter });

⚠️ 错误处理

// multer 错误处理 app.post('/upload', upload.single('file'), (req, res) => { res.json(req.file); }); // 错误处理中间件 app.use((err, req, res, next) => { if (err instanceof multer.MulterError) { // Multer 错误 switch (err.code) { case 'LIMIT_FILE_SIZE': return res.status(400).json({ error: 'File too large', message: 'Maximum file size is 5MB' }); case 'LIMIT_FILE_COUNT': return res.status(400).json({ error: 'Too many files', message: 'Maximum 10 files allowed' }); case 'LIMIT_UNEXPECTED_FILE': return res.status(400).json({ error: 'Unexpected field', message: err.message }); default: return res.status(400).json({ error: 'Upload error', message: err.message }); } } // 其他错误 if (err.message === 'Only image files are allowed') { return res.status(400).json({ error: 'Invalid file type', message: 'Only image files (jpeg, png, gif, webp) are allowed' }); } next(err); });

☁️ 上传到云存储

const AWS = require('aws-sdk'); const multer = require('multer'); const multerS3 = require('multer-s3'); // 配置 S3 const s3 = new AWS.S3({ accessKeyId: process.env.AWS_ACCESS_KEY, secretAccessKey: process.env.AWS_SECRET_KEY, region: process.env.AWS_REGION }); // S3 存储 const uploadS3 = multer({ storage: multerS3({ s3: s3, bucket: process.env.S3_BUCKET, acl: 'public-read', metadata: (req, file, cb) => { cb(null, { fieldName: file.fieldname }); }, key: (req, file, cb) => { const fileName = `${Date.now()}-${file.originalname}`; cb(null, fileName); } }), limits: { fileSize: 5 * 1024 * 1024 } }); // 上传路由 app.post('/upload/s3', uploadS3.single('file'), (req, res) => { res.json({ message: 'File uploaded to S3', url: req.file.location }); });

📊 完整示例

const express = require('express'); const multer = require('multer'); const path = require('path'); const fs = require('fs'); const app = express(); // 确保上传目录存在 const uploadDir = 'uploads'; fs.mkdirSync(uploadDir, { recursive: true }); // 存储配置 const storage = multer.diskStorage({ destination: (req, file, cb) => { const typeDir = path.join(uploadDir, file.mimetype.split('/')[0]); fs.mkdirSync(typeDir, { recursive: true }); cb(null, typeDir); }, filename: (req, file, cb) => { const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1E9); const ext = path.extname(file.originalname); cb(null, uniqueSuffix + ext); } }); // 文件过滤器 const fileFilter = (req, file, cb) => { const allowedTypes = ['image/jpeg', 'image/png', 'image/gif', 'image/webp']; if (allowedTypes.includes(file.mimetype)) { cb(null, true); } else { cb(new Error(`File type ${file.mimetype} is not allowed`)); } }; // multer 实例 const upload = multer({ storage, fileFilter, limits: { fileSize: 5 * 1024 * 1024, files: 10 } }); // 路由 app.post('/upload/single', upload.single('file'), (req, res) => { res.json({ file: req.file }); }); app.post('/upload/multiple', upload.array('files', 5), (req, res) => { res.json({ files: req.files }); }); // 错误处理 app.use((err, req, res, next) => { if (err instanceof multer.MulterError || err.message.includes('not allowed')) { return res.status(400).json({ error: err.message }); } next(err); }); module.exports = app;
