📚 函数式编程概述

// 函数式编程核心概念 const concepts = { 纯函数: '相同输入总是返回相同输出，无副作用', 不可变性: '数据一旦创建就不能修改', 函数是一等公民: '函数可以作为参数、返回值、赋值给变量', 高阶函数: '接受函数作为参数或返回函数的函数', 组合: '将简单函数组合成复杂函数', 柯里化: '将多参数函数转换为单参数函数链' }; // 命令式 vs 声明式 // 命令式：告诉计算机怎么做 const numbers = [1, 2, 3, 4, 5]; const doubled = []; for (let i = 0; i < numbers.length; i++) { doubled.push(numbers[i] * 2); } // 声明式：告诉计算机要什么 const doubledFP = numbers.map(n => n * 2); // 函数式优势 const advantages = { 可预测性: '纯函数输出可预测，易于测试', 可组合性: '小函数组合成大函数', 可测试性: '纯函数易于单元测试', 并发安全: '无共享状态，避免竞态条件', 调试友好: '调用链清晰，问题易定位' };
💡
函数式思想

用函数描述"做什么"而非"怎么做"，避免状态变化和副作用。

🔧 副作用与引用透明

// 副作用：函数对外部状态的修改 // 有副作用 let counter = 0; function increment() { counter++; // 修改外部状态 return counter; } // 无副作用（纯函数） function pureIncrement(count: number): number { return count + 1; // 不修改外部状态 } // 引用透明：函数调用可以用返回值替换 // 引用透明的函数 function add(a: number, b: number): number { return a + b; } const result = add(1, 2); // 可以直接替换为 3 const result2 = 3; // 不透明的函数 function addRandom(a: number): number { return a + Math.random(); // 每次调用结果不同 } // 避免副作用的方法 const practices = { 不修改参数: '函数内不修改传入的对象', 不修改外部变量: '函数内不修改外部作用域变量', 不做IO操作: '不在纯函数中做IO操作', 返回新对象: '需要修改时返回新对象而非修改原对象' }; // 不可变数据实践 interface User { name: string; age: number; } // 错误：直接修改 function updateAgeBad(user: User, newAge: number): void { user.age = newAge; // 修改原对象 } // 正确：返回新对象 function updateAgeGood(user: User, newAge: number): User { return { ...user, age: newAge }; // 返回新对象 } const user = { name: 'Alice', age: 30 }; const updatedUser = updateAgeGood(user, 31); console.log(user); // { name: 'Alice', age: 30 } console.log(updatedUser); // { name: 'Alice', age: 31 }
