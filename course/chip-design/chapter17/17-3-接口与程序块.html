17.3 接口与程序块

SystemVerilog 语言精讲

  • 学会program块的使用

🔌 接口（Interface）

// 定义接口
interface bus_if #(parameter WIDTH = 8);
    logic [WIDTH-1:0] data;
    logic              valid;
    logic              ready;

    // modport定义接口方向
    modport master (
        output data,
        output valid,
        input  ready
    );

    modport slave (
        input  data,
        input  valid,
        output ready
    );
endinterface

⏰ Clocking Block

// Clocking block用于同步接口
interface clk_if;
    logic clk;
    logic data;
    logic valid;

    clocking cb @(posedge clk);
        default input #1step output #0;
        input data, valid;
    endclocking
endinterface
