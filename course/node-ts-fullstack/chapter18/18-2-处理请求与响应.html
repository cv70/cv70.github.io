18.2 处理请求与响应

深入理解请求处理和响应发送

🎯 学习目标

  • 掌握请求体解析
  • 学会处理不同类型请求
  • 掌握响应发送技巧
  • 了解 Cookie 和 Session

📚 解析请求体

const http = require('http'); const querystring = require('querystring'); const server = http.createServer((req, res) => { // 解析 JSON 请求体 if (req.headers['content-type'] === 'application/json') { let body = ''; req.on('data', (chunk) => body += chunk); req.on('end', () => { const data = JSON.parse(body); console.log('JSON 数据:', data); res.end(JSON.stringify({ received: data })); }); } // 解析表单数据 if (req.headers['content-type'] === 'application/x-www-form-urlencoded') { let body = ''; req.on('data', (chunk) => body += chunk); req.on('end', () => { const data = querystring.parse(body); console.log('表单数据:', data); res.end('表单已接收'); }); } // 处理大文件上传 if (req.headers['content-type']?.startsWith('multipart/form-data')) { let size = 0; req.on('data', (chunk) => { size += chunk.length; console.log(`已接收 ${size} 字节`); }); req.on('end', () => { res.end(`上传完成，大小: ${size} 字节`); }); } }); // 通用请求体解析函数 function parseBody(req) { return new Promise((resolve, reject) => { let body = ''; req.on('data', (chunk) => body += chunk); req.on('end', () => { try { const contentType = req.headers['content-type']; if (contentType?.includes('application/json')) { resolve(JSON.parse(body)); } else if (contentType?.includes('form-urlencoded')) { resolve(querystring.parse(body)); } else { resolve(body); } } catch (err) { reject(err); } }); req.on('error', reject); }); }

🔧 处理不同请求方法

const http = require('http'); const server = http.createServer(async (req, res) => { const method = req.method; const url = new URL(req.url, `http://${req.headers.host}`); // 根据方法分发 switch (method) { case 'GET': handleGet(req, res, url); break; case 'POST': handlePost(req, res); break; case 'PUT': handlePut(req, res); break; case 'DELETE': handleDelete(req, res); break; default: res.writeHead(405); res.end('Method Not Allowed'); } }); function handleGet(req, res, url) { const path = url.pathname; const query = Object.fromEntries(url.searchParams); if (path === '/users') { res.writeHead(200, { 'Content-Type': 'application/json' }); res.end(JSON.stringify({ users: [], query })); } else { res.writeHead(404); res.end('Not Found'); } } async function handlePost(req, res) { const body = await parseBody(req); res.writeHead(201, { 'Content-Type': 'application/json' }); res.end(JSON.stringify({ created: body })); } async function handlePut(req, res) { const body = await parseBody(req); res.writeHead(200, { 'Content-Type': 'application/json' }); res.end(JSON.stringify({ updated: body })); } function handleDelete(req, res) { res.writeHead(204); res.end(); }

📦 发送响应

const http = require('http'); const fs = require('fs'); const server = http.createServer((req, res) => { // 发送 JSON 响应 function sendJson(data, status = 200) { res.writeHead(status, { 'Content-Type': 'application/json' }); res.end(JSON.stringify(data)); } // 发送 HTML 响应 function sendHtml(html, status = 200) { res.writeHead(status, { 'Content-Type': 'text/html; charset=utf-8' }); res.end(html); } // 发送文件 function sendFile(filepath) { const stream = fs.createReadStream(filepath); stream.on('error', () => { res.writeHead(404); res.end('File Not Found'); }); stream.pipe(res); } // 设置 Cookie function setCookie(name, value, options = {}) { let cookie = `${name}=${encodeURIComponent(value)}`; if (options.maxAge) cookie += `; Max-Age=${options.maxAge}`; if (options.path) cookie += `; Path=${options.path}`; if (options.httpOnly) cookie += '; HttpOnly'; if (options.secure) cookie += '; Secure'; res.setHeader('Set-Cookie', cookie); } // 读取 Cookie function getCookies() { const cookies = {}; req.headers.cookie?.split(';').forEach(cookie => { const [name, value] = cookie.trim().split('='); cookies[name] = decodeURIComponent(value); }); return cookies; } // 使用示例 if (req.url === '/') { sendHtml('

⚙️ CORS 处理

const http = require('http'); const server = http.createServer((req, res) => { // 设置 CORS 头 res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type, Authorization'); res.setHeader('Access-Control-Max-Age', '86400'); // 处理预检请求 if (req.method === 'OPTIONS') { res.writeHead(204); res.end(); return; } // 正常处理请求 res.writeHead(200, { 'Content-Type': 'application/json' }); res.end(JSON.stringify({ message: 'Hello CORS' })); }); // 指定允许的域名 function setCORS(res, origin) { const allowedOrigins = ['http://localhost:3000', 'https://example.com']; if (allowedOrigins.includes(origin)) { res.setHeader('Access-Control-Allow-Origin', origin); } } // 带凭证的 CORS function setCORSWithCredentials(res, origin) { res.setHeader('Access-Control-Allow-Origin', origin); res.setHeader('Access-Control-Allow-Credentials', 'true'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); }
