71.5 测试数据库操作

第十部分：测试篇 - 数据库集成测试最佳实践

1. 测试数据库策略

内存数据库

使用 SQLite 内存数据库，速度快，适合单元/集成测试。

  • 快速执行
  • 无需外部依赖
  • 每次测试全新状态

测试数据库

使用真实的测试数据库，更接近生产环境。

  • 真实数据库行为
  • 验证 SQL 语法
  • 测试数据库特定功能

2. 配置测试数据库

// tests/setup/database.ts import { Connection, createConnection, ConnectionOptions } from 'typeorm'; import { User } from '../../src/entities/User'; import { Post } from '../../src/entities/Post'; let connection: Connection; export async function setupTestDatabase(): Promise<Connection> { const config: ConnectionOptions = { type: 'sqlite', database: ':memory:', entities: [User, Post], synchronize: true, dropSchema: true }; connection = await createConnection(config); return connection; } export async function teardownTestDatabase(): Promise<void> { if (connection && connection.isConnected) { await connection.close(); } } export function getConnection(): Connection { return connection; }
// tests/setup/global-setup.ts import { setupTestDatabase, teardownTestDatabase } from './database'; export = async () => { await setupTestDatabase(); // global teardown return async () => { await teardownTestDatabase(); }; };

3. Repository 测试

// tests/repositories/user.repository.test.ts import { getConnection } from 'typeorm'; import { UserRepository } from '../../src/repositories/user.repository'; import { User } from '../../src/entities/User'; import { setupTestDatabase, teardownTestDatabase } from '../setup/database'; describe('UserRepository', () => { let repository: UserRepository; beforeAll(async () => { await setupTestDatabase(); const connection = getConnection(); repository = new UserRepository(connection.getRepository(User)); }); afterAll(async () => { await teardownTestDatabase(); }); beforeEach(async () => { // 清空数据 await getConnection().getRepository(User).clear(); }); describe('findById', () => { it('should find user by id', async () => { // 创建测试数据 const user = await repository.create({ name: 'John Doe', email: 'john@example.com' }); const found = await repository.findById(user.id); expect(found).toBeDefined(); expect(found?.name).toBe('John Doe'); }); it('should return null for non-existent user', async () => { const found = await repository.findById('non-existent-id'); expect(found).toBeNull(); }); }); describe('create', () => { it('should create a new user', async () => { const user = await repository.create({ name: 'Jane Doe', email: 'jane@example.com' }); expect(user.id).toBeDefined(); expect(user.name).toBe('Jane Doe'); }); it('should reject duplicate email', async () => { await repository.create({ name: 'User 1', email: 'same@example.com' }); await expect(repository.create({ name: 'User 2', email: 'same@example.com' })).rejects.toThrow(); }); }); describe('update', () => { it('should update user', async () => { const user = await repository.create({ name: 'Original Name', email: 'original@example.com' }); const updated = await repository.update(user.id, { name: 'Updated Name' }); 4. Service 层测试

// tests/services/user.service.test.ts import { getConnection } from 'typeorm'; import { UserService } from '../../src/services/user.service'; import { UserRepository } from '../../src/repositories/user.repository'; import { User } from '../../src/entities/User'; import { setupTestDatabase, teardownTestDatabase } from '../setup/database'; describe('UserService', () => { let service: UserService; let repository: UserRepository; beforeAll(async () => { await setupTestDatabase(); const connection = getConnection(); repository = new UserRepository(connection.getRepository(User)); service = new UserService(repository); }); afterAll(async () => { await teardownTestDatabase(); }); beforeEach(async () => { await getConnection().getRepository(User).clear(); }); describe('createUser', () => { it('should create user with hashed password', async () => { const user = await service.createUser({ name: 'John', email: 'john@example.com', password: 'plainPassword' }); expect(user.password).not.toBe('plainPassword'); expect(user.name).toBe('John'); }); it('should validate email format', async () => { await expect(service.createUser({ name: 'John', email: 'invalid-email', password: 'password' })).rejects.toThrow('Invalid email format'); }); }); describe('authenticate', () => { it('should authenticate with correct credentials', async () => { await service.createUser({ name: 'John', email: 'john@example.com', password: 'password123' }); const result = await service.authenticate({ email: 'john@example.com', password: 'password123' }); expect(result).toBeDefined(); expect(result.token).toBeDefined(); }); it('should reject wrong password', async () => { await service.createUser({ name: 'John', email: 'john@example.com', password: 'password123' }); await expect(service.authenticate({ email: 'john@example.com', password: 'wrongpassword' })).rejects.toThrow('Invalid credentials'); }); }); });

5. 数据库事务测试

describe('Transaction Tests', () => { it('should rollback on error', async () => { const connection = getConnection(); const queryRunner = connection.createQueryRunner(); await queryRunner.startTransaction(); try { // 创建用户 await queryRunner.manager.save(User, { name: 'User 1', email: 'user1@example.com' }); // 模拟错误 throw new Error('Simulated error'); } catch (error) { await queryRunner.rollbackTransaction(); // 验证回滚 const users = await queryRunner.manager.find(User); expect(users).toHaveLength(0); } finally { await queryRunner.release(); } }); it('should commit on success', async () => { const connection = getConnection(); const queryRunner = connection.createQueryRunner(); await queryRunner.startTransaction(); try { await queryRunner.manager.save(User, { name: 'User 1', email: 'user1@example.com' }); await queryRunner.commitTransaction(); const users = await connection.manager.find(User); expect(users).toHaveLength(1); } finally { await queryRunner.release(); } }); });

6. 测试数据工厂

// tests/factories/user.factory.ts import { Connection } from 'typeorm'; import { User } from '../../src/entities/User'; type UserFactoryOptions = Partial<User>; export async function createUser( connection: Connection, options: UserFactoryOptions = {} ): Promise<User> { const defaultOptions = { name: 'Test User', email: `test${Date.now()}@example.com`, password: 'hashedPassword', isActive: true }; const user = connection.manager.create(User, { ...defaultOptions, ...options }); return connection.manager.save(user); } export async function createUsers( connection: Connection, count: number, options: UserFactoryOptions = {} ): Promise<User[]> { const users: User[] = []; for (let i = 0; i < count; i++) { const user = await createUser(connection, { ...options, email: options.email || `user${i}@example.com` }); users.push(user); } return users; }
// 使用工厂 import { createUser, createUsers } from '../factories/user.factory'; import { getConnection } from 'typeorm'; describe('with test data', () => { it('works with factory-created data', async () => { const connection = getConnection(); const user = await createUser(connection, { name: 'Specific Name' }); expect(user.name).toBe('Specific Name'); }); it('can create multiple users', async () => { const connection = getConnection(); const users = await createUsers(connection, 5); expect(users).toHaveLength(5); }); });

7. 数据库测试最佳实践

✅ 推荐

  • 每次测试后清理数据
  • 使用内存数据库加速测试
  • 创建测试数据工厂
  • 测试边界条件
  • 使用事务隔离测试

❌ 避免

  • 测试间共享数据
  • 依赖测试执行顺序
  • 使用生产数据库
  • 忘记清理数据
  • 硬编码测试数据
