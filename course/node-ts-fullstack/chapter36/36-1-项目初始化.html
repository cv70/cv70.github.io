36.1 项目初始化

从零开始搭建 Express + TypeScript 项目

🎯 学习目标

  • 掌握 TypeScript 项目初始化流程
  • 学会配置 tsconfig.json
  • 理解 TypeScript 项目结构
  • 掌握开发环境配置

📦 项目初始化步骤

# 1. 创建项目目录 mkdir express-ts-project cd express-ts-project # 2. 初始化 package.json npm init -y # 3. 安装 Express npm install express # 4. 安装 TypeScript 和类型定义 npm install -D typescript @types/node @types/express # 5. 安装开发工具 npm install -D ts-node nodemon # 6. 初始化 tsconfig.json npx tsc --init

⚙️ tsconfig.json 配置

{ "compilerOptions": { "target": "ES2020", // 编译目标 "module": "commonjs", // 模块系统 "lib": ["ES2020"], // 包含的库 "outDir": "./dist", // 输出目录 "rootDir": "./src", // 源码目录 "strict": true, // 严格模式 "esModuleInterop": true, // ES 模块互操作 "skipLibCheck": true, // 跳过库检查 "forceConsistentCasingInFileNames": true, // 路径别名 "baseUrl": ".", "paths": { "@/*": ["src/*"], "@routes/*": ["src/routes/*"], "@controllers/*": ["src/controllers/*"], "@middleware/*": ["src/middleware/*"] }, // 装饰器支持（如需要） "experimentalDecorators": true, "emitDecoratorMetadata": true }, "include": ["src/**/*"], "exclude": ["node_modules", "dist"] }

📁 项目结构

express-ts-project/ ├── src/ │ ├── config/ │ │ └── index.ts # 配置 │ ├── controllers/ │ │ └── userController.ts # 控制器 │ ├── middleware/ │ │ ├── auth.ts # 认证中间件 │ │ └── errorHandler.ts # 错误处理 │ ├── models/ │ │ └── User.ts # 数据模型 │ ├── routes/ │ │ ├── index.ts # 路由汇总 │ │ └── userRoutes.ts # 用户路由 │ ├── services/ │ │ └── userService.ts # 业务逻辑 │ ├── types/ │ │ └── express.d.ts # 类型扩展 │ ├── utils/ │ │ └── helpers.ts # 工具函数 │ ├── app.ts # Express 配置 │ └── server.ts # 入口文件 ├── dist/ # 编译输出 ├── nodemon.json ├── tsconfig.json └── package.json

📝 package.json 脚本

{ "name": "express-ts-project", "version": "1.0.0", "main": "dist/server.js", "scripts": { "dev": "nodemon", "build": "tsc", "start": "node dist/server.js", "lint": "eslint src/**/*.ts", "format": "prettier --write src/**/*.ts" }, "dependencies": { "express": "^4.18.2" }, "devDependencies": { "@types/express": "^4.17.17", "@types/node": "^20.0.0", "nodemon": "^3.0.0", "ts-node": "^10.9.0", "typescript": "^5.0.0" } }

🔄 nodemon 配置

// nodemon.json { "watch": ["src"], "ext": "ts,json", "ignore": ["src/**/*.spec.ts"], "exec": "ts-node src/server.ts" }

🚀 基本代码

// src/app.ts import express, { Express, Request, Response, NextFunction } from 'express'; const app: Express = express(); // 中间件 app.use(express.json()); app.use(express.urlencoded({ extended: true })); // 路由 app.get('/', (req: Request, res: Response) => { res.json({ message: 'Hello, TypeScript!' }); }); // 404 处理 app.use((req: Request, res: Response) => { res.status(404).json({ error: 'Not Found' }); }); // 错误处理 app.use((err: Error, req: Request, res: Response, next: NextFunction) => { console.error(err.stack); res.status(500).json({ error: 'Internal Server Error' }); }); export default app; // src/server.ts import app from './app'; const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server running on port ${PORT}`); });
