🧪 测试平台概述

Testbench

测试平台（Testbench）是用于验证设计功能的Verilog代码。它不对应实际硬件，仅用于仿真。测试平台实例化被测设计（DUT），施加激励，并检查响应。

// 测试平台基本结构
module tb;
    // 信号声明
    reg clk, rst_n;
    reg [7:0] data_in;
    wire [7:0] data_out;

    // 实例化被测设计
    dut u_dut (
        .clk(clk),
        .rst_n(rst_n),
        .data_in(data_in),
        .data_out(data_out)
    );

    // 生成激励
    initial begin
        // 初始化
        clk = 0;
        rst_n = 0;
        data_in = 0;

        // 复位
        #100;
        rst_n = 1;

        // 施加激励
        @(posedge clk);
        data_in = 8'hAA;
        @(posedge clk);
        data_in = 8'h55;

        // 结束仿真
        #100;
        $finish;
    end

    // 生成时钟
    always #10 clk = ~clk;

endmodule

🕐 时间控制

// 延迟控制
#10;           // 延迟10个时间单位
#5 a = b;      // 延迟5个时间单位后赋值

// 事件控制
@(posedge clk);      // 等待时钟上升沿
@(negedge rst_n);    // 等待复位下降沿
@(a or b);           // 等待a或b变化
@(posedge clk iff enable);  // 条件事件控制

// 等待表达式
wait (ready);        // 等待ready为真
wait (count == 10);  // 等待count等于10

⏰ 时钟与复位生成

// 方法1：always块生成时钟
reg clk = 0;
always #10 clk = ~clk;  // 50MHz时钟（周期20ns）

// 方法2：initial块生成时钟
reg clk;
initial begin
    clk = 0;
    forever #10 clk = ~clk;
end

// 方法3：可配置时钟
parameter CLK_PERIOD = 20;
reg clk = 0;
always #(CLK_PERIOD/2) clk = ~clk;

// 复位信号生成
reg rst_n;
initial begin
    rst_n = 0;      // 复位有效
    #100;          // 复位持续100ns
    rst_n = 1;      // 释放复位
end

// 带释放的复位
initial begin
    rst_n = 0;
    repeat(5) @(posedge clk);  // 等待5个时钟周期
    rst_n = 1;
end

📊 系统任务与函数

// 显示任务
$display("Hello World");                    // 显示字符串
$display("Time=%0t, Value=%h", $time, data); // 格式化显示
$display("Data: %b %d %h %o", a, a, a, a);   // 多种格式

// 监控任务
$monitor("Time=%0t, clk=%b, data=%h", $time, clk, data);
// 当列表中任意信号变化时自动显示

// 文件操作
integer fd;
initial begin
    fd = $fopen("results.txt", "w");  // 打开文件
    $fdisplay(fd, "Data=%h", data);    // 写入文件
    $fclose(fd);                      // 关闭文件
end

// 读取文件
$readmemh("data.hex", memory);  // 读取十六进制文件到存储器
$readmemb("data.bin", memory);  // 读取二进制文件到存储器

// 仿真控制
$finish;       // 结束仿真
$stop;         // 暂停仿真（交互式）

// 时间函数
$time;         // 返回当前仿真时间
$realtime;     // 返回实数时间
$stime;        // 返回短整数时间

🎲 随机数生成

// 随机数生成
reg [7:0] rand_data;
rand_data = $random;           // 32位有符号随机数
rand_data = $random % 256;     // 0-255的随机数
rand_data = {$random} % 256;   // 无符号随机数

// $urandom：无符号随机数
rand_data = $urandom % 256;    // 0-255的随机数

// $urandom_range：范围内的随机数
rand_data = $urandom_range(0, 255);  // 0-255的随机数
addr = $urandom_range(0, 1023);      // 0-1023的随机地址

// 示例：随机数据流
initial begin
    for (integer i = 0; i < 100; i = i + 1) begin
        @(posedge clk);
        data_in = $urandom_range(0, 255);
        valid = $urandom_range(0, 1);  // 50%概率valid
    end
end

✅ 完整测试平台示例

// 被测设计：4位计数器
module counter_4bit(
    input        clk,
    input        rst_n,
    input        enable,
    output reg [3:0] count
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            count <= 4'b0000;
        else if (enable)
            count <= count + 1;
    end
endmodule

// 测试平台
module tb_counter;
    reg clk, rst_n, enable;
    wire [3:0] count;

    // 实例化DUT
    counter_4bit u_dut (
        .clk(clk),
        .rst_n(rst_n),
        .enable(enable),
        .count(count)
    );

    // 生成时钟
    always #10 clk = ~clk;

    // 测试激励
    initial begin
        // 初始化
        clk = 0;
        rst_n = 0;
        enable = 0;

        // 打开文件记录结果
        integer fd = $fopen("counter_results.txt", "w");
        $fdisplay(fd, "Time, Count");

        // 复位
        #100;
        rst_n = 1;

        // 测试1：使能计数
        $display("Test 1: Enable counting");
        enable = 1;
        repeat(20) begin
            @(posedge clk);
            $fdisplay(fd, "%0t, %d", $time, count);
        end

        // 测试2：禁止计数
        $display("Test 2: Disable counting");
        enable = 0;
        repeat(5) begin
            @(posedge clk);
            $fdisplay(fd, "%0t, %d", $time, count);
        end

        // 测试3：重新使能
        $display("Test 3: Re-enable");
        enable = 1;
        repeat(10) begin
            @(posedge clk);
            $fdisplay(fd, "%0t, %d", $time, count);
        end

        // 关闭文件并结束仿真
        $fclose(fd);
        #100;
        $display("Simulation completed!");
        $finish;
    end

    // 监控输出
    initial begin
        $monitor("Time=%0t, Count=%d", $time, count);
    end

endmodule
