电商系统特点

高并发、事务一致性、库存扣减、订单状态机、支付安全。

🔧 核心业务流程

// 下单核心流程 const orderFlow = ` 1. 用户浏览商品 ↓ 2. 加入购物车 ↓ 3. 选择商品结算 ↓ 4. 填写收货地址 ↓ 5. 选择支付方式 ↓ 6. 创建订单 ↓ 7. 预扣库存 ↓ 8. 发起支付 ↓ 9. 支付成功回调 ↓ 10. 扣减库存 ↓ 11. 生成发货单 ↓ 12. 物流配送 ↓ 13. 确认收货 ↓ 14. 订单完成 `; // 订单状态机 enum OrderStatus { PENDING = 'pending', // 待支付 PAID = 'paid', // 已支付 SHIPPED = 'shipped', // 已发货 DELIVERED = 'delivered', // 已送达 COMPLETED = 'completed', // 已完成 CANCELLED = 'cancelled', // 已取消 REFUNDING = 'refunding', // 退款中 REFUNDED = 'refunded' // 已退款 } const statusTransitions = { pending: ['paid', 'cancelled'], paid: ['shipped', 'refunding'], shipped: ['delivered', 'refunding'], delivered: ['completed', 'refunding'], completed: [], cancelled: [], refunding: ['refunded', 'paid'], refunded: [] };
