40.1 项目配置

搭建 TypeScript + Koa 项目

🎯 学习目标

  • 学会配置 TypeScript 项目
  • 掌握 tsconfig.json 配置
  • 理解开发和生产环境配置
  • 学会使用 nodemon + ts-node

📦 项目初始化

# 创建项目目录 mkdir koa-ts-project cd koa-ts-project # 初始化 package.json npm init -y # 安装 Koa npm install koa @koa/router koa-bodyparser # 安装 TypeScript 相关 npm install -D typescript ts-node nodemon npm install -D @types/koa @types/koa-router @types/koa-bodyparser npm install -D @types/node
💡
类型定义

使用 @types/* 包为 JavaScript 库提供 TypeScript 类型支持。

⚙️ tsconfig.json 配置

{ "compilerOptions": { // 基本配置 "target": "ES2020", "module": "commonjs", "outDir": "./dist", "rootDir": "./src", "baseUrl": "./src", // 严格类型检查 "strict": true, "noImplicitAny": true, "strictNullChecks": true, "strictFunctionTypes": true, // 模块解析 "moduleResolution": "node", "esModuleInterop": true, "allowSyntheticDefaultImports": true, // 源码映射（调试用） "sourceMap": true, // 路径别名 "paths": { "@/*": ["./*"], "@routes/*": ["routes/*"], "@controllers/*": ["controllers/*"], "@middlewares/*": ["middlewares/*"] }, // 装饰器支持（可选） "experimentalDecorators": true, "emitDecoratorMetadata": true }, "include": ["src/**/*"], "exclude": ["node_modules", "dist"] }

📁 项目结构

koa-ts-project/ ├── src/ │ ├── index.ts # 入口文件 │ ├── app.ts # Koa 应用配置 │ ├── config/ │ │ ├── index.ts # 配置入口 │ │ └── database.ts # 数据库配置 │ ├── routes/ │ │ ├── index.ts # 路由入口 │ │ └── userRoutes.ts # 用户路由 │ ├── controllers/ │ │ └── userController.ts │ ├── middlewares/ │ │ ├── errorHandler.ts │ │ └── auth.ts │ ├── models/ │ │ └── User.ts │ ├── services/ │ │ └── userService.ts │ └── types/ │ └── index.ts # 类型定义 ├── dist/ # 编译输出 ├── node_modules/ ├── package.json ├── tsconfig.json └── .env

📝 package.json 配置

{ "name": "koa-ts-project", "version": "1.0.0", "main": "dist/index.js", "scripts": { "dev": "nodemon --exec ts-node src/index.ts", "build": "tsc", "start": "node dist/index.js", "lint": "eslint src/**/*.ts", "test": "jest" }, "dependencies": { "koa": "^2.14.0", "@koa/router": "^12.0.0", "koa-bodyparser": "^4.4.0" }, "devDependencies": { "typescript": "^5.0.0", "ts-node": "^10.9.0", "nodemon": "^3.0.0", "@types/koa": "^2.13.0", "@types/koa-router": "^7.4.0", "@types/koa-bodyparser": "^4.3.0", "@types/node": "^20.0.0" } }

🚀 入口文件

// src/app.ts import Koa from 'koa'; import Router from '@koa/router'; import bodyParser from 'koa-bodyparser'; const app = new Koa(); const router = new Router(); // 中间件 app.use(bodyParser()); // 路由 router.get('/', async (ctx) => { ctx.body = { message: 'Hello Koa + TypeScript!' }; }); app.use(router.routes()); app.use(router.allowedMethods()); export default app;
// src/index.ts import app from './app'; const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server running at http://localhost:${PORT}`); });

🔧 nodemon 配置

// nodemon.json { "watch": ["src"], "ext": "ts", "ignore": ["src/**/*.spec.ts"], "exec": "ts-node src/index.ts" }

或者使用 ts-node-dev 替代 nodemon + ts-node：

# 安装 ts-node-dev npm install -D ts-node-dev # 使用 "scripts": { "dev": "ts-node-dev --respawn --transpile-only src/index.ts" }

🌐 环境变量配置

# .env PORT=3000 NODE_ENV=development DB_HOST=localhost DB_PORT=5432 DB_NAME=myapp DB_USER=postgres DB_PASSWORD=password JWT_SECRET=your-secret-key
# 安装 npm install dotenv # 使用环境变量 npm install -D @types/dotenv
// src/config/index.ts import dotenv from 'dotenv'; import path from 'path'; // 加载环境变量 dotenv.config({ path: path.resolve(__dirname, '../../.env') }); export const config = { port: parseInt(process.env.PORT || '3000', 10), env: process.env.NODE_ENV || 'development', database: { host: process.env.DB_HOST || 'localhost', port: parseInt(process.env.DB_PORT || '5432', 10), name: process.env.DB_NAME || 'myapp', user: process.env.DB_USER || 'postgres', password: process.env.DB_PASSWORD || '' }, jwt: { secret: process.env.JWT_SECRET || 'secret', expiresIn: process.env.JWT_EXPIRES_IN || '7d' } };

📊 构建生产版本

# 编译 TypeScript npm run build # 运行生产版本 NODE_ENV=production npm start
// package.json { "scripts": { "build": "tsc", "start": "node dist/index.js", "dev": "nodemon --exec ts-node src/index.ts" } }
✅
生产部署

生产环境使用编译后的 JS 文件，性能更好。确保在部署前运行 npm run build。
