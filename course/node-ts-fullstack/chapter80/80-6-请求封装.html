80.6 请求封装

构建可复用的 HTTP 请求模块

📖 本节概述

在实际项目中，我们需要将 HTTP 请求进行统一封装，以便于维护、复用和统一管理。本节将介绍如何构建一个完善的 HTTP 请求模块，包括类型定义、错误处理、API 组织等方面。

🎯 学习目标

  • 掌握 HTTP 请求模块的设计思路
  • 学会定义类型安全的 API 接口
  • 了解如何组织和管理 API 模块
  • 掌握请求封装的最佳实践

🏗️ 基础封装结构

// src/utils/http/types.ts import type { AxiosRequestConfig, AxiosResponse } from 'axios'; // 通用响应结构 export interface ApiResponse<T = any> { code: number; message: string; data: T; } // 分页响应 export interface PageResponse<T> { list: T[]; total: number; page: number; pageSize: number; } // 请求配置扩展 export interface RequestOptions extends AxiosRequestConfig { // 是否显示加载状态 loading?: boolean; // 是否显示错误提示 showError?: boolean; // 自定义错误消息 errorMessage?: string; // 重试次数 retryCount?: number; } // HTTP 实例接口 export interface HttpInstance { get<T>(url: string, config?: RequestOptions): Promise<T>; post<T>(url: string, data?: any, config?: RequestOptions): Promise<T>; put<T>(url: string, data?: any, config?: RequestOptions): Promise<T>; delete<T>(url: string, config?: RequestOptions): Promise<T>; }

🔧 HTTP 类封装

// src/utils/http/index.ts import axios, { AxiosInstance, AxiosError } from 'axios'; import type { ApiResponse, RequestOptions } from './types'; class HttpClient { private instance: AxiosInstance; private loadingCount = 0; constructor(baseURL: string) { this.instance = axios.create({ baseURL, timeout: 10000, headers: { 'Content-Type': 'application/json' } }); this.setupInterceptors(); } private setupInterceptors() { // 请求拦截器 this.instance.interceptors.request.use( config => { const token = localStorage.getItem('token'); if (token) { config.headers.Authorization = `Bearer ${token}`; } return config; }, error => Promise.reject(error) ); // 响应拦截器 this.instance.interceptors.response.use( response => { const { data } = response; if (data.code === 0 || data.code === 200) { return data.data; } return Promise.reject(new Error(data.message)); }, error => { this.handleError(error); return Promise.reject(error); } ); } private handleError(error: AxiosError) { if (error.response?.status === 401) { localStorage.removeItem('token'); window.location.href = '/login'; } } async get<T>(url: string, config?: RequestOptions): Promise<T> { return this.instance.get(url, config); } async post<T>(url: string, data?: any, config?: RequestOptions): Promise<T> { return this.instance.post(url, data, config); } async put<T>(url: string, data?: any, config?: RequestOptions): Promise<T> { return this.instance.put(url, data, config); } async delete<T>(url: string, config?: RequestOptions): Promise<T> { return this.instance.delete(url, config); } } // 导出实例 export const http = new HttpClient('/api');

📁 API 模块组织

// src/api/types/user.ts export interface User { id: number; name: string; email: string; avatar?: string; createdAt: string; } export interface LoginParams { email: string; password: string; } export interface LoginResult { token: string; user: User; } export interface UserQueryParams { page?: number; pageSize?: number; name?: string; }
// src/api/user.ts import { http } from '@/utils/http'; import type { User, LoginParams, LoginResult, UserQueryParams, PageResponse } from './types/user'; export const userApi = { // 用户登录 login(params: LoginParams) { return http.post<LoginResult>('/auth/login', params); }, // 获取当前用户信息 getCurrentUser() { return http.get<User>('/user/current'); }, // 获取用户列表 getUsers(params?: UserQueryParams) { return http.get<PageResponse<User>>('/users', { params }); }, // 获取单个用户 getUser(id: number) { return http.get<User>(`/users/${id}`); }, // 创建用户 createUser(data: Partial<User>) { return http.post<User>('/users', data); }, // 更新用户 updateUser(id: number, data: Partial<User>) { return http.put<User>(`/users/${id}`, data); }, // 删除用户 deleteUser(id: number) { return http.delete<void>(`/users/${id}`); } };

🔄 React Hooks 封装

// src/hooks/useRequest.ts import { useState, useCallback } from 'react'; interface RequestState<T> { data: T | null; loading: boolean; error: Error | null; } export function useRequest<T, P = void>( requestFn: (params: P) => Promise<T> ) { const [state, setState] = useState<RequestState<T>>({ data: null, loading: false, error: null }); const execute = useCallback(async (params: P) => { setState({ data: null, loading: true, error: null }); try { const data = await requestFn(params); setState({ data, loading: false, error: null }); return data; } catch (error) { setState({ data: null, loading: false, error: error as Error }); throw error; } }, [requestFn]); return { ...state, execute }; } // 使用示例 function UserList() { const { data: users, loading, error, execute } = useRequest( userApi.getUsers ); useEffect(() => { execute({ page: 1, pageSize: 10 }); }, []); if (loading) return <div>Loading...</div>; if (error) return <div>Error: {error.message}</div>; return ( <ul> {users?.list.map(user => ( <li key={user.id}>{user.name}</li> ))} </ul> ); }

⚡ Vue Composables 封装

// src/composables/useRequest.ts import { ref, Ref } from 'vue'; interface RequestResult<T> { data: Ref<T | null>; loading: Ref<boolean>; error: Ref<Error | null>; execute: () => Promise<T | null>; } export function useRequest<T>( requestFn: () => Promise<T>, immediate = false ): RequestResult<T> { const data = ref<T | null>(null) as Ref<T | null>; const loading = ref(false); const error = ref<Error | null>(null); const execute = async () => { loading.value = true; error.value = null; try { data.value = await requestFn(); return data.value; } catch (e) { error.value = e as Error; return null; } finally { loading.value = false; } }; if (immediate) { execute(); } return { data, loading, error, execute }; } // 使用示例 const { data: users, loading, execute } = useRequest( () => userApi.getUsers({ page: 1 }), true // 立即执行 );
