管理数据库连接和生命周期

📚 DataSource 配置

// src/data-source.ts import { DataSource, DataSourceOptions } from 'typeorm'; import * as dotenv from 'dotenv'; dotenv.config(); export const dataSourceOptions: DataSourceOptions = { type: 'mysql', host: process.env.DB_HOST || 'localhost', port: parseInt(process.env.DB_PORT) || 3306, username: process.env.DB_USERNAME || 'root', password: process.env.DB_PASSWORD, database: process.env.DB_DATABASE || 'myapp', entities: ['dist/**/*.entity.js'], // 生产环境使用编译后的文件 migrations: ['dist/migrations/*.js'], synchronize: false, // 生产环境必须为 false logging: process.env.NODE_ENV === 'development', }; export const AppDataSource = new DataSource(dataSourceOptions); // 连接池配置 export const poolOptions: DataSourceOptions = { type: 'mysql', // ...其他配置 extra: { connectionLimit: 10, // 最大连接数 waitForConnections: true, queueLimit: 0, // 无限排队 acquireTimeout: 30000, // 获取连接超时 }, poolSize: 10, // TypeORM 连接池大小 };
⚠️
synchronize 警告

synchronize: true 会自动同步实体定义到数据库表结构，这在开发环境很方便，但在生产环境可能导致数据丢失。生产环境务必使用 migrations。

🔧 连接初始化

// 方式一：直接初始化 import { AppDataSource } from './data-source'; async function bootstrap() { try { await AppDataSource.initialize(); console.log('Database connected successfully'); // 启动应用 app.listen(3000, () => { console.log('Server running on port 3000'); }); } catch (error) { console.error('Database connection failed:', error); process.exit(1); } } bootstrap(); // 方式二：使用依赖注入（推荐） // app.module.ts import { Module } from '@nestjs/common'; import { TypeOrmModule } from '@nestjs/typeorm'; import { dataSourceOptions } from './data-source'; @Module({ imports: [ TypeOrmModule.forRoot(dataSourceOptions), ], }) export class AppModule {} // 方式三：Express 集成 import express from 'express'; import { AppDataSource } from './data-source'; const app = express(); AppDataSource.initialize().then(() => { app.listen(3000); }); // 优雅关闭 process.on('SIGINT', async () => { await AppDataSource.destroy(); process.exit(0); }); process.on('SIGTERM', async () => { await AppDataSource.destroy(); process.exit(0); });

📊 多数据库连接

// 配置多个数据源 import { DataSource } from 'typeorm'; // 主数据库 export const PrimaryDataSource = new DataSource({ name: 'primary', type: 'mysql', host: 'primary-db.example.com', username: 'root', password: 'password', database: 'primary_db', entities: ['src/entity/*.ts'], }); // 只读副本 export const ReadOnlyDataSource = new DataSource({ name: 'readonly', type: 'mysql', host: 'readonly-db.example.com', username: 'readonly', password: 'password', database: 'primary_db', entities: ['src/entity/*.ts'], }); // 使用指定数据源 async function getUser(id: number) { const repo = PrimaryDataSource.getRepository(User); return repo.findOne({ where: { id } }); } async function searchUsers(query: string) { const repo = ReadOnlyDataSource.getRepository(User); return repo.find({ where: { name: Like(`%${query}%`) } }); } // NestJS 多数据库配置 @Module({ imports: [ TypeOrmModule.forRoot({ name: 'primary', type: 'mysql', host: 'primary-db.example.com', entities: [User], }), TypeOrmModule.forRoot({ name: 'readonly', type: 'mysql', host: 'readonly-db.example.com', entities: [User], }), TypeOrmModule.forFeature([User], 'primary'), ], }) export class AppModule {}

⚙️ 连接状态管理

import { AppDataSource } from './data-source'; class DatabaseService { private static instance: DatabaseService; private isConnected = false; private constructor() {} static getInstance(): DatabaseService { if (!DatabaseService.instance) { DatabaseService.instance = new DatabaseService(); } return DatabaseService.instance; } async connect(): Promise { if (this.isConnected) return; try { await AppDataSource.initialize(); this.isConnected = true; console.log('Database connected'); } catch (error) { this.isConnected = false; throw error; } } async disconnect(): Promise { if (!this.isConnected) return; await AppDataSource.destroy(); this.isConnected = false; console.log('Database disconnected'); } isReady(): boolean { return this.isConnected && AppDataSource.isInitialized; } // 健康检查 async healthCheck(): Promise { try { await AppDataSource.query('SELECT 1'); return true; } catch { return false; } } } export const databaseService = DatabaseService.getInstance();
