# Prometheus告警规则
groups:
- name: ai_platform_alerts
  rules:
  # 节点资源告警
  - alert: HighCPUUsage
    expr: (1 - avg(rate(node_cpu_seconds_total{mode="idle"}[5m]))) by (instance) > 0.8
    for: 5m
    labels:
      severity: warning
    annotations:
      summary: "High CPU usage on {{ $labels.instance }}"
      description: "CPU usage is {{ $value }}%"

  # GPU温度告警
  - alert: HighGPUTemperature
    expr: DCGM_FI_DEV_GPU_TEMP > 85
    for: 2m
    labels:
      severity: critical
    annotations:
      summary: "High GPU temperature"
      description: "GPU {{ $labels.gpu }} temperature is {{ $value }}°C"

  # GPU显存告警
  - alert: HighGPUMemoryUsage
    expr: DCGM_FI_DEV_FB_USED / DCGM_FI_DEV_FB_TOTAL > 0.95
    for: 5m
    labels:
      severity: warning
    annotations:
      summary: "High GPU memory usage"
      description: "GPU {{ $labels.gpu }} memory usage is {{ $value }}%"

  # 任务失败告警
  - alert: HighTaskFailureRate
    expr: rate(kube_job_status_failed[5m]) > 0.05
    for: 5m
    labels:
      severity: critical
    annotations:
      summary: "High task failure rate"
      description: "Task failure rate is {{ $value }}"

  # 存储空间告警
  - alert: LowDiskSpace
    expr: (node_filesystem_avail_bytes / node_filesystem_size_bytes) < 0.1
    for: 5m
    labels:
      severity: warning
    annotations:
      summary: "Low disk space"
      description: "Disk {{ $labels.device }} has less than 10% free space"

🔄 自动化运维

# 自动化运维脚本
import requests
import time

class PlatformOps:
    def __init__(self, api_url):
        self.api_url = api_url

    def check_health(self):
        """健康检查"""
        endpoints = [
            '/health',
            '/api/v1/health',
            '/training/health',
            '/inference/health'
        ]
        for endpoint in endpoints:
            try:
                response = requests.get(f"{self.api_url}{endpoint}")
                if response.status_code != 200:
                    print(f"Health check failed: {endpoint}")
            except Exception as e:
                print(f"Health check error: {endpoint}, {e}")

    def restart_failed_tasks(self):
        """重启失败任务"""
        response = requests.get(f"{self.api_url}/api/v1/tasks?status=failed")
        tasks = response.json()

        for task in tasks:
            print(f"Restarting task: {task['id']}")
            requests.post(f"{self.api_url}/api/v1/tasks/{task['id']}/restart")

    def scale_inference(self, target_replicas):
        """扩缩容推理服务"""
        response = requests.patch(
            f"{self.api_url}/api/v1/deployments/inference",
            json={"replicas": target_replicas}
        )
        return response.json()

    def cleanup_old_logs(self, days=30):
        """清理旧日志"""
        cutoff_date = time.time() - days * 86400
        # 调用日志清理API
        requests.delete(f"{self.api_url}/api/v1/logs?before={cutoff_date}")

💾 备份恢复策略

备份策略

# Kubernetes资源备份
kubectl get all,configmaps,secrets,pvc --all-namespaces -o yaml > backup.yaml

# Etcd备份
ETCDCTL_API=3 etcdctl snapshot save snapshot.db \
  --endpoints=https://127.0.0.1:2379 \
  --cacert=/etc/kubernetes/pki/etcd/ca.crt \
  --cert=/etc/kubernetes/pki/etcd/server.crt \
  --key=/etc/kubernetes/pki/etcd/server.key

# 数据库备份
mysqldump -u root -p database_name > backup.sql

恢复流程

# Etcd恢复
ETCDCTL_API=3 etcdctl snapshot restore snapshot.db \
  --data-dir /var/lib/etcd-from-backup

# K8s资源恢复
kubectl apply -f backup.yaml

# 数据库恢复
mysql -u root -p database_name < backup.sql

运维最佳实践

监控先行：建立全面的监控体系，提前发现问题。文档完善：详细记录运维流程和应急方案。定期演练：定期进行故障演练和恢复测试。持续改进：根据故障经验不断优化运维体系。
