🔧 实现事件驱动类

const EventEmitter = require('events'); // 文件下载器 class Downloader extends EventEmitter { constructor() { super(); this.downloads = new Map(); } async download(url, filename) { const id = Date.now().toString(36); this.emit('download-started', { id, url, filename }); try { const response = await fetch(url); const total = parseInt(response.headers.get('content-length'), 10); let downloaded = 0; const file = fs.createWriteStream(filename); const reader = response.body.getReader(); while (true) { const { done, value } = await reader.read(); if (done) break; file.write(value); downloaded += value.length; this.emit('download-progress', { id, downloaded, total, percent: ((downloaded / total) * 100).toFixed(2) }); } file.end(); this.emit('download-completed', { id, filename, size: downloaded }); } catch (err) { this.emit('download-error', { id, error: err.message }); } } cancel(id) { this.emit('download-cancelled', { id }); } } // 使用 const downloader = new Downloader(); downloader.on('download-started', (info) => { console.log(`开始下载: ${info.filename}`); }); downloader.on('download-progress', (info) => { process.stdout.write(`\r下载进度: ${info.percent}%`); }); downloader.on('download-completed', (info) => { console.log(`\n下载完成: ${info.filename}`); }); downloader.on('download-error', (info) => { console.error(`下载失败: ${info.error}`); }); downloader.download('https://example.com/file.zip', 'file.zip');

📦 事件总线模式

const EventEmitter = require('events'); // 全局事件总线 class EventBus extends EventEmitter { constructor() { super(); this.setMaxListeners(100); // 增加监听器上限 } // 订阅事件 subscribe(event, listener) { this.on(event, listener); return () => this.off(event, listener); // 返回取消订阅函数 } // 发布事件 publish(event, data) { this.emit(event, data); } // 一次性订阅 once(event, listener) { return super.once(event, listener); } } // 全局单例 const eventBus = new EventBus(); // 模块 A - 发布事件 class UserService { login(username, password) { // 登录逻辑... eventBus.publish('user-logged-in', { username, time: Date.now() }); } logout() { eventBus.publish('user-logged-out', { time: Date.now() }); } } // 模块 B - 订阅事件（无需知道模块 A 的存在） class AnalyticsService { constructor() { eventBus.subscribe('user-logged-in', (data) => { this.track('login', data); }); eventBus.subscribe('user-logged-out', (data) => { this.track('logout', data); }); } track(event, data) { console.log(`[Analytics] ${event}:`, data); } } // 模块 C - 订阅事件 class NotificationService { constructor() { eventBus.subscribe('user-logged-in', ({ username }) => { this.sendWelcome(username); }); } sendWelcome(username) { console.log(`欢迎回来，${username}！`); } }
💡
事件总线的作用

事件总线实现模块间解耦，发布者和订阅者互不依赖，适合大型应用的模块通信。

⚙️ 状态机示例

const EventEmitter = require('events'); // 订单状态机 class OrderStateMachine extends EventEmitter { constructor(orderId) { super(); this.orderId = orderId; this.state = 'pending'; this.history = []; } // 状态转换 transitions = { pending: ['confirmed', 'cancelled'], confirmed: ['shipped', 'cancelled'], shipped: ['delivered', 'returned'], delivered: ['returned'], cancelled: [], returned: [] }; canTransitionTo(newState) { return this.transitions[this.state].includes(newState); } transition(newState, metadata = {}) { if (!this.canTransitionTo(newState)) { this.emit('transition-failed', { from: this.state, to: newState, reason: 'Invalid transition' }); return false; } const oldState = this.state; this.state = newState; this.history.push({ from: oldState, to: newState, time: Date.now(), ...metadata }); this.emit('state-changed', { orderId: this.orderId, from: oldState, to: newState, metadata }); return true; } confirm() { return this.transition('confirmed'); } ship() { return this.transition('shipped'); } deliver() { return this.transition('delivered'); } cancel() { return this.transition('cancelled'); } return() { return this.transition('returned'); } } // 使用 const order = new OrderStateMachine('ORDER-123'); order.on('state-changed', (data) => { console.log(`订单 ${data.orderId}: ${data.from} -> ${data.to}`); }); order.on('transition-failed', (data) => { console.error(`状态转换失败: ${data.from} -> ${data.to}, 原因: ${data.reason}`); }); order.confirm(); // pending -> confirmed order.deliver(); // 失败：confirmed 不能直接到 delivered order.ship(); // confirmed -> shipped order.deliver(); // shipped -> delivered
