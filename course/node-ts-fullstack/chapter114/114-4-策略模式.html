114.4 策略模式

算法族的封装与切换

📚 模式定义

// 策略模式：定义一系列算法，把它们封装起来，并使它们可以互相替换 // 应用场景 const useCases = [ '支付方式选择', '排序算法选择', '折扣策略', '路由策略', '验证策略' ]; // 策略接口 interface Strategy { execute(input: T): R; }
💡
策略模式优点

避免条件语句、易于扩展新策略、算法可独立变化。

🔧 TypeScript实现

// 折扣策略示例 interface DiscountStrategy { calculate(price: number): number; } // 正常价格 class NoDiscount implements DiscountStrategy { calculate(price: number): number { return price; } } // 百分比折扣 class PercentageDiscount implements DiscountStrategy { constructor(private percentage: number) {} calculate(price: number): number { return price * (1 - this.percentage / 100); } } // 固定金额折扣 class FixedDiscount implements DiscountStrategy { constructor(private amount: number) {} calculate(price: number): number { return Math.max(0, price - this.amount); } } // 买X送Y折扣 class BuyXGetYDiscount implements DiscountStrategy { constructor(private x: number, private y: number) {} calculate(price: number): number { // 实现... return price; } } // 折扣上下文 class PriceCalculator { private strategy: DiscountStrategy; setStrategy(strategy: DiscountStrategy) { this.strategy = strategy; } calculate(price: number): number { if (!this.strategy) { throw new Error('Strategy not set'); } return this.strategy.calculate(price); } } // 使用 const calculator = new PriceCalculator(); calculator.setStrategy(new NoDiscount()); console.log(calculator.calculate(100)); // 100 calculator.setStrategy(new PercentageDiscount(20)); console.log(calculator.calculate(100)); // 80 calculator.setStrategy(new FixedDiscount(30)); console.log(calculator.calculate(100)); // 70 // 策略工厂 class DiscountStrategyFactory { private strategies: Map = new Map(); constructor() { this.strategies.set('none', new NoDiscount()); this.strategies.set('vip10', new PercentageDiscount(10)); this.strategies.set('vip20', new PercentageDiscount(20)); this.strategies.set('coupon50', new FixedDiscount(50)); } getStrategy(type: string): DiscountStrategy { const strategy = this.strategies.get(type); if (!strategy) { throw new Error(`Unknown strategy: ${type}`); } return strategy; } } // 验证策略示例 interface ValidationStrategy { validate(value: any): boolean; } class EmailValidation implements ValidationStrategy { validate(value: any): boolean { return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value); } } class PhoneValidation implements ValidationStrategy { validate(value: any): boolean { return /^1[3-9]\d{9}$/.test(value); } } class Validator { constructor(private strategy: ValidationStrategy) {} validate(value: any): boolean { return this.strategy.validate(value); } } const emailValidator = new Validator(new EmailValidation()); console.log(emailValidator.validate('test@example.com')); // true const phoneValidator = new Validator(new PhoneValidation()); console.log(phoneValidator.validate('13800138000')); // true
