🚪 时钟门控

// 时钟门控单元
module clock_gating(
    input  clk,
    input  enable,
    output gated_clk
);
    reg latch_q;
    wire latch_en;

    // 锁存器使能
    assign latch_en = enable | !clk;

    // 锁存器
    always @(*) begin
        if (clk)
            latch_q = latch_en;
    end

    // 与门
    assign gated_clk = clk & latch_q;
endmodule

💤 模块使能

// 带使能的模块
module gated_module(
    input  clk,
    input  rst_n,
    input  enable,
    input  [7:0] data_in,
    output [7:0] data_out
);
    reg [7:0] data_reg;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            data_reg <= 0;
        else if (enable)
            data_reg <= data_in;
    end

    assign data_out = data_reg;
endmodule
