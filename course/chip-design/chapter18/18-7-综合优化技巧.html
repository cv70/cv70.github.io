🚀 资源共享

// 资源共享：多个操作共享一个加法器
module shared_resource(
    input  [7:0] a, b, c, d,
    input        sel,
    output [7:0] y
);
    wire [7:0] sum1 = a + b;
    wire [7:0] sum2 = c + d;
    
    // 综合工具可能会共享加法器
    assign y = sel ? sum1 : sum2;
endmodule

⏱️ 流水线设计

// 2级流水线乘法器
module pipelined_multiplier(
    input  clk,
    input  rst_n,
    input  [7:0] a, b,
    output [15:0] product
);
    reg [15:0] product_reg;
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            product_reg <= 0;
        else
            product_reg <= a * b;
    end
    
    assign product = product_reg;
endmodule
