115.2 纯函数

函数式编程的基石

📚 纯函数定义

// 纯函数的两个条件： // 1. 相同输入总是返回相同输出 // 2. 没有副作用 // 纯函数示例 function add(a: number, b: number): number { return a + b; } add(1, 2); // 总是返回 3 add(1, 2); // 总是返回 3 // 非纯函数示例 // 依赖外部状态 let multiplier = 2; function multiplyWithExternal(a: number): number { return a * multiplier; // 依赖外部变量 } multiplyWithExternal(3); // 6 multiplier = 3; multiplyWithExternal(3); // 9 // 结果变了 // 修改外部状态 let total = 0; function addToTotal(n: number): number { total += n; // 修改外部状态 return total; } addToTotal(1); // 1 addToTotal(1); // 2 // 相同输入不同输出 // 执行IO操作 function getUser(id: string): User { return database.findUser(id); // 数据库查询 } // 当前时间 function getCurrentTimestamp(): number { return Date.now(); // 每次调用结果不同 } // 随机数 function getRandom(): number { return Math.random(); // 不可预测 }

🔧 纯函数实践

// 将非纯函数转为纯函数 // 非纯函数：依赖外部配置 let config = { taxRate: 0.1 }; function calculatePrice(price: number): number { return price * (1 + config.taxRate); } // 纯函数：配置作为参数 function calculatePricePure(price: number, taxRate: number): number { return price * (1 + taxRate); } // 非纯函数：修改对象 function updateUser(user: User, updates: Partial): void { Object.assign(user, updates); // 修改原对象 } // 纯函数：返回新对象 function updateUserPure(user: User, updates: Partial): User { return { ...user, ...updates }; // 返回新对象 } // 非纯函数：修改数组 function addItem(arr: T[], item: T): void { arr.push(item); // 修改原数组 } // 纯函数：返回新数组 function addItemPure(arr: T[], item: T): T[] { return [...arr, item]; // 返回新数组 } // 纯函数的优势 // 1. 易于测试 function testAdd() { assert(add(1, 2) === 3); assert(add(-1, 1) === 0); assert(add(0, 0) === 0); // 不需要 mock，不需要设置环境 } // 2. 易于缓存 function memoize(fn: (arg: T) => R): (arg: T) => R { const cache = new Map(); return (arg: T) => { if (cache.has(arg)) { return cache.get(arg)!; } const result = fn(arg); cache.set(arg, result); return result; }; } const expensiveCalculation = (n: number): number => { console.log('Calculating...'); return n * n; }; const memoizedCalc = memoize(expensiveCalculation); memoizedCalc(5); // Calculating... 25 memoizedCalc(5); // 25 (缓存命中) // 3. 并发安全 // 纯函数没有共享状态，不会产生竞态条件 // 4. 可预测性 function formatName(firstName: string, lastName: string): string { return `${firstName} ${lastName}`.trim(); } formatName('John', 'Doe'); // 总是 'John Doe'

📝 本节小结

✅
  • • 纯函数相同输入总是返回相同输出
  • • 纯函数没有副作用
  • • 将依赖和修改作为参数传入
  • • 纯函数易于测试、缓存和并发
