69.4 测试异步代码

第十部分：测试篇 - 掌握异步代码的测试技巧

1. 异步测试的重要性

Node.js 开发中大量使用异步代码，如数据库操作、API 调用、文件读写等。正确测试异步代码是后端开发的关键技能。

⚠️
常见错误

忘记等待异步操作完成是最常见的错误，导致测试提前结束，无法验证异步代码的结果。

2. 回调函数测试

// 被测试的函数 function fetchData(callback: (data: string) => void) { setTimeout(() => { callback('peanut butter'); }, 100); }

错误写法

// ❌ 错误：测试会在回调执行前结束 test('the data is peanut butter', () => { functionCallback(data => { expect(data).toBe('peanut butter'); }); // 测试在这里就结束了！ });

正确写法：使用 done 参数

// ✅ 正确：使用 done 参数 test('the data is peanut butter', done => { functionCallback(data => { expect(data).toBe('peanut butter'); done(); // 告诉 Jest 测试完成 }); });
// 处理异步错误 test('the fetch fails with an error', done => { fetchWithError((error, data) => { if (error) { expect(error.message).toBe('Network error'); done(); return; } done.fail('Should have thrown an error'); }); });
💡
done 参数

Jest 会等待 done() 被调用才结束测试。如果 done() 没有被调用，测试会超时失败。

3. Promise 测试

// 被测试的函数 function fetchData(): Promise<string> { return new Promise(resolve => { setTimeout(() => resolve('peanut butter'), 100); }); } function fetchError(): Promise<never> { return Promise.reject(new Error('Network error')); }

方式一：返回 Promise

// ✅ 返回 Promise test('the data is peanut butter', () => { return fetchData().then(data => { expect(data).toBe('peanut butter'); }); });

方式二：使用 async/await（推荐）

// ✅ 使用 async/await（推荐） test('the data is peanut butter', async () => { const data = await fetchData(); expect(data).toBe('peanut butter'); });

方式三：使用 .resolves / .rejects

// ✅ 使用 resolves test('the data is peanut butter', () => { return expect(fetchData()).resolves.toBe('peanut butter'); }); // ✅ 使用 rejects test('the fetch fails with an error', () => { return expect(fetchError()).rejects.toThrow('Network error'); }); // ✅ 结合 async/await test('the data is peanut butter', async () => { await expect(fetchData()).resolves.toBe('peanut butter'); });

4. async/await 最佳实践

describe('UserService', () => { let service: UserService; beforeEach(() => { service = new UserService(); }); test('should create user', async () => { // Arrange const userData = { name: 'John', email: 'john@example.com' }; // Act const user = await service.create(userData); // Assert expect(user).toMatchObject({ name: 'John', email: 'john@example.com' }); expect(user.id).toBeDefined(); }); test('should throw error for duplicate email', async () => { // 先创建一个用户 await service.create({ name: 'John', email: 'john@example.com' }); // 尝试创建相同邮箱的用户 await expect(service.create({ name: 'Jane', email: 'john@example.com' })).rejects.toThrow('Email already exists'); }); });

并发测试

// 默认：测试顺序执行 test('test 1', async () => { await someAsyncOperation(); }); test('test 2', async () => { await anotherAsyncOperation(); }); // 并发执行测试 test.concurrent('concurrent test 1', async () => { await someAsyncOperation(); }); test.concurrent('concurrent test 2', async () => { await anotherAsyncOperation(); });

5. 测试定时器

// 被测试的代码 function delayedMessage(callback: (msg: string) => void) { setTimeout(() => { callback('Hello after delay'); }, 1000); } function intervalCounter(callback: (count: number) => void) { let count = 0; const id = setInterval(() => { count++; callback(count); if (count === 3) clearInterval(id); }, 100); return id; }

使用 Jest 假定时器

describe('Timer tests', () => { beforeEach(() => { // 使用假定时器 jest.useFakeTimers(); }); afterEach(() => { // 恢复真实定时器 jest.useRealTimers(); }); test('calls callback after delay', () => { const callback = jest.fn(); delayedMessage(callback); // 此时回调还没被调用 expect(callback).not.toHaveBeenCalled(); // 快进时间 jest.advanceTimersByTime(1000); // 现在回调被调用了 expect(callback).toHaveBeenCalledWith('Hello after delay'); }); test('runs all timers', () => { const callback = jest.fn(); delayedMessage(callback); // 运行所有定时器 jest.runAllTimers(); expect(callback).toHaveBeenCalled(); }); test('interval counter', () => { const callback = jest.fn(); intervalCounter(callback); // 快进到第一个间隔 jest.advanceTimersByTime(100); expect(callback).toHaveBeenCalledWith(1); // 快进到第二个间隔 jest.advanceTimersByTime(100); expect(callback).toHaveBeenCalledWith(2); // 快进剩余时间 jest.advanceTimersByTime(100); expect(callback).toHaveBeenCalledWith(3); }); test('only pending timers', () => { const callback = jest.fn(); delayedMessage(callback); // 只运行待处理的定时器 jest.runOnlyPendingTimers(); expect(callback).toHaveBeenCalled(); }); });

定时器方法一览

方法 说明
jest.useFakeTimers() 使用假定时器
jest.useRealTimers() 恢复真实定时器
jest.runAllTimers() 运行所有定时器
jest.runOnlyPendingTimers() 只运行待处理的定时器
jest.advanceTimersByTime(ms) 快进指定时间
jest.advanceTimersToNextTimer() 快进到下一个定时器
jest.getTimerCount() 获取定时器数量

6. 测试超时设置

// 设置测试超时时间 test('slow operation', async () => { const result = await slowOperation(); expect(result).toBeDefined(); }, 10000); // 10 秒超时 // 设置整个 describe 的超时 describe('slow tests', () => { test('test 1', async () => { // ... }); test('test 2', async () => { // ... }); }, 30000); // 30 秒超时 // 在配置中设置默认超时 // jest.config.ts export default { testTimeout: 10000 // 默认 10 秒 };

7. 完整示例：数据库操作测试

// tests/services/user.service.test.ts import { UserService } from '@/services/user.service'; import { createConnection, Connection } from 'typeorm'; import { User } from '@/entities/user.entity'; describe('UserService', () => { let connection: Connection; let service: UserService; beforeAll(async () => { // 建立测试数据库连接 connection = await createConnection({ type: 'sqlite', database: ':memory:', entities: [User], synchronize: true }); service = new UserService(connection); }); afterAll(async () => { // 关闭连接 await connection.close(); }); beforeEach(async () => { // 清空数据 await connection.getRepository(User).clear(); }); describe('create', () => { it('should create a new user', async () => { const userData = { name: 'John Doe', email: 'john@example.com' }; const user = await service.create(userData); expect(user).toMatchObject(userData); expect(user.id).toBeDefined(); expect(user.createdAt).toBeInstanceOf(Date); }); it('should throw error for duplicate email', async () => { await service.create({ name: 'John', email: 'john@example.com' }); await expect(service.create({ name: 'Jane', email: 'john@example.com' })).rejects.toThrow(); }); }); describe('findByEmail', () => { it('should return user when found', async () => { await service.create({ name: 'John', email: 'john@example.com' }); const user = await service.findByEmail('john@example.com'); expect(user).not.toBeNull(); expect(user?.name).toBe('John'); }); it('should return null when not found', async () => { const user = await service.findByEmail('nonexistent@example.com'); expect(user).toBeNull(); }); }); });
