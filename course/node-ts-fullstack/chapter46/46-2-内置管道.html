46.2 内置管道

使用 NestJS 提供的内置管道

📚 内置管道概览

管道 功能 错误响应
ParseIntPipe 转换为整数 400 Bad Request
ParseFloatPipe 转换为浮点数 400 Bad Request
ParseBoolPipe 转换为布尔值 400 Bad Request
ParseArrayPipe 转换为数组 400 Bad Request
ParseUUIDPipe 验证 UUID 格式 400 Bad Request
ParseEnumPipe 验证枚举值 400 Bad Request
DefaultValuePipe 设置默认值 不抛出异常
ValidationPipe DTO 验证 400 Bad Request

🔧 ParseIntPipe / ParseFloatPipe

import { Controller, Get, Param, ParseIntPipe, ParseFloatPipe } from '@nestjs/common'; @Controller('numbers') export class NumbersController { @Get('int/:id') parseInt(@Param('id', ParseIntPipe) id: number) { console.log(typeof id); // number return { id, type: typeof id }; } @Get('float/:value') parseFloat(@Param('value', ParseFloatPipe) value: number) { return { value, type: typeof value }; } // 自定义错误消息 @Get('custom/:id') customParse( @Param('id', new ParseIntPipe({ errorHttpStatusCode: HttpStatus.NOT_ACCEPTABLE })) id: number, ) { return { id }; } }

🌐 ParseBoolPipe / ParseUUIDPipe

import { Controller, Get, Param, Query, ParseBoolPipe, ParseUUIDPipe } from '@nestjs/common'; @Controller('validation') export class ValidationController { @Get('bool') parseBool(@Query('active', ParseBoolPipe) active: boolean) { return { active, type: typeof active }; } // GET /validation/bool?active=true => { active: true, type: 'boolean' } @Get('uuid/:id') parseUUID(@Param('id', ParseUUIDPipe) id: string) { return { id }; } // GET /validation/uuid/123e4567-e89b-12d3-a456-426614174000 // 指定 UUID 版本 @Get('uuid-v4/:id') parseUUIDV4( @Param('id', new ParseUUIDPipe({ version: '4' })) id: string, ) { return { id }; } }

📦 ParseArrayPipe

import { Controller, Get, Query, ParseArrayPipe } from '@nestjs/common'; @Controller('arrays') export class ArraysController { // 解析逗号分隔的字符串为数组 @Get('tags') parseArray( @Query('tags', new ParseArrayPipe({ items: String, separator: ',' })) tags: string[], ) { return { tags }; } // GET /arrays/tags?tags=node,nest,typescript => { tags: ['node', 'nest', 'typescript'] } // 必需数组 @Get('required') parseRequired( @Query('ids', new ParseArrayPipe({ items: Number, separator: ',', optional: false })) ids: number[], ) { return { ids }; } // 转换为数字数组 @Get('numbers') parseNumbers( @Query('nums', new ParseArrayPipe({ items: Number, separator: ',' })) nums: number[], ) { return { nums }; } // GET /arrays/numbers?nums=1,2,3 => { nums: [1, 2, 3] } }

✨ DefaultValuePipe

import { Controller, Get, Query, DefaultValuePipe } from '@nestjs/common'; @Controller('defaults') export class DefaultsController { @Get() find( @Query('page', new DefaultValuePipe(1)) page: number, @Query('limit', new DefaultValuePipe(10)) limit: number, @Query('sort', new DefaultValuePipe('createdAt')) sort: string, ) { return { page, limit, sort }; } // GET /defaults => { page: 1, limit: 10, sort: 'createdAt' } // GET /defaults?page=2&limit=20 => { page: '2', limit: '20', sort: 'createdAt' } // 组合使用 @Get('combined') combined( @Query('page', new DefaultValuePipe(1), ParseIntPipe) page: number, @Query('limit', new DefaultValuePipe(10), ParseIntPipe) limit: number, ) { return { page, limit, type: typeof page }; } }
💡
管道组合

可以在同一个参数上使用多个管道，按顺序执行。DefaultValuePipe 应该放在前面，为空值设置默认值后再由其他管道处理。
