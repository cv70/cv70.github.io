module and_gate(

    output y
);

endmodule

    output y
);

endmodule

    input  a,
    output y

module xor_gate(

🔢 优先编码器

// 8-3优先编码器
module priority_encoder_8to3(
    input  [7:0] in,
    output [2:0] out,
    output       valid
);
    reg [2:0] out_reg;
    reg valid_reg;

    always @(*) begin
        valid_reg = 0;
        out_reg = 3'b000;

        if (in[7]) begin
            out_reg = 3'b111;
            valid_reg = 1;
        end
        else if (in[6]) begin
            out_reg = 3'b110;
            valid_reg = 1;
        end
        else if (in[5]) begin
            out_reg = 3'b101;
            valid_reg = 1;
        end
        else if (in[4]) begin
            out_reg = 3'b100;
            valid_reg = 1;
        end
        else if (in[3]) begin
            out_reg = 3'b011;
            valid_reg = 1;
        end
        else if (in[2]) begin
            out_reg = 3'b010;
            valid_reg = 1;
        end
        else if (in[1]) begin
            out_reg = 3'b001;
            valid_reg = 1;
        end
        else if (in[0]) begin
            out_reg = 3'b000;
            valid_reg = 1;
        end
    end

    assign out = out_reg;
    assign valid = valid_reg;
endmodule

📡 3-8译码器

// 3-8译码器
module decoder_3to8(
    input  [2:0] sel,
    input        en,
    output [7:0] y
);
    reg [7:0] y_reg;

    always @(*) begin
        if (!en)
            y_reg = 8'b0000_0000;
        else begin
            case (sel)
                3'b000: y_reg = 8'b0000_0001;
                3'b001: y_reg = 8'b0000_0010;
                3'b010: y_reg = 8'b0000_0100;
                3'b011: y_reg = 8'b0000_1000;
                3'b100: y_reg = 8'b0001_0000;
                3'b101: y_reg = 8'b0010_0000;
                3'b110: y_reg = 8'b0100_0000;
                3'b111: y_reg = 8'b1000_0000;
                default: y_reg = 8'b0000_0000;
            endcase
        end
    end

    assign y = y_reg;
endmodule

🔀 4-1多路选择器

// 4-1多路选择器
module mux4to1(
    input  [3:0] d,
    input  [1:0] sel,
    output       y
);
    reg y_reg;

    always @(*) begin
        case (sel)
            2'b00: y_reg = d[0];
            2'b01: y_reg = d[1];
            2'b10: y_reg = d[2];
            2'b11: y_reg = d[3];
            default: y_reg = 1'b0;
        endcase
    end

    assign y = y_reg;
endmodule

// 使用条件运算符实现
module mux4to1_cond(
    input  [3:0] d,
    input  [1:0] sel,
    output       y
);
    assign y = (sel == 2'b00) ? d[0] :
               (sel == 2'b01) ? d[1] :
               (sel == 2'b10) ? d[2] : d[3];
endmodule

// 4位数值比较器
module comparator_4bit(

    input  [3:0] b,
    output       a_gt_b,
    output       a_lt_b,
    output       a_eq_b

    assign a_gt_b = (a > b);

    assign a_eq_b = (a == b);
endmodule
