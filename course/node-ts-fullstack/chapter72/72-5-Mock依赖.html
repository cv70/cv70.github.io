72.5 Mock 依赖

第十部分：测试篇 - NestJS 测试中的依赖 Mock 技巧

1. Mock 策略

单元测试

Mock 所有外部依赖，只测试被测单元的逻辑。

  • Mock 数据库访问
  • Mock 外部服务
  • Mock 其他服务

集成测试

Mock 部分依赖，测试组件间的交互。

  • 使用内存数据库
  • Mock 外部 API
  • Mock 消息队列

2. Mock TypeORM

import { Test, TestingModule } from '@nestjs/testing'; import { getRepositoryToken } from '@nestjs/typeorm'; import { Repository } from 'typeorm'; import { User } from './user.entity'; import { UsersService } from './users.service'; describe('UsersService with Mock Repository', () => { let service: UsersService; let repository: jest.Mocked<Repository<User>>; beforeEach(async () => { const module = await Test.createTestingModule({ providers: [ UsersService, { provide: getRepositoryToken(User), useValue: { find: jest.fn(), findOne: jest.fn(), create: jest.fn(), save: jest.fn(), update: jest.fn(), delete: jest.fn(), count: jest.fn(), createQueryBuilder: jest.fn(() => ({ where: jest.fn().mockReturnThis(), andWhere: jest.fn().mockReturnThis(), getMany: jest.fn(), getOne: jest.fn() })) } } ] }).compile(); service = module.get(UsersService); repository = module.get(getRepositoryToken(User)); }); it('should find all users', async () => { const mockUsers = [{ id: '1', name: 'John' }]; repository.find.mockResolvedValue(mockUsers as any); const result = await service.findAll(); expect(result).toEqual(mockUsers); }); });

describe('Mocking Custom Providers', () => { it('should mock useValue provider', async () => { const module = await Test.createTestingModule({ providers: [ UsersService, { provide: 'DATABASE_CONNECTION', useValue: { query: jest.fn().mockResolvedValue([{ id: 1 }]), execute: jest.fn().mockResolvedValue({ affected: 1 }) } } ] }).compile(); const connection = module.get('DATABASE_CONNECTION'); expect(connection.query).toBeDefined(); }); it('should mock useFactory provider', async () => { const module = await Test.createTestingModule({ providers: [ { provide: 'CONFIG', useFactory: () => ({ get: jest.fn((key: string) => { if (key === 'DB_HOST') return 'localhost'; return null; }) }) } ] }).compile(); const config = module.get('CONFIG'); expect(config.get('DB_HOST')).toBe('localhost'); }); it('should mock useClass provider', async () => { class MockEmailService { send = jest.fn().mockResolvedValue({ success: true }); } const module = await Test.createTestingModule({ providers: [ UsersService, { provide: EmailService, useClass: MockEmailService } ] }).compile(); const emailService = module.get(EmailService); await emailService.send({ to: 'test@test.com' }); expect(emailService.send).toHaveBeenCalled(); }); });

4. Mock 外部服务

import { HttpService } from '@nestjs/axios'; import { of, throwError } from 'rxjs'; import { AxiosResponse } from 'axios'; describe('Mocking HttpService', () => { let service: ExternalApiService; let httpService: jest.Mocked<HttpService>; beforeEach(async () => { const module = await Test.createTestingModule({ providers: [ ExternalApiService, { provide: HttpService, useValue: { get: jest.fn(), post: jest.fn(), put: jest.fn(), delete: jest.fn() } } ] }).compile(); service = module.get(ExternalApiService); httpService = module.get(HttpService); }); it('should fetch data from external API', async () => { const mockResponse: AxiosResponse = { data: { id: 1, name: 'Test' }, status: 200, statusText: 'OK', headers: {}, config: {} as any }; httpService.get.mockReturnValue(of(mockResponse)); const result = await service.fetchData(); expect(result).toEqual({ id: 1, name: 'Test' }); }); it('should handle API error', async () => { httpService.get.mockReturnValue( throwError(() => new Error('Network Error')) ); await expect(service.fetchData()).rejects.toThrow('Network Error'); }); });

5. Mock ConfigService

describe('Mocking ConfigService', () => { let service: MyService; let configService: jest.Mocked<ConfigService>; beforeEach(async () => { const module = await Test.createTestingModule({ providers: [ MyService, { provide: ConfigService, useValue: { get: jest.fn((key: string) => { const config = { 'DATABASE_URL': 'sqlite::memory:', 'JWT_SECRET': 'test-secret', 'PORT': 3000, 'NODE_ENV': 'test' }; return config[key]; }), getOrThrow: jest.fn((key: string) => { const value = { 'JWT_SECRET': 'test-secret' }[key]; if (!value) throw new Error(`${key} not found`); return value; }) } } ] }).compile(); service = module.get(MyService); configService = module.get(ConfigService); }); it('should get config values', () => { expect(configService.get('DATABASE_URL')).toBe('sqlite::memory:'); expect(configService.get('PORT')).toBe(3000); }); });

7. overrideProvider 技巧

describe('Override Provider', () => { it('should override provider in imported module', async () => { const module = await Test.createTestingModule({ imports: [AppModule] }) .overrideProvider(UsersService) .useValue({ findAll: jest.fn().mockResolvedValue([{ id: 'mock' }]), findById: jest.fn().mockResolvedValue({ id: 'mock' }) }) .overrideProvider(getRepositoryToken(User)) .useValue(createMockRepository<User>()) .overrideGuard(AuthGuard) .useValue({ canActivate: () => true }) .overrideInterceptor(LoggingInterceptor) .useValue({ intercept: (context, next) => next.handle() }) .compile(); const service = module.get(UsersService); const result = await service.findAll(); expect(result).toEqual([{ id: 'mock' }]); }); });

8. 本节小结

📚
关键要点
  • 使用 getRepositoryToken 获取 Repository 的注入 token
  • useValue/useClass/useFactory 提供不同的 Mock 方式
  • overrideProvider 可以覆盖导入模块中的依赖
  • 创建 Mock 工具函数减少重复代码
  • Mock HttpService 需要使用 RxJS 的 Observable
