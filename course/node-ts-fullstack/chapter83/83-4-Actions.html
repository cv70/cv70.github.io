83.4 Actions

Pinia 中的业务逻辑处理

📖 本节概述

Actions 相当于组件中的方法，用于处理业务逻辑。与 Vuex 不同，Pinia 的 Actions 支持异步操作，可以替代 Vuex 中的 mutations 和 actions。

🎯 学习目标

  • 掌握 Actions 的定义方式
  • 学会异步 Actions 的处理
  • 理解 Actions 之间的调用
  • 掌握 Actions 的订阅

🔧 定义 Actions

Options Store

export const useUserStore = defineStore('user', { state: () => ({ user: null as User | null, token: null as string | null, loading: false, error: null as string | null }), actions: { // 同步 Action setUser(user: User) { this.user = user; }, setToken(token: string) { this.token = token; }, clearError() { this.error = null; }, // 异步 Action async login(credentials: LoginCredentials) { this.loading = true; this.error = null; try { const response = await api.login(credentials); this.setUser(response.user); this.setToken(response.token); return response; } catch (error) { this.error = error.message; throw error; } finally { this.loading = false; } }, async logout() { try { await api.logout(); } finally { this.user = null; this.token = null; } }, // 调用其他 Action async fetchUser() { const user = await api.getCurrentUser(); this.setUser(user); } } });

Setup Store

export const useUserStore = defineStore('user', () => { const user = ref<User | null>(null); const token = ref<string | null>(null); const loading = ref(false); const error = ref<string | null>(null); // 同步 Action function setUser(newUser: User) { user.value = newUser; } function setToken(newToken: string) { token.value = newToken; } function clearError() { error.value = null; } // 异步 Action async function login(credentials: LoginCredentials) { loading.value = true; error.value = null; try { const response = await api.login(credentials); setUser(response.user); setToken(response.token); return response; } catch (e) { error.value = e.message; throw e; } finally { loading.value = false; } } return { user, token, loading, error, setUser, setToken, clearError, login }; });

📖 调用 Actions

<script setup lang="ts"> import { useUserStore } from '@/stores/user'; const userStore = useUserStore(); // 调用同步 Action userStore.setUser({ id: 1, name: 'John' }); // 调用异步 Action const handleLogin = async () => { try { await userStore.login({ email: 'user@example.com', password: 'password' }); // 登录成功 } catch (error) { // 处理错误 } }; // 解构 Actions（不需要 storeToRefs） const { login, logout } = userStore; </script>

🔗 Actions 间调用

同一 Store 内调用

// Options Store export const useUserStore = defineStore('user', { actions: { async login(credentials) { const response = await api.login(credentials); this.setToken(response.token); // 调用其他 action await this.fetchUser(); // 调用异步 action }, setToken(token) { this.token = token; }, async fetchUser() { const user = await api.getCurrentUser(); this.user = user; } } }); // Setup Store export const useUserStore = defineStore('user', () => { async function login(credentials) { const response = await api.login(credentials); setToken(response.token); await fetchUser(); } function setToken(token) { /* ... */ } async function fetchUser() { /* ... */ } return { login, setToken, fetchUser }; });

跨 Store 调用

// stores/cart.ts import { useUserStore } from './user'; import { useProductsStore } from './products'; export const useCartStore = defineStore('cart', () => { const items = ref<CartItem[]>([]); async function checkout() { // 使用其他 store const userStore = useUserStore(); const productsStore = useProductsStore(); if (!userStore.isLoggedIn) { throw new Error('Please login first'); } const orderItems = items.value.map(item => ({ ...item, product: productsStore.getById(item.productId) })); const order = await api.createOrder({ userId: userStore.user!.id, items: orderItems }); items.value = []; return order; } return { items, checkout }; });

实际应用示例

// 全局错误处理 const userStore = useUserStore(); userStore.$onAction(({ name, onError }) => { onError(error => { if (error.code === 'UNAUTHORIZED') { userStore.logout(); router.push('/login'); } // 显示错误提示 toast.error(error.message); }); });

💡 最佳实践

// 1. 将复杂逻辑放在 Actions 中 export const useCartStore = defineStore('cart', () => { // 好的做法：将业务逻辑封装在 action 中 async function addToCart(productId: number, quantity: number = 1) { const existingItem = items.value.find(item => item.productId === productId); if (existingItem) { existingItem.quantity += quantity; } else { items.value.push({ productId, quantity }); } // 同步到服务器 await api.syncCart(items.value); } return { items, addToCart }; }); // 2. 使用 TypeScript 类型约束 interface LoginCredentials { email: string; password: string; remember?: boolean; } async function login(credentials: LoginCredentials) { // 类型安全的调用 } // 3. 处理加载状态和错误 async function fetchData() { loading.value = true; error.value = null; try { const data = await api.getData(); this.data = data; } catch (e) { error.value = e.message; throw e; } finally { loading.value = false; } }

📝 本节小结

  • Actions 用于处理业务逻辑，支持异步操作
  • 可以使用 this 或直接调用访问 state 和其他 actions
  • 可以跨 Store 调用其他 Store 的 actions
  • $onAction 可以订阅 actions 的调用
  • 推荐将复杂逻辑封装在 actions 中
