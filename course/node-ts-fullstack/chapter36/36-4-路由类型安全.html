🛣️ 类型安全路由

// src/routes/userRoutes.ts import { Router, Request, Response } from 'express'; import { User, CreateUserInput, UpdateUserInput, ApiResponse, PaginatedResponse } from '../types/api'; import * as userController from '../controllers/userController'; const router = Router(); // GET /users - 获取用户列表 router.get('/', async (req: Request, res: Response>) => { const { page = 1, limit = 10 } = req.query; const result = await userController.getUsers(Number(page), Number(limit)); res.json({ success: true, data: result.users, pagination: { page: Number(page), limit: Number(limit), total: result.total, totalPages: Math.ceil(result.total / Number(limit)) } }); }); // GET /users/:id - 获取单个用户 router.get('/:id', async (req: Request<{ id: string }>, res: Response>) => { const user = await userController.getUserById(req.params.id); if (!user) { return res.status(404).json({ success: false, error: 'User not found' }); } res.json({ success: true, data: user }); }); // POST /users - 创建用户 router.post('/', async (req: Request<{}, ApiResponse, CreateUserInput>, res: Response>) => { const user = await userController.createUser(req.body); res.status(201).json({ success: true, data: user, message: 'User created' }); } ); // PUT /users/:id - 更新用户 router.put('/:id', async (req: Request<{ id: string }, ApiResponse, UpdateUserInput>, res: Response>) => { const user = await userController.updateUser(req.params.id, req.body); if (!user) { return res.status(404).json({ success: false, error: 'User not found' }); } res.json({ success: true, data: user }); } ); // DELETE /users/:id - 删除用户 router.delete('/:id', async (req: Request<{ id: string }>, res: Response) => { await userController.deleteUser(req.params.id); res.json({ success: true, message: 'User deleted' }); }); export default router;

🎮 类型安全控制器

// src/controllers/userController.ts import { User, CreateUserInput, UpdateUserInput } from '../types/api'; import { NotFoundError } from '../utils/errors'; // 模拟数据库 const users: User[] = []; export async function getUsers(page: number, limit: number): Promise<{ users: User[], total: number }> { const start = (page - 1) * limit; const paginatedUsers = users.slice(start, start + limit); return { users: paginatedUsers, total: users.length }; } export async function getUserById(id: string): Promise { return users.find(u => u.id === id) || null; } export async function createUser(input: CreateUserInput): Promise { const user: User = { id: crypto.randomUUID(), name: input.name, email: input.email, role: 'user', createdAt: new Date() }; users.push(user); return user; } export async function updateUser(id: string, input: UpdateUserInput): Promise { const index = users.findIndex(u => u.id === id); if (index === -1) { return null; } users[index] = { ...users[index], ...input }; return users[index]; } export async function deleteUser(id: string): Promise { const index = users.findIndex(u => u.id === id); if (index === -1) { throw new NotFoundError('User not found'); } users.splice(index, 1); return true; }

( fn: TypedRequestHandler ): RequestHandler => { return (req, res, next) => { Promise.resolve(fn(req, res, next)).catch(next); }; }; // 使用 import { asyncHandler } from '../utils/routeTypes'; router.get<{ id: string }, User>('/:id', asyncHandler(async (req, res) => { const user = await getUserById(req.params.id); res.json(user); }));
