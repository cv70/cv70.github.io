🎯 对抗攻击

# 对抗攻击类型 ## 1. 白盒攻击 # 知道模型结构和参数 ### FGSM攻击 import torch import torch.nn.functional as F def fgsm_attack(model, image, label, epsilon=0.01): image.requires_grad = True output = model(image) loss = F.nll_loss(output, label) model.zero_grad() loss.backward() # 生成对抗样本 perturbed_image = image + epsilon * image.grad.data.sign() return perturbed_image ### PGD攻击 def pgd_attack(model, image, label, epsilon=0.1, alpha=0.01, num_iter=40): perturbed_image = image.clone().detach() perturbed_image.requires_grad = True for _ in range(num_iter): output = model(perturbed_image) loss = F.nll_loss(output, label) model.zero_grad() loss.backward() # 生成对抗样本 perturbed_image = perturbed_image + alpha * perturbed_image.grad.data.sign() perturbed_image = torch.clamp(perturbed_image, image - epsilon, image + epsilon) perturbed_image = torch.clamp(perturbed_image, 0, 1) return perturbed_image ## 2. 黑盒攻击 # 只知道模型输入输出 ### 查询攻击 def query_attack(model, target_image, target_class, max_queries=1000): current_image = target_image.clone() best_loss = float('inf') for _ in range(max_queries): output = model(current_image) loss = 1 - output[0, target_class].item() if loss < best_loss: best_loss = loss best_image = current_image.clone() # 简单的梯度估计 perturbation = torch.randn_like(current_image) * 0.1 output_plus = model(current_image + perturbation) output_minus = model(current_image - perturbation) gradient = (output_plus - output_minus) / 0.2 current_image = current_image + 0.01 * gradient[0, target_class] * perturbation return best_image

# AI系统安全评估 ## 1. 对抗鲁棒性测试 ```python import numpy as np from cleverhans.future.tf2.attacks.fast_gradient_method import fast_gradient_method def evaluate_adversarial_robustness(model, test_dataset, epsilon=0.1): correct = 0 total = 0 for data, target in test_dataset: # 生成对抗样本 adv_data = fast_gradient_method(model, data, eps=epsilon, norm=np.inf) # 测试模型 output = model(adv_data) pred = output.argmax(axis=1) if pred == target: correct += 1 total += 1 accuracy = correct / total return accuracy ``` ## 2. 隐私保护评估 ```python import torch from torch.utils.data import DataLoader def evaluate_privacy_leakage(model, training_data, test_samples): # 成员推断攻击 member_scores = [] non_member_scores = [] # 训练集样本 for data in training_data: output = model(data) confidence = output.max().item() member_scores.append(confidence) # 非训练集样本 for data in test_samples: output = model(data) confidence = output.max().item() non_member_scores.append(confidence) # 评估泄露风险 avg_member = np.mean(member_scores) avg_non_member = np.mean(non_member_scores) return { 'member_confidence': avg_member, 'non_member_confidence': avg_non_member, 'leakage_risk': abs(avg_member - avg_non_member) } ``` ## 3. 模型完整性检查 ```python import hashlib def verify_model_integrity(model_path, expected_hash): with open(model_path, 'rb') as f: model_data = f.read() actual_hash = hashlib.sha256(model_data).hexdigest() return actual_hash == expected_hash ```
