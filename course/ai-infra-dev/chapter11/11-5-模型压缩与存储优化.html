模型压缩的价值

随着模型规模的快速增长，存储成本和传输带宽成为关键瓶颈。 模型压缩技术可以在保持模型性能的同时，显著减少模型大小， 从而降低存储成本、加快加载速度、提升推理效率。

✂️ 剪枝压缩

非结构化剪枝

import torch.nn.utils.prune as prune # 全局非结构化剪枝 def global_prune(model, pruning_ratio=0.3): """全局剪枝所有参数""" parameters_to_prune = [] # 收集所有可剪枝的参数 for name, module in model.named_modules(): if isinstance(module, (torch.nn.Linear, torch.nn.Conv2d)): parameters_to_prune.append((module, 'weight')) # 全局剪枝 prune.global_unstructured( parameters_to_prune, pruning_method=prune.L1Unstructured, amount=pruning_ratio ) # 移除剪枝掩码 for module, _ in parameters_to_prune: prune.remove(module, 'weight') return model # 局部非结构化剪枝 def local_prune(model, layer_name, pruning_ratio=0.5): """局部剪枝特定层""" for name, module in model.named_modules(): if name == layer_name: prune.l1_unstructured(module, name='weight', amount=pruning_ratio) prune.remove(module, 'weight') break return model

结构化剪枝

import torch.nn.utils.prune as prune # 结构化剪枝（剪枝整个通道） def structured_prune(model, pruning_ratio=0.2): """结构化剪枝，移除整个通道""" for name, module in model.named_modules(): if isinstance(module, torch.nn.Conv2d): # 按L2范数排序通道 l2_norm = torch.norm(module.weight.data, dim=(1, 2, 3)) num_channels = module.out_channels num_pruned = int(pruning_ratio * num_channels) # 找到要剪枝的通道索引 indices_to_prune = torch.argsort(l2_norm)[:num_pruned] # 创建结构化剪枝 mask = torch.ones(num_channels) mask[indices_to_pruned] = 0 prune.custom_from_mask(module, 'weight', mask.unsqueeze(1).unsqueeze(1).unsqueeze(1)) prune.remove(module, 'weight') return model

迭代式剪枝

class IterativePruner: def __init__(self, model, initial_sparsity=0.2, final_sparsity=0.7, steps=10): self.model = model self.initial_sparsity = initial_sparsity self.final_sparsity = final_sparsity self.steps = steps self.current_step = 0 def prune_step(self): """执行一次剪枝步骤""" if self.current_step >= self.steps: return # 计算当前剪枝率 current_sparsity = self._calculate_sparsity( self.initial_sparsity, self.final_sparsity, self.current_step, self.steps ) # 执行剪枝 self._apply_pruning(current_sparsity) self.current_step += 1 print(f"Step {self.current_step}: sparsity = {current_sparsity:.2%}") def _calculate_sparsity(self, initial, final, step, total_steps): """计算当前剪枝率（多项式衰减）""" return final + (initial - final) * (1 - step / total_steps) ** 3 def _apply_pruning(self, sparsity): """应用剪枝""" for module in self.model.modules(): if isinstance(module, (torch.nn.Linear, torch.nn.Conv2d)): prune.l1_unstructured(module, name='weight', amount=sparsity) prune.remove(module, 'weight')

📉 知识蒸馏

基本原理

蒸馏训练

import torch import torch.nn as nn import torch.nn.functional as F class DistillationLoss(nn.Module): def __init__(self, temperature=4.0, alpha=0.7): super().__init__() self.temperature = temperature self.alpha = alpha def forward(self, student_logits, teacher_logits, labels): """计算蒸馏损失""" # 软标签损失（KL散度） soft_teacher = F.softmax(teacher_logits / self.temperature, dim=-1) soft_student = F.log_softmax(student_logits / self.temperature, dim=-1) distill_loss = F.kl_div( soft_student, soft_teacher, reduction='batchmean' ) * (self.temperature ** 2) # 硬标签损失（交叉熵） hard_loss = F.cross_entropy(student_logits, labels) # 组合损失 return self.alpha * distill_loss + (1 - self.alpha) * hard_loss # 训练循环 def distill_train(student_model, teacher_model, train_loader, epochs=10, temperature=4.0, alpha=0.7): teacher_model.eval() # 教师模型不训练 student_model.train() criterion = DistillationLoss(temperature, alpha) optimizer = torch.optim.Adam(student_model.parameters()) for epoch in range(epochs): total_loss = 0 for batch_idx, (data, labels) in enumerate(train_loader): # 前向传播 with torch.no_grad(): teacher_logits = teacher_model(data) student_logits = student_model(data) # 计算损失 loss = criterion(student_logits, teacher_logits, labels) # 反向传播 optimizer.zero_grad() loss.backward() optimizer.step() total_loss += loss.item() print(f"Epoch {epoch}, Loss: {total_loss / len(train_loader)}")

多层蒸馏

class MultiLayerDistillation(nn.Module): def __init__(self, student_model, teacher_model): super().__init__() self.student = student_model self.teacher = teacher_model # 注册钩子获取中间层输出 self.teacher_features = {} self.student_features = {} self._register_hooks() def _register_hooks(self): """注册前向钩子""" def teacher_hook(module, input, output): self.teacher_features[module] = output def student_hook(module, input, output): self.student_features[module] = output # 为对应层注册钩子 for teacher_layer, student_layer in zip(self.teacher.layers, self.student.layers): teacher_layer.register_forward_hook(teacher_hook) student_layer.register_forward_hook(student_hook) def feature_distillation_loss(self): """计算特征蒸馏损失""" loss = 0 for teacher_layer, student_layer in zip(self.teacher.layers, self.student.layers): teacher_feat = self.teacher_features[teacher_layer] student_feat = self.student_features[student_layer] # 使用MSE损失 loss += F.mse_loss(student_feat, teacher_feat.detach()) return loss / len(self.teacher.layers)

🗜️ 存储格式优化

压缩格式

import gzip import lz4.frame import zstandard as zstd def compress_model(model_path, algorithm='zstd', level=3): """压缩模型文件""" with open(model_path, 'rb') as f: model_data = f.read() if algorithm == 'gzip': compressed = gzip.compress(model_data, compresslevel=level) suffix = '.gz' elif algorithm == 'lz4': compressed = lz4.frame.compress(model_data, compression_level=level) suffix = '.lz4' elif algorithm == 'zstd': cctx = zstd.ZstdCompressor(level=level) compressed = cctx.compress(model_data) suffix = '.zst' else: raise ValueError(f"Unknown compression algorithm: {algorithm}") compressed_path = model_path + suffix with open(compressed_path, 'wb') as f: f.write(compressed) original_size = len(model_data) compressed_size = len(compressed) ratio = (1 - compressed_size / original_size) * 100 print(f"Compression ratio: {ratio:.1f}%") print(f"Original: {original_size / 1024 / 1024:.1f} MB") print(f"Compressed: {compressed_size / 1024 / 1024:.1f} MB") return compressed_path

增量存储

import numpy as np class DeltaStorage: def __init__(self, base_version_path): self.base_version = np.load(base_version_path) def save_delta(self, new_version, delta_path): """保存增量差异""" delta = new_version - self.base_version # 保存差异 np.save(delta_path, delta) # 计算压缩比 original_size = new_version.nbytes delta_size = delta.nbytes ratio = (1 - delta_size / original_size) * 100 return ratio def load_delta(self, delta_path): """加载增量版本""" delta = np.load(delta_path) return self.base_version + delta # 使用示例 delta_storage = DeltaStorage('model_v1.0.npy') compression_ratio = delta_storage.save_delta(model_v2.0, 'delta_v2.0.npy') print(f"Delta compression ratio: {compression_ratio:.1f}%")

共享参数存储

class SharedParameterStorage: def __init__(self): self.parameter_cache = {} def save_model(self, model, model_name): """保存模型，共享相同参数""" saved_state = {} for name, param in model.named_parameters(): param_hash = hash_param(param.data) if param_hash in self.parameter_cache: # 参数已存在，只保存引用 saved_state[name] = { 'reference': self.parameter_cache[param_hash], 'shape': param.shape } else: # 保存新参数 param_path = f"shared_params/{param_hash}.npy" np.save(param_path, param.data.cpu().numpy()) saved_state[name] = { 'path': param_path, 'hash': param_hash, 'shape': param.shape } self.parameter_cache[param_hash] = param_path # 保存模型配置 np.save(f"{model_name}_config.npy", saved_state) def hash_param(tensor): """计算参数的哈希值""" return hash(tensor.data.tobytes())

📈 压缩效果评估

评估指标

存储指标

  • 模型大小 (MB)
  • 压缩比
  • 加载时间
  • 存储成本

性能指标

  • 准确率下降
  • 推理延迟
  • 吞吐量
  • 能耗

评估脚本

import time import torch class CompressionEvaluator: def __init__(self, original_model, compressed_model, test_data): self.original = original_model self.compressed = compressed_model self.test_data = test_data def evaluate(self): """全面评估压缩效果""" results = {} # 1. 模型大小 results['original_size'] = self._get_model_size(self.original) results['compressed_size'] = self._get_model_size(self.compressed) results['compression_ratio'] = ( results['original_size'] / results['compressed_size'] ) # 2. 准确率 results['original_acc'] = self._evaluate_accuracy(self.original) results['compressed_acc'] = self._evaluate_accuracy(self.compressed) results['acc_drop'] = ( results['original_acc'] - results['compressed_acc'] ) # 3. 推理性能 results['original_latency'] = self._measure_latency(self.original) results['compressed_latency'] = self._measure_latency(self.compressed) results['speedup'] = ( results['original_latency'] / results['compressed_latency'] ) # 4. 加载时间 results['load_time'] = self._measure_load_time() return results def _get_model_size(self, model): """获取模型大小""" param_size = sum(p.numel() * p.element_size() for p in model.parameters()) return param_size / 1024 / 1024 # MB def _evaluate_accuracy(self, model): """评估准确率""" model.eval() correct = 0 total = 0 with torch.no_grad(): for data, target in self.test_data: output = model(data) pred = output.argmax(dim=1) correct += (pred == target).sum().item() total += target.size(0) return correct / total def _measure_latency(self, model): """测量推理延迟""" model.eval() latencies = [] with torch.no_grad(): for i in range(100): # 预热 model(self.test_data[0][0:1]) for data, _ in self.test_data: start = time.time() _ = model(data[0:1]) end = time.time() latencies.append(end - start) return np.mean(latencies) * 1000 # ms def print_report(self, results): """打印评估报告""" print("=" * 50) print("压缩效果评估报告") print("=" * 50) print(f"模型大小:") print(f" 原始: {results['original_size']:.2f} MB") print(f" 压缩: {results['compressed_size']:.2f} MB") print(f" 压缩比: {results['compression_ratio']:.2f}x") print() print(f"准确率:") print(f" 原始: {results['original_acc']:.4f}") print(f" 压缩: {results['compressed_acc']:.4f}") print(f" 下降: {results['acc_drop']:.4f}") print() print(f"推理延迟:") print(f" 原始: {results['original_latency']:.2f} ms") print(f" 压缩: {results['compressed_latency']:.2f} ms") print(f" 加速: {results['speedup']:.2f}x")
⚠️
压缩注意事项

1）在压缩前评估模型对精度的敏感度；2）保留原始模型作为基准；3）在真实数据上验证压缩后的模型；4）监控压缩后的模型性能；5）定期评估是否需要重新压缩。
