💻 融资成本计算

杠杆交易成本示例

import numpy as np

def calculate_margin_cost(principal, margin_rate, days, annual_rate=0.06):
    """计算融资成本"""
    daily_rate = annual_rate / 365
    margin_debt = principal * margin_rate
    cost = margin_debt * daily_rate * days
    return cost

# 示例
principal = 1000000  # 本金
margin_rate = 0.5  # 50%保证金
hold_days = 90  # 持仓天数

cost = calculate_margin_cost(principal, margin_rate, hold_days)
print(f"融资成本: ¥{cost:,.2f}")
print(f"年化成本率: {(cost/principal)*365/hold_days*100:.2f}%")
