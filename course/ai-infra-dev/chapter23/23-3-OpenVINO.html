什么是OpenVINO

OpenVINO（Open Visual Inference and Neural Network Optimization）是Intel推出的AI推理优化工具包。 它可以加速深度学习模型在Intel硬件（CPU、GPU、VPU、FPGA）上的推理性能， 广泛应用于边缘计算、工业自动化和智能视频分析等领域。

💻 使用示例

import openvino as ov
import numpy as np

# 将PyTorch模型转换为OpenVINO IR
def convert_to_openvino(model_path, output_dir):
    """
    转换模型到OpenVINO IR格式
    """
    from openvino.tools import mo

    # 转换ONNX模型
    model = mo.convert_model(
        input_model=model_path,  # ONNX/PB/TorchScript
        output_dir=output_dir,
        model_name="model",
        data_type="FP16"  # FP16或FP32
    )

    # 保存IR格式
    ov.save_model(model, f"{output_dir}/model.xml")

# 创建推理核心
def create_inference_core(device="CPU"):
    """
    创建推理核心
    """
    core = ov.Core()

    # 可用设备
    available_devices = core.available_devices
    print(f"可用设备: {available_devices}")

    # 设备属性
    device_properties = core.get_property(device, "SUPPORTED_PROPERTIES")
    print(f"设备属性: {device_properties}")

    return core

# 加载模型并创建推理请求
def load_model(core, model_xml, device="CPU"):
    """
    加载模型
    """
    # 读取模型
    model = core.read_model(model_xml)

    # 编译模型
    compiled_model = core.compile_model(
        model,
        device_name=device,
        config={
            "PERFORMANCE_HINT": "LATENCY"  # 或"THROUGHPUT"
        }
    )

    # 创建推理请求
    infer_request = compiled_model.create_infer_request()

    return infer_request, compiled_model

# 执行推理
def openvino_infer(infer_request, input_data):
    """
    执行推理
    """
    # 设置输入
    infer_request.infer({0: input_data})

    # 获取输出
    output = infer_request.get_output_tensor(0)
    result = output.data

    return result

# 异步推理
def async_infer(infer_request, input_data, callback):
    """
    异步推理
    """
    # 设置回调函数
    infer_request.set_callback(
        lambda *args: callback(infer_request.get_output_tensor(0).data)
    )

    # 启动异步推理
    infer_request.start_async({0: input_data})

# 量化模型
def quantize_model(model_xml, calibration_data):
    """
    INT8量化
    """
    from openvino.tools import pot

    # 配置量化
    config = {
        "model": {"model_name": "model", "model_xml": model_xml},
        "engine": {"device": "CPU", "stat_requests_number": 100},
        "compression": {"algorithm": "DefaultQuantization"}
    }

    # 运行量化
    quantized_model = pot.run(config)

    return quantized_model

# 完整示例
if __name__ == "__main__":
    # 1. 转换模型（首次）
    convert_to_openvino("model.onnx", "openvino_model")

    # 2. 创建推理核心
    core = create_inference_core(device="CPU")

    # 3. 加载模型
    infer_request, compiled_model = load_model(core, "openvino_model/model.xml")

    # 4. 获取输入输出信息
    input_shape = compiled_model.input().shape
    output_shape = compiled_model.output().shape
    print(f"输入形状: {input_shape}")
    print(f"输出形状: {output_shape}")

    # 5. 执行推理
    input_data = np.random.randn(*input_shape).astype(np.float32)
    output = openvino_infer(infer_request, input_data)

    print(f"推理结果: {output}")

OpenVINO使用建议
  • CPU推理使用INT8量化可获得最佳性能
  • 根据场景选择LATENCY或THROUGHPUT模式
