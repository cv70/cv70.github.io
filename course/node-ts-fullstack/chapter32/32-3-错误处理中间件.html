32.3 错误处理中间件

掌握 Express 错误处理的完整机制

🎯 学习目标

  • 理解 Express 错误处理中间件的工作原理
  • 学会自定义错误类型和错误处理
  • 掌握异步错误的捕获方法
  • 学会优雅地处理和返回错误信息

📖 错误处理中间件基础

Express 错误处理中间件有四个参数 (err, req, res, next)，Express 会自动识别并调用它来处理错误。

基本语法

// 错误处理中间件有四个参数 app.use((err, req, res, next) => { console.error(err.stack); res.status(500).send('Something broke!'); });
💡
关键区别

普通中间件有 3 个参数，错误处理中间件有 4 个参数。Express 根据参数数量识别中间件类型。

🔄 错误传递机制

同步错误

// 同步代码中的错误会自动被 Express 捕获 app.get('/sync-error', (req, res) => { throw new Error('Sync error!'); }); // 或者手动传递给 next() app.get('/manual-error', (req, res, next) => { try { // 一些操作 throw new Error('Manual error'); } catch (error) { next(error); // 传递给错误处理中间件 } }); // 错误处理中间件 app.use((err, req, res, next) => { res.status(500).json({ success: false, message: err.message }); });

异步错误

// ❌ 异步错误不会被自动捕获 app.get('/async-error', async (req, res) => { const user = await User.findById('invalid'); // 如果出错，Express 不会捕获 }); // ✅ 方式一：手动捕获 app.get('/async-error-1', async (req, res, next) => { try { const user = await User.findById('invalid'); res.json(user); } catch (error) { next(error); } }); // ✅ 方式二：使用异步包装器 const asyncHandler = fn => (req, res, next) => { Promise.resolve(fn(req, res, next)).catch(next); }; app.get('/async-error-2', asyncHandler(async (req, res) => { const user = await User.findById('invalid'); res.json(user); }));

🏗️ 自定义错误类

// utils/errors.js // 基础应用错误 class AppError extends Error { constructor(message, statusCode = 500) { super(message); this.statusCode = statusCode; this.status = `${statusCode}`.startsWith('4') ? 'fail' : 'error'; this.isOperational = true; Error.captureStackTrace(this, this.constructor); } } // 404 错误 class NotFoundError extends AppError { constructor(message = 'Resource not found') { super(message, 404); } } // 验证错误 class ValidationError extends AppError { constructor(message = 'Validation failed', errors = []) { super(message, 400); this.errors = errors; } } // 认证错误 class AuthenticationError extends AppError { constructor(message = 'Authentication failed') { super(message, 401); } } // 权限错误 class AuthorizationError extends AppError { constructor(message = 'Access denied') { super(message, 403); } } // 冲突错误 class ConflictError extends AppError { constructor(message = 'Resource conflict') { super(message, 409); } } module.exports = { AppError, NotFoundError, ValidationError, AuthenticationError, AuthorizationError, ConflictError };

🛠️ 完整的错误处理系统

// middleware/errorHandler.js const { AppError } = require('../utils/errors'); // 开发环境错误响应 const sendErrorDev = (err, res) => { res.status(err.statusCode).json({ success: false, status: err.status, message: err.message, errors: err.errors, stack: err.stack, error: err }); }; // 生产环境错误响应 const sendErrorProd = (err, res) => { // 操作性错误：可以安全地发送给客户端 if (err.isOperational) { res.status(err.statusCode).json({ success: false, status: err.status, message: err.message, errors: err.errors }); } else { // 编程错误或未知错误：不泄露细节 console.error('ERROR 💥:', err); res.status(500).json({ success: false, status: 'error', message: 'Something went wrong' }); } }; // 处理特定错误类型 const handleCastErrorDB = (err) => { return new AppError(`Invalid ${err.path}: ${err.value}`, 400); }; const handleDuplicateFieldsDB = (err) => { const value = err.errmsg.match(/(["'])(\\?.)*?\1/)[0]; return new AppError(`Duplicate field value: ${value}`, 400); }; const handleValidationErrorDB = (err) => { const errors = Object.values(err.errors).map(el => el.message); return new AppError(`Invalid input data: ${errors.join('. ')}`, 400); }; const handleJWTError = () => { return new AppError('Invalid token. Please log in again', 401); }; const handleJWTExpiredError = () => { return new AppError('Your token has expired. Please log in again', 401); }; // 全局错误处理中间件 module.exports = (err, req, res, next) => { err.statusCode = err.statusCode || 500; err.status = err.status || 'error'; if (process.env.NODE_ENV === 'development') { sendErrorDev(err, res); } else { let error = { ...err }; error.message = err.message; error.stack = err.stack; // MongoDB CastError if (err.name === 'CastError') { error = handleCastErrorDB(err); } // MongoDB 重复键错误 if (err.code === 11000) { error = handleDuplicateFieldsDB(err); } // MongoDB 验证错误 if (err.name === 'ValidationError') { error = handleValidationErrorDB(err); } // JWT 错误 if (err.name === 'JsonWebTokenError') { error = handleJWTError(); } if (err.name === 'TokenExpiredError') { error = handleJWTExpiredError(); } sendErrorProd(error, res); } };

🚀 使用错误处理

在路由中使用自定义错误

// controllers/userController.js const { NotFoundError, ValidationError } = require('../utils/errors'); exports.getUser = async (req, res, next) => { const user = await User.findById(req.params.id); if (!user) { throw new NotFoundError('User not found'); } res.json(user); }; exports.createUser = async (req, res, next) => { const { name, email } = req.body; if (!name || !email) { throw new ValidationError('Name and email are required', [ { field: 'name', message: 'Name is required' }, { field: 'email', message: 'Email is required' } ]); } const user = await User.create({ name, email }); res.status(201).json(user); }; // 使用 asyncHandler 包装 const asyncHandler = require('../middleware/asyncHandler'); exports.getUser = asyncHandler(async (req, res, next) => { const user = await User.findById(req.params.id); if (!user) throw new NotFoundError('User not found'); res.json(user); });

异步处理包装器

// middleware/asyncHandler.js const asyncHandler = fn => (req, res, next) => { Promise.resolve(fn(req, res, next)).catch(next); }; module.exports = asyncHandler; // 在路由中使用 const asyncHandler = require('../middleware/asyncHandler'); router.get('/users/:id', asyncHandler(userController.getUser) ); router.post('/users', asyncHandler(userController.createUser) );

📋 404 错误处理

// 处理未匹配的路由 app.all('*', (req, res, next) => { next(new NotFoundError(`Cannot find ${req.originalUrl} on this server`)); }); // 或者 app.use((req, res) => { res.status(404).json({ success: false, message: `Route ${req.originalUrl} not found` }); });

📊 完整应用示例

// app.js const express = require('express'); const app = express(); // 中间件 app.use(express.json()); // 路由 app.use('/api/users', require('./routes/users')); app.use('/api/posts', require('./routes/posts')); // 404 处理 app.all('*', (req, res, next) => { next(new NotFoundError(`Cannot find ${req.originalUrl}`)); }); // 全局错误处理 - 必须放在最后 app.use(errorHandler); module.exports = app;

✅ 小结

📌
本节要点
  • 错误处理中间件有四个参数 (err, req, res, next)
  • 同步错误自动捕获，异步错误需要手动传递给 next()
  • 使用 asyncHandler 包装器简化异步错误处理
  • 创建自定义错误类，区分不同类型的错误
  • 开发和生产环境返回不同的错误详情
