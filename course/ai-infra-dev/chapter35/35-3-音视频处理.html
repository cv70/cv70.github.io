🔧 Python音频处理

音频处理实现

import librosa
import numpy as np
import soundfile as sf
from pydub import AudioSegment

class AudioProcessor:
    def __init__(self, sr=16000):
        self.sr = sr

    def load_audio(self, audio_path: str):
        """加载音频"""
        audio, sr = librosa.load(audio_path, sr=self.sr)
        return audio, sr

    def resample(self, audio: np.ndarray, original_sr: int):
        """重采样"""
        if original_sr != self.sr:
            audio = librosa.resample(audio, orig_sr=original_sr,
                                    target_sr=self.sr)
        return audio

    def denoise(self, audio: np.ndarray):
        """降噪处理"""
        # 使用频谱减法
        stft = librosa.stft(audio)
        magnitude, phase = librosa.magphase(stft)

        # 估计噪声谱
        noise = np.mean(magnitude[:, :10], axis=1, keepdims=True)

        # 减去噪声
        magnitude_denoised = magnitude - noise
        magnitude_denoised = np.maximum(magnitude_denoised, 0)

        # 重建信号
        stft_denoised = magnitude_denoised * phase
        audio_denoised = librosa.istft(stft_denoised)

        return audio_denoised

    def extract_mfcc(self, audio: np.ndarray, n_mfcc=13):
        """提取MFCC特征"""
        mfcc = librosa.feature.mfcc(
            y=audio,
            sr=self.sr,
            n_mfcc=n_mfcc
        )
        return mfcc

    def extract_mel(self, audio: np.ndarray, n_mels=80):
        """提取Mel频谱"""
        mel = librosa.feature.melspectrogram(
            y=audio,
            sr=self.sr,
            n_mels=n_mels
        )
        log_mel = librosa.power_to_db(mel)
        return log_mel

    def segment_audio(self, audio: np.ndarray,
                    segment_length=10.0):
        """分段处理"""
        samples_per_segment = int(segment_length * self.sr)
        segments = []

        for i in range(0, len(audio), samples_per_segment):
            segment = audio[i:i + samples_per_segment]
            if len(segment) == samples_per_segment:
                segments.append(segment)

        return segments

    def process(self, audio_path: str):
        """完整处理流程"""
        # 加载音频
        audio, sr = self.load_audio(audio_path)

        # 重采样
        audio = self.resample(audio, sr)

        # 降噪
        audio = self.denoise(audio)

        # 提取特征
        mfcc = self.extract_mfcc(audio)
        mel = self.extract_mel(audio)

        return {
            "audio": audio,
            "mfcc": mfcc,
            "mel": mel
        }

# 使用示例
processor = AudioProcessor(sr=16000)
result = processor.process("input.wav")

视频处理实现

import cv2
import numpy as np
from typing import List
import imageio

class VideoProcessor:
    def __init__(self, target_size=(224, 224)):
        self.target_size = target_size

    def extract_frames(self, video_path: str,
                     max_frames=None,
                     fps=None) -> List[np.ndarray]:
        """提取视频帧"""
        cap = cv2.VideoCapture(video_path)
        frames = []

        if fps:
            # 按FPS采样
            video_fps = cap.get(cv2.CAP_PROP_FPS)
            frame_interval = int(video_fps / fps)
        else:
            frame_interval = 1

        frame_count = 0
        while True:
            ret, frame = cap.read()
            if not ret:
                break

            if frame_count % frame_interval == 0:
                frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                frame = cv2.resize(frame, self.target_size)
                frames.append(frame)

                if max_frames and len(frames) >= max_frames:
                    break

            frame_count += 1

        cap.release()
        return frames

    def extract_keyframes(self, video_path: str,
                         threshold=30.0):
        """提取关键帧"""
        cap = cv2.VideoCapture(video_path)
        frames = []

        prev_frame = None
        while True:
            ret, frame = cap.read()
            if not ret:
                break

            # 转换为灰度图
            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

            if prev_frame is not None:
                # 计算帧间差异
                diff = cv2.absdiff(gray, prev_frame)
                diff_score = np.mean(diff)

                # 如果差异超过阈值，保存关键帧
                if diff_score > threshold:
                    frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                    frame = cv2.resize(frame, self.target_size)
                    frames.append(frame)

            prev_frame = gray

        cap.release()
        return frames

    def create_video(self, frames: List[np.ndarray],
                    output_path: str, fps=30):
        """创建视频"""
        if not frames:
            return

        height, width = frames[0].shape[:2]
        fourcc = cv2.VideoWriter_fourcc(*'mp4v')
        writer = cv2.VideoWriter(output_path, fourcc,
                              fps, (width, height))

        for frame in frames:
            frame_bgr = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR)
            writer.write(frame_bgr)

        writer.release()

    def extract_temporal_features(self, frames: List[np.ndarray]):
        """提取时序特征"""
        # 计算光流
        if len(frames) < 2:
            return None

        prev_gray = cv2.cvtColor(frames[0], cv2.COLOR_RGB2GRAY)
        optical_flows = []

        for frame in frames[1:]:
            curr_gray = cv2.cvtColor(frame, cv2.COLOR_RGB2GRAY)

            # 计算光流
            flow = cv2.calcOpticalFlowFarneback(
                prev_gray, curr_gray, None, 0.5, 3, 15, 3, 5, 1.2, 0
            )

            optical_flows.append(flow)
            prev_gray = curr_gray

        return optical_flows

    def process(self, video_path: str):
        """完整处理流程"""
        # 提取帧
        frames = self.extract_frames(video_path, max_frames=16)

        # 提取时序特征
        optical_flows = self.extract_temporal_features(frames)

        return {
            "frames": frames,
            "optical_flows": optical_flows
        }

# 使用示例
processor = VideoProcessor()
result = processor.process("input.mp4")

处理建议

对于大视频文件，建议使用关键帧提取减少处理量。音频处理使用分段处理避免内存溢出。使用GPU加速提升处理速度。
