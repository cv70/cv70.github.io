🔐 数据加密

# 数据加密方案 ## 1. 传输加密 ### TLS/SSL加密 import ssl import socket def create_secure_connection(host, port): context = ssl.create_default_context() context.verify_mode = ssl.CERT_REQUIRED context.check_hostname = True sock = socket.create_connection((host, port)) secure_sock = context.wrap_socket(sock, server_hostname=host) return secure_sock ### API调用加密 import requests def secure_api_call(url, data, cert_path, key_path): response = requests.post( url, json=data, cert=(cert_path, key_path), verify=True ) return response ## 2. 存储加密 ### 文件加密 from cryptography.fernet import Fernet import os def encrypt_file(file_path, key): cipher = Fernet(key) with open(file_path, 'rb') as f: file_data = f.read() encrypted_data = cipher.encrypt(file_data) with open(f"{file_path}.enc", 'wb') as f: f.write(encrypted_data) def decrypt_file(encrypted_path, key, output_path): cipher = Fernet(key) with open(encrypted_path, 'rb') as f: encrypted_data = f.read() decrypted_data = cipher.decrypt(encrypted_data) with open(output_path, 'wb') as f: f.write(decrypted_data) ### 数据库加密 import pymysql from cryptography.fernet import Fernet def store_encrypted_data(data, key): cipher = Fernet(key) encrypted_data = cipher.encrypt(data.encode()) conn = pymysql.connect( host='localhost', user='user', password='password', database='database' ) with conn.cursor() as cursor: cursor.execute( "INSERT INTO encrypted_data (data) VALUES (%s)", (encrypted_data,) ) conn.commit() conn.close() ## 3. 内存加密 import os import mmap def secure_memory_alloc(size): # 使用mlock锁定内存 mem = mmap.mmap(-1, size, flags=mmap.MAP_SHARED | mmap.MAP_ANONYMOUS) os.mlock(mem, size) return mem

🔐 差分隐私

# 差分隐私实现 ## 1. 拉普拉斯机制 import numpy as np def laplace_mechanism(data, epsilon=1.0, sensitivity=1.0): """ 拉普拉斯机制：用于数值型数据 """ noise = np.random.laplace(0, sensitivity / epsilon) return data + noise ## 2. 指数机制 import numpy as np def exponential_mechanism(data, score_function, epsilon=1.0): """ 指数机制：用于离散型数据 """ scores = np.array([score_function(data, r) for r in range(len(data))]) scaled_scores = epsilon * scores / (2 * np.max(scores) - 2 * np.min(scores)) probabilities = np.exp(scaled_scores) / np.sum(np.exp(scaled_scores)) return np.random.choice(len(data), p=probabilities) ## 3. DP-SGD训练 import torch import torch.nn as nn import torch.optim as optim def dp_sgd_step(model, optimizer, data, target, epsilon=1.0, delta=1e-5): """ 差分隐私SGD """ # 计算梯度 optimizer.zero_grad() output = model(data) loss = nn.CrossEntropyLoss()(output, target) loss.backward() # 裁剪梯度 max_grad_norm = 1.0 torch.nn.utils.clip_grad_norm_(model.parameters(), max_grad_norm) # 添加噪声 for param in model.parameters(): noise = torch.randn_like(param.grad) * max_grad_norm / epsilon param.grad += noise # 更新参数 optimizer.step() return epsilon ## 4. PyTorch差分隐私 from opacus import PrivacyEngine def train_with_dp(model, train_loader, optimizer, epochs, delta=1e-5): """ 使用Opacus进行差分隐私训练 """ privacy_engine = PrivacyEngine() model, optimizer, train_loader = privacy_engine.make_private( module=model, optimizer=optimizer, data_loader=train_loader, noise_multiplier=1.1, max_grad_norm=1.0, ) for epoch in range(epochs): for data, target in train_loader: optimizer.zero_grad() output = model(data) loss = nn.CrossEntropyLoss()(output, target) loss.backward() optimizer.step() epsilon = privacy_engine.get_epsilon(delta=delta) return model, epsilon

🌐 联邦学习

# 联邦学习实现 import torch import torch.nn as nn import torch.optim as optim class FederatedClient: def __init__(self, model, train_data): self.model = model self.train_data = train_data self.optimizer = optim.SGD(self.model.parameters(), lr=0.01) def train(self, epochs=5): self.model.train() for epoch in range(epochs): for data, target in self.train_data: self.optimizer.zero_grad() output = self.model(data) loss = nn.CrossEntropyLoss()(output, target) loss.backward() self.optimizer.step() return self.model.state_dict() class FederatedServer: def __init__(self, model): self.model = model self.client_models = [] def aggregate(self, client_updates): """ FedAvg聚合 """ aggregated_state = {} # 获取模型参数 state_dicts = [update.state_dict() for update in client_updates] # 计算平均 for key in state_dicts[0].keys(): aggregated_state[key] = torch.stack( [state[key] for state in state_dicts] ).mean(0) # 更新服务器模型 self.model.load_state_dict(aggregated_state) return self.model def evaluate(self, test_data): self.model.eval() correct = 0 total = 0 with torch.no_grad(): for data, target in test_data: output = self.model(data) pred = output.argmax(dim=1) correct += (pred == target).sum().item() total += target.size(0) accuracy = correct / total return accuracy # 联邦学习流程 def federated_learning(clients, server, rounds=10): for round in range(rounds): print(f"Round {round + 1}") # 本地训练 client_updates = [] for client in clients: client_model = type(server.model)(**client.model.init_params) client_model.load_state_dict(server.model.state_dict()) client.model = client_model client_updates.append(client.train()) # 服务端聚合 server.aggregate(client_updates)

🔍 数据脱敏

# 数据脱敏技术 ## 1. 基本脱敏 import re import hashlib def mask_email(email): """邮箱脱敏""" name, domain = email.split('@') masked_name = name[0] + '*' * (len(name) - 2) + name[-1] return f"{masked_name}@{domain}" def mask_phone(phone): """电话脱敏""" return phone[:3] + '****' + phone[-4:] def mask_id_number(id_number): """身份证号脱敏""" return id_number[:6] + '********' + id_number[-4:] def hash_data(data): """数据哈希""" return hashlib.sha256(data.encode()).hexdigest() ## 2. 数据匿名化 import pandas as pd def anonymize_dataset(df): """ 数据集匿名化 """ # k-匿名化 k = 5 # 泛化数值型数据 if 'age' in df.columns: df['age_group'] = pd.cut(df['age'], bins=[0, 18, 30, 45, 60, 100], labels=['0-18', '19-30', '31-45', '46-60', '60+']) df = df.drop('age', axis=1) # 泛化地理位置 if 'location' in df.columns: df['region'] = df['location'].apply(lambda x: x.split()[0]) df = df.drop('location', axis=1) # 删除直接标识符 identifiers = ['name', 'phone', 'email', 'id_number'] df = df.drop(columns=[col for col in identifiers if col in df.columns]) return df ## 3. 数据交换 from sklearn.model_selection import train_test_split def data_swapping(df, sensitive_col='salary', n_swaps=100): """ 数据交换保护数值型敏感数据 """ df_swapped = df.copy() indices = np.random.choice(len(df), n_swaps * 2, replace=False) for i in range(0, len(indices), 2): idx1, idx2 = indices[i], indices[i + 1] df_swapped.loc[idx1, sensitive_col], df_swapped.loc[idx2, sensitive_col] = \ df_swapped.loc[idx2, sensitive_col], df_swapped.loc[idx1, sensitive_col] return df_swapped

# 数据访问控制实现 from functools import wraps from flask import Flask, request, jsonify app = Flask(__name__) # 角色定义 ROLES = { 'admin': ['read', 'write', 'delete'], 'user': ['read', 'write'], 'viewer': ['read'] } def require_role(role): """ 角色检查装饰器 """ def decorator(f): @wraps(f) def decorated_function(*args, **kwargs): token = request.headers.get('Authorization') # 验证token if not token: return jsonify({'error': 'Unauthorized'}), 401 # 获取用户角色 user_role = get_user_role(token) # 检查角色权限 if user_role not in ROLES: return jsonify({'error': 'Invalid role'}), 403 return f(*args, **kwargs) return decorated_function return decorator def require_permission(permission): """ 权限检查装饰器 """ def decorator(f): @wraps(f) def decorated_function(*args, **kwargs): token = request.headers.get('Authorization') user_role = get_user_role(token) # 检查权限 if permission not in ROLES.get(user_role, []): return jsonify({'error': 'Permission denied'}), 403 return f(*args, **kwargs) return decorated_function return decorator @app.route('/data', methods=['GET']) @require_role('user') def get_data(): """获取数据""" return jsonify({'data': 'sensitive data'}) @app.route('/data', methods=['POST']) @require_permission('write') def update_data(): """更新数据""" return jsonify({'status': 'updated'}) @app.route('/data/', methods=['DELETE']) @require_permission('delete') def delete_data(id): """删除数据""" return jsonify({'status': 'deleted'})

# 数据审计日志 import logging from datetime import datetime class DataAuditLogger: def __init__(self, log_file='data_audit.log'): self.logger = logging.getLogger('DataAudit') self.logger.setLevel(logging.INFO) handler = logging.FileHandler(log_file) formatter = logging.Formatter( '%(asctime)s - %(levelname)s - %(message)s' ) handler.setFormatter(formatter) self.logger.addHandler(handler) def log_access(self, user_id, action, data_id, details=None): """ 记录数据访问 """ log_entry = { 'user_id': user_id, 'action': action, 'data_id': data_id, 'timestamp': datetime.now().isoformat(), 'details': details } self.logger.info(log_entry) def log_modification(self, user_id, data_id, changes): """ 记录数据修改 """ log_entry = { 'user_id': user_id, 'action': 'MODIFY', 'data_id': data_id, 'changes': changes, 'timestamp': datetime.now().isoformat() } self.logger.info(log_entry) def log_deletion(self, user_id, data_id): """ 记录数据删除 """ log_entry = { 'user_id': user_id, 'action': 'DELETE', 'data_id': data_id, 'timestamp': datetime.now().isoformat() } self.logger.info(log_entry) # 使用示例 audit_logger = DataAuditLogger() # 记录访问 audit_logger.log_access('user123', 'READ', 'data_456') # 记录修改 audit_logger.log_modification( 'user123', 'data_456', {'field': 'value', 'old': 'old_value', 'new': 'new_value'} ) # 记录删除 audit_logger.log_deletion('admin', 'data_456')
