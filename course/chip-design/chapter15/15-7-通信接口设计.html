📡 UART发送器

// UART发送器（8N1格式）
module uart_tx(
    input  clk,
    input  rst_n,
    input        tx_start,
    input  [7:0] tx_data,
    output       tx,
    output       tx_busy
);
    parameter CLK_FREQ = 50_000_000;
    parameter BAUD_RATE = 115200;
    parameter BAUD_DIV = CLK_FREQ / BAUD_RATE;

    reg [15:0] baud_counter;
    reg [3:0]  bit_counter;
    reg [9:0]  shift_reg;
    reg        busy;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            baud_counter <= 0;
            bit_counter <= 0;
            shift_reg <= 0;
            busy <= 0;
            tx <= 1;
        end
        else if (tx_start && !busy) begin
            shift_reg <= {1'b1, tx_data, 1'b0};
            bit_counter <= 10;
            baud_counter <= 0;
            busy <= 1;
        end
        else if (busy) begin
            if (baud_counter == BAUD_DIV - 1) begin
                baud_counter <= 0;
                shift_reg <= {1'b1, shift_reg[9:1]};
                tx <= shift_reg[0];
                if (bit_counter > 0)
                    bit_counter <= bit_counter - 1;
                else
                    busy <= 0;
            end
            else
                baud_counter <= baud_counter + 1;
        end
    end

    assign tx_busy = busy;
endmodule
