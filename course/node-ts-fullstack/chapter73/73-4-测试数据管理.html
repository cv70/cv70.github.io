73.4 测试数据管理

第十部分：测试篇 - 高效管理测试数据

1. 测试数据的重要性

良好的测试数据管理可以提高测试的可读性、可维护性和可靠性。

常见问题

  • 测试数据散落各处
  • 硬编码的魔法值
  • 重复创建相同数据
  • 数据难以理解和维护

解决思路

  • 集中管理测试数据
  • 使用工厂模式创建
  • 使用 Fixture 文件
  • 参数化生成数据

2. 测试数据工厂

// tests/factories/user.factory.ts import { User } from '@/entities/user.entity'; type UserOverrides = Partial<User>; export function createUser(overrides: UserOverrides = {}): User { return { id: generateId(), name: 'Test User', email: `test${Date.now()}@example.com`, password: 'hashedPassword', isActive: true, createdAt: new Date(), updatedAt: new Date(), ...overrides }; } export function createAdminUser(overrides: UserOverrides = {}): User { return createUser({ role: 'admin', ...overrides }); } // tests/factories/post.factory.ts export function createPost(overrides: Partial<Post> = {}): Post { return { id: generateId(), title: 'Test Post', content: 'Test content', authorId: '1', status: 'published', createdAt: new Date(), ...overrides }; }

使用工厂

import { createUser, createAdminUser } from '../factories/user.factory'; describe('UserService', () => { it('should activate user', () => { const inactiveUser = createUser({ isActive: false }); const result = service.activate(inactiveUser); expect(result.isActive).toBe(true); }); it('should allow admin to delete', () => { const admin = createAdminUser(); expect(service.canDelete(admin)).toBe(true); }); });

4. 数据库数据管理

// tests/helpers/database-helpers.ts import { Connection } from 'typeorm'; import { User } from '@/entities/user.entity'; export class DatabaseHelper { constructor(private connection: Connection) {} async createUser(overrides: Partial<User> = {}): Promise<User> { const repo = this.connection.getRepository(User); const user = repo.create({ name: 'Test User', email: `test${Date.now()}@example.com`, ...overrides }); return repo.save(user); } async createUsers(count: number): Promise<User[]> { const users: User[] = []; for (let i = 0; i < count; i++) { users.push(await this.createUser({ name: `User ${i}` })); } return users; } async clearUsers(): Promise<void> { await this.connection.getRepository(User).clear(); } async clearAll(): Promise<void> { const entities = this.connection.entityMetadatas; for (const entity of entities) { await this.connection.getRepository(entity.name).clear(); } } }

使用数据库辅助函数

describe('UserRepository', () => { let dbHelper: DatabaseHelper; beforeEach(async () => { const connection = await createTestConnection(); dbHelper = new DatabaseHelper(connection); }); afterEach(async () => { await dbHelper.clearAll(); }); it('should find user by email', async () => { const user = await dbHelper.createUser({ email: 'findme@example.com' }); const found = await repository.findByEmail('findme@example.com'); expect(found.id).toBe(user.id); }); });

5. 使用 Faker 生成数据

import { faker } from '@faker-js/faker'; describe('With Faker', () => { it('should create user with realistic data', async () => { const userData = { name: faker.person.fullName(), email: faker.internet.email(), phone: faker.phone.number(), address: { street: faker.location.streetAddress(), city: faker.location.city(), country: faker.location.country() } }; const user = await service.create(userData); expect(user.email).toMatch(/@/); }); it('should handle various names', () => { const names = Array.from({ length: 10 }, () => faker.person.fullName()); names.forEach(name => { expect(validateName(name)).toBe(true); }); }); });

6. 测试数据最佳实践

// ✅ 好的做法：使用有意义的变量名 it('should calculate discount for premium user', () => { const premiumUser = createUser({ level: 'premium' }); const regularUser = createUser({ level: 'regular' }); expect(calculateDiscount(premiumUser)).toBe(0.2); expect(calculateDiscount(regularUser)).toBe(0.1); }); // ❌ 不好的做法：使用魔法值 it('should calculate discount', () => { const u1 = { id: '1', level: 'premium' }; const u2 = { id: '2', level: 'regular' }; expect(calculateDiscount(u1)).toBe(0.2); expect(calculateDiscount(u2)).toBe(0.1); }); // ✅ 好的做法：只在必要时覆盖 it('should validate email format', () => { const user = createUser({ email: 'invalid-email' }); expect(validateEmail(user)).toBe(false); }); // ❌ 不好的做法：重复定义 it('should validate email', () => { const user = { id: '1', name: 'Test User', email: 'invalid-email', // ... 重复所有字段 }; });
