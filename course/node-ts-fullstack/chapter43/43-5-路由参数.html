43.5 路由参数

深入理解各种参数装饰器的使用

🎯 学习目标

  • 掌握路径参数的使用
  • 学会查询参数处理
  • 了解参数验证和转换
  • 掌握参数装饰器自定义

📚 路径参数 @Param

import { Controller, Get, Param } from '@nestjs/common'; @Controller('users') export class UsersController { // 单个参数 @Get(':id') findOne(@Param('id') id: string) { return `User ID: ${id}`; } // 多个参数 @Get(':userId/posts/:postId') findPost( @Param('userId') userId: string, @Param('postId') postId: string, ) { return `User: ${userId}, Post: ${postId}`; } // 获取所有参数 @Get('category/:category/item/:item') findByCategory(@Param() params: { category: string; item: string }) { return params; } // 使用管道转换类型 @Get(':id/details') findDetails(@Param('id', ParseIntPipe) id: number) { console.log(typeof id); // number return `User ID: ${id}`; } }

🔧 查询参数 @Query

import { Controller, Get, Query, DefaultValuePipe } from '@nestjs/common'; @Controller('products') export class ProductsController { // 单个查询参数 @Get() findAll(@Query('keyword') keyword: string) { return { keyword }; } // 默认值 @Get('list') list( @Query('page', new DefaultValuePipe(1)) page: number, @Query('limit', new DefaultValuePipe(10)) limit: number, ) { return { page, limit }; } // 多个查询参数 @Get('search') search( @Query('keyword') keyword: string, @Query('category') category: string, @Query('sort') sort: 'asc' | 'desc' = 'asc', ) { return { keyword, category, sort }; } // 使用 DTO @Get('filter') filter(@Query() query: FilterDto) { return query; } } // filter.dto.ts import { IsOptional, IsString, IsNumber, Min } from 'class-validator'; class FilterDto { @IsOptional() @IsString() keyword?: string; @IsOptional() @IsNumber() @Min(1) page?: number = 1; @IsOptional() @IsNumber() @Min(1) limit?: number = 10; }
💡
查询参数类型

查询参数默认都是字符串类型。可以使用 DefaultValuePipe 设置默认值，使用 ParseIntPipe 转换为数字。

🌐 参数管道

import { Controller, Get, Param, Query, ParseIntPipe, ParseBoolPipe, ParseUUIDPipe, ParseArrayPipe } from '@nestjs/common'; @Controller('examples') export class ExamplesController { // ParseIntPipe: 转换为整数，失败则抛出 400 @Get('int/:num') parseNum(@Param('num', ParseIntPipe) num: number) { return { value: num, type: typeof num }; } // ParseBoolPipe: 转换为布尔值 @Get('bool') parseBool(@Query('active', ParseBoolPipe) active: boolean) { return { active }; } // ParseUUIDPipe: 验证 UUID @Get('uuid/:id') parseUUID(@Param('id', ParseUUIDPipe) id: string) { return { id }; } // ParseArrayPipe: 解析数组 @Get('tags') parseArray( @Query('tags', new ParseArrayPipe({ items: String, separator: ',' })) tags: string[], ) { return { tags }; } // GET /tags?tags=a,b,c => { tags: ['a', 'b', 'c'] } }

✨ 自定义参数装饰器

import { createParamDecorator, ExecutionContext } from '@nestjs/common'; // 获取当前用户 export const User = createParamDecorator( (data: string, ctx: ExecutionContext) => { const request = ctx.switchToHttp().getRequest(); const user = request.user; return data ? user?.[data] : user; }, ); // 使用自定义装饰器 @Controller('profile') export class ProfileController { @Get() getProfile(@User() user: UserEntity) { return user; } @Get('name') getUserName(@User('name') name: string) { return { name }; } @Get('email') getUserEmail(@User('email') email: string) { return { email }; } }
⚠️
自定义装饰器

自定义参数装饰器可以封装常用的参数获取逻辑，使代码更加简洁。常用于获取当前用户、提取特定请求头等场景。
