📚 RBAC模型

// RBAC (Role-Based Access Control) // 用户-角色-权限关系 /* User ←→ Role ←→ Permission 用户可以有多个角色 角色可以有多个权限 权限控制具体操作 */ // 权限类型 enum PermissionAction { CREATE = 'create', READ = 'read', UPDATE = 'update', DELETE = 'delete', PUBLISH = 'publish', MANAGE = 'manage' // 管理权限 } // 权限资源 interface Permission { resource: string; // 资源类型：article, user, media action: PermissionAction; conditions?: { own?: boolean; // 仅自己的内容 roles?: string[]; // 特定角色 }; } // 示例权限配置 const roles = { admin: { name: '管理员', permissions: [ { resource: '*', action: PermissionAction.MANAGE } ] }, editor: { name: '编辑', permissions: [ { resource: 'article', action: PermissionAction.CREATE }, { resource: 'article', action: PermissionAction.READ }, { resource: 'article', action: PermissionAction.UPDATE, conditions: { own: true } }, { resource: 'article', action: PermissionAction.DELETE, conditions: { own: true } }, { resource: 'media', action: PermissionAction.CREATE }, { resource: 'media', action: PermissionAction.READ }, ] }, author: { name: '作者', permissions: [ { resource: 'article', action: PermissionAction.CREATE }, { resource: 'article', action: PermissionAction.READ }, { resource: 'article', action: PermissionAction.UPDATE, conditions: { own: true } }, ] }, viewer: { name: '访客', permissions: [ { resource: 'article', action: PermissionAction.READ }, ] } };
💡
RBAC优势

简化权限管理、角色继承、易于维护。

🔧 权限实体与中间件

// role.entity.ts @Entity('roles') export class Role { @PrimaryGeneratedColumn('uuid') id: string; @Column({ unique: true }) name: string; @Column({ nullable: true }) description: string; @Column({ type: 'json' }) permissions: Permission[]; @ManyToMany(() => User, user => user.roles) users: User[]; } // user.entity.ts 添加角色关联 @Entity('users') export class User { // ...其他字段 @ManyToMany(() => Role, role => role.users) @JoinTable() roles: Role[]; } // permission.guard.ts import { Injectable, CanActivate, ExecutionContext, ForbiddenException } from '@nestjs/common'; @Injectable() export class PermissionGuard implements CanActivate { constructor(private permissionService: PermissionService) {} async canActivate(context: ExecutionContext): Promise { const request = context.switchToHttp().getRequest(); const user = request.user; if (!user) { throw new ForbiddenException('未登录'); } // 获取需要的权限 const requiredPermission = this.reflector.get( 'permission', context.getHandler() ); if (!requiredPermission) { return true; // 无权限要求 } // 检查权限 const hasPermission = await this.permissionService.checkPermission( user.id, requiredPermission, request.params // 传入请求参数用于own检查 ); if (!hasPermission) { throw new ForbiddenException('无权限'); } return true; } } // permission.service.ts @Injectable() export class PermissionService { constructor( @InjectRepository(User) private userRepo: Repository, ) {} async checkPermission( userId: string, required: Permission, params?: any ): Promise { // 获取用户角色 const user = await this.userRepo.findOne({ where: { id: userId }, relations: ['roles'], }); if (!user || !user.roles) { return false; } // 遍历角色权限 for (const role of user.roles) { for (const permission of role.permissions) { // 检查资源匹配 if (permission.resource !== '*' && permission.resource !== required.resource) { continue; } // 检查操作匹配 if (permission.action !== PermissionAction.MANAGE && permission.action !== required.action) { continue; } // 检查条件 if (permission.conditions?.own && required.conditions?.own) { // 需要检查是否是自己的内容 const contentId = params?.id; const content = await this.getContent(contentId); if (content.createdBy !== userId) { continue; // 不是自己的内容，继续检查下一个权限 } } return true; // 找到匹配的权限 } } return false; } } // 使用权限守卫 @Controller('articles') @UseGuards(JwtAuthGuard, PermissionGuard) export class ArticleController { @Post() @RequirePermission({ resource: 'article', action: PermissionAction.CREATE }) async create() {} @Put(':id') @RequirePermission({ resource: 'article', action: PermissionAction.UPDATE, conditions: { own: true } }) async update() {} @Delete(':id') @RequirePermission({ resource: 'article', action: PermissionAction.DELETE }) async remove() {} }
