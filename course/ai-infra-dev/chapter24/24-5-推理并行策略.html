🔢 张量并行

Tensor Parallelism

张量并行将每一层的张量切分到多个GPU上，每个GPU计算张量的一部分。

列并行

  • 权重按列切分
  • 输入复制到所有GPU
  • 每个GPU计算部分输出
  • 需要AllReduce聚合

行并行

  • 权重按行切分
  • 输入按维度切分
  • 每个GPU计算独立输出
  • 不需要通信
特点
  • 通信：需要频繁通信
  • 显存：单GPU显存需求减少
  • 延迟：低延迟推理最佳选择
  • 适用：单一大模型推理

📊 流水线并行

Pipeline Parallelism

流水线并行将模型的不同层分布到不同的GPU上，形成计算流水线。

GPU 0: 层1-10
→
GPU 1: 层11-20
→
GPU 2: 层21-30
→
GPU 3: 层31-40
特点
  • 通信：GPU间通信较少
  • 显存：单GPU显存需求大幅减少
  • 延迟：流水线气泡导致延迟增加
  • 适用：大模型多GPU推理
图：多GPU并行推理架构示意图

📦 数据并行

Data Parallelism

数据并行将模型复制到每个GPU，每个GPU处理不同的输入batch。

特点
  • 通信：需要AllReduce同步梯度（训练）
  • 显存：每个GPU需要完整模型
  • 延迟：不适合低延迟推理
  • 适用：高吞吐量推理

💻 推理并行实现

import torch
import torch.distributed as dist
from transformers import AutoModelForCausalLM, AutoTokenizer

# 张量并行实现（简化版）
class TensorParallelModel:
    """
    张量并行模型
    """
    def __init__(self, model, world_size, rank):
        self.model = model
        self.world_size = world_size
        self.rank = rank

        # 切分线性层权重
        self._parallelize_linear_layers()

    def _parallelize_linear_layers(self):
        """
        并行化线性层
        """
        for name, module in self.model.named_modules():
            if isinstance(module, torch.nn.Linear):
                weight = module.weight.data

                # 列并行：按列切分
                weight_chunk = torch.chunk(weight, self.world_size, dim=1)[self.rank]

                # 更新权重
                module.weight.data = weight_chunk

                # 调整输出维度
                module.out_features = weight_chunk.shape[0]

    def forward(self, x):
        """
        前向传播
        """
        # 前向传播
        output = self.model(x)

        # 聚合输出（AllReduce）
        if self.world_size > 1:
            dist.all_reduce(output, op=dist.ReduceOp.SUM)

        return output

# 流水线并行实现（简化版）
class PipelineParallelModel:
    """
    流水线并行模型
    """
    def __init__(self, model, stages, rank):
        self.stages = stages
        self.rank = rank

        # 分配层到当前GPU
        self._assign_layers_to_gpu(model)

    def _assign_layers_to_gpu(self, model):
        """
        将层分配到GPU
        """
        layers_per_stage = len(model.layers) // self.stages

        start_idx = self.rank * layers_per_stage
        end_idx = start_idx + layers_per_stage

        # 只保留当前阶段的层
        self.model_layers = model.layers[start_idx:end_idx]

    def forward(self, hidden_states):
        """
        前向传播
        """
        # 通过当前阶段的层
        for layer in self.model_layers:
            hidden_states = layer(hidden_states)

        # 如果不是最后一个阶段，发送给下一个GPU
        if self.rank < self.stages - 1:
            dist.send(hidden_states, dst=self.rank + 1)
            return None

        return hidden_states

# 使用vLLM的张量并行
def vllm_tensor_parallel():
    """
    vLLM中的张量并行
    """
    from vllm import LLM, SamplingParams

    # 初始化LLM（4个GPU）
    llm = LLM(
        model="meta-llama/Llama-2-70b-hf",
        tensor_parallel_size=4,  # 使用4个GPU进行张量并行
        trust_remote_code=True
    )

    sampling_params = SamplingParams(
        temperature=0.8,
        max_tokens=100
    )

    prompts = ["写一个Python函数"]

    outputs = llm.generate(prompts, sampling_params)

    for output in outputs:
        print(f"Output: {output.outputs[0].text}")

# 使用DeepSpeed的流水线并行
def deepspeed_pipeline_parallel():
    """
    DeepSpeed中的流水线并行
    """
    import deepspeed

    # 加载模型
    model = AutoModelForCausalLM.from_pretrained("meta-llama/Llama-2-70b-hf")

    # 配置流水线并行
    config = {
        "train_batch_size": 1,
        "gradient_accumulation_steps": 1,
        "fp16": {
            "enabled": True
        },
        "tensor_parallel": {
            "enabled": False
        },
        "pipeline": {
            "enabled": True,
            "parallel_size": 4,  # 4个GPU流水线并行
            "schedule": "interleaved"
        }
    }

    # 初始化DeepSpeed
    model_engine, _, _, _ = deepspeed.initialize(
        model=model,
        config=config
    )

    # 推理
    inputs = torch.tensor([1, 2, 3]).cuda()
    outputs = model_engine(inputs)

    return outputs

# 使用Megatron-LM的张量并行
def megatron_tensor_parallel():
    """
    Megatron-LM中的张量并行
    """
    from megatron import initialize_megatron
    from megatron.model import GPTModel
    from megatron.utils import get_args

    # 初始化Megatron
    args = get_args()
    initialize_megatron(args)

    # 创建模型（自动张量并行）
    model = GPTModel(
        num_layers=32,
        hidden_size=4096,
        num_attention_heads=32,
        tensor_model_parallel_size=args.tensor_model_parallel_size
    )

    # 推理
    inputs = torch.tensor([1, 2, 3]).cuda()
    outputs = model(inputs)

    return outputs

⚖️ 并行策略对比

策略 GPU数量 显存需求 延迟 吞吐量 适用场景
张量并行 2-8 模型/N 低 高 低延迟推理
流水线并行 2-16 模型/N 中高 中高 大模型推理
张量+流水线 4-64 模型/(M×N) 中 最高 超大模型
数据并行 任意 完整模型 高 最高 高吞吐量
💡
推理并行实践建议
  • 7B以下模型通常单GPU即可
  • 7B-70B模型推荐张量并行（2-8个GPU）
  • 70B以上模型需要结合张量和流水线并行
  • 使用NVLink加速GPU间通信
