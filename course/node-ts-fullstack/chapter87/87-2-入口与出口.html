87.2 入口与出口

Webpack 入口出口配置详解

🎯 学习目标

  • 理解入口和出口的作用
  • 掌握单入口和多入口配置
  • 学会输出文件命名技巧
  • 了解公共路径配置

📚 入口配置

// 单入口配置 module.exports = { entry: './src/index.js', }; // 单入口对象形式 module.exports = { entry: { main: './src/index.js', }, }; // 多入口配置 module.exports = { entry: { home: './src/pages/home.js', about: './src/pages/about.js', contact: './src/pages/contact.js', }, }; // 数组形式入口（合并多个文件） module.exports = { entry: { main: ['./src/polyfills.js', './src/index.js'], }, }; // 动态入口 module.exports = { entry: () => './src/index.js', }; // Promise 形式 module.exports = { entry: () => new Promise((resolve) => resolve('./src/index.js')), };
💡
多入口场景

多入口常用于多页面应用（MPA），每个页面有独立的入口文件，生成独立的打包文件。

🔧 出口配置

const path = require('path'); module.exports = { output: { // 输出目录（绝对路径） path: path.resolve(__dirname, 'dist'), // 输出文件名 filename: 'bundle.js', // 清理输出目录 clean: true, }, }; // 多入口时的文件名 module.exports = { entry: { home: './src/pages/home.js', about: './src/pages/about.js', }, output: { path: path.resolve(__dirname, 'dist'), // 使用 [name] 占位符 filename: '[name].bundle.js', // 输出: home.bundle.js, about.bundle.js }, }; // 文件名占位符 // [name] - 入口名称 // [id] - 入口 ID // [hash] - 构建哈希 // [chunkhash] - chunk 哈希 // [contenthash] - 内容哈希（推荐用于缓存） module.exports = { output: { filename: '[name].[contenthash:8].js', // 输出: main.abc12345.js }, };

⚙️ 高级输出配置

const path = require('path'); module.exports = { output: { // 输出目录 path: path.resolve(__dirname, 'dist'), // 文件名 filename: '[name].[contenthash:8].js', // chunk 文件名（动态导入的文件） chunkFilename: '[name].[contenthash:8].chunk.js', // 资源文件名（图片、字体等） assetModuleFilename: 'assets/[hash][ext][query]', // 公共路径（CDN 或子目录） publicPath: 'https://cdn.example.com/', // 或 publicPath: '/app/', // 库配置（打包成库时使用） library: { name: 'MyLibrary', type: 'umd', // 'var' | 'module' | 'umd' | 'commonjs2' }, // 清理配置 clean: { keep: /important/, // 保留匹配的文件 }, // source map 文件名 sourceMapFilename: '[file].map', }, }; // 不同类型资源的输出 module.exports = { output: { path: path.resolve(__dirname, 'dist'), filename: 'js/[name].[contenthash:8].js', }, module: { rules: [ { test: /\.(png|jpg|gif)$/, type: 'asset/resource', generator: { filename: 'images/[hash][ext][query]', }, }, { test: /\.(woff|woff2|eot|ttf)$/, type: 'asset/resource', generator: { filename: 'fonts/[hash][ext][query]', }, }, ], }, };

🚀 实战配置示例

// 完整的多页面应用配置 const path = require('path'); const HtmlWebpackPlugin = require('html-webpack-plugin'); module.exports = { entry: { home: './src/pages/home/index.js', about: './src/pages/about/index.js', admin: './src/pages/admin/index.js', }, output: { path: path.resolve(__dirname, 'dist'), filename: 'js/[name].[contenthash:8].js', chunkFilename: 'js/[name].[contenthash:8].chunk.js', assetModuleFilename: 'assets/[hash][ext][query]', publicPath: '/', clean: true, }, plugins: [ new HtmlWebpackPlugin({ template: './src/pages/home/index.html', filename: 'home.html', chunks: ['home'], }), new HtmlWebpackPlugin({ template: './src/pages/about/index.html', filename: 'about.html', chunks: ['about'], }), new HtmlWebpackPlugin({ template: './src/pages/admin/index.html', filename: 'admin.html', chunks: ['admin'], }), ], };
