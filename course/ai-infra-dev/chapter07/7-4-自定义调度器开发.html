7.4 自定义调度器开发

构建专属的AI作业调度系统

🎯 学习目标

  • 理解自定义调度器的应用场景
  • 掌握调度器的开发框架
  • 学会实现Gang调度等AI特殊需求
  • 了解调度器的部署和测试方法

为什么需要自定义调度器

AI工作负载具有独特的调度需求：Gang调度（所有Pod同时启动）、 拓扑感知（GPU互联）、公平调度（多租户）、资源切分（GPU共享）等。 通用调度器无法满足这些需求，需要开发专用调度器。

⚙️ 核心组件实现

调度器核心代码

调度器主程序：

package main

import (
    "context"
    "fmt"
    "sync"
    "time"

    "k8s.io/client-go/informers"
    "k8s.io/client-go/kubernetes"
    "k8s.io/client-go/tools/cache"
    "k8s.io/client-go/tools/clientcmd"
    "k8s.io/klog/v2"
)

// AIClusterScheduler AI集群调度器
type AIClusterScheduler struct {
    kubeClient    kubernetes.Interface
    informerFactory informers.SharedInformerFactory
    schedulerQueue chan *Job
    jobManager    *JobManager
    resourceMgr   *ResourceManager
    policyEngine  *PolicyEngine
    stopCh        chan struct{}
}

// NewScheduler 创建调度器
func NewScheduler(config *rest.Config) (*AIClusterScheduler, error) {
    kubeClient, err := kubernetes.NewForConfig(config)
    if err != nil {
        return nil, err
    }

    factory := informers.NewSharedInformerFactory(kubeClient, 0)

    return &AIClusterScheduler{
        kubeClient:       kubeClient,
        informerFactory:  factory,
        schedulerQueue:  make(chan *Job, 1000),
        stopCh:          make(chan struct{}),
    }, nil
}

// Run 启动调度器
func (s *AIClusterScheduler) Run(workers int) error {
    // 启动Informer
    s.informerFactory.Start(s.stopCh)

    // 启动调度器
    for i := 0; i < workers; i++ {
        go s.schedulerWorker()
    }

    // 启动资源监控
    go s.monitorResources()

    // 启动策略引擎
    go s.runPolicyEngine()

    <-s.stopCh
    return nil
}

func main() {
    kubeconfig := os.Getenv("KUBECONFIG")
    config, err := clientcmd.BuildConfigFromFlags("", kubeconfig)
    if err != nil {
        klog.Fatalf("Error building kubeconfig: %v", err)
    }

    scheduler, err := NewScheduler(config)
    if err != nil {
        klog.Fatalf("Error creating scheduler: %v", err)
    }

    if err := scheduler.Run(4); err != nil {
        klog.Fatalf("Error running scheduler: %v", err)
    }
}

🎯 Gang调度实现

组调度插件

Gang调度策略：

package gang

import (
    "context"
    "fmt"

    "k8s.io/api/core/v1"
    "k8s.io/kubernetes/pkg/scheduler/framework"
)

// GangScheduler Gang调度器
type GangScheduler struct {
    handle framework.Handle
    gangMap map[string]*GangGroup
    mutex  sync.RWMutex
}

// GangGroup Pod组
type GangGroup struct {
    GroupName string
    TotalPods int
    ReadyPods int
    PodList   []*v1.Pod
    Phase     GroupPhase
}

// GroupPhase 组状态
type GroupPhase int

const (
    PhasePending GroupPhase = iota
    PhaseReady
    PhaseRunning
    PhaseCompleted
    PhaseFailed
)

// NewGangScheduler 创建Gang调度器
func NewGangScheduler(handle framework.Handle) *GangScheduler {
    return &GangScheduler{
        handle:  handle,
        gangMap: make(map[string]*GangGroup),
    }
}

// Filter 过滤阶段
func (gs *GangScheduler) Filter(
    ctx context.Context,
    state *framework.CycleState,
    pod *v1.Pod,
    nodeInfo *framework.NodeInfo,
) *framework.Status {
    gangName := getGangName(pod)

    gs.mutex.RLock()
    group, exists := gs.gangMap[gangName]
    gs.mutex.RUnlock()

    if !exists {
        // 新组，允许调度
        return nil
    }

    // 检查组状态
    if group.Phase == PhaseReady || group.Phase == PhaseRunning {
        // 组已就绪，正常调度
        return nil
    }

    // 检查资源是否足够
    if !hasEnoughResources(group, nodeInfo) {
        return framework.NewStatus(
            framework.Unschedulable,
            fmt.Sprintf("Not enough resources for gang %s", gangName),
        )
    }

    return nil
}

// Score 打分阶段
func (gs *GangScheduler) Score(
    ctx context.Context,
    state *framework.CycleState,
    pod *v1.Pod,
    nodeName string,
) (int64, *framework.Status) {
    gangName := getGangName(pod)

    gs.mutex.RLock()
    group, exists := gs.gangMap[gangName]
    gs.mutex.RUnlock()

    if !exists {
        return 50, nil
    }

    // 优先选择已有同组Pod的节点
    nodeInfo, err := gs.handle.SnapshotSharedLister().NodeInfos().Get(nodeName)
    if err != nil {
        return 0, framework.NewStatus(framework.Error, err.Error())
    }

    score := int64(0)
    for _, p := range nodeInfo.Pods {
        if getGangName(p.Pod) == gangName {
            score += 10
        }
    }

    return score, nil
}

// Bind 绑定阶段
func (gs *GangScheduler) Bind(
    ctx context.Context,
    state *framework.CycleState,
    pod *v1.Pod,
    nodeName string,
) *framework.Status {
    gangName := getGangName(pod)

    gs.mutex.Lock()
    defer gs.mutex.Unlock()

    group := gs.gangMap[gangName]
    group.ReadyPods++

    // 检查是否所有Pod都就绪
    if group.ReadyPods == group.TotalPods {
        group.Phase = PhaseReady
        // 触发组启动
        gs.startGang(group)
    }

    return nil
}

// startGang 启动组
func (gs *GangScheduler) startGang(group *GangGroup) {
    group.Phase = PhaseRunning
    klog.Infof("Gang %s started with %d pods", group.GroupName, group.TotalPods)
}

🌐 拓扑感知调度

GPU拓扑优化

拓扑感知插件：

package topology

import (
    "context"
    "math"
)

// TopologyAware 拓扑感知调度
type TopologyAware struct {
    handle framework.Handle
    topologyMap map[string]*TopologyInfo
}

// TopologyInfo 拓扑信息
type TopologyInfo struct {
    Zone      string
    Rack      string
    Host      string
    GPUCount  int
    NVLink    bool
}

// Score 拓扑打分
func (ta *TopologyAware) Score(
    ctx context.Context,
    state *framework.CycleState,
    pod *v1.Pod,
    nodeName string,
) (int64, *framework.Status) {
    gpuCount := getGPURequest(pod)

    // 单GPU，不考虑拓扑
    if gpuCount <= 1 {
        return 50, nil
    }

    // 多GPU，需要考虑拓扑
    nodeInfo, err := ta.handle.SnapshotSharedLister().NodeInfos().Get(nodeName)
    if err != nil {
        return 0, framework.NewStatus(framework.Error, err.Error())
    }

    score := int64(0)

    // NVLink加分
    if hasNVLink(nodeInfo) {
        score += 30
    }

    // 同机架加分
    rackNodes := countNodesInRack(nodeInfo, ta.topologyMap)
    if rackNodes >= gpuCount {
        score += 20
    }

    // 同可用区分
    zoneNodes := countNodesInZone(nodeInfo, ta.topologyMap)
    if zoneNodes >= gpuCount {
        score += 10
    }

    return score, nil
}
