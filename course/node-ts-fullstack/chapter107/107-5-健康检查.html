📚 健康检查概述

// 健康检查用于判断服务是否正常运行 // 为什么需要健康检查 const reasons = { faultIsolation: '故障隔离，自动剔除不健康实例', loadBalancing: '负载均衡，只分发到健康实例', scaling: '弹性伸缩，感知实例上下线', monitoring: '监控告警，及时发现服务异常' }; // 健康检查类型 const checkTypes = { active: { description: '主动探测', method: '注册中心定期探测服务', examples: ['HTTP探针', 'TCP探针', '自定义脚本'] }, passive: { description: '被动上报', method: '服务主动上报心跳', examples: ['心跳包', '健康状态上报'] } }; // 健康状态定义 interface HealthStatus { status: 'healthy' | 'unhealthy' | 'degraded'; checks: { database: CheckResult; cache: CheckResult; external: CheckResult; }; timestamp: Date; } interface CheckResult { status: 'up' | 'down'; latency?: number; message?: string; }
💡
健康检查目标

及时发现故障实例、自动摘除不健康服务、保障系统可用性。

🔧 实现健康检查端点

import express from 'express'; import { HealthCheck, HealthCheckResult } from './health-check'; const app = express(); // ==================== 简单健康检查 ==================== // Kubernetes liveness probe app.get('/health/live', (req, res) => { // 只检查进程是否存活 res.status(200).json({ status: 'ok' }); }); // Kubernetes readiness probe app.get('/health/ready', async (req, res) => { // 检查服务是否准备好接收请求 const isReady = await checkDependencies(); if (isReady) { res.status(200).json({ status: 'ready' }); } else { res.status(503).json({ status: 'not ready' }); } }); // ==================== 详细健康检查 ==================== class HealthChecker { private checks: Map Promise> = new Map(); // 注册检查项 register(name: string, check: () => Promise) { this.checks.set(name, check); } // 执行所有检查 async check(): Promise { const checks: Record = {}; let overallStatus: 'healthy' | 'unhealthy' | 'degraded' = 'healthy'; for (const [name, check] of this.checks) { try { const start = Date.now(); const result = await check(); result.latency = Date.now() - start; checks[name] = result; if (result.status === 'down') { // 核心依赖故障则整体不健康 if (['database', 'cache'].includes(name)) { overallStatus = 'unhealthy'; } else { overallStatus = 'degraded'; } } } catch (error) { checks[name] = { status: 'down', message: error.message }; overallStatus = 'unhealthy'; } } return { status: overallStatus, checks, timestamp: new Date() }; } } // 使用健康检查器 const healthChecker = new HealthChecker(); // 注册数据库检查 healthChecker.register('database', async () => { const client = await mongoose.connect(process.env.MONGO_URL); await client.connection.db.admin().ping(); return { status: 'up' }; }); // 注册Redis检查 healthChecker.register('cache', async () => { const redis = new Redis(process.env.REDIS_URL); await redis.ping(); return { status: 'up' }; }); // 注册外部API检查 healthChecker.register('payment-gateway', async () => { const response = await fetch('https://payment.api/health', { timeout: 5000 }); if (response.ok) { return { status: 'up' }; } return { status: 'down', message: `HTTP ${response.status}` }; }); // 健康检查端点 app.get('/health', async (req, res) => { const health = await healthChecker.check(); const statusCode = health.status === 'healthy' ? 200 : health.status === 'degraded' ? 200 : 503; res.status(statusCode).json(health); });

🔀 心跳与健康检查集成

// ==================== 心跳上报 ==================== class HeartbeatReporter { private serviceName: string; private instanceId: string; private registryClient: RegistryClient; private interval: NodeJS.Timeout | null = null; constructor( serviceName: string, instanceId: string, registryClient: RegistryClient ) { this.serviceName = serviceName; this.instanceId = instanceId; this.registryClient = registryClient; } // 启动心跳 start(intervalMs: number = 10000) { this.interval = setInterval(async () => { try { await this.sendHeartbeat(); } catch (error) { console.error('Heartbeat failed:', error); } }, intervalMs); } // 发送心跳 private async sendHeartbeat() { await this.registryClient.sendHeartbeat({ serviceName: this.serviceName, instanceId: this.instanceId, timestamp: Date.now() }); } // 停止心跳 stop() { if (this.interval) { clearInterval(this.interval); this.interval = null; } } } // ==================== 注册中心健康检查 ==================== class ServiceRegistry { private instances: Map = new Map(); private heartbeatTimeout = 30000; // 30秒无心跳则下线 // 处理心跳 async handleHeartbeat(data: HeartbeatData) { const key = `${data.serviceName}:${data.instanceId}`; const instance = this.instances.get(key); if (instance) { instance.lastHeartbeat = Date.now(); instance.status = 'healthy'; } } // 定期检查过期实例 startHealthCheckTask() { setInterval(() => { const now = Date.now(); for (const [key, instance] of this.instances) { if (now - instance.lastHeartbeat > this.heartbeatTimeout) { console.log(`Instance ${key} expired, removing...`); this.instances.delete(key); } } }, 10000); } // 主动健康检查（TCP/HTTP探针） async activeCheck(instance: ServiceInstance): Promise { try { const response = await fetch( `http://${instance.host}:${instance.port}/health`, { timeout: 5000 } ); return response.ok; } catch (error) { return false; } } } // ==================== 客户端侧健康检查 ==================== class LoadBalancerWithHealthCheck { private instances: ServiceInstance[] = []; private healthStatus: Map = new Map(); // 获取健康实例 getHealthyInstance(): ServiceInstance | null { const healthy = this.instances.filter( i => this.healthStatus.get(i.id) !== false ); if (healthy.length === 0) { return null; } // 负载均衡选择 return this.selectInstance(healthy); } // 记录调用结果（被动健康检查） recordCallResult(instanceId: string, success: boolean, latency: number) { if (!success) { this.healthStatus.set(instanceId, false); } else if (latency > 1000) { // 延迟过高标记为不健康 this.healthStatus.set(instanceId, false); } else { this.healthStatus.set(instanceId, true); } } // 定期恢复检查 startRecoveryCheck() { setInterval(async () => { for (const [instanceId, isHealthy] of this.healthStatus) { if (!isHealthy) { // 尝试重新探测 const instance = this.instances.find(i => i.id === instanceId); if (instance) { const healthy = await this.checkInstance(instance); this.healthStatus.set(instanceId, healthy); } } } }, 30000); } }
