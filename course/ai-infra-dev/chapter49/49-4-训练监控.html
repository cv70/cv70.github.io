# GPU监控脚本
import subprocess
import time
import json

def get_gpu_stats():
    """获取GPU状态"""
    result = subprocess.run(
        ['nvidia-smi', '--query-gpu=index,name,temperature.gpu,utilization.gpu,'
         'utilization.memory,memory.used,memory.total,power.draw',
         '--format=csv,noheader,nounits'],
        capture_output=True,
        text=True
    )

    stats = []
    for line in result.stdout.split('\n'):
        if not line:
            continue
        parts = [p.strip() for p in line.split(',')]
        stats.append({
            'index': int(parts[0]),
            'name': parts[1],
            'temperature': float(parts[2]),
            'gpu_util': float(parts[3]),
            'mem_util': float(parts[4]),
            'mem_used': float(parts[5]),
            'mem_total': float(parts[6]),
            'power': float(parts[7])
        })
    return stats

# 持续监控
while True:
    gpu_stats = get_gpu_stats()
    for gpu in gpu_stats:
        print(f"GPU {gpu['index']}: "
              f"Temp {gpu['temperature']}°C, "
              f"Util {gpu['gpu_util']}%, "
              f"Mem {gpu['mem_used']}/{gpu['mem_total']}MB")
    time.sleep(5)

🔔 告警配置

# 训练告警规则
import requests

class TrainingMonitor:
    def __init__(self, webhook_url):
        self.webhook_url = webhook_url
        self.thresholds = {
            'loss_spike': 10.0,        # Loss异常
            'gpu_low_util': 10.0,      # GPU利用率过低
            'gpu_high_temp': 85.0,      # GPU温度过高
            'oom': 0.95               # 显存使用过高
        }

    def check_and_alert(self, metrics):
        """检查指标并发送告警"""
        alerts = []

        # 检查Loss
        if metrics['loss'] > self.thresholds['loss_spike']:
            alerts.append({
                'type': 'loss_spike',
                'value': metrics['loss'],
                'message': f"Loss spike detected: {metrics['loss']}"
            })

        # 检查GPU利用率
        if metrics['gpu_util'] < self.thresholds['gpu_low_util']:
            alerts.append({
                'type': 'gpu_low_util',
                'value': metrics['gpu_util'],
                'message': f"Low GPU utilization: {metrics['gpu_util']}%"
            })

        # 检查GPU温度
        if metrics['gpu_temp'] > self.thresholds['gpu_high_temp']:
            alerts.append({
                'type': 'gpu_high_temp',
                'value': metrics['gpu_temp'],
                'message': f"High GPU temperature: {metrics['gpu_temp']}°C"
            })

        # 发送告警
        for alert in alerts:
            self.send_alert(alert)

    def send_alert(self, alert):
        """发送告警消息"""
        payload = {
            'text': f"⚠️ Training Alert: {alert['message']}",
            'attachments': [{
                'color': 'danger',
                'fields': [
                    {'title': 'Type', 'value': alert['type']},
                    {'title': 'Value', 'value': str(alert['value'])}
                ]
            }]
        }
        requests.post(self.webhook_url, json=payload)
