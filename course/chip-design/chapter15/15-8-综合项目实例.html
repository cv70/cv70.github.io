🏗️ 数字频率计设计

数字频率计综合应用了计数器、分频器、控制器和显示驱动等多个模块，是一个典型的综合项目。

// 频率计顶层模块
module frequency_counter(
    input  clk,          // 系统时钟
    input  rst_n,        // 复位
    input  signal_in,    // 待测信号
    output [3:0] digit0,  // 显示 digit0-3
    output [3:0] digit1,
    output [3:0] digit2,
    output [3:0] digit3
);
    wire clk_1hz;        // 1Hz闸门信号
    wire [31:0] count;
    wire [3:0]  bcd0, bcd1, bcd2, bcd3;

    // 分频器：生成1Hz闸门信号
    clock_divider #(.DIV(50_000_000)) u_div (
        .clk(clk),
        .rst_n(rst_n),
        .clk_div(clk_1hz)
    );

    // 计数器：计算1秒内的脉冲数
    counter_32bit u_counter (
        .clk(signal_in),
        .rst_n(rst_n),
        .gate(clk_1hz),
        .count(count)
    );

    // 二进制转BCD
    binary_to_bcd u_bcd (
        .binary_in(count),
        .bcd0(bcd0),
        .bcd1(bcd1),
        .bcd2(bcd2),
        .bcd3(bcd3)
    );

    // 七段译码器
    seg7_decoder u_seg0 (.bcd(bcd0), .seg(digit0));
    seg7_decoder u_seg1 (.bcd(bcd1), .seg(digit1));
    seg7_decoder u_seg2 (.bcd(bcd2), .seg(digit2));
    seg7_decoder u_seg3 (.bcd(bcd3), .seg(digit3));

endmodule
