OIDC/OAuth2.0认证流程

# OAuth2.0认证流程

# 1. 用户发起登录请求
GET /auth/authorize?
    response_type=code&
    client_id=ai-platform&
    redirect_uri=https://platform.ai/callback&
    scope=openid profile email&
    state=random_string

# 2. 用户在认证服务器登录
# 输入用户名/密码或使用第三方登录

# 3. 认证服务器返回授权码
302 Found
Location: https://platform.ai/callback?
    code=authorization_code&
    state=random_string

# 4. 平台用授权码换取访问令牌
POST /oauth/token
{
    "grant_type": "authorization_code",
    "code": "authorization_code",
    "redirect_uri": "https://platform.ai/callback",
    "client_id": "ai-platform",
    "client_secret": "client_secret"
}

# 5. 返回访问令牌
{
    "access_token": "eyJhbGciOiJSUzI1NiIsInR5cCI6IkpXVCJ9...",
    "token_type": "Bearer",
    "expires_in": 3600,
    "refresh_token": "def50200f9..."
}

# 6. 使用访问令牌访问资源
GET /api/v1/models
Authorization: Bearer eyJhbGciOiJSUzI1NiIsInR5cCI6IkpXVCJ9...

🏢 多租户架构

# 多租户数据模型设计

# 租户表
CREATE TABLE tenants (
    id VARCHAR(36) PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    status ENUM('active', 'suspended', 'deleted'),
    resource_quota JSON,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

# 用户表
CREATE TABLE users (
    id VARCHAR(36) PRIMARY KEY,
    username VARCHAR(255) UNIQUE NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    status ENUM('active', 'inactive', 'locked'),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

# 租户用户关联
CREATE TABLE tenant_users (
    id VARCHAR(36) PRIMARY KEY,
    tenant_id VARCHAR(36) REFERENCES tenants(id),
    user_id VARCHAR(36) REFERENCES users(id),
    role ENUM('admin', 'developer', 'viewer'),
    UNIQUE(tenant_id, user_id)
);

# 资源命名空间
CREATE TABLE namespaces (
    id VARCHAR(36) PRIMARY KEY,
    tenant_id VARCHAR(36) REFERENCES tenants(id),
    name VARCHAR(255) NOT NULL,
    resource_quota JSON,
    UNIQUE(tenant_id, name)
);

🔑 API密钥管理

# API密钥生成与验证

# 密钥生成
import secrets
import hashlib
from datetime import datetime, timedelta

def generate_api_key():
    # 生成随机密钥
    key_id = secrets.token_urlsafe(16)
    key_secret = secrets.token_urlsafe(32)

    # 存储哈希值
    key_hash = hashlib.sha256(key_secret.encode()).hexdigest()

    return {
        'key_id': key_id,
        'key_secret': key_secret,
        'key_hash': key_hash
    }

# 密钥验证
def verify_api_key(key_id, key_secret, stored_hash):
    key_hash = hashlib.sha256(key_secret.encode()).hexdigest()
    return secrets.compare_digest(key_hash, stored_hash)

# JWT Token生成
import jwt

def generate_jwt_token(user_id, tenant_id, role):
    payload = {
        'user_id': user_id,
        'tenant_id': tenant_id,
        'role': role,
        'exp': datetime.utcnow() + timedelta(hours=24),
        'iat': datetime.utcnow()
    }
    token = jwt.encode(payload, SECRET_KEY, algorithm='HS256')
    return token

🔒 资源隔离

Kubernetes命名空间隔离

# 为每个租户创建命名空间
kubectl create namespace tenant-001
kubectl create namespace tenant-002

# 配置资源配额
kubectl apply -f - << EOF
apiVersion: v1
kind: ResourceQuota
metadata:
  name: tenant-quota
  namespace: tenant-001
spec:
  hard:
    requests.cpu: "16"
    requests.memory: 64Gi
    limits.cpu: "32"
    limits.memory: 128Gi
    nvidia.com/gpu: "8"
    persistentvolumeclaims: "10"
EOF

网络策略隔离

# 网络隔离策略
kubectl apply -f - << EOF
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: tenant-isolation
  namespace: tenant-001
spec:
  podSelector: {}
  policyTypes:
  - Ingress
  - Egress
  ingress:
  - from:
    - podSelector: {}
  egress:
  - to:
    - podSelector: {}
EOF

用户体系最佳实践

最小权限原则：用户只获得完成任务所需的最小权限。审计日志：记录所有敏感操作，便于追溯。定期审查：定期检查和清理过期权限。多因素认证：为关键操作添加MFA保护。
