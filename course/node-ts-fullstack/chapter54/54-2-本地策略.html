54.2 本地策略

实现用户名密码认证

🎯 学习目标

  • 掌握 passport-local 的配置
  • 学会用户名密码验证流程
  • 理解自定义选项配置
  • 学会错误处理和消息提示

📦 安装依赖

# 安装 passport-local npm install passport-local # 安装类型定义 npm install -D @types/passport-local

🔧 配置本地策略

// src/config/passport.ts import passport from 'passport'; import { Strategy as LocalStrategy } from 'passport-local'; import { User, IUser } from '../models/User'; import bcrypt from 'bcryptjs'; // 本地策略配置 passport.use(new LocalStrategy( { usernameField: 'email', // 使用邮箱作为用户名 passwordField: 'password', passReqToCallback: false, }, async (email: string, password: string, done: Function) => { try { // 1. 查找用户 const user = await User.findOne({ email: email.toLowerCase() }); if (!user) { return done(null, false, { message: '邮箱或密码错误' }); } // 2. 验证密码 const isMatch = await bcrypt.compare(password, user.password); if (!isMatch) { return done(null, false, { message: '邮箱或密码错误' }); } // 3. 检查用户状态 if (!user.isActive) { return done(null, false, { message: '账户已被禁用' }); } // 4. 更新登录时间 user.lastLogin = new Date(); await user.save(); // 5. 认证成功 return done(null, user); } catch (error) { return done(error); } } )); export default passport;
💡
安全提示

认证失败时不要提示"用户不存在"或"密码错误"，统一使用"邮箱或密码错误"，防止攻击者枚举用户。

🔐 登录路由实现

// src/routes/auth.routes.ts import express, { Request, Response, NextFunction } from 'express'; import passport from '../config/passport'; const router = express.Router(); // 登录接口（返回 JSON） router.post('/login', (req: Request, res: Response, next: NextFunction) => { passport.authenticate('local', (err: any, user: IUser | false, info: any) => { if (err) { return next(err); } if (!user) { return res.status(401).json({ success: false, message: info?.message || '认证失败', }); } req.login(user, (err) => { if (err) { return next(err); } return res.json({ success: true, message: '登录成功', user: { id: user._id, email: user.email, username: user.username, role: user.role, }, }); }); })(req, res, next); }); // 登录接口（Session 版本） router.post('/login-session', passport.authenticate('local', { successRedirect: '/dashboard', failureRedirect: '/login', failureFlash: true, }) ); // 登出 router.post('/logout', (req: Request, res: Response, next: NextFunction) => { req.logout((err) => { if (err) { return next(err); } res.json({ success: true, message: '登出成功' }); }); }); // 检查认证状态 router.get('/me', (req: Request, res: Response) => { if (req.isAuthenticated()) { res.json({ authenticated: true, user: req.user, }); } else { res.status(401).json({ authenticated: false, message: '未登录', }); } }); export default router;

🛡️ 认证中间件

// src/middleware/auth.middleware.ts import { Request, Response, NextFunction } from 'express'; // 必须登录 export function requireAuth(req: Request, res: Response, next: NextFunction) { if (req.isAuthenticated()) { return next(); } if (req.xhr || req.headers.accept?.indexOf('json') > -1) { return res.status(401).json({ error: '请先登录', code: 'UNAUTHORIZED', }); } res.redirect('/login'); } // 必须是特定角色 export function requireRole(...roles: string[]) { return (req: Request, res: Response, next: NextFunction) => { if (!req.isAuthenticated()) { return res.status(401).json({ error: '请先登录' }); } const user = req.user as any; if (!roles.includes(user.role)) { return res.status(403).json({ error: '权限不足', required: roles, current: user.role, }); } next(); }; } // 使用示例 import express from 'express'; import { requireAuth, requireRole } from './middleware/auth.middleware'; const app = express(); // 需要登录 app.get('/profile', requireAuth, (req, res) => { res.json({ user: req.user }); }); // 需要管理员权限 app.get('/admin/users', requireAuth, requireRole('admin'), (req, res) => { res.json({ message: '管理员面板' }); } );

🔄 注册与密码重置

// src/controllers/auth.controller.ts import { Request, Response } from 'express'; import { User } from '../models/User'; import bcrypt from 'bcryptjs'; import crypto from 'crypto'; export class AuthController { // 用户注册 async register(req: Request, res: Response) { try { const { email, username, password } = req.body; // 检查用户是否已存在 const existingUser = await User.findOne({ $or: [{ email }, { username }] }); if (existingUser) { return res.status(400).json({ success: false, message: '邮箱或用户名已被注册', }); } // 创建用户 const user = new User({ email, username, password: await bcrypt.hash(password, 12), role: 'user', isActive: true, }); await user.save(); // 自动登录 req.login(user, (err) => { if (err) { return res.status(500).json({ message: '注册成功但自动登录失败' }); } res.status(201).json({ success: true, message: '注册成功', user: { id: user._id, email: user.email, username: user.username, }, }); }); } catch (error) { res.status(500).json({ message: '服务器错误' }); } } // 请求密码重置 async forgotPassword(req: Request, res: Response) { try { const { email } = req.body; const user = await User.findOne({ email }); if (!user) { return res.json({ success: true, message: '如果邮箱存在，重置链接已发送', }); } // 生成重置令牌 const resetToken = crypto.randomBytes(32).toString('hex'); const resetTokenHash = crypto .createHash('sha256') .update(resetToken) .digest('hex'); user.resetPasswordToken = resetTokenHash; user.resetPasswordExpires = new Date(Date.now() + 3600000); await user.save(); // 发送邮件 // await sendEmail({ to: user.email, ... }); res.json({ success: true, message: '如果邮箱存在，重置链接已发送', }); } catch (error) { res.status(500).json({ message: '服务器错误' }); } } }
