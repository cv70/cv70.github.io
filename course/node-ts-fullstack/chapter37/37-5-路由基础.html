37.5 路由基础

使用 koa-router 构建路由系统

🎯 学习目标

  • 学会安装和配置 koa-router
  • 掌握 RESTful 路由设计
  • 理解路由参数和查询参数的区别
  • 学会路由模块化组织

📦 安装 koa-router

# 安装 koa-router npm install koa-router
💡
注意

Koa 本身不内置路由功能，需要使用 koa-router 或其他路由中间件。

🚀 基本使用

const Koa = require('koa'); const Router = require('koa-router'); const app = new Koa(); const router = new Router(); // 定义路由 router.get('/', async ctx => { ctx.body = 'Home Page'; }); router.get('/about', async ctx => { ctx.body = 'About Page'; }); // 注册路由中间件 app.use(router.routes()); app.use(router.allowedMethods()); app.listen(3000);

📡 HTTP 方法路由

const router = new Router(); // GET 请求 router.get('/users', async ctx => { ctx.body = 'Get all users'; }); // POST 请求 router.post('/users', async ctx => { ctx.body = 'Create user'; }); // PUT 请求 - 完整更新 router.put('/users/:id', async ctx => { ctx.body = `Update user ${ctx.params.id}`; }); // PATCH 请求 - 部分更新 router.patch('/users/:id', async ctx => { ctx.body = `Patch user ${ctx.params.id}`; }); // DELETE 请求 router.delete('/users/:id', async ctx => { ctx.body = `Delete user ${ctx.params.id}`; }); // 支持多个 HTTP 方法 router.all('/api', async ctx => { ctx.body = `Method: ${ctx.method}`; });

🔗 动态路由参数

// 单个参数 router.get('/users/:id', async ctx => { const { id } = ctx.params; ctx.body = { userId: id }; }); // 多个参数 router.get('/posts/:postId/comments/:commentId', async ctx => { const { postId, commentId } = ctx.params; ctx.body = { postId, commentId }; }); // 可选参数 (需要正则支持) router.get('/books/:id?', async ctx => { const { id } = ctx.params; if (id) { ctx.body = { bookId: id }; } else { ctx.body = { books: [] }; } }); // 参数约束 (正则表达式) router.get('/users/:id(\\d+)', async ctx => { // id 只匹配数字 ctx.body = { userId: ctx.params.id }; });

📁 路由模块化

// routes/userRoutes.js const Router = require('koa-router'); const router = new Router({ prefix: '/api/users' }); router.get('/', async ctx => { ctx.body = { users: [] }; }); router.get('/:id', async ctx => { ctx.body = { user: { id: ctx.params.id } }; }); router.post('/', async ctx => { ctx.body = { message: 'User created' }; }); module.exports = router;
// app.js const Koa = require('koa'); const userRoutes = require('./routes/userRoutes'); const postRoutes = require('./routes/postRoutes'); const app = new Koa(); // 注册路由 app.use(userRoutes.routes()); app.use(postRoutes.routes()); app.listen(3000);

🔀 路由前缀

// 创建带前缀的路由器 const router = new Router({ prefix: '/api/v1' }); // 实际路径: /api/v1/users router.get('/users', async ctx => { ctx.body = { users: [] }; }); // 实际路径: /api/v1/posts router.get('/posts', async ctx => { ctx.body = { posts: [] }; });
✅
API 版本管理

使用路由前缀可以方便地管理 API 版本，如 /api/v1, /api/v2。

🔀 路由重定向

const router = new Router(); // 路由重定向 router.redirect('/old-path', '/new-path', 301); // 或者使用中间件方式 router.get('/old-page', async ctx => { ctx.redirect('/new-page'); });

🔐 路由中间件

// 单个路由的中间件 router.get('/protected', authMiddleware, async ctx => { ctx.body = { message: 'Protected content' }; }); // 多个中间件 router.get('/admin', authMiddleware, adminMiddleware, async ctx => { ctx.body = { message: 'Admin area' }; } ); // 示例中间件 async function authMiddleware(ctx, next) { const token = ctx.headers.authorization; if (!token) { ctx.status = 401; ctx.body = { error: 'Unauthorized' }; return; } await next(); }

📊 RESTful API 设计规范

HTTP 方法 路径 作用
GET /api/users 获取用户列表
GET /api/users/:id 获取单个用户
POST /api/users 创建用户
PUT /api/users/:id 更新用户（完整）
PATCH /api/users/:id 更新用户（部分）
DELETE /api/users/:id 删除用户

💻 完整示例

const Koa = require('koa'); const Router = require('koa-router'); const bodyParser = require('koa-bodyparser'); const app = new Koa(); const router = new Router({ prefix: '/api' }); app.use(bodyParser()); // 模拟数据 let users = [ { id: 1, name: 'Alice' }, { id: 2, name: 'Bob' } ]; // RESTful API router .get('/users', async ctx => { ctx.body = { success: true, data: users }; }) .get('/users/:id', async ctx => { const user = users.find(u => u.id === +ctx.params.id); if (!user) { ctx.status = 404; ctx.body = { success: false, message: 'User not found' }; return; } ctx.body = { success: true, data: user }; }) .post('/users', async ctx => { const user = { id: Date.now(), ...ctx.request.body }; users.push(user); ctx.status = 201; ctx.body = { success: true, data: user }; }) .put('/users/:id', async ctx => { const index = users.findIndex(u => u.id === +ctx.params.id); if (index === -1) { ctx.status = 404; ctx.body = { success: false, message: 'User not found' }; return; } users[index] = { ...users[index], ...ctx.request.body }; ctx.body = { success: true, data: users[index] }; }) .delete('/users/:id', async ctx => { const index = users.findIndex(u => u.id === +ctx.params.id); if (index === -1) { ctx.status = 404; ctx.body = { success: false, message: 'User not found' }; return; } users.splice(index, 1); ctx.body = { success: true, message: 'User deleted' }; }); app.use(router.routes()); app.use(router.allowedMethods()); app.listen(3000, () => { console.log('Server running at http://localhost:3000'); });
