💾 D触发器

// 基本D触发器
module dff(
    input  clk,
    input  d,
    output reg q
);
    always @(posedge clk) begin
        q <= d;
    end
endmodule

// 带异步复位的D触发器
module dff_async_rst(
    input  clk,
    input  rst_n,
    input  d,
    output reg q
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            q <= 1'b0;
        else
            q <= d;
    end
endmodule

// 带同步复位的D触发器
module dff_sync_rst(
    input  clk,
    input  rst_n,
    input  d,
    output reg q
);
    always @(posedge clk) begin
        if (!rst_n)
            q <= 1'b0;
        else
            q <= d;
    end
endmodule

🔄 JK触发器

// JK触发器
module jk_ff(
    input  clk,
    input  rst_n,
    input  j,
    input  k,
    output reg q,
    output reg q_n
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            q <= 1'b0;
            q_n <= 1'b1;
        end
        else begin
            case ({j, k})
                2'b00: q <= q;          // 保持
                2'b01: q <= 1'b0;        // 复位
                2'b10: q <= 1'b1;        // 置位
                2'b11: q <= ~q;         // 翻转
            endcase
            q_n <= ~q;
        end
    end
endmodule

⏯️ T触发器

// T触发器
module t_ff(
    input  clk,
    input  rst_n,
    input  t,
    output reg q
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            q <= 1'b0;
        else if (t)
            q <= ~q;
    end
endmodule

📊 寄存器

// 8位寄存器
module register_8bit(
    input  clk,
    input  rst_n,
    input  load,
    input  [7:0] d,
    output reg [7:0] q
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            q <= 8'b0000_0000;
        else if (load)
            q <= d;
    end
endmodule

🔀 移位寄存器

// 8位移位寄存器（左移）
module shift_register_left(
    input  clk,
    input  rst_n,
    input  shift_en,
    input  serial_in,
    output [7:0] parallel_out
);
    reg [7:0] shift_reg;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            shift_reg <= 8'b0000_0000;
        else if (shift_en)
            shift_reg <= {shift_reg[6:0], serial_in};
    end

    assign parallel_out = shift_reg;
endmodule

// 双向移位寄存器
module bidirectional_shift(
    input  clk,
    input  rst_n,
    input  shift_en,
    input  dir,        // 0:右移, 1:左移
    input  serial_in,
    output [7:0] parallel_out
);
    reg [7:0] shift_reg;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            shift_reg <= 8'b0000_0000;
        else if (shift_en) begin
            if (dir)
                shift_reg <= {shift_reg[6:0], serial_in};  // 左移
            else
                shift_reg <= {serial_in, shift_reg[7:1]};  // 右移
        end
    end

    assign parallel_out = shift_reg;
endmodule
