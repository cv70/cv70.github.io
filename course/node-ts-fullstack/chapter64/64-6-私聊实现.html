点对点消息发送

私聊原理

Socket.IO 中实现私聊的核心思路是：每个 Socket 连接时会自动加入一个以其 socket.id 为名称的房间。通过向这个特定房间发送消息，就能实现点对点通信。

🔗 基本私聊实现

最简单的私聊

io.on('connection', (socket) => { // 向特定 socket 发送消息 socket.on('private-message', (targetSocketId, message) => { // 方式一：直接向目标 socket 的房间发送 io.to(targetSocketId).emit('private-message', { from: socket.id, message, timestamp: Date.now() }); }); });

基于用户 ID 的私聊

// 用户 ID 与 Socket ID 的映射 const userSockets = new Map<string, Set<string>>(); // userId -> Set<socketId> const socketUsers = new Map<string, string>(); // socketId -> userId io.on('connection', (socket) => { const userId = socket.handshake.query.userId as string; // 建立映射 socketUsers.set(socket.id, userId); if (!userSockets.has(userId)) { userSockets.set(userId, new Set()); } userSockets.get(userId)!.add(socket.id); // 用户可能同时登录多个设备，加入用户专属房间 socket.join(`user:${userId}`); // 断开时清理 socket.on('disconnect', () => { socketUsers.delete(socket.id); const sockets = userSockets.get(userId); if (sockets) { sockets.delete(socket.id); if (sockets.size === 0) { userSockets.delete(userId); } } }); }); // 向特定用户发送消息 function sendToUser(userId: string, event: string, data: any) { io.to(`user:${userId}`).emit(event, data); } // 使用示例 socket.on('private-message', (targetUserId: string, content: string) => { const message = { from: socketUsers.get(socket.id), to: targetUserId, content, timestamp: Date.now() }; sendToUser(targetUserId, 'private-message', message); // 同时发给发送者（同步到其他设备） sendToUser(socketUsers.get(socket.id)!, 'private-message', message); });

💻 完整私聊系统实现

// ==================== 类型定义 ==================== interface User { id: string; username: string; avatar?: string; status: 'online' | 'offline' | 'busy'; } interface PrivateMessage { id: string; from: User; to: User; content: string; type: 'text' | 'image' | 'file'; timestamp: number; read: boolean; } interface Conversation { id: string; participants: string[]; lastMessage?: PrivateMessage; unreadCount: number; } // ==================== 用户管理 ==================== class UserManager { private users = new Map<string, User>(); private userSockets = new Map<string, Set<string>>(); private socketUsers = new Map<string, string>(); connect(socketId: string, userId: string, userInfo: Partial<User>) { // 更新或创建用户 if (!this.users.has(userId)) { this.users.set(userId, { id: userId, username: userInfo.username || 'Anonymous', avatar: userInfo.avatar, status: 'online' }); } else { const user = this.users.get(userId)!; user.status = 'online'; } // 建立映射 this.socketUsers.set(socketId, userId); if (!this.userSockets.has(userId)) { this.userSockets.set(userId, new Set()); } this.userSockets.get(userId)!.add(socketId); return this.users.get(userId)!; } disconnect(socketId: string) { const userId = this.socketUsers.get(socketId); if (!userId) return; this.socketUsers.delete(socketId); const sockets = this.userSockets.get(userId); if (sockets) { sockets.delete(socketId); if (sockets.size === 0) { this.userSockets.delete(userId); const user = this.users.get(userId); if (user) { user.status = 'offline'; } } } } getUser(userId: string): User | undefined { return this.users.get(userId); } getOnlineUsers(): User[] { return Array.from(this.users.values()).filter(u => u.status === 'online'); } isOnline(userId: string): boolean { const sockets = this.userSockets.get(userId); return sockets !== undefined && sockets.size > 0; } getSockets(userId: string): string[] { return Array.from(this.userSockets.get(userId) || []); } } // ==================== 消息管理 ==================== class MessageManager { private messages: PrivateMessage[] = []; private conversations = new Map<string, Conversation>(); // 生成会话 ID（两人之间的唯一标识） getConversationId(userId1: string, userId2: string): string { return [userId1, userId2].sort().join('-'); } async saveMessage(message: PrivateMessage): Promise<PrivateMessage> { this.messages.push(message); // 更新会话 const conversationId = this.getConversationId(message.from.id, message.to.id); const conversation = this.conversations.get(conversationId); if (conversation) { conversation.lastMessage = message; conversation.unreadCount++; } else { this.conversations.set(conversationId, { id: conversationId, participants: [message.from.id, message.to.id], lastMessage: message, unreadCount: 1 }); } // 存入数据库 await this.saveToDatabase(message); return message; } async getConversation(userId: string, withUserId: string): Promise<PrivateMessage[]> { return this.messages.filter(m => (m.from.id === userId && m.to.id === withUserId) || (m.from.id === withUserId && m.to.id === userId) ); } markAsRead(conversationId: string, userId: string) { const conversation = this.conversations.get(conversationId); if (conversation) { conversation.unreadCount = 0; } this.messages.forEach(m => { if (m.to.id === userId) { m.read = true; } }); } } // ==================== 初始化 ==================== const userManager = new UserManager(); const messageManager = new MessageManager(); io.on('connection', (socket) => { const userId = socket.handshake.auth.userId; if (!userId) { return socket.disconnect(); } // 用户连接 const user = userManager.connect(socket.id, userId, { username: socket.handshake.auth.username, avatar: socket.handshake.auth.avatar }); // 加入用户专属房间 socket.join(`user:${userId}`); // 广播用户上线 socket.broadcast.emit('user-online', user); // ==================== 私聊消息 ==================== socket.on('send-private-message', async (data: { to: string; content: string; type?: 'text' | 'image' | 'file' }, callback) => { const targetUser = userManager.getUser(data.to); if (!targetUser) { return callback({ success: false, error: '用户不存在' }); } const message: PrivateMessage = { id: generateId(), from: user, to: targetUser, content: data.content, type: data.type || 'text', timestamp: Date.now(), read: false }; // 保存消息 await messageManager.saveMessage(message); // 发送给目标用户 io.to(`user:${data.to}`).emit('private-message', message); // 发送给发送者（多设备同步） socket.emit('private-message', message); callback({ success: true, message }); }); // ==================== 获取会话列表 ==================== socket.on('get-conversations', async (callback) => { const conversations = await getConversationsForUser(userId); callback(conversations); }); // ==================== 获取聊天记录 ==================== socket.on('get-messages', async (withUserId: string, callback) => { const messages = await messageManager.getConversation(userId, withUserId); callback(messages); }); // ==================== 标记已读 ==================== socket.on('mark-read', async (withUserId: string) => { const conversationId = messageManager.getConversationId(userId, withUserId); messageManager.markAsRead(conversationId, userId); socket.emit('messages-read', { withUserId }); }); // ==================== 输入状态 ==================== socket.on('typing', (toUserId: string) => { io.to(`user:${toUserId}`).emit('user-typing', { userId, username: user.username }); }); socket.on('stop-typing', (toUserId: string) => { io.to(`user:${toUserId}`).emit('user-stop-typing', { userId }); }); // ==================== 获取在线用户 ==================== socket.on('get-online-users', (callback) => { callback(userManager.getOnlineUsers()); }); // ==================== 断开连接 ==================== socket.on('disconnect', () => { userManager.disconnect(socket.id); // 检查用户是否完全离线 if (!userManager.isOnline(userId)) { socket.broadcast.emit('user-offline', { userId }); } }); });

💾 离线消息处理

存储离线消息

// 用户上线时推送离线消息 socket.on('connection', async (socket) => { const userId = socket.handshake.auth.userId; // 获取离线消息 const offlineMessages = await getOfflineMessages(userId); if (offlineMessages.length > 0) { // 分批发送，避免阻塞 for (const message of offlineMessages) { socket.emit('offline-message', message); await sleep(10); // 短暂延迟 } // 发送完成通知 socket.emit('offline-messages-done', { count: offlineMessages.length }); // 标记已送达 await markMessagesDelivered(offlineMessages.map(m => m.id)); } }); // 发送消息时检查在线状态 async function sendPrivateMessage(from: string, to: string, content: string) { const message = { id: generateId(), from, to, content, timestamp: Date.now(), delivered: false, read: false }; // 保存消息 await saveMessage(message); if (userManager.isOnline(to)) { // 用户在线，直接发送 io.to(`user:${to}`).emit('private-message', message); message.delivered = true; await updateMessageDelivered(message.id); } // 用户离线，消息已存储，等待上线推送 return message; }

🔒 安全考虑

⚠️
私聊安全注意事项
  • 验证发送者身份，防止伪造消息
  • 验证接收者是否存在，防止信息泄露
  • 敏感消息内容加密存储
  • 实现消息撤回功能时注意时间限制
  • 考虑举报和屏蔽功能
// 消息发送验证中间件 io.use((socket, next) => { const userId = socket.handshake.auth.userId; const token = socket.handshake.auth.token; // 验证 token 和用户身份 verifyUser(userId, token) .then(user => { socket.data.user = user; next(); }) .catch(() => { next(new Error('认证失败')); }); }); // 发送消息时验证 socket.on('private-message', async (to, content, callback) => { // 验证接收者 if (!await userExists(to)) { return callback({ error: '接收者不存在' }); } // 验证是否被屏蔽 if (await isBlocked(to, socket.data.user.id)) { return callback({ error: '您已被对方屏蔽' }); } // 内容过滤 if (containsForbiddenContent(content)) { return callback({ error: '消息内容不合规' }); } // 发送消息... });
