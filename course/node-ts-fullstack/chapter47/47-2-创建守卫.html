47.2 创建守卫

实现自定义守卫

🎯 学习目标

  • 掌握守卫的创建方式
  • 学会使用 Reflector 获取元数据
  • 了解守卫的作用范围
  • 掌握自定义装饰器与守卫配合

📚 基本守卫实现

import { Injectable, CanActivate, ExecutionContext, UnauthorizedException } from '@nestjs/common'; import { Request } from 'express'; @Injectable() export class AuthGuard implements CanActivate { constructor(private readonly authService: AuthService) {} async canActivate(context: ExecutionContext): Promise { const request = context.switchToHttp().getRequest(); const token = this.extractTokenFromHeader(request); if (!token) { throw new UnauthorizedException('No token provided'); } try { const user = await this.authService.validateToken(token); // 将用户信息附加到请求对象 request['user'] = user; return true; } catch (error) { throw new UnauthorizedException('Invalid token'); } } private extractTokenFromHeader(request: Request): string | undefined { const [type, token] = request.headers.authorization?.split(' ') ?? []; return type === 'Bearer' ? token : undefined; } }
💡
抛出异常

守卫返回 false 时返回 403 Forbidden。如果需要自定义错误响应，可以抛出异常，如 UnauthorizedException。

🔧 使用 Reflector 获取元数据

import { Injectable, CanActivate, ExecutionContext, Reflector } from '@nestjs/common'; import { IS_PUBLIC_KEY } from './public.decorator'; @Injectable() export class AuthGuard implements CanActivate { constructor(private reflector: Reflector) {} canActivate(context: ExecutionContext): boolean { // 检查是否标记为公开 const isPublic = this.reflector.getAllAndOverride( IS_PUBLIC_KEY, [context.getHandler(), context.getClass()], ); if (isPublic) { return true; } // 执行认证检查 const request = context.switchToHttp().getRequest(); return this.validateRequest(request); } private validateRequest(request: Request): boolean { // 认证逻辑 return !!request.headers.authorization; } } // 公开路由装饰器 import { SetMetadata } from '@nestjs/common'; export const IS_PUBLIC_KEY = 'isPublic'; export const Public = () => SetMetadata(IS_PUBLIC_KEY, true); // 使用 @Controller('auth') export class AuthController { @Public() // 标记为公开，不需要认证 @Post('login') login() {} @Post('logout') // 需要认证 logout() {} }

📍 守卫的作用范围

// 1. 方法级 @Controller('users') export class UsersController { @UseGuards(AuthGuard) @Get('profile') getProfile() {} } // 2. 控制器级 @Controller('admin') @UseGuards(AuthGuard, AdminGuard) // 可以应用多个守卫 export class AdminController { @Get('dashboard') getDashboard() {} @Get('users') getUsers() {} } // 3. 全局级 async function bootstrap() { const app = await NestFactory.create(AppModule); app.useGlobalGuards(new AuthGuard()); await app.listen(3000); } // 4. 模块级提供者 @Module({ providers: [ { provide: APP_GUARD, useClass: AuthGuard, }, ], }) export class AppModule {}

🔄 多个守卫执行

@Controller('admin') @UseGuards(AuthGuard) // 第一个执行 @UseGuards(AdminGuard) // 第二个执行 export class AdminController { @UseGuards(SuperAdminGuard) // 第三个执行 @Get('sensitive') getSensitiveData() {} } // 执行顺序：AuthGuard -> AdminGuard -> SuperAdminGuard // 任何一个守卫返回 false 或抛出异常，后续守卫不会执行
⚠️
守卫执行顺序

守卫按顺序执行：全局守卫 -> 控制器守卫 -> 方法守卫。任何一个守卫拒绝请求，后续守卫都不会执行。
