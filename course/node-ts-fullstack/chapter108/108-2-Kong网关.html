🔧 Kong配置示例

// 使用Kong Admin API配置 // 1. 创建Service curl -X POST http://localhost:8001/services \ -d "name=user-service" \ -d "url=http://user-service:3001" // 2. 创建Route curl -X POST http://localhost:8001/services/user-service/routes \ -d "paths[]=/api/users" // 3. 添加插件 - JWT认证 curl -X POST http://localhost:8001/services/user-service/plugins \ -d "name=jwt" // 4. 添加插件 - 限流 curl -X POST http://localhost:8001/services/user-service/plugins \ -d "name=rate-limiting" \ -d "config.second=100" \ -d "config.policy=local" // 使用Kong Deck配置（YAML） /* _format_version: "3.0" services: - name: user-service url: http://user-service:3001 routes: - name: users-route paths: - /api/users plugins: - name: jwt - name: rate-limiting config: second: 100 policy: local - name: order-service url: http://order-service:3002 routes: - name: orders-route paths: - /api/orders plugins: - name: key-auth - name: request-size-limiting config: allowed_payload_size: 10 */

🔀 Kong插件使用

// 常用Kong插件 const plugins = { authentication: { 'jwt': 'JWT认证', 'key-auth': 'API Key认证', 'basic-auth': 'Basic认证', 'oauth2': 'OAuth2认证' }, security: { 'cors': '跨域处理', 'ip-restriction': 'IP白名单/黑名单', 'bot-detection': '爬虫检测' }, traffic: { 'rate-limiting': '限流', 'request-size-limiting': '请求大小限制', 'response-ratelimiting': '响应限流' }, transformation: { 'request-transformer': '请求转换', 'response-transformer': '响应转换', 'correlation-id': '请求ID注入' } }; // Node.js管理Kong import axios from 'axios'; class KongAdminClient { private baseUrl = 'http://localhost:8001'; // 创建服务 async createService(name: string, url: string) { const response = await axios.post(`${this.baseUrl}/services`, { name, url }); return response.data; } // 创建路由 async createRoute(serviceName: string, paths: string[]) { const response = await axios.post( `${this.baseUrl}/services/${serviceName}/routes`, { paths } ); return response.data; } // 添加插件 async addPlugin(serviceName: string, pluginName: string, config: any) { const response = await axios.post( `${this.baseUrl}/services/${serviceName}/plugins`, { name: pluginName, config } ); return response.data; } // 创建消费者 async createConsumer(username: string) { const response = await axios.post(`${this.baseUrl}/consumers`, { username }); return response.data; } // 为消费者生成API Key async createApiKey(consumerId: string) { const response = await axios.post( `${this.baseUrl}/consumers/${consumerId}/key-auth` ); return response.data; } } // 使用示例 const kong = new KongAdminClient(); // 创建服务和路由 await kong.createService('user-service', 'http://user-service:3001'); await kong.createRoute('user-service', ['/api/users']); // 添加限流插件 await kong.addPlugin('user-service', 'rate-limiting', { second: 100, policy: 'local' }); // 创建消费者 const consumer = await kong.createConsumer('client-app'); const apiKey = await kong.createApiKey(consumer.id); console.log('API Key:', apiKey.key);
