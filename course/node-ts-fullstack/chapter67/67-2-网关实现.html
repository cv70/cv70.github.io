📡 完整聊天网关示例

// chat/chat.gateway.ts import { WebSocketGateway, WebSocketServer, SubscribeMessage, OnGatewayConnection, OnGatewayDisconnect, ConnectedSocket, MessageBody } from '@nestjs/websockets'; import { Server, Socket } from 'socket.io'; import { ChatService } from './chat.service'; import { Logger, UseFilters, UsePipes, UseGuards } from '@nestjs/common'; // 类型定义 interface JoinRoomDto { roomId: string; } interface SendMessageDto { roomId: string; content: string; } interface Message { id: string; content: string; senderId: string; senderName: string; roomId: string; timestamp: Date; } @WebSocketGateway({ namespace: 'chat', cors: { origin: ['http://localhost:3000'], credentials: true } }) export class ChatGateway implements OnGatewayConnection, OnGatewayDisconnect { @WebSocketServer() server: Server; private logger = new Logger(ChatGateway.name); constructor(private readonly chatService: ChatService) {} // 连接处理 handleConnection(client: Socket) { const userId = client.handshake.auth.userId; const username = client.handshake.auth.username; if (!userId || !username) { this.logger.warn(`客户端缺少认证信息: ${client.id}`); client.disconnect(true); return; } // 存储用户信息 client.data.userId = userId; client.data.username = username; this.logger.log(`用户连接: ${username} (${client.id})`); // 通知所有客户端用户上线 this.server.emit('user:online', { userId, username, onlineCount: this.getOnlineCount() }); } // 断开处理 handleDisconnect(client: Socket) { const userId = client.data.userId; const username = client.data.username; this.logger.log(`用户断开: ${username} (${client.id})`); // 通知所有客户端用户下线 this.server.emit('user:offline', { userId, username, onlineCount: this.getOnlineCount() }); } // 加入房间 @SubscribeMessage('room:join') async handleJoinRoom( @ConnectedSocket() client: Socket, @MessageBody() data: JoinRoomDto ) { const { roomId } = data; const { userId, username } = client.data; // 加入房间 client.join(roomId); // 获取房间消息历史 const messages = await this.chatService.getRoomMessages(roomId); // 发送房间信息给加入者 client.emit('room:joined', { roomId, messages, users: this.getRoomUsers(roomId) }); // 通知房间其他人 client.to(roomId).emit('room:user-joined', { userId, username, timestamp: new Date() }); this.logger.log(`${username} 加入房间 ${roomId}`); } // 离开房间 @SubscribeMessage('room:leave') handleLeaveRoom( @ConnectedSocket() client: Socket, @MessageBody() data: JoinRoomDto ) { const { roomId } = data; const { userId, username } = client.data; client.leave(roomId); // 通知房间其他人 client.to(roomId).emit('room:user-left', { userId, username, timestamp: new Date() }); this.logger.log(`${username} 离开房间 ${roomId}`); } // 发送消息 @SubscribeMessage('message:send') async handleSendMessage( @ConnectedSocket() client: Socket, @MessageBody() data: SendMessageDto ): Promise<Message> { const { roomId, content } = data; const { userId, username } = client.data; // 创建消息 const message: Message = { id: this.generateId(), content, senderId: userId, senderName: username, roomId, timestamp: new Date() }; // 保存消息 await this.chatService.saveMessage(message); // 广播给房间所有人 this.server.to(roomId).emit('message:new', message); this.logger.log(`消息 [${roomId}]: ${username}: ${content}`); return message; } // 输入状态 @SubscribeMessage('typing:start') handleTypingStart( @ConnectedSocket() client: Socket, @MessageBody() roomId: string ) { const { userId, username } = client.data; client.to(roomId).emit('user:typing', { userId, username, isTyping: true }); } @SubscribeMessage('typing:stop') handleTypingStop( @ConnectedSocket() client: Socket, @MessageBody() roomId: string ) { const { userId } = client.data; client.to(roomId).emit('user:typing', { userId, isTyping: false }); } // 私有方法 private getOnlineCount(): number { return this.server.sockets.sockets.size; } private getRoomUsers(roomId: string): Array<{ userId: string; username: string }> { const room = this.server.sockets.adapter.rooms.get(roomId); if (!room) return []; const users: Array<{ userId: string; username: string }> = []; room.forEach(socketId => { const socket = this.server.sockets.sockets.get(socketId); if (socket) { users.push({ userId: socket.data.userId, username: socket.data.username }); } }); return users; } private generateId(): string { return Date.now().toString(36) + Math.random().toString(36).substr(2, 9); } }

🔧 装饰器详解

装饰器 说明
@WebSocketServer() 注入 Socket.IO Server 实例
@SubscribeMessage('event') 订阅指定事件
@ConnectedSocket() 注入当前连接的 Socket 实例
@MessageBody() 获取消息体数据
// 装饰器使用示例 @SubscribeMessage('event-name') handleEvent( @ConnectedSocket() client: Socket, // Socket 实例 @MessageBody() data: any, // 整个消息体 @MessageBody('field') field: string // 消息体的特定字段 ) { // ... }

💾 服务注入

Chat Service

// chat/chat.service.ts import { Injectable, Logger } from '@nestjs/common'; import { Message } from './interfaces/message.interface'; @Injectable() export class ChatService { private messages: Message[] = []; private logger = new Logger(ChatService.name); async saveMessage(message: Message): Promise<void> { this.messages.push(message); this.logger.log(`消息已保存: ${message.id}`); // 实际项目中应该保存到数据库 } async getRoomMessages(roomId: string, limit: number = 50): Promise<Message[]> { return this.messages .filter(m => m.roomId === roomId) .slice(-limit); } async getUserMessages(userId: string): Promise<Message[]> { return this.messages.filter(m => m.senderId === userId); } } // chat/chat.module.ts import { Module } from '@nestjs/common'; import { ChatGateway } from './chat.gateway'; import { ChatService } from './chat.service'; @Module({ providers: [ChatGateway, ChatService], exports: [ChatService] }) export class ChatModule {}

🌐 WebSocket 与 HTTP 混合

同时提供 HTTP 和 WebSocket

// chat/chat.controller.ts import { Controller, Get, Post, Body, Param } from '@nestjs/common'; import { ChatService } from './chat.service'; @Controller('chat') export class ChatController { constructor(private readonly chatService: ChatService) {} @Get('rooms/:roomId/messages') async getMessages(@Param('roomId') roomId: string) { return this.chatService.getRoomMessages(roomId); } @Get('users/:userId/messages') async getUserMessages(@Param('userId') userId: string) { return this.chatService.getUserMessages(userId); } } // chat.module.ts @Module({ controllers: [ChatController], providers: [ChatGateway, ChatService], exports: [ChatService] }) export class ChatModule {}
💡
最佳实践

将业务逻辑放在 Service 中，Gateway 只负责处理 WebSocket 事件和消息转发。这样可以在 HTTP Controller 中复用相同的业务逻辑。

📝 本节小结

✅
  • • @SubscribeMessage 装饰器订阅客户端事件
  • • @ConnectedSocket 注入 Socket 实例
  • • @MessageBody 获取消息数据
  • • Gateway 可以注入 Service 实现业务逻辑
  • • WebSocket 和 HTTP 可以共享同一个 Service
  • • 使用 Logger 记录日志便于调试
