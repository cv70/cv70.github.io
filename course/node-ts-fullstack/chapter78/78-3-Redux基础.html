📚 Redux 核心概念

// Redux 三大原则 // 1. 单一数据源：整个应用的 state 存储在一个 store 中 // 2. State 是只读的：只能通过 dispatch action 来修改 // 3. 使用纯函数修改：Reducer 是纯函数 // 核心概念 // Action：描述发生了什么 interface Action { type: string; payload?: any; } const incrementAction = { type: 'INCREMENT' }; const addTodoAction = { type: 'ADD_TODO', payload: { text: 'Learn Redux' } }; // Reducer：根据 action 返回新 state function counterReducer(state = 0, action: Action) { switch (action.type) { case 'INCREMENT': return state + 1; case 'DECREMENT': return state - 1; case 'INCREMENT_BY': return state + action.payload; default: return state; } } // Store：保存 state，提供 API import { createStore } from 'redux'; const store = createStore(counterReducer); // 获取 state console.log(store.getState()); // 0 // 订阅变化 store.subscribe(() => { console.log('State changed:', store.getState()); }); // 派发 action store.dispatch({ type: 'INCREMENT' }); // State changed: 1 store.dispatch({ type: 'INCREMENT_BY', payload: 5 }); // State changed: 6

🔧 Redux 工作流程

// Redux 工作流程 // View ──dispatch(action)──> Store ──action──> Reducer ──newState──> Store ──notify──> View // 完整示例：Todo 应用 // 定义类型 interface Todo { id: number; text: string; completed: boolean; } interface TodoState { todos: Todo[]; filter: 'all' | 'active' | 'completed'; } type TodoAction = | { type: 'ADD_TODO'; payload: string } | { type: 'TOGGLE_TODO'; payload: number } | { type: 'DELETE_TODO'; payload: number } | { type: 'SET_FILTER'; payload: 'all' | 'active' | 'completed' }; // 初始状态 const initialState: TodoState = { todos: [], filter: 'all', }; // Reducer function todoReducer(state = initialState, action: TodoAction): TodoState { switch (action.type) { case 'ADD_TODO': return { ...state, todos: [ ...state.todos, { id: Date.now(), text: action.payload, completed: false, }, ], }; case 'TOGGLE_TODO': return { ...state, todos: state.todos.map(todo => todo.id === action.payload ? { ...todo, completed: !todo.completed } : todo ), }; case 'DELETE_TODO': return { ...state, todos: state.todos.filter(todo => todo.id !== action.payload), }; case 'SET_FILTER': return { ...state, filter: action.payload, }; default: return state; } } // 创建 store const store = createStore(todoReducer);

📊 Redux 中间件

// 中间件：在 action 到达 reducer 之前处理 action // 常用于处理异步操作 import { createStore, applyMiddleware } from 'redux'; import thunk from 'redux-thunk'; // Thunk 中间件：允许 dispatch 函数 const fetchUser = (userId: string) => async (dispatch) => { dispatch({ type: 'FETCH_USER_START' }); try { const response = await fetch(`/api/users/${userId}`); const user = await response.json(); dispatch({ type: 'FETCH_USER_SUCCESS', payload: user }); } catch (error) { dispatch({ type: 'FETCH_USER_ERROR', payload: error.message }); } }; // 使用中间件 const store = createStore(userReducer, applyMiddleware(thunk)); // 派发异步 action store.dispatch(fetchUser('123')); // 自定义中间件 const logger = store => next => action => { console.log('Dispatching:', action); const result = next(action); console.log('New state:', store.getState()); return result; }; // 应用多个中间件 const store = createStore( reducer, applyMiddleware(thunk, logger) );
