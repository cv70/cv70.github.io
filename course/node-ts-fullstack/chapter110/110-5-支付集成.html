🔧 支付宝集成

// alipay.service.ts import AlipaySdk from 'alipay-sdk'; @Injectable() export class AlipayService { private alipaySdk: AlipaySdk; constructor() { this.alipaySdk = new AlipaySdk({ appId: process.env.ALIPAY_APP_ID, privateKey: process.env.ALIPAY_PRIVATE_KEY, alipayPublicKey: process.env.ALIPAY_PUBLIC_KEY, gateway: 'https://openapi.alipay.com/gateway.do', }); } // 创建支付 async createPayment(order: Order): Promise { const result = await this.alipaySdk.exec('alipay.trade.page.pay', { notify_url: process.env.ALIPAY_NOTIFY_URL, return_url: process.env.ALIPAY_RETURN_URL, bizContent: { out_trade_no: order.id, total_amount: order.payAmount.toFixed(2), subject: `订单${order.id}`, product_code: 'FAST_INSTANT_TRADE_PAY', }, }); return result; } // 处理异步通知 async handleNotify(params: any): Promise { // 1. 验证签名 const signValid = this.alipaySdk.checkNotifySign(params); if (!signValid) { console.error('Alipay notify sign invalid'); return false; } // 2. 验证金额 const orderId = params.out_trade_no; const tradeStatus = params.trade_status; const totalAmount = parseFloat(params.total_amount); const order = await this.orderService.findById(orderId); if (!order || order.payAmount !== totalAmount) { console.error('Amount mismatch'); return false; } // 3. 处理支付结果 if (tradeStatus === 'TRADE_SUCCESS' || tradeStatus === 'TRADE_FINISHED') { await this.orderService.handlePaymentSuccess( params.trade_no, orderId, 'alipay' ); } return true; } // 退款 async refund(orderId: string, amount: number, reason: string) { const result = await this.alipaySdk.exec('alipay.trade.refund', { bizContent: { out_trade_no: orderId, refund_amount: amount.toFixed(2), refund_reason: reason, }, }); return result; } }

🔀 微信支付集成

// wechat-pay.service.ts import { Wechatpay } from 'wechatpay-node-v3'; @Injectable() export class WechatPayService { private pay: Wechatpay; constructor() { this.pay = new Wechatpay({ appid: process.env.WECHAT_APP_ID, mchid: process.env.WECHAT_MCH_ID, serial_no: process.env.WECHAT_SERIAL_NO, privateKey: process.env.WECHAT_PRIVATE_KEY, apiv3_private_key: process.env.WECHAT_API_V3_KEY, }); } // JSAPI支付（小程序/公众号） async createJsapiPayment(order: Order, openid: string) { const result = await this.pay.transactions_jsapi({ description: `订单${order.id}`, out_trade_no: order.id, notify_url: process.env.WECHAT_NOTIFY_URL, amount: { total: Math.round(order.payAmount * 100), // 分 }, payer: { openid, }, }); return result; } // Native支付（扫码支付） async createNativePayment(order: Order) { const result = await this.pay.transactions_native({ description: `订单${order.id}`, out_trade_no: order.id, notify_url: process.env.WECHAT_NOTIFY_URL, amount: { total: Math.round(order.payAmount * 100), }, }); return result.code_url; // 返回二维码链接 } // 处理异步通知 async handleNotify(headers: any, body: any): Promise { // 验证签名 const { id, create_time, resource_type, event_type, resource } = await this.pay.decipher_notify(headers, body); // 处理支付成功事件 if (event_type === 'TRANSACTION.SUCCESS') { const { out_trade_no, transaction_id, amount } = resource; await this.orderService.handlePaymentSuccess( transaction_id, out_trade_no, 'wechat' ); } return { code: 'SUCCESS', message: '成功' }; } // 退款 async refund(orderId: string, amount: number, reason: string) { const result = await this.pay.refunds({ out_trade_no: orderId, out_refund_no: `refund_${Date.now()}`, amount: { total: Math.round(amount * 100), refund: Math.round(amount * 100), }, reason, }); return result; } }
