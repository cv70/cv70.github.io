15.5 文件流操作

高效处理大文件的流式读写

🎯 学习目标

  • 理解流的概念和优势
  • 掌握 createReadStream 和 createWriteStream
  • 学会使用管道 pipe
  • 了解流的错误处理

📚 为什么使用流

// 传统方式读取大文件的问题 // 1. 全部加载到内存 // 2. 内存占用大 // 3. 等待时间长 // 不好的方式 const fs = require('fs'); const data = fs.readFileSync('large-file.mp4'); // 占用大量内存 fs.writeFileSync('copy.mp4', data); // 使用流的方式 // 1. 分块读取 // 2. 内存占用小 // 3. 可以边读边处理 // 好的方式 const readStream = fs.createReadStream('large-file.mp4'); const writeStream = fs.createWriteStream('copy.mp4'); readStream.pipe(writeStream); // 内存对比 // 传统方式: 文件大小 1GB = 内存占用 1GB+ // 流方式: 固定块大小，内存占用约 64KB（默认 highWaterMark）
💡
流的优势

流式处理将大文件分成小块处理，显著降低内存使用，适合处理大文件或实时数据。

🔧 可读流

const fs = require('fs'); // 创建可读流 const readStream = fs.createReadStream('file.txt', { encoding: 'utf8', // 编码 highWaterMark: 64 * 1024, // 缓冲区大小（默认 64KB） start: 0, // 开始位置 end: 1000 // 结束位置 }); // 事件监听 readStream.on('open', () => { console.log('文件打开'); }); readStream.on('data', (chunk) => { console.log('读取到数据:', chunk.length, '字节'); // 处理数据块 }); readStream.on('end', () => { console.log('读取完成'); }); readStream.on('close', () => { console.log('文件关闭'); }); readStream.on('error', (err) => { console.error('读取错误:', err); }); // 暂停和恢复 readStream.pause(); setTimeout(() => { readStream.resume(); }, 1000); // 手动读取 readStream.on('readable', () => { let chunk; while (null !== (chunk = readStream.read())) { console.log('读取:', chunk.length); } });

📦 可写流

const fs = require('fs'); // 创建可写流 const writeStream = fs.createWriteStream('output.txt', { encoding: 'utf8', highWaterMark: 16 * 1024 }); // 写入数据 writeStream.write('第一行\n'); writeStream.write('第二行\n'); writeStream.write('第三行\n'); // 结束写入 writeStream.end('最后一行\n'); // 事件监听 writeStream.on('finish', () => { console.log('写入完成'); }); writeStream.on('error', (err) => { console.error('写入错误:', err); }); // 检查是否可以继续写入 const canContinue = writeStream.write('大量数据'); if (!canContinue) { // 缓冲区已满，等待 drain 事件 writeStream.once('drain', () => { console.log('缓冲区已清空，可以继续写入'); }); } // 追加写入 const appendStream = fs.createWriteStream('log.txt', { flags: 'a' }); appendStream.write(new Date().toISOString() + '\n'); appendStream.end();

⚙️ 管道操作

const fs = require('fs'); const zlib = require('zlib'); const crypto = require('crypto'); // 基本管道：复制文件 const readStream = fs.createReadStream('source.txt'); const writeStream = fs.createWriteStream('dest.txt'); readStream.pipe(writeStream); // 链式管道：压缩文件 fs.createReadStream('source.txt') .pipe(zlib.createGzip()) .pipe(fs.createWriteStream('source.txt.gz')); // 解压文件 fs.createReadStream('source.txt.gz') .pipe(zlib.createGunzip()) .pipe(fs.createWriteStream('source-copy.txt')); // 加密后压缩 fs.createReadStream('secret.txt') .pipe(crypto.createCipher('aes192', 'password')) .pipe(zlib.createGzip()) .pipe(fs.createWriteStream('secret.txt.gz')); // HTTP 响应流 const http = require('http'); http.createServer((req, res) => { const stream = fs.createReadStream('video.mp4'); res.writeHead(200, { 'Content-Type': 'video/mp4' }); stream.pipe(res); }).listen(3000); // 管道错误处理 const pipeline = require('stream').pipeline; pipeline( fs.createReadStream('source.txt'), zlib.createGzip(), fs.createWriteStream('dest.txt.gz'), (err) => { if (err) { console.error('管道失败:', err); } else { console.log('管道成功'); } } );

🔄 实际应用示例

const fs = require('fs'); const readline = require('readline'); // 逐行读取大文件 async function processLineByLine() { const fileStream = fs.createReadStream('large-file.txt'); const rl = readline.createInterface({ input: fileStream, crlfDelay: Infinity }); for await (const line of rl) { // 处理每一行 console.log(`行: ${line}`); } } // 复制目录 function copyDirectory(src, dest) { fs.mkdirSync(dest, { recursive: true }); const entries = fs.readdirSync(src, { withFileTypes: true }); for (const entry of entries) { const srcPath = path.join(src, entry.name); const destPath = path.join(dest, entry.name); if (entry.isDirectory()) { copyDirectory(srcPath, destPath); } else { fs.createReadStream(srcPath).pipe(fs.createWriteStream(destPath)); } } } // 进度显示 function copyWithProgress(src, dest) { const stats = fs.statSync(src); const totalSize = stats.size; let copied = 0; const readStream = fs.createReadStream(src); const writeStream = fs.createWriteStream(dest); readStream.on('data', (chunk) => { copied += chunk.length; const percent = ((copied / totalSize) * 100).toFixed(2); process.stdout.write(`\r复制进度: ${percent}%`); }); readStream.pipe(writeStream); }
