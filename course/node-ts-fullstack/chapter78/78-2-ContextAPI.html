78.2 Context API

React 内置的状态共享方案

🎯 学习目标

  • 深入理解 Context API
  • 掌握 Context 最佳实践
  • 学会拆分 Context 优化性能
  • 了解 Context 的局限性

📚 Context 基础回顾

import { createContext, useContext, useState, ReactNode } from 'react'; // 创建 Context interface User { id: string; name: string; email: string; } interface UserContextType { user: User | null; login: (user: User) => void; logout: () => void; } const UserContext = createContext<UserContextType | undefined>(undefined); // Provider 组件 function UserProvider({ children }: { children: ReactNode }) { const [user, setUser] = useState<User | null>(null); const login = (user: User) => setUser(user); const logout = () => setUser(null); return ( <UserContext.Provider value={{ user, login, logout }}> {children} </UserContext.Provider> ); } // 自定义 Hook function useUser() { const context = useContext(UserContext); if (context === undefined) { throw new Error('useUser must be used within UserProvider'); } return context; } // 使用 function App() { return ( <UserProvider> <Header /> <Main /> </UserProvider> ); } function Header() { const { user, logout } = useUser(); return ( <header> {user ? ( <> <span>{user.name}</span> <button onClick={logout}>Logout</button> </> ) : ( <a href="/login">Login</a> )} </header> ); }

🔧 性能优化：拆分 Context

// 问题：单一 Context 导致不必要的渲染 // 当 user 更新时，所有消费组件都会重新渲染 // 解决：拆分为多个 Context // 1. 状态 Context（数据） // 2. 更新 Context（方法） // 状态 Context const UserStateContext = createContext<User | null>(null); // 更新 Context const UserDispatchContext = createContext<{ login: (user: User) => void; logout: () => void; } | undefined>(undefined); // Provider function UserProvider({ children }: { children: ReactNode }) { const [user, setUser] = useState<User | null>(null); const dispatch = useMemo(() => ({ login: (user: User) => setUser(user), logout: () => setUser(null), }), []); return ( <UserStateContext.Provider value={user}> <UserDispatchContext.Provider value={dispatch}> {children} </UserDispatchContext.Provider> </UserStateContext.Provider> ); } // 自定义 Hooks function useUserState() { return useContext(UserStateContext); } function useUserDispatch() { const context = useContext(UserDispatchContext); if (!context) { throw new Error('useUserDispatch must be used within UserProvider'); } return context; } // 使用：只订阅需要的数据 function UserAvatar() { const user = useUserState(); // 只订阅状态 return <img src={user?.avatar} />; } function LogoutButton() { const { logout } = useUserDispatch(); // 只订阅方法 // user 更新不会导致此组件重新渲染 return <button onClick={logout}>Logout</button>; }

📊 Context 局限性

// Context 的局限性 // 1. 不适合高频更新 // - Context 值更新时，所有消费者都会重新渲染 // - 不适合频繁变化的数据（如鼠标位置） // 2. 不适合复杂状态逻辑 // - 没有内置的状态管理机制 // - 需要手动实现 action、reducer // 3. 调试困难 // - 没有时间旅行调试 // - 状态变化难以追踪 // 4. 不支持选择性订阅 // - 必须订阅整个 Context 值 // - 细粒度更新需要拆分 Context // 高频更新的替代方案 // 1. Zustand：细粒度订阅 const useStore = create((set) => ({ count: 0, increment: () => set((state) => ({ count: state.count + 1 })), })); // 组件可以选择性订阅 function CountDisplay() { const count = useStore((state) => state.count); // 只订阅 count return <div>{count}</div>; } function IncrementButton() { const increment = useStore((state) => state.increment); // 只订阅方法 // count 更新不会导致重新渲染 return <button onClick={increment}>+</button>; }
