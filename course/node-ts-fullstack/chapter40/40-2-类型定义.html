📚 Koa 基本类型

import Koa, { Context, Next, Middleware } from 'koa'; import Router from '@koa/router'; // Context 类型 app.use(async (ctx: Context, next: Next) => { // ctx.request - Koa Request // ctx.response - Koa Response // ctx.state - 状态对象 // ctx.app - Koa 实例 await next(); }); // Middleware 类型 const loggerMiddleware: Middleware = async (ctx, next) => { console.log(`${ctx.method} ${ctx.url}`); await next(); }; app.use(loggerMiddleware);

🔧 扩展 Context 类型

// src/types/index.ts import 'koa'; // 扩展 Context 的 state 属性 declare module 'koa' { interface Context { // 自定义属性 userId?: string; } interface Request { // 扩展请求 body?: any; } } // 扩展 state declare module 'koa' { interface DefaultState { user?: { id: string; username: string; role: string; }; } }
// 使用扩展后的类型 app.use(async (ctx: Context, next: Next) => { // 可以访问扩展的属性 ctx.state.user = { id: '1', username: 'admin', role: 'admin' }; ctx.userId = '1'; await next(); });

👤 用户类型定义

// src/types/user.ts export interface User { id: string; username: string; email: string; role: 'admin' | 'user' | 'guest'; createdAt: Date; updatedAt: Date; } export interface CreateUserDTO { username: string; email: string; password: string; role?: 'admin' | 'user'; } export interface UpdateUserDTO { username?: string; email?: string; password?: string; } export interface UserResponse { id: string; username: string; email: string; role: string; createdAt: string; }

📡 API 响应类型

// src/types/response.ts // 通用 API 响应 export interface ApiResponse { success: boolean; data?: T; error?: string; message?: string; } // 分页响应 export interface PaginatedResponse { success: true; data: T[]; pagination: { page: number; limit: number; total: number; totalPages: number; }; } // 错误响应 export interface ErrorResponse { success: false; error: string; code: string; details?: any; }
// 使用 import { ApiResponse, UserResponse } from '@/types'; router.get('/users/:id', async (ctx: Context) => { const user = await UserService.findById(ctx.params.id); const response: ApiResponse = { success: true, data: user }; ctx.body = response; });

📝 请求体类型

// src/types/request.ts import { RouterContext } from '@koa/router'; // 登录请求 export interface LoginRequest { username: string; password: string; } // 注册请求 export interface RegisterRequest { username: string; email: string; password: string; confirmPassword: string; } // 获取请求体的辅助类型 export type RequestBody = T extends infer U ? U : never; // 使用 router.post('/login', async (ctx: RouterContext) => { const { username, password } = ctx.request.body as LoginRequest; // 类型安全的访问 const user = await AuthService.login(username, password); ctx.body = { success: true, data: user }; });

🔍 查询参数类型

// src/types/query.ts // 分页查询 export interface PaginationQuery { page?: string; limit?: string; sort?: 'asc' | 'desc'; sortBy?: string; } // 用户查询 export interface UserQuery extends PaginationQuery { username?: string; email?: string; role?: string; status?: 'active' | 'inactive'; } // 解析查询参数 function parsePagination(query: PaginationQuery) { return { page: parseInt(query.page || '1', 10), limit: parseInt(query.limit || '10', 10), sort: query.sort || 'desc', sortBy: query.sortBy || 'createdAt' }; } // 使用 router.get('/users', async (ctx: RouterContext) => { const query = ctx.query as UserQuery; const { page, limit, sort, sortBy } = parsePagination(query); const result = await UserService.findAll({ page, limit, sort, sortBy }); ctx.body = result; });

🔗 路由参数类型

📦 泛型控制器类型

// src/types/controller.ts import { RouterContext, Next } from '@koa/router'; // 控制器处理器类型 export type ControllerHandler< Params = any, Query = any, Body = any, Response = any > = ( ctx: RouterContext & { request: { body: Body }; }, next: Next ) => Promise; // 基础控制器类 export abstract class BaseController { protected success(ctx: RouterContext, data: T, message?: string) { ctx.body = { success: true, data, message }; } protected error(ctx: RouterContext, error: string, status = 400) { ctx.status = status; ctx.body = { success: false, error }; } }

💻 完整类型示例

// src/controllers/userController.ts import { RouterContext } from '@koa/router'; import { BaseController } from '@/types/controller'; import { User, CreateUserDTO, UpdateUserDTO } from '@/types/user'; import { ApiResponse, PaginatedResponse } from '@/types/response'; class UserController extends BaseController { // 获取用户列表 async index(ctx: RouterContext) { const { page = 1, limit = 10 } = ctx.query; const result = await UserService.findAll({ page: Number(page), limit: Number(limit) }); const response: PaginatedResponse = { success: true, ...result }; ctx.body = response; } // 创建用户 async create(ctx: RouterContext) { const data = ctx.request.body; const user = await UserService.create(data); this.success(ctx, user, 'User created'); } // 更新用户 async update(ctx: RouterContext<{ id: string }, any, any, UpdateUserDTO>) { const { id } = ctx.params; const data = ctx.request.body; const user = await UserService.update(id, data); this.success(ctx, user); } } export default new UserController();
