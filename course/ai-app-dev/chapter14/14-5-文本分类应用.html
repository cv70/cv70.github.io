🔧 步骤二：构建分类模型

class SentimentClassifier(nn.Module): def __init__(self, vocab_size, embed_dim, hidden_dim, output_dim, n_layers=2, dropout=0.5): super().__init__() # 词嵌入层 self.embedding = nn.Embedding(vocab_size, embed_dim, padding_idx=vocab['']) # 双向LSTM self.lstm = nn.LSTM(embed_dim, hidden_dim, num_layers=n_layers, bidirectional=True, dropout=dropout, batch_first=True) # 注意力层 self.attention = nn.Linear(hidden_dim * 2, 1) # 分类头 self.fc = nn.Sequential( nn.Linear(hidden_dim * 2, hidden_dim), nn.ReLU(), nn.Dropout(dropout), nn.Linear(hidden_dim, output_dim) ) def attention_pooling(self, lstm_out): # 计算注意力权重 weights = torch.softmax(self.attention(lstm_out), dim=1) # 加权求和 context = torch.sum(weights * lstm_out, dim=1) return context def forward(self, text): embedded = self.embedding(text) lstm_out, _ = self.lstm(embedded) # 注意力池化 pooled = self.attention_pooling(lstm_out) return self.fc(pooled)

🏋️ 步骤三：训练与评估

def train_epoch(model, dataloader, optimizer, criterion): model.train() total_loss, total_correct, total = 0, 0, 0 for texts, labels in dataloader: optimizer.zero_grad() predictions = model(texts) loss = criterion(predictions, labels) loss.backward() optimizer.step() total_loss += loss.item() total_correct += (predictions.argmax(1) == labels).sum().item() total += labels.size(0) return total_loss / len(dataloader), total_correct / total def evaluate(model, dataloader, criterion): model.eval() total_loss, total_correct, total = 0, 0, 0 with torch.no_grad(): for texts, labels in dataloader: predictions = model(texts) loss = criterion(predictions, labels) total_loss += loss.item() total_correct += (predictions.argmax(1) == labels).sum().item() total += labels.size(0) return total_loss / len(dataloader), total_correct / total

注意力机制的优势

加入注意力机制后，模型可以自动学习关注文本中的关键词汇。这不仅提升了性能，还提供了可解释性， 我们可以通过可视化注意力权重来理解模型的决策依据。
