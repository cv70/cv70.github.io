15.4 文件监听

监控文件和目录变化

🎯 学习目标

  • 掌握 fs.watch 和 fs.watchFile
  • 理解两种监听方式的区别
  • 学会实现自动重载功能
  • 了解监听的注意事项

📚 fs.watch

const fs = require('fs'); // 监听文件变化（推荐） const watcher = fs.watch('file.txt', (eventType, filename) => { console.log(`事件类型: ${eventType}`); console.log(`文件名: ${filename}`); }); // eventType 可能是: 'rename' 或 'change' // rename: 文件被重命名或删除 // change: 文件内容被修改 // 监听目录 fs.watch('./src', (eventType, filename) => { console.log(`检测到变化: ${filename} - ${eventType}`); }); // 使用选项 fs.watch('file.txt', { persistent: true, // 持续监听（默认 true） recursive: false, // 递归监听子目录（仅 Windows/macOS） encoding: 'utf8' // 文件名编码 }, (eventType, filename) => { console.log(`${filename} 发生了 ${eventType} 事件`); }); // 关闭监听 watcher.close(); // 监听错误事件 watcher.on('error', (err) => { console.error('监听出错:', err); });

🔧 fs.watchFile

const fs = require('fs'); // 监听文件变化（使用轮询） fs.watchFile('file.txt', (curr, prev) => { console.log('当前文件状态:', curr); console.log('之前文件状态:', prev); if (curr.mtime !== prev.mtime) { console.log('文件被修改了'); } if (curr.size !== prev.size) { console.log(`文件大小从 ${prev.size} 变为 ${curr.size}`); } }); // 设置轮询间隔（默认 5007ms） fs.watchFile('file.txt', { interval: 1000 }, (curr, prev) => { console.log('每秒检查一次文件变化'); }); // 停止监听 fs.unwatchFile('file.txt'); // 使用 stat 对象属性 fs.watchFile('file.txt', (curr, prev) => { console.log('修改时间:', curr.mtime); console.log('访问时间:', curr.atime); console.log('文件大小:', curr.size); console.log('文件权限:', curr.mode); });
💡
watch vs watchFile

watch 基于事件，效率高但不稳定；watchFile 基于轮询，稳定但消耗资源。优先使用 watch。

📦 实现自动重载

const fs = require('fs'); const path = require('path'); const { spawn } = require('child_process'); class AutoReload { constructor(entryFile) { this.entryFile = entryFile; this.process = null; this.watcher = null; this.restart(); this.watch(); } start() { this.process = spawn('node', [this.entryFile], { stdio: 'inherit' }); this.process.on('exit', (code) => { if (code !== null) { console.log(`进程退出，代码: ${code}`); } }); } restart() { if (this.process) { this.process.kill(); } console.log('\n正在重启...'); this.start(); } watch() { let debounceTimer; this.watcher = fs.watch('./', { recursive: true }, (eventType, filename) => { if (filename && filename.endsWith('.js')) { // 防抖处理 clearTimeout(debounceTimer); debounceTimer = setTimeout(() => { console.log(`检测到 ${filename} 变化`); this.restart(); }, 100); } }); } close() { if (this.process) this.process.kill(); if (this.watcher) this.watcher.close(); } } // 使用 const reloader = new AutoReload('app.js'); // 退出时清理 process.on('SIGINT', () => { reloader.close(); process.exit(); });

⚙️ 监听配置文件

const fs = require('fs'); const path = require('path'); class ConfigWatcher { constructor(configPath) { this.configPath = configPath; this.config = null; this.listeners = []; this.load(); this.watch(); } load() { try { const data = fs.readFileSync(this.configPath, 'utf8'); this.config = JSON.parse(data); this.notify(); } catch (err) { console.error('配置加载失败:', err); } } watch() { let debounceTimer; fs.watch(this.configPath, (eventType) => { if (eventType === 'change') { clearTimeout(debounceTimer); debounceTimer = setTimeout(() => { console.log('配置文件已更新'); this.load(); }, 100); } }); } onChange(callback) { this.listeners.push(callback); } notify() { this.listeners.forEach(cb => cb(this.config)); } get() { return this.config; } } // 使用 const configWatcher = new ConfigWatcher('./config.json'); configWatcher.onChange((config) => { console.log('配置已更新:', config); }); // 获取当前配置 console.log(configWatcher.get());

⚠️ 注意事项

⚠️
  • • fs.watch 在不同系统上行为可能不同
  • • filename 参数在某些系统可能为 null
  • • 一次修改可能触发多次事件
  • • 使用防抖避免频繁触发
  • • 生产环境建议使用 chokidar 库

📝 本节小结

✅
  • • fs.watch 基于事件，效率高
  • • fs.watchFile 基于轮询，更稳定
  • • 使用防抖处理频繁变化
  • • 生产环境推荐 chokidar 库
