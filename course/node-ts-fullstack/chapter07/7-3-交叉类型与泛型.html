7.3 交叉类型与泛型

结合泛型实现灵活的类型组合

🎯 学习目标

  • 掌握交叉类型与泛型的结合使用
  • 学会创建高阶类型工具
  • 理解类型推断与交叉类型
  • 了解 mixins 模式的类型实现

🔧 泛型函数中的交叉类型

// 合并两个对象的泛型函数 function merge(a: T, b: U): T & U { return { ...a, ...b }; } const defaults = { theme: 'light', lang: 'en' }; const userConfig = { lang: 'zh', debug: true }; const config = merge(defaults, userConfig); // 类型推断为 { theme: string; lang: string; } & { lang: string; debug: boolean; } // 扩展对象类型 function withTimestamp(obj: T): T & { timestamp: Date } { return { ...obj, timestamp: new Date() }; } const message = withTimestamp({ text: 'Hello', author: 'Alice' }); // 类型：{ text: string; author: string; } & { timestamp: Date } console.log(message.text); // string console.log(message.timestamp); // Date // 添加 ID 属性 function withId(obj: T, id: number): T & { id: number } { return { ...obj, id }; } const entity = withId({ name: 'Product', price: 99.9 }, 1); // 类型：{ name: string; price: number; } & { id: number }

🔀 Mixins 模式

// 使用交叉类型实现 mixins type Constructor = new (...args: any[]) => T; // 时间戳 mixin function Timestamped(Base: TBase) { return class extends Base { timestamp = new Date(); }; } // 可序列化 mixin function Serializable(Base: TBase) { return class extends Base { serialize() { return JSON.stringify(this); } }; } // 基础类 class User { constructor(public name: string, public email: string) {} } // 应用 mixins const TimestampedUser = Timestamped(User); const FullUser = Serializable(TimestampedUser); const user = new FullUser('Alice', 'alice@example.com'); console.log(user.timestamp); // Date console.log(user.serialize()); // JSON string // 类型定义辅助 type MixinType = T & M; interface TimestampMixin { timestamp: Date; } interface SerializableMixin { serialize(): string; } type EnhancedUser = User & TimestampMixin & SerializableMixin;

📚 条件类型与交叉类型

// 在条件类型中使用交叉类型 type OptionalKeys = { [K in keyof T]-?: {} extends Pick ? K : never }[keyof T]; // 扩展类型工具 type Extend = T & U; type Base = { id: number }; type WithName = Extend; // { id: number; name: string } // 条件性添加属性 type AddOptional = T & { [P in K]?: V }; type ConfigWithDebug = AddOptional<{ port: number }, 'debug', boolean>; // { port: number; debug?: boolean }

最佳实践

结合泛型和交叉类型可以创建灵活的类型工具，但要避免过度复杂的类型定义，保持代码可读性。
