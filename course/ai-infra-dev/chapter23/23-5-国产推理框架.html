💻 使用示例

# MindSpore推理示例
import mindspore as ms
from mindspore import Model, context
import numpy as np

# 配置运行环境
context.set_context(mode=context.GRAPH_MODE,
                   device_target="Ascend")

# 加载模型
def mindspore_infer(model_path, input_data):
    """
    MindSpore推理
    """
    # 加载模型
    net = MyNet()  # 定义网络
    param_dict = ms.load_checkpoint(model_path)
    ms.load_param_into_net(net, param_dict)

    # 创建模型
    model = Model(net)

    # 推理
    output = model.predict(ms.Tensor(input_data))
    return output.asnumpy()

# PaddlePaddle推理示例
import paddle
from paddle.inference import Config, create_predictor

def paddle_infer(model_path, input_data):
    """
    PaddlePaddle推理
    """
    # 配置推理
    config = Config(model_path)

    # 启用GPU
    config.enable_use_gpu(memory_pool_init_size_mb=1000,
                         device_id=0)

    # 启用TensorRT
    config.enable_tensorrt_engine(
        workspace_size=1 << 30,
        max_batch_size=32,
        min_subgraph_size=3
    )

    # 创建预测器
    predictor = create_predictor(config)

    # 获取输入输出
    input_names = predictor.get_input_names()
    output_names = predictor.get_output_names()

    # 设置输入
    input_tensor = predictor.get_input_handle(input_names[0])
    input_tensor.copy_from_cpu(input_data)

    # 执行推理
    predictor.run()

    # 获取输出
    output_tensor = predictor.get_output_handle(output_names[0])
    output = output_tensor.copy_to_cpu()

    return output

# MNN推理示例（Python）
import MNN

def mnn_infer(mnn_path, input_data):
    """
    MNN推理
    """
    # 创建推理器
    interpreter = MNN.Interpreter(mnn_path)
    session = interpreter.createSession()

    # 获取输入输出
    input_tensor = interpreter.getSessionInput(session)
    output_tensor = interpreter.getSessionOutput(session)

    # 设置输入
    tmp = MNN.Tensor(
        MNN.Halide_Type_Float,
        input_data.shape,
        input_data,
        MNN.Tensor_DimensionType_Caffe
    )
    input_tensor.copyFromHostTensor(tmp)

    # 执行推理
    interpreter.runSession(session)

    # 获取输出
    output = output_tensor.getData()
    return output

# Paddle Lite移动端推理示例
from paddlelite import Place, CxxConfig, Predictor

def paddle_lite_infer(model_dir, input_data):
    """
    Paddle Lite推理
    """
    # 配置推理
    place = Place(TargetType.ARM, PrecisionType.FP32)
    config = CxxConfig()
    config.set_model_dir(model_dir)
    config.set_valid_places(place)

    # 创建预测器
    predictor = Predictor.create_predictor(config)

    # 设置输入
    input_tensor = predictor.get_input(0)
    input_tensor.from_numpy(input_data)

    # 执行推理
    predictor.run()

    # 获取输出
    output_tensor = predictor.get_output(0)
    output = output_tensor.numpy()

    return output
