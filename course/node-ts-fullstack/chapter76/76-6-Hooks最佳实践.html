📚 Hooks 使用规则

// 规则 1：只在顶层调用 Hooks // 错误示例 function BadComponent({ isLoggedIn }) { if (isLoggedIn) { const [user, setUser] = useState(null); // 错误！ } } // 正确示例 function GoodComponent({ isLoggedIn }) { const [user, setUser] = useState(null); if (isLoggedIn) { // 使用 user } } // 规则 2：只在 React 函数中调用 Hooks // 错误示例 function badHelper() { const [value, setValue] = useState(0); // 错误！ } // 正确示例 function GoodComponent() { const [value, setValue] = useState(0); // 正确 } // 自定义 Hook 中可以调用其他 Hooks function useCustomHook() { const [value, setValue] = useState(0); // 正确 return [value, setValue]; } // 规则 3：使用 ESLint 插件 // 安装：npm install eslint-plugin-react-hooks --save-dev // .eslintrc.json { "plugins": ["react-hooks"], "rules": { "react-hooks/rules-of-hooks": "error", "react-hooks/exhaustive-deps": "warn" } }

🔧 性能优化

// useMemo：缓存计算结果 import { useMemo } from 'react'; function ExpensiveComponent({ items, filter }) { // 每次渲染都重新计算 // const filtered = items.filter(item => item.includes(filter)); // 只在依赖变化时重新计算 const filtered = useMemo(() => { console.log('Filtering...'); return items.filter(item => item.includes(filter)); }, [items, filter]); return <ul>{filtered.map(item => <li key={item}>{item}</li>)}</ul>; } // useCallback：缓存函数引用 import { useCallback } from 'react'; function ParentComponent() { const [count, setCount] = useState(0); // 每次渲染都创建新函数 // const handleClick = () => console.log('clicked'); // 缓存函数引用 const handleClick = useCallback(() => { console.log('clicked'); }, []); return <ChildComponent onClick={handleClick} />; } // 配合 React.memo const ChildComponent = React.memo(function ChildComponent({ onClick }) { console.log('Child rendered'); return <button onClick={onClick}>Click</button>; }); // 使用场景 // 1. 传递给使用 React.memo 的子组件 // 2. 作为其他 Hook 的依赖 // 3. 计算开销大的操作 // 不要过度优化 // 简单计算不需要 useMemo function SimpleComponent({ a, b }) { const sum = a + b; // 简单计算，不需要 useMemo return <div>{sum}</div>; }

📊 依赖管理

// 正确声明依赖 function UserProfile({ userId }) { const [user, setUser] = useState(null); useEffect(() => { fetchUser(userId).then(setUser); }, [userId]); // 正确 return <div>{user?.name}</div>; } // 避免遗漏依赖 function SearchResults({ query, pageSize }) { const [results, setResults] = useState([]); useEffect(() => { search(query, pageSize).then(setResults); }, [query]); // 错误：缺少 pageSize // 正确 useEffect(() => { search(query, pageSize).then(setResults); }, [query, pageSize]); } // 使用函数式更新避免依赖 function Counter() { const [count, setCount] = useState(0); useEffect(() => { const timer = setInterval(() => { setCount(c => c + 1); // 函数式更新 }, 1000); return () => clearInterval(timer); }, []); // 不需要 count 依赖 } // 使用 useRef 存储可变值 function useLatest<T>(value: T) { const ref = useRef(value); ref.current = value; return ref; } function Component({ onChange }) { const latestOnChange = useLatest(onChange); useEffect(() => { const handler = () => latestOnChange.current(); window.addEventListener('resize', handler); return () => window.removeEventListener('resize', handler); }, []); // 不需要 onChange 依赖 }
