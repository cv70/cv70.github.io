79.3 Hooks 类型

React Hooks 的 TypeScript 支持

📚 useState 类型

import { useState } from 'react'; // 类型推断 const [count, setCount] = useState(0); // number const [name, setName] = useState(''); // string const [isActive, setIsActive] = useState(false); // boolean // 显式类型 const [user, setUser] = useState<User | null>(null); const [items, setItems] = useState<string[]>([]); const [map, setMap] = useState<Map<string, number>>(new Map()); // 联合类型 const [status, setStatus] = useState<'idle' | 'loading' | 'success' | 'error'>('idle'); // 对象类型 interface User { id: string; name: string; email: string; } const [user, setUser] = useState<User>({ id: '', name: '', email: '', }); // 惰性初始化类型 const [data, setData] = useState<Data>(() => { // 类型推断为 Data return computeExpensiveInitialValue(); }); // 更新函数类型 const updateUser = (updates: Partial<User>) => { setUser(prev => ({ ...prev, ...updates })); }; // 错误示例 const [value, setValue] = useState<string | null>('initial'); // setValue(123); // 错误：number 不能赋值给 string | null setValue(null); // 正确 setValue('new'); // 正确 interface User { id: string; name: string; }

🔧 useRef 类型

import { useRef, RefObject, MutableRefObject } from 'react'; // DOM 元素引用 const inputRef = useRef<HTMLInputElement>(null); const divRef = useRef<HTMLDivElement>(null); const videoRef = useRef<HTMLVideoElement>(null); function Form() { const inputRef = useRef<HTMLInputElement>(null); const focusInput = () => { // 需要检查 null inputRef.current?.focus(); }; return ( <> <input ref={inputRef} type="text" /> <button onClick={focusInput}>Focus</button> </> ); } // 可变值引用 const countRef = useRef<number>(0); const timerRef = useRef<NodeJS.Timeout | null>(null); const previousValueRef = useRef<string | undefined>(); function Timer() { const timerRef = useRef<NodeJS.Timeout | null>(null); useEffect(() => { timerRef.current = setInterval(() => { console.log('tick'); }, 1000); return () => { if (timerRef.current) { clearInterval(timerRef.current); } }; }, []); return <div>Timer running</div>; } // 区分 RefObject 和 MutableRefObject // RefObject<T>：{ readonly current: T | null }，用于 DOM 引用 // MutableRefObject<T>：{ current: T }，用于可变值 // forwardRef 类型 import { forwardRef } from 'react'; interface InputProps { label?: string; } const Input = forwardRef<HTMLInputElement, InputProps>( ({ label, ...props }, ref) => { return ( <div> {label && <label>{label}</label>} <input ref={ref} {...props} /> </div> ); } ); // 使用 function Form() { const inputRef = useRef<HTMLInputElement>(null); return <Input ref={inputRef} label="Username" />; }

📊 自定义 Hook 类型

// 自定义 Hook 类型定义 interface UseFormReturn<T> { values: T; handleChange: (e: ChangeEvent<HTMLInputElement>) => void; reset: () => void; setValues: (values: T | ((prev: T) => T)) => void; } function useForm<T extends Record<string, unknown>>( initialValues: T ): UseFormReturn<T> { const [values, setValues] = useState<T>(initialValues); const handleChange = (e: ChangeEvent<HTMLInputElement>) => { const { name, value, type, checked } = e.target; setValues(prev => ({ ...prev, [name]: type === 'checkbox' ? checked : value, })); }; const reset = () => setValues(initialValues); return { values, handleChange, reset, setValues }; } // 使用 interface FormData { username: string; email: string; subscribe: boolean; } function MyForm() { const { values, handleChange, reset } = useForm<FormData>({ username: '', email: '', subscribe: false, }); // values 类型推断为 FormData return ( <form> <input name="username" value={values.username} onChange={handleChange} /> </form> ); } // 返回元组类型（类似 useState） function useToggle(initial = false): [boolean, () => void] { const [value, setValue] = useState(initial); const toggle = () => setValue(v => !v); return [value, toggle]; } // 使用 const [isOpen, toggleOpen] = useToggle();
