序列建模概述

序列建模是深度学习中最具挑战性的任务之一。无论是预测股票走势、生成文本，还是语音识别， 都需要处理具有时间依赖关系的数据。本节将通过一个完整的实战项目，带你掌握序列建模的核心技术。

💻 步骤一：数据预处理

import torch import numpy as np from sklearn.preprocessing import MinMaxScaler class StockDataset(torch.utils.data.Dataset): def __init__(self, data, seq_length=60): self.data = data self.seq_length = seq_length self.scaler = MinMaxScaler() self.scaled_data = self.scaler.fit_transform(data.reshape(-1, 1)) def __len__(self): return len(self.scaled_data) - self.seq_length def __getitem__(self, idx): # 获取序列窗口 x = self.scaled_data[idx:idx + self.seq_length] # 预测下一个值 y = self.scaled_data[idx + self.seq_length] return torch.FloatTensor(x), torch.FloatTensor(y) # 加载和准备数据 def prepare_data(prices, train_ratio=0.8): train_size = int(len(prices) * train_ratio) train_data = prices[:train_size] test_data = prices[train_size:] train_dataset = StockDataset(train_data) test_dataset = StockDataset(test_data) return train_dataset, test_dataset

🔧 步骤二：构建LSTM模型

class StockPredictor(torch.nn.Module): def __init__(self, input_size=1, hidden_size=64, num_layers=2, dropout=0.2): super().__init__() self.hidden_size = hidden_size self.num_layers = num_layers # LSTM层 self.lstm = torch.nn.LSTM( input_size=input_size, hidden_size=hidden_size, num_layers=num_layers, batch_first=True, dropout=dropout ) # 全连接层 self.fc1 = torch.nn.Linear(hidden_size, 32) self.relu = torch.nn.ReLU() self.fc2 = torch.nn.Linear(32, 1) self.dropout = torch.nn.Dropout(dropout) def forward(self, x): # LSTM处理 lstm_out, _ = self.lstm(x) # 取最后一个时间步 out = lstm_out[:, -1, :] # 全连接预测 out = self.dropout(self.relu(self.fc1(out))) out = self.fc2(out) return out

🏋️ 步骤三：训练循环

def train_model(model, train_loader, epochs=100, lr=0.001): criterion = torch.nn.MSELoss() optimizer = torch.optim.Adam(model.parameters(), lr=lr) scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau( optimizer, patience=5, factor=0.5 ) model.train() for epoch in range(epochs): total_loss = 0 for batch_x, batch_y in train_loader: optimizer.zero_grad() predictions = model(batch_x) loss = criterion(predictions.squeeze(), batch_y) loss.backward() # 梯度裁剪防止爆炸 torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0) optimizer.step() total_loss += loss.item() avg_loss = total_loss / len(train_loader) scheduler.step(avg_loss) if (epoch + 1) % 10 == 0: print(f'Epoch [{epoch+1}/{epochs}], Loss: {avg_loss:.6f}')

📈 步骤四：预测与评估

def evaluate_and_predict(model, test_dataset, scaler): model.eval() predictions = [] actuals = [] with torch.no_grad(): for i in range(len(test_dataset)): x, y = test_dataset[i] x = x.unsqueeze(0) # 添加batch维度 pred = model(x) predictions.append(pred.item()) actuals.append(y.item()) # 反归一化 predictions = scaler.inverse_transform( np.array(predictions).reshape(-1, 1) ) actuals = scaler.inverse_transform( np.array(actuals).reshape(-1, 1) ) # 计算评估指标 mse = np.mean((predictions - actuals) ** 2) mae = np.mean(np.abs(predictions - actuals)) rmse = np.sqrt(mse) print(f'MSE: {mse:.4f}, MAE: {mae:.4f}, RMSE: {rmse:.4f}') return predictions, actuals

⚖️ 序列填充技术对比

技术 描述 适用场景
Zero Padding 用0填充短序列 简单快速，适合均匀分布长度
PackedSequence 打包变长序列，跳过填充 PyTorch推荐，节省计算
Masking 使用掩码标记有效位置 灵活，适合复杂模型
Truncation 截断超长序列 固定长度需求

💡
注意事项

处理变长序列时，务必使用PackedSequence配合pack_padded_sequence和pad_packed_sequence， 这样可以避免在填充位置进行无效计算，显著提升训练效率。
