📝 审计日志管理

# 审计日志系统 import logging import json from datetime import datetime from typing import Dict, Any class AuditLogger: def __init__(self, log_file='audit.log'): self.logger = logging.getLogger('AuditLogger') self.logger.setLevel(logging.INFO) # 文件处理器 file_handler = logging.FileHandler(log_file) formatter = logging.Formatter('%(asctime)s - %(levelname)s - %(message)s') file_handler.setFormatter(formatter) self.logger.addHandler(file_handler) def log_event(self, event_type: str, user_id: str, details: Dict[str, Any]): """ 记录审计事件 """ log_entry = { 'timestamp': datetime.utcnow().isoformat(), 'event_type': event_type, 'user_id': user_id, 'details': details } self.logger.info(json.dumps(log_entry)) def log_access(self, user_id: str, resource: str, action: str, success: bool): """ 记录访问事件 """ details = { 'resource': resource, 'action': action, 'success': success } self.log_event('ACCESS', user_id, details) def log_model_access(self, user_id: str, model_id: str, action: str): """ 记录模型访问 """ details = { 'model_id': model_id, 'action': action } self.log_event('MODEL_ACCESS', user_id, details) def log_data_access(self, user_id: str, data_id: str, action: str): """ 记录数据访问 """ details = { 'data_id': data_id, 'action': action } self.log_event('DATA_ACCESS', user_id, details) def log_security_event(self, event_type: str, details: Dict[str, Any]): """ 记录安全事件 """ self.log_event('SECURITY', 'system', details) # 使用示例 audit_logger = AuditLogger() # 记录用户访问 audit_logger.log_access('user123', '/api/model/train', 'POST', True) # 记录模型访问 audit_logger.log_model_access('user123', 'model_001', 'inference') # 记录数据访问 audit_logger.log_data_access('user123', 'dataset_001', 'read') # 记录安全事件 audit_logger.log_security_event('FAILED_LOGIN', { 'user_id': 'user456', 'ip_address': '192.168.1.100', 'attempt': 3 })

🔍 安全监控

# 安全监控系统 import psutil import time from collections import defaultdict from datetime import datetime, timedelta class SecurityMonitor: def __init__(self): self.failed_login_attempts = defaultdict(list) self.unusual_access_patterns = defaultdict(list) self.resource_usage_history = defaultdict(list) def monitor_failed_logins(self, user_id: str, ip_address: str): """ 监控失败登录 """ now = datetime.utcnow() # 记录失败尝试 self.failed_login_attempts[user_id].append({ 'timestamp': now, 'ip_address': ip_address }) # 清理旧记录 self.failed_login_attempts[user_id] = [ attempt for attempt in self.failed_login_attempts[user_id] if now - attempt['timestamp'] < timedelta(minutes=15) ] # 检查是否超过阈值 if len(self.failed_login_attempts[user_id]) >= 5: raise SecurityException(f"Multiple failed login attempts for user {user_id}") def monitor_unusual_access(self, user_id: str, resource: str): """ 监控异常访问 """ now = datetime.utcnow() # 记录访问 self.unusual_access_patterns[user_id].append({ 'timestamp': now, 'resource': resource }) # 清理旧记录 self.unusual_access_patterns[user_id] = [ access for access in self.unusual_access_patterns[user_id] if now - access['timestamp'] < timedelta(hours=1) ] # 检查访问频率 recent_accesses = self.unusual_access_patterns[user_id] if len(recent_accesses) > 1000: raise SecurityException(f"Unusual access pattern detected for user {user_id}") def monitor_resource_usage(self, user_id: str, resource_type: str, usage: float): """ 监控资源使用 """ now = datetime.utcnow() # 记录使用情况 self.resource_usage_history[user_id].append({ 'timestamp': now, 'resource_type': resource_type, 'usage': usage }) # 清理旧记录 self.resource_usage_history[user_id] = [ record for record in self.resource_usage_history[user_id] if now - record['timestamp'] < timedelta(hours=24) ] # 检查异常使用 recent_usage = [ record['usage'] for record in self.resource_usage_history[user_id] if record['resource_type'] == resource_type ] if recent_usage: avg_usage = sum(recent_usage) / len(recent_usage) if usage > avg_usage * 10: # 超过平均值10倍 raise SecurityException(f"Unusual resource usage detected for user {user_id}") def monitor_system_resources(self): """ 监控系统资源 """ # CPU使用率 cpu_percent = psutil.cpu_percent(interval=1) if cpu_percent > 90: audit_logger.log_security_event('HIGH_CPU_USAGE', { 'cpu_percent': cpu_percent }) # 内存使用率 memory = psutil.virtual_memory() if memory.percent > 90: audit_logger.log_security_event('HIGH_MEMORY_USAGE', { 'memory_percent': memory.percent }) # 磁盘使用率 disk = psutil.disk_usage('/') if disk.percent > 90: audit_logger.log_security_event('HIGH_DISK_USAGE', { 'disk_percent': disk.percent }) # 定期监控任务 def monitoring_task(): monitor = SecurityMonitor() while True: try: monitor.monitor_system_resources() except Exception as e: print(f"Monitoring error: {e}") time.sleep(60) # 每分钟检查一次

📊 审计报告

# 审计报告生成 from datetime import datetime, timedelta from typing import List, Dict import pandas as pd class AuditReport: def __init__(self, audit_logger): self.audit_logger = audit_logger def generate_daily_report(self, date: datetime) -> Dict: """ 生成每日审计报告 """ start_date = date.replace(hour=0, minute=0, second=0, microsecond=0) end_date = start_date + timedelta(days=1) # 读取日志 audit_events = self._read_audit_logs(start_date, end_date) # 统计数据 report = { 'date': date.strftime('%Y-%m-%d'), 'total_events': len(audit_events), 'access_events': self._count_events(audit_events, 'ACCESS'), 'model_access_events': self._count_events(audit_events, 'MODEL_ACCESS'), 'data_access_events': self._count_events(audit_events, 'DATA_ACCESS'), 'security_events': self._count_events(audit_events, 'SECURITY'), 'top_users': self._get_top_users(audit_events), 'top_resources': self._get_top_resources(audit_events), 'failed_logins': self._get_failed_logins(audit_events) } return report def _read_audit_logs(self, start_date: datetime, end_date: datetime) -> List[Dict]: """ 读取审计日志 """ # 实际实现中从日志文件或数据库读取 pass return [] def _count_events(self, events: List[Dict], event_type: str) -> int: """ 统计特定类型事件 """ return len([e for e in events if e['event_type'] == event_type]) def _get_top_users(self, events: List[Dict], top_n: int = 10) -> List[Dict]: """ 获取活跃用户 """ user_counts = defaultdict(int) for event in events: user_id = event['user_id'] user_counts[user_id] += 1 sorted_users = sorted(user_counts.items(), key=lambda x: x[1], reverse=True) return [{'user_id': user, 'count': count} for user, count in sorted_users[:top_n]] def _get_top_resources(self, events: List[Dict], top_n: int = 10) -> List[Dict]: """ 获取热门资源 """ resource_counts = defaultdict(int) for event in events: if 'details' in event and 'resource' in event['details']: resource = event['details']['resource'] resource_counts[resource] += 1 sorted_resources = sorted(resource_counts.items(), key=lambda x: x[1], reverse=True) return [{'resource': resource, 'count': count} for resource, count in sorted_resources[:top_n]] def _get_failed_logins(self, events: List[Dict]) -> List[Dict]: """ 获取失败登录 """ failed_logins = [] for event in events: if event['event_type'] == 'SECURITY' and 'FAILED_LOGIN' in event['details']: failed_logins.append(event['details']['FAILED_LOGIN']) return failed_logins def generate_security_report(self) -> Dict: """ 生成安全报告 """ today = datetime.now() yesterday = today - timedelta(days=1) # 生成报告 daily_report = self.generate_daily_report(yesterday) # 安全指标 security_metrics = { 'security_score': self._calculate_security_score(daily_report), 'risk_level': self._assess_risk_level(daily_report), 'recommendations': self._generate_recommendations(daily_report) } return { 'daily_report': daily_report, 'security_metrics': security_metrics } def _calculate_security_score(self, report: Dict) -> float: """ 计算安全评分 """ # 简单的评分逻辑 base_score = 100.0 # 扣分项 deductions = 0 deductions += len(report['failed_logins']) * 5 deductions += len([e for e in report.get('security_events', [])]) * 10 score = max(0, base_score - deductions) return score def _assess_risk_level(self, report: Dict) -> str: """ 评估风险等级 """ score = self._calculate_security_score(report) if score >= 80: return 'LOW' elif score >= 60: return 'MEDIUM' elif score >= 40: return 'HIGH' else: return 'CRITICAL' def _generate_recommendations(self, report: Dict) -> List[str]: """ 生成安全建议 """ recommendations = [] if len(report['failed_logins']) > 5: recommendations.append('Consider implementing rate limiting for login attempts') if self._calculate_security_score(report) < 70: recommendations.append('Review security logs and investigate potential security incidents') if report['total_events'] > 100000: recommendations.append('Consider implementing real-time security monitoring') return recommendations
