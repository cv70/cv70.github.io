内核优化

# 调整内核参数
cat > /etc/sysctl.d/99-ai-cluster.conf << EOF
# 网络优化
net.core.rmem_max = 2147483647
net.core.wmem_max = 2147483647
net.ipv4.tcp_rmem = 4096 87380 16777216
net.ipv4.tcp_wmem = 4096 65536 16777216

# 文件描述符
fs.file-max = 2097152

# 共享内存
kernel.shmmax = 68719476736
kernel.shmall = 4294967296
EOF

# 应用配置
sysctl -p /etc/sysctl.d/99-ai-cluster.conf

🎮 GPU驱动与CUDA安装

# NVIDIA驱动安装
# 添加NVIDIA仓库
distribution=$(. /etc/os-release;echo $ID$VERSION_ID)
curl -s -L https://nvidia.github.io/nvidia-docker/gpgkey | \
  apt-key add -
curl -s -L https://nvidia.github.io/nvidia-docker/$distribution/nvidia-docker.list | \
  tee /etc/apt/sources.list.d/nvidia-docker.list

# 安装NVIDIA驱动
apt update
apt install -y nvidia-driver-535

# 验证安装
nvidia-smi

# CUDA Toolkit安装
wget https://developer.download.nvidia.com/compute/cuda/12.1.0/local_installers/cuda_12.1.0_530.30.02_linux.run
sh cuda_12.1.0_530.30.02_linux.run --toolkit --silent

# 设置环境变量
echo 'export PATH=/usr/local/cuda-12.1/bin:$PATH' >> ~/.bashrc
echo 'export LD_LIBRARY_PATH=/usr/local/cuda-12.1/lib64:$LD_LIBRARY_PATH' >> ~/.bashrc
source ~/.bashrc

# 验证CUDA
nvcc --version

🐳 容器环境部署

# Docker安装
curl -fsSL https://get.docker.com -o get-docker.sh
sh get-docker.sh

# NVIDIA Container Toolkit
distribution=$(. /etc/os-release;echo $ID$VERSION_ID)
curl -s -L https://nvidia.github.io/nvidia-docker/gpgkey | \
  apt-key add -
curl -s -L https://nvidia.github.io/nvidia-docker/$distribution/nvidia-docker.list | \
  tee /etc/apt/sources.list.d/nvidia-docker.list

apt update
apt install -y nvidia-container-toolkit

# 配置Docker运行时
nvidia-ctk runtime configure --runtime=docker
systemctl restart docker

# 测试GPU容器
docker run --rm --gpus all nvidia/cuda:12.1.0-base-ubuntu22.04 nvidia-smi

192.168.0.201  storage-01-data
EOF

🔧 集群管理工具部署

# SSH免密登录配置
ssh-keygen -t ed25519 -C "cluster-admin"
ssh-copy-id root@gpu-01
ssh-copy-id root@gpu-02
ssh-copy-id root@storage-01

# 使用Ansible批量管理
# /etc/ansible/hosts
[training_nodes]
gpu-01 ansible_user=root
gpu-02 ansible_user=root

[storage_nodes]
storage-01 ansible_user=root

# 测试连接
ansible all -m ping

📊 存储配置

NFS共享存储

# 存储节点配置（storage-01）
apt install -y nfs-kernel-server
mkdir -p /data/nfs
chown nobody:nogroup /data/nfs
chmod 777 /data/nfs

# /etc/exports
/data/nfs *(rw,sync,no_subtree_check,no_root_squash)

# 启动NFS服务
systemctl enable nfs-kernel-server
systemctl start nfs-kernel-server

# 计算节点挂载
apt install -y nfs-common
mkdir -p /mnt/data
mount storage-01:/data/nfs /mnt/data

# 自动挂载 /etc/fstab
storage-01:/data/nfs  /mnt/data  nfs  defaults  0 0
⚠️
部署注意事项

版本一致性：所有节点的CUDA、驱动版本必须一致。防火墙配置：合理配置防火墙规则，避免阻断集群通信。时间同步：所有节点必须时间同步，建议使用NTP服务。文档记录：详细记录部署过程，便于故障排查。
