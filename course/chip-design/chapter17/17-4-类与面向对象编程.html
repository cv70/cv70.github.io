📦 类的基本语法

// 类定义
class Transaction;
    logic [31:0] addr;
    logic [31:0] data;
    logic        valid;

    // 构造函数
    function new();
        addr = 0;
        data = 0;
        valid = 0;
    endfunction

    // 方法
    function void display();
        $display("Addr: %h, Data: %h", addr, data);
    endfunction
endclass
