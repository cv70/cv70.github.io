// API网关是系统的统一入口 // 没有网关的架构 /* Client -----> User Service (3001) Client -----> Order Service (3002) Client -----> Product Service (3003) 问题： 1. 客户端需要知道所有服务地址 2. 跨域问题 3. 认证需要在每个服务重复实现 4. 限流、监控分散 */ // 有网关的架构 /* Client -----> API Gateway -----> User Service -----> Order Service -----> Product Service 优势： 1. 统一入口，简化客户端 2. 统一认证授权 3. 统一限流熔断 4. 统一监控日志 */ // 网关核心功能 const coreFeatures = { routing: '请求路由，根据路径转发到不同服务', loadBalancing: '负载均衡，分发请求到多个实例', authentication: '认证授权，统一鉴权入口', rateLimiting: '限流控制，保护后端服务', circuitBreaking: '熔断降级，防止雪崩', caching: '缓存响应，提升性能', logging: '日志记录，统一收集', monitoring: '监控统计，性能分析' };

网关价值

统一入口、简化客户端、集中治理、安全防护、性能优化。

🔧 网关核心功能详解

// 1. 请求路由 const routingConfig = { routes: [ { path: '/api/users/*', service: 'user-service', rewrite: '/users/$1' }, { path: '/api/orders/*', service: 'order-service' }, { path: '/api/products/*', service: 'product-service' } ] }; // 2. 负载均衡 const loadBalancing = { algorithms: { roundRobin: '轮询，依次分配', weightedRoundRobin: '加权轮询，根据权重分配', leastConnections: '最少连接，分配给连接最少的服务', ipHash: 'IP哈希，同一IP分配到同一服务' } }; // 3. 认证授权 async function authenticationMiddleware(req, res, next) { const token = req.headers.authorization; if (!token) { return res.status(401).json({ error: 'Unauthorized' }); } try { const user = await verifyToken(token); req.user = user; next(); } catch (error) { return res.status(401).json({ error: 'Invalid token' }); } } // 4. 限流控制 class RateLimiter { private requests: Map = new Map(); async check(key: string, limit: number, window: number): Promise { const now = Date.now(); const requests = this.requests.get(key) || []; // 过滤窗口外的请求 const validRequests = requests.filter(t => now - t < window); if (validRequests.length >= limit) { return false; // 超过限制 } validRequests.push(now); this.requests.set(key, validRequests); return true; } }
