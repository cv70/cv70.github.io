🔧 签名验证

// 签名验证中间件 @Injectable() export class SignatureMiddleware implements NestMiddleware { async use(req: Request, res: Response, next: NextFunction) { const { appkey, timestamp, nonce, signature } = req.headers; // 1. 验证必要参数 if (!appkey || !timestamp || !nonce || !signature) { throw new UnauthorizedException('Missing signature parameters'); } // 2. 验证时间戳（防止重放攻击） const now = Date.now(); const requestTime = parseInt(timestamp as string); if (Math.abs(now - requestTime) > 300000) { // 5分钟 throw new UnauthorizedException('Request expired'); } // 3. 验证nonce是否已使用（防止重放） const nonceUsed = await this.checkNonce(nonce as string); if (nonceUsed) { throw new UnauthorizedException('Nonce already used'); } // 4. 获取应用信息 const app = await this.getApplication(appkey as string); if (!app || app.status !== 'active') { throw new UnauthorizedException('Invalid app key'); } // 5. 构造签名字符串 const stringToSign = this.buildSignString(req); // 6. 计算签名 const expectedSignature = crypto .createHmac('sha256', app.appSecret) .update(stringToSign) .digest('hex'); // 7. 验证签名 if (signature !== expectedSignature) { throw new UnauthorizedException('Invalid signature'); } // 8. 记录nonce await this.recordNonce(nonce as string); // 9. 将应用信息挂载到请求 req['app'] = app; next(); } private buildSignString(req: Request): string { const method = req.method.toUpperCase(); const path = req.path; const timestamp = req.headers['timestamp']; const nonce = req.headers['nonce']; // GET请求使用query参数，POST使用body let params = ''; if (method === 'GET') { params = this.sortParams(req.query as Record); } else { params = JSON.stringify(req.body); } return `${method}\n${path}\n${timestamp}\n${nonce}\n${params}`; } private sortParams(params: Record): string { return Object.keys(params) .sort() .map(key => `${key}=${params[key]}`) .join('&'); } private async checkNonce(nonce: string): Promise { const key = `nonce:${nonce}`; const exists = await this.redis.get(key); return !!exists; } private async recordNonce(nonce: string): Promise { const key = `nonce:${nonce}`; await this.redis.set(key, '1', 'EX', 300); // 5分钟过期 } } // 客户端签名示例 class ApiClient { private appKey: string; private appSecret: string; constructor(appKey: string, appSecret: string) { this.appKey = appKey; this.appSecret = appSecret; } async request(method: string, path: string, params?: any): Promise { const timestamp = Date.now().toString(); const nonce = uuidv4(); const stringToSign = this.buildSignString(method, path, timestamp, nonce, params); const signature = crypto .createHmac('sha256', this.appSecret) .update(stringToSign) .digest('hex'); const config: any = { method, url: path, headers: { 'X-App-Key': this.appKey, 'X-Timestamp': timestamp, 'X-Nonce': nonce, 'X-Signature': signature, }, }; if (method === 'GET') { config.params = params; } else { config.data = params; } return axios(config); } }
