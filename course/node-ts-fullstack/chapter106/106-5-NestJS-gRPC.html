106.5 NestJS gRPC

在NestJS中使用gRPC服务

🎯 学习目标

  • 掌握NestJS gRPC模块配置
  • 学会定义gRPC控制器
  • 学习gRPC客户端的使用
  • 了解gRPC与REST混合使用

📚 模块配置

// 安装依赖 // npm install @nestjs/microservices @grpc/grpc-js @grpc/proto-loader // app.module.ts import { Module } from '@nestjs/common'; import { ClientsModule, Transport } from '@nestjs/microservices'; import { join } from 'path'; @Module({ imports: [ ClientsModule.register([ { name: 'USER_SERVICE', transport: Transport.GRPC, options: { package: 'user', protoPath: join(__dirname, '../proto/user.proto'), url: 'localhost:5000', loader: { keepCase: true, enums: String, defaults: true, oneofs: true } } } ]) ], controllers: [UserController], providers: [UserService] }) export class AppModule {} // user.proto syntax = "proto3"; package user; service UserService { rpc GetUser(GetUserRequest) returns (User); rpc CreateUser(CreateUserRequest) returns (User); rpc ListUsers(ListUsersRequest) returns (stream User); } message User { int32 id = 1; string name = 2; string email = 3; } message GetUserRequest { int32 id = 1; } message CreateUserRequest { string name = 1; string email = 2; } message ListUsersRequest { int32 page = 1; int32 page_size = 2; }
💡
NestJS gRPC优势

装饰器定义服务、依赖注入、拦截器支持、与REST混合使用。

🔧 gRPC服务端实现

// user.controller.ts import { Controller } from '@nestjs/common'; import { GrpcMethod, GrpcStreamMethod } from '@nestjs/microservices'; import { Observable, from } from 'rxjs'; @Controller() export class UserGrpcController { constructor(private readonly userService: UserService) {} // 一元RPC @GrpcMethod('UserService', 'GetUser') async getUser(data: GetUserRequest): Promise { const user = await this.userService.findById(data.id); return { id: user.id, name: user.name, email: user.email }; } @GrpcMethod('UserService', 'CreateUser') async createUser(data: CreateUserRequest): Promise { const user = await this.userService.create(data); return { id: user.id, name: user.name, email: user.email }; } // 服务端流RPC @GrpcMethod('UserService', 'ListUsers') listUsers(data: ListUsersRequest): Observable { const users = this.userService.findAll(data.page, data.pageSize); // 返回Observable流 return from(users.then(userList => userList.map(user => ({ id: user.id, name: user.name, email: user.email })) )); } // 双向流RPC @GrpcStreamMethod('ChatService', 'Chat') chat(data$: Observable): Observable { return new Observable(observer => { data$.subscribe({ next: (message) => { // 处理消息并广播 this.chatService.broadcast(message); observer.next({ sender: 'Server', content: `Received: ${message.content}`, timestamp: Date.now() }); }, complete: () => observer.complete() }); }); } } // main.ts - 启动gRPC服务 import { NestFactory } from '@nestjs/core'; import { Transport, MicroserviceOptions } from '@nestjs/microservices'; import { AppModule } from './app.module'; async function bootstrap() { const app = await NestFactory.createMicroservice( AppModule, { transport: Transport.GRPC, options: { package: 'user', protoPath: join(__dirname, '../proto/user.proto'), url: 'localhost:5000' } } ); await app.listen(); } bootstrap();

🔀 gRPC客户端使用

// user.service.ts import { Inject, Injectable, OnModuleInit } from '@nestjs/common'; import { ClientGrpc } from '@nestjs/microservices'; interface UserServiceGrpc { getUser(data: GetUserRequest): Observable; createUser(data: CreateUserRequest): Observable; listUsers(data: ListUsersRequest): Observable; } @Injectable() export class UserService implements OnModuleInit { private userService: UserServiceGrpc; constructor( @Inject('USER_SERVICE') private client: ClientGrpc ) {} onModuleInit() { this.userService = this.client.getService('UserService'); } async getUser(id: number): Promise { return this.userService.getUser({ id }).toPromise(); } async createUser(data: CreateUserRequest): Promise { return this.userService.createUser(data).toPromise(); } async listUsers(page: number, pageSize: number): Promise { return new Promise((resolve) => { const users: User[] = []; const stream = this.userService.listUsers({ page, pageSize }); stream.subscribe({ next: (user) => users.push(user), complete: () => resolve(users) }); }); } } // 混合使用gRPC和REST // main.ts async function bootstrap() { // 创建REST应用 const app = await NestFactory.create(AppModule); app.enableCors(); await app.listen(3000); // 同时启动gRPC微服务 app.connectMicroservice({ transport: Transport.GRPC, options: { package: 'user', protoPath: join(__dirname, '../proto/user.proto'), url: 'localhost:5000' } }); await app.startAllMicroservices(); }
