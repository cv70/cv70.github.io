实时游戏特点

实时多人游戏对网络通信有特殊要求：

⚡

低延迟

延迟需控制在 100ms 以内

🔄

高频同步

每秒 20-60 次状态更新

🎯

一致性

所有玩家看到相同的游戏状态

游戏类型定义

// types/game.ts // 玩家 interface Player { id: string; name: string; position: { x: number; y: number }; velocity: { x: number; y: number }; rotation: number; color: string; score: number; health: number; isAlive: boolean; } // 游戏房间 interface GameRoom { id: string; name: string; maxPlayers: number; players: Map<string, Player>; gameState: GameState; tickRate: number; // 每秒更新次数 lastTick: number; } // 游戏状态 interface GameState { status: 'waiting' | 'playing' | 'ended'; countdown: number; items: GameItem[]; obstacles: Obstacle[]; startTime: number; duration: number; } // 游戏物品 interface GameItem { id: string; type: 'coin' | 'powerup' | 'weapon'; position: { x: number; y: number }; collected: boolean; } // 玩家输入 interface PlayerInput { up: boolean; down: boolean; left: boolean; right: boolean; action: boolean; timestamp: number; } // 服务端事件 interface ServerToClientEvents { 'game:joined': (room: GameRoom) => void; 'game:state': (state: { players: Player[]; items: GameItem[] }) => void; 'game:player-joined': (player: Player) => void; 'game:player-left': (playerId: string) => void; 'game:start': () => void; 'game:end': (results: { winner: string; scores: Record<string, number> }) => void; } // 客户端事件 interface ClientToServerEvents { 'game:join': (roomId: string, callback: (success: boolean) => void) => void; 'game:input': (input: PlayerInput) => void; 'game:ready': () => void; 'game:leave': () => void; }

游戏服务端

import { Server } from 'socket.io'; class GameServer { private rooms = new Map<string, GameRoom>(); private tickInterval?: NodeJS.Timeout; private readonly TICK_RATE = 20; // 每秒 20 次 constructor(private io: Server) { this.startGameLoop(); } private startGameLoop() { this.tickInterval = setInterval(() => { this.tick(); }, 1000 / this.TICK_RATE); } private tick() { this.rooms.forEach((room, roomId) => { if (room.gameState.status !== 'playing') return; // 更新所有玩家 room.players.forEach((player) => { this.updatePlayer(player); }); // 碰撞检测 this.checkCollisions(room); // 发送状态更新 const state = { players: Array.from(room.players.values()), items: room.gameState.items.filter(i => !i.collected) }; // 使用 volatile 发送（可丢失） this.io.to(roomId).volatile.emit('game:state', state); }); } private updatePlayer(player: Player) { // 应用速度更新位置 player.position.x += player.velocity.x; player.position.y += player.velocity.y; // 边界检查 player.position.x = Math.max(0, Math.min(800, player.position.x)); player.position.y = Math.max(0, Math.min(600, player.position.y)); // 摩擦力 player.velocity.x *= 0.9; player.velocity.y *= 0.9; } private checkCollisions(room: GameRoom) { room.players.forEach((player) => { // 检查物品碰撞 room.gameState.items.forEach((item) => { if (!item.collected && this.checkDistance(player.position, item.position, 20)) { item.collected = true; player.score += 10; this.io.to(room.id).emit('item:collected', { itemId: item.id, playerId: player.id }); } }); // 检查玩家碰撞 room.players.forEach((other) => { if (player.id !== other.id && this.checkDistance(player.position, other.position, 30)) { // 弹开 const dx = player.position.x - other.position.x; const dy = player.position.y - other.position.y; const dist = Math.sqrt(dx * dx + dy * dy); if (dist > 0) { player.velocity.x += (dx / dist) * 2; player.velocity.y += (dy / dist) * 2; } } }); }); } private checkDistance(a: { x: number; y: number }, b: { x: number; y: number }, threshold: number): boolean { const dx = a.x - b.x; const dy = a.y - b.y; return Math.sqrt(dx * dx + dy * dy) < threshold; } // 加入游戏 joinGame(socket: any, roomId: string) { let room = this.rooms.get(roomId); if (!room) { room = this.createRoom(roomId); this.rooms.set(roomId, room); } if (room.players.size >= room.maxPlayers) { return false; } const player: Player = { id: socket.data.userId, name: socket.data.username, position: { x: 100 + Math.random() * 600, y: 100 + Math.random() * 400 }, velocity: { x: 0, y: 0 }, rotation: 0, color: this.generateColor(), score: 0, health: 100, isAlive: true }; room.players.set(player.id, player); socket.join(roomId); // 通知其他玩家 socket.to(roomId).emit('game:player-joined', player); // 发送当前游戏状态 socket.emit('game:joined', room); return true; } // 处理输入 handleInput(socket: any, input: PlayerInput) { const roomId = this.findPlayerRoom(socket.data.userId); if (!roomId) return; const room = this.rooms.get(roomId); if (!room || room.gameState.status !== 'playing') return; const player = room.players.get(socket.data.userId); if (!player) return; // 应用输入 const speed = 0.5; if (input.up) player.velocity.y -= speed; if (input.down) player.velocity.y += speed; if (input.left) player.velocity.x -= speed; if (input.right) player.velocity.x += speed; } // 开始游戏 startGame(roomId: string) { const room = this.rooms.get(roomId); if (!room) return; room.gameState.status = 'playing'; room.gameState.startTime = Date.now(); // 生成物品 this.spawnItems(room); // 通知所有玩家 this.io.to(roomId).emit('game:start'); } // 生成物品 private spawnItems(room: GameRoom) { for (let i = 0; i < 10; i++) { room.gameState.items.push({ id: `item-${i}`, type: 'coin', position: { x: Math.random() * 700 + 50, y: Math.random() * 500 + 50 }, collected: false }); } } // 创建房间 private createRoom(id: string): GameRoom { return { id, name: `Room ${id}`, maxPlayers: 10, players: new Map(), gameState: { status: 'waiting', countdown: 5, items: [], obstacles: [], startTime: 0, duration: 60 }, tickRate: this.TICK_RATE, lastTick: Date.now() }; } private generateColor(): string { const colors = ['#FF6B6B', '#4ECDC4', '#45B7D1', '#96CEB4', '#FFEAA7', '#DDA0DD']; return colors[Math.floor(Math.random() * colors.length)]; } private findPlayerRoom(userId: string): string | null { for (const [roomId, room] of this.rooms) { if (room.players.has(userId)) return roomId; } return null; } }

⏱️ 客户端预测与插值

客户端预测

客户端预测让玩家操作立即响应，无需等待服务端确认：

// client/game.ts class GameClient { private socket: Socket; private localPlayer: Player; private otherPlayers = new Map<string, Player>(); private pendingInputs: { input: PlayerInput; sequence: number }[] = []; private inputSequence = 0; private serverState: GameState | null = null; constructor(socket: Socket) { this.socket = socket; // 接收服务端状态 socket.on('game:state', (state) => { this.serverState = state; this.reconcile(); }); } // 处理玩家输入 handleInput(input: PlayerInput) { // 1. 立即应用输入（客户端预测） this.applyInput(this.localPlayer, input); // 2. 保存输入序列号 const sequence = this.inputSequence++; this.pendingInputs.push({ input, sequence }); // 3. 发送给服务端 this.socket.emit('game:input', { ...input, sequence }); } // 应用输入 private applyInput(player: Player, input: PlayerInput) { const speed = 0.5; if (input.up) player.velocity.y -= speed; if (input.down) player.velocity.y += speed; if (input.left) player.velocity.x -= speed; if (input.right) player.velocity.x += speed; // 更新位置 player.position.x += player.velocity.x; player.position.y += player.velocity.y; // 摩擦力 player.velocity.x *= 0.9; player.velocity.y *= 0.9; } // 状态和解（校正客户端预测） private reconcile() { if (!this.serverState) return; // 找到本地玩家在服务端的状态 const serverPlayer = this.serverState.players.find(p => p.id === this.localPlayer.id); if (!serverPlayer) return; // 移除已确认的输入 const lastProcessedSequence = serverPlayer.lastProcessedInput || 0; this.pendingInputs = this.pendingInputs.filter( pending => pending.sequence > lastProcessedSequence ); // 重放到当前位置 this.localPlayer.position = { ...serverPlayer.position }; this.localPlayer.velocity = { ...serverPlayer.velocity }; // 重新应用未确认的输入 this.pendingInputs.forEach(({ input }) => { this.applyInput(this.localPlayer, input); }); // 更新其他玩家（使用插值） this.serverState.players.forEach(player => { if (player.id !== this.localPlayer.id) { this.interpolateOtherPlayer(player); } }); } // 插值其他玩家位置 private interpolateOtherPlayer(serverPlayer: Player) { const clientPlayer = this.otherPlayers.get(serverPlayer.id); if (!clientPlayer) { this.otherPlayers.set(serverPlayer.id, { ...serverPlayer }); return; } // 平滑插值到服务端位置 clientPlayer.position.x += (serverPlayer.position.x - clientPlayer.position.x) * 0.3; clientPlayer.position.y += (serverPlayer.position.y - clientPlayer.position.y) * 0.3; } // 渲染循环 render() { // 渲染本地玩家 this.renderPlayer(this.localPlayer); // 渲染其他玩家（插值后的位置） this.otherPlayers.forEach(player => { this.renderPlayer(player); }); } private renderPlayer(player: Player) { // Canvas 渲染逻辑 } }

🛡️ 防作弊基础

⚠️
防作弊策略
  • 服务端权威：所有关键逻辑在服务端执行
  • 输入验证：检查输入是否合理（速度、频率）
  • 状态同步：定期同步权威状态给客户端
  • 延迟补偿：处理网络延迟对公平性的影响
// 服务端防作弊检查 function validateInput(player: Player, input: PlayerInput): boolean { // 检查移动速度是否合理 const maxSpeed = 10; const speed = Math.sqrt(input.velocityX ** 2 + input.velocityY ** 2); if (speed > maxSpeed) { console.warn(`玩家 ${player.id} 速度异常: ${speed}`); return false; } // 检查输入频率 const timeSinceLastInput = Date.now() - player.lastInputTime; if (timeSinceLastInput < 16) { // 约 60 FPS console.warn(`玩家 ${player.id} 输入频率异常`); return false; } // 检查位置跳变 const distance = Math.sqrt( (input.position.x - player.position.x) ** 2 + (input.position.y - player.position.y) ** 2 ); if (distance > 100) { console.warn(`玩家 ${player.id} 位置跳变: ${distance}`); return false; } return true; }
