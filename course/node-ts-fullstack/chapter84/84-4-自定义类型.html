84.4 自定义类型

Vue 项目中的 TypeScript 类型管理

📖 本节概述

本节将介绍如何在 Vue 项目中组织和管理 TypeScript 类型定义，包括类型文件的目录结构、全局类型声明、模块扩展以及类型复用等最佳实践。

🎯 学习目标

  • 掌握类型文件的组织结构
  • 学会声明全局类型
  • 掌握模块扩展的使用
  • 了解类型复用的最佳实践

📂 类型文件组织

src/ ├── types/ │ ├── index.ts // 统一导出 │ ├── global.d.ts // 全局类型声明 │ ├── api.ts // API 相关类型 │ ├── user.ts // 用户相关类型 │ ├── product.ts // 产品相关类型 │ └── components.ts // 组件相关类型 └── ... // types/index.ts export * from './api'; export * from './user'; export * from './product'; export * from './components';

👤 用户相关类型

// types/user.ts // 基础用户类型 export interface User { id: number; name: string; email: string; avatar?: string; role: UserRole; createdAt: string; updatedAt: string; } // 用户角色枚举 export type UserRole = 'admin' | 'user' | 'guest'; // 登录参数 export interface LoginParams { email: string; password: string; remember?: boolean; } // 登录响应 export interface LoginResponse { user: User; token: string; refreshToken: string; } // 用户查询参数 export interface UserQueryParams { page?: number; pageSize?: number; keyword?: string; role?: UserRole; } // 用户表单 export interface UserForm { name: string; email: string; password?: string; role: UserRole; } // 部分更新 export type UserUpdate = Partial<Pick<User, 'name' | 'email' | 'role'>>;

🌐 API 响应类型

// types/api.ts // 通用 API 响应 export interface ApiResponse<T = any> { code: number; message: string; data: T; } // 分页响应 export interface PageResponse<T> { list: T[]; total: number; page: number; pageSize: number; totalPages: number; } // 分页请求参数 export interface PaginationParams { page: number; pageSize: number; sortBy?: string; sortOrder?: 'asc' | 'desc'; } // API 错误 export interface ApiError { code: number; message: string; errors?: Record<string, string[]>; } // 请求配置 export interface RequestOptions { loading?: boolean; showError?: boolean; retryCount?: number; }

🌍 全局类型声明

// types/global.d.ts // 扩展 Window 接口 declare global { interface Window { __APP_VERSION__: string; __API_URL__: string; } } // 环境变量类型 interface ImportMetaEnv { readonly VITE_APP_TITLE: string; readonly VITE_API_URL: string; readonly VITE_MODE: 'development' | 'production'; } interface ImportMeta { readonly env: ImportMetaEnv; } // 全局通用类型 type Maybe<T> = T | null | undefined; type AsyncResult<T> = Promise<{ data: T | null; error: Error | null; }>; export;

扩展 Vue Router

扩展 Pinia

// types/pinia.d.ts import 'pinia'; declare module 'pinia' { interface DefineStoreOptionsBase<S, Store> { // 添加自定义选项 persistence?: { enabled: boolean; key?: string; paths?: string[]; }; } }

扩展组件实例

// types/components.d.ts import { RouterLink } from 'vue-router'; declare module '@vue/runtime-core' { export interface GlobalComponents { RouterLink: typeof RouterLink; // 添加全局组件类型 AppButton: typeof import('@/components/AppButton.vue').default; AppInput: typeof import('@/components/AppInput.vue').default; AppModal: typeof import('@/components/AppModal.vue').default; } } export {};

🔧 类型工具函数

// types/utils.ts // 获取函数参数类型 type Parameters<T extends (...args: any) => any> = T extends ( ...args: infer P ) => any ? P : never; // 获取函数返回类型 type ReturnType<T extends (...args: any) => any> = T extends ( ...args: any ) => infer R ? R : never; // 将对象所有属性变为可选 type DeepPartial<T> = { [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P]; }; // 将对象所有属性变为只读 type DeepReadonly<T> = { readonly [P in keyof T]: T[P] extends object ? DeepReadonly<T[P]> : T[P]; }; // 提取对象特定类型的键 type PickByType<T, U> = { [P in keyof T as T[P] extends U ? P : never]: T[P]; }; // 忽略对象特定类型的键 type OmitByType<T, U> = { [P in keyof T as T[P] extends U ? never : P]: T[P]; }; // 示例使用 interface Example { name: string; age: number; active: boolean; } type StringKeys = PickByType<Example, string>; // { name: string } type NumberKeys = PickByType<Example, number>; // { age: number }

📋 组件 Props 类型复用

// types/components.ts // 基础按钮 Props export interface BaseButtonProps { size?: 'small' | 'medium' | 'large'; variant?: 'primary' | 'secondary' | 'danger'; disabled?: boolean; loading?: boolean; block?: boolean; } // 基础输入 Props export interface BaseInputProps { modelValue?: string | number; label?: string; placeholder?: string; disabled?: boolean; readonly?: boolean; error?: string; required?: boolean; } // 表单项 Props export interface FormItemProps { label: string; name: string; rules?: FormRule[]; required?: boolean; } // 在组件中复用 // components/Button.vue import type { BaseButtonProps } from '@/types/components'; interface Props extends BaseButtonProps { // 额外的 props icon?: string; } const props = defineProps<Props>();
