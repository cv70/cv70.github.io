📚 模型实体设计

// content-model.entity.ts @Entity('content_models') export class ContentModel { @PrimaryGeneratedColumn('uuid') id: string; @Column({ unique: true }) name: string; @Column({ unique: true }) slug: string; @Column({ nullable: true }) description: string; @OneToMany(() => ContentField, field => field.model, { cascade: true }) fields: ContentField[]; @Column({ type: 'json', nullable: true }) permissions: string[]; @Column({ default: true }) active: boolean; @CreateDateColumn() createdAt: Date; @UpdateDateColumn() updatedAt: Date; } // content-field.entity.ts @Entity('content_fields') export class ContentField { @PrimaryGeneratedColumn('uuid') id: string; @Column() name: string; @Column() label: string; @Column() type: FieldType; @Column({ default: false }) required: boolean; @Column({ type: 'json', nullable: true }) validation: { min?: number; max?: number; pattern?: string; message?: string; }; @Column({ type: 'json', nullable: true }) options: { multiple?: boolean; relationTo?: string; choices?: { label: string; value: string }[]; default?: any; }; @Column() order: number; @ManyToOne(() => ContentModel, model => model.fields) model: ContentModel; @Column() modelId: string; } // content.entity.ts @Entity('contents') export class Content { @PrimaryGeneratedColumn('uuid') id: string; @Column() modelId: string; @ManyToOne(() => ContentModel) model: ContentModel; @Column({ type: 'jsonb' }) data: Record; // 动态字段值 @Column({ default: 'draft' }) status: ContentStatus; @Column() createdBy: string; @Column({ nullable: true }) updatedBy: string; @Column({ nullable: true }) publishedAt: Date; @CreateDateColumn() createdAt: Date; @UpdateDateColumn() updatedAt: Date; } enum ContentStatus { DRAFT = 'draft', PENDING = 'pending', PUBLISHED = 'published', ARCHIVED = 'archived' }

🔧 模型服务实现

// content-model.service.ts @Injectable() export class ContentModelService { constructor( @InjectRepository(ContentModel) private modelRepo: Repository, @InjectRepository(ContentField) private fieldRepo: Repository, ) {} // 创建内容模型 async createModel(dto: CreateModelDto): Promise { const model = this.modelRepo.create({ name: dto.name, slug: dto.slug, description: dto.description, permissions: dto.permissions, }); // 创建字段 model.fields = dto.fields.map((fieldDto, index) => this.fieldRepo.create({ ...fieldDto, order: index, modelId: model.id, }) ); return this.modelRepo.save(model); } // 获取模型及其字段 async getModelWithFields(slug: string): Promise { return this.modelRepo.findOne({ where: { slug }, relations: ['fields'], order: { fields: { order: 'ASC' } }, }); } // 验证内容数据 async validateContent(modelSlug: string, data: Record): Promise { const model = await this.getModelWithFields(modelSlug); const errors: string[] = []; for (const field of model.fields) { const value = data[field.name]; // 必填验证 if (field.required && (value === undefined || value === null || value === '')) { errors.push(`${field.label} is required`); continue; } // 类型验证 if (value !== undefined) { switch (field.type) { case FieldType.NUMBER: if (typeof value !== 'number') { errors.push(`${field.label} must be a number`); } else if (field.validation) { if (field.validation.min !== undefined && value < field.validation.min) { errors.push(`${field.label} must be at least ${field.validation.min}`); } if (field.validation.max !== undefined && value > field.validation.max) { errors.push(`${field.label} must be at most ${field.validation.max}`); } } break; case FieldType.TEXT: if (typeof value !== 'string') { errors.push(`${field.label} must be a string`); } else if (field.validation?.pattern) { const regex = new RegExp(field.validation.pattern); if (!regex.test(value)) { errors.push(field.validation.message || `${field.label} format is invalid`); } } break; } } } return errors; } } // content.service.ts @Injectable() export class ContentService { constructor( @InjectRepository(Content) private contentRepo: Repository, private modelService: ContentModelService, ) {} // 创建内容 async createContent(modelSlug: string, dto: CreateContentDto, userId: string): Promise { // 验证数据 const errors = await this.modelService.validateContent(modelSlug, dto.data); if (errors.length > 0) { throw new BadRequestException(errors); } const model = await this.modelService.getModelWithFields(modelSlug); const content = this.contentRepo.create({ modelId: model.id, data: dto.data, status: ContentStatus.DRAFT, createdBy: userId, }); return this.contentRepo.save(content); } // 查询内容 async getContents(modelSlug: string, options: { page: number; limit: number; status?: ContentStatus; filters?: Record; }) { const model = await this.modelService.getModelWithFields(modelSlug); const query = this.contentRepo.createQueryBuilder('content') .where('content.modelId = :modelId', { modelId: model.id }); if (options.status) { query.andWhere('content.status = :status', { status: options.status }); } // 动态过滤 if (options.filters) { for (const [key, value] of Object.entries(options.filters)) { query.andWhere(`content.data->>'${key}' = :${key}`, { [key]: value }); } } return query .skip((options.page - 1) * options.limit) .take(options.limit) .getManyAndCount(); } }
