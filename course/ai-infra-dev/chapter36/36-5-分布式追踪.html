分布式追踪的价值

分布式追踪记录请求在多个服务间的完整路径，帮助定位性能瓶颈和故障根因。

📊 核心概念

Trace（追踪）

一个请求的完整执行路径，包含多个Span

Span（跨度）

单个服务或操作的执行单元

Trace ID

唯一标识一个追踪的全局ID

Span ID

唯一标识一个Span的ID

🐍 Python集成示例

# 安装依赖 # pip install opentelemetry-api opentelemetry-sdk # pip install opentelemetry-instrumentation # pip install opentelemetry-instrumentation-fastapi # pip install opentelemetry-exporter-jaeger import opentelemetry.trace as trace from opentelemetry import trace as trace_api from opentelemetry.sdk.trace import TracerProvider from opentelemetry.sdk.trace.export import BatchSpanProcessor from opentelemetry.exporter.jaeger import JaegerExporter from opentelemetry.instrumentation.fastapi import FastAPIInstrumentor # 配置追踪 def setup_tracing(service_name: str): # 设置提供者 trace.set_tracer_provider(TracerProvider()) tracer = trace.get_tracer(service_name) # 配置Jaeger导出器 jaeger_exporter = JaegerExporter( agent_host_name="localhost", agent_port=6831, ) # 添加批量处理器 span_processor = BatchSpanProcessor(jaeger_exporter) trace.get_tracer_provider().add_span_processor(span_processor) return tracer # 在训练任务中使用 import torch from torch.utils.data import DataLoader class TrainingTracer: def __init__(self): self.tracer = setup_tracing("training-service") def train_epoch(self, model, dataloader, optimizer, epoch): with self.tracer.start_as_current_span("train_epoch") as span: span.set_attribute("epoch", epoch) span.set_attribute("batch_count", len(dataloader)) for batch_idx, (data, target) in enumerate(dataloader): with self.tracer.start_as_current_span("train_batch") as batch_span: batch_span.set_attribute("batch_idx", batch_idx) # 前向传播 with self.tracer.start_as_current_span("forward"): output = model(data) # 计算损失 with self.tracer.start_as_current_span("compute_loss"): loss = criterion(output, target) batch_span.set_attribute("loss", float(loss)) # 反向传播 with self.tracer.start_as_current_span("backward"): loss.backward() # 优化器步骤 with self.tracer.start_as_current_span("optimizer_step"): optimizer.step() optimizer.zero_grad() # 推理服务追踪 from fastapi import FastAPI app = FastAPI() FastAPIInstrumentor.instrument_app(app, tracer_provider=trace_api.get_tracer_provider()) @app.post("/predict") async def predict(request: dict): tracer = trace.get_tracer(__name__) with tracer.start_as_current_span("predict") as span: # 数据预处理 with tracer.start_as_current_span("preprocess") as prep_span: data = preprocess(request['data']) prep_span.set_attribute("input_size", len(data)) # 模型推理 with tracer.start_as_current_span("inference") as inf_span: result = model(data) inf_span.set_attribute("model", model.__class__.__name__) # 后处理 with tracer.start_as_current_span("postprocess"): output = postprocess(result) return {"result": output}

# 训练流水线追踪 class TrainingPipeline: def __init__(self): self.tracer = setup_tracing("training-pipeline") def run(self, config): with self.tracer.start_as_current_span("training_pipeline") as span: span.set_attribute("model", config.model) span.set_attribute("batch_size", config.batch_size) span.set_attribute("epochs", config.epochs) # 数据加载 with self.tracer.start_as_current_span("data_loading"): train_loader = self.load_data(config.train_path) val_loader = self.load_data(config.val_path) # 模型初始化 with self.tracer.start_as_current_span("model_init"): model = self.create_model(config.model) # 训练循环 with self.tracer.start_as_current_span("training_loop"): for epoch in range(config.epochs): self.train_epoch(model, train_loader, epoch) self.validate(model, val_loader, epoch) # 模型保存 with self.tracer.start_as_current_span("model_save"): self.save_model(model) # 分布式训练追踪 class DistributedTracing: def __init__(self, rank, world_size): self.rank = rank self.world_size = world_size self.tracer = setup_tracing(f"distributed-training-rank-{rank}") def allreduce_trace(self, tensor): with self.tracer.start_as_current_span("allreduce") as span: span.set_attribute("tensor_size", tensor.numel()) span.set_attribute("rank", self.rank) # 记录通信时间 start = time.time() dist.all_reduce(tensor) duration = time.time() - start span.set_attribute("duration_ms", duration * 1000) return tensor

🚀 推理服务追踪

# 推理服务流水线追踪 class InferenceService: def __init__(self, model): self.model = model self.tracer = setup_tracing("inference-service") async def handle_request(self, request): with self.tracer.start_as_current_span("inference_request") as span: # 添加请求属性 span.set_attribute("request_id", request.id) span.set_attribute("model_name", request.model) span.set_attribute("input_size", len(request.data)) # 模型加载 with self.tracer.start_as_current_span("model_loading") as load_span: if request.model != self.current_model: self.load_model(request.model) load_span.set_attribute("loaded", True) # 预处理 with self.tracer.start_as_current_span("preprocessing") as prep_span: inputs = self.preprocess(request.data) prep_span.set_attribute("preprocessed", True) # 批处理 with self.tracer.start_as_current_span("batching") as batch_span: batch = self.batch_requests(inputs) batch_span.set_attribute("batch_size", len(batch)) # 推理 with self.tracer.start_as_current_span("model_inference") as inf_span: outputs = self.model(batch) inf_span.set_attribute("inference_time", time.time()) # 后处理 with self.tracer.start_as_current_span("postprocessing") as post_span: results = self.postprocess(outputs) post_span.set_attribute("output_size", len(results)) return results # RAG系统追踪 class RAGTracing: def __init__(self): self.tracer = setup_tracing("rag-service") def query(self, question): with self.tracer.start_as_current_span("rag_query") as span: span.set_attribute("question", question) # 检索 with self.tracer.start_as_current_span("retrieval") as ret_span: documents = self.retriever.search(question) ret_span.set_attribute("doc_count", len(documents)) # 上下文构建 with self.tracer.start_as_current_span("context_building"): context = self.build_context(documents) # LLM生成 with self.tracer.start_as_current_span("llm_generation") as llm_span: answer = self.llm.generate(question, context) llm_span.set_attribute("tokens", len(answer)) return answer

📊 追踪数据分析

关键指标

  • 端到端延迟 - 完整请求的耗时
  • Span延迟 - 各个操作的耗时分布
  • 调用链深度 - Span的嵌套层级
  • 错误率 - 失败的Span占比
  • 关键路径 - 影响整体性能的Span
  • 服务依赖 - 服务间的调用关系
