116.2 领域建模

构建业务领域的概念模型

  • 理解领域建模的过程

📚 实体与值对象

// 实体（Entity）：有唯一标识，状态可变 // 用户实体 - 由ID唯一标识 class User { private constructor( private readonly id: UserId, // 唯一标识 private name: string, // 可变属性 private email: string, private status: UserStatus ) {} // 通过ID判断相等性 equals(other: User): boolean { return this.id.equals(other.id); } // 业务方法改变状态 changeName(newName: string): void { if (!newName || newName.trim().length < 2) { throw new Error('Name must be at least 2 characters'); } this.name = newName; } activate(): void { if (this.status === UserStatus.ACTIVE) { throw new Error('User is already active'); } this.status = UserStatus.ACTIVE; } } // 值对象（Value Object）：无唯一标识，不可变，通过属性值判断相等 // 地址值对象 class Address { constructor( readonly province: string, readonly city: string, readonly district: string, readonly detail: string ) { // 验证 if (!province || !city) { throw new Error('Province and city are required'); } } // 通过属性值判断相等 equals(other: Address): boolean { return this.province === other.province && this.city === other.city && this.district === other.district && this.detail === other.detail; } // 返回新实例而非修改 withDetail(newDetail: string): Address { return new Address(this.province, this.city, this.district, newDetail); } // 值对象应该是不可变的 // 所有修改操作都返回新实例 } // 金额值对象 class Money { private constructor( readonly amount: number, readonly currency: string ) { if (amount < 0) { throw new Error('Amount cannot be negative'); } } static create(amount: number, currency: string = 'CNY'): Money { return new Money(amount, currency); } add(other: Money): Money { if (this.currency !== other.currency) { throw new Error('Cannot add money with different currencies'); } return new Money(this.amount + other.amount, this.currency); } multiply(factor: number): Money { return new Money(this.amount * factor, this.currency); } equals(other: Money): boolean { return this.amount === other.amount && this.currency === other.currency; } } // 实体 vs 值对象的选择 const choice = { 实体: '需要追踪生命周期，关心"是谁"', 值对象: '只关心"是什么"，描述、度量、数量' };
💡
值对象优势

不可变、线程安全、易于测试、可自由替换、无副作用。

🔧 领域建模实践

// 订单领域建模示例 // 订单ID值对象 class OrderId { private constructor(readonly value: string) { if (!value || value.length === 0) { throw new Error('Order ID cannot be empty'); } } static generate(): OrderId { return new OrderId(`ORD-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`); } static from(value: string): OrderId { return new OrderId(value); } equals(other: OrderId): boolean { return this.value === other.value; } } // 订单项实体 class OrderItem { constructor( readonly productId: string, readonly productName: string, readonly unitPrice: Money, private quantity: number ) { if (quantity <= 0) { throw new Error('Quantity must be positive'); } } get Quantity(): number { return this.quantity; } getTotalPrice(): Money { return this.unitPrice.multiply(this.quantity); } increaseQuantity(amount: number): void { if (amount <= 0) { throw new Error('Amount must be positive'); } this.quantity += amount; } } // 订单状态枚举 enum OrderStatus { PENDING = 'pending', PAID = 'paid', SHIPPED = 'shipped', DELIVERED = 'delivered', CANCELLED = 'cancelled' } // 订单实体（聚合根） class Order { private constructor( readonly id: OrderId, readonly userId: string, private items: OrderItem[], private status: OrderStatus, private shippingAddress: Address, readonly createdAt: Date ) {} // 工厂方法 static create(userId: string, shippingAddress: Address): Order { return new Order( OrderId.generate(), userId, [], OrderStatus.PENDING, shippingAddress, new Date() ); } // 业务方法 addItem(item: OrderItem): void { if (this.status !== OrderStatus.PENDING) { throw new Error('Cannot add items to non-pending order'); } const existingItem = this.items.find(i => i.productId === item.productId); if (existingItem) { existingItem.increaseQuantity(item.Quantity); } else { this.items.push(item); } } removeItem(productId: string): void { if (this.status !== OrderStatus.PENDING) { throw new Error('Cannot remove items from non-pending order'); } this.items = this.items.filter(i => i.productId !== productId); } pay(): void { if (this.status !== OrderStatus.PENDING) { throw new Error('Order is not pending'); } if (this.items.length === 0) { throw new Error('Cannot pay for empty order'); } this.status = OrderStatus.PAID; } cancel(): void { if (this.status === OrderStatus.SHIPPED || this.status === OrderStatus.DELIVERED) { throw new Error('Cannot cancel shipped or delivered order'); } this.status = OrderStatus.CANCELLED; } getTotalAmount(): Money { return this.items.reduce( (total, item) => total.add(item.getTotalPrice()), Money.create(0) ); } get Items(): ReadonlyArray<OrderItem> { return this.items; } get Status(): OrderStatus { return this.status; } }

📝 本节小结

✅
  • • 实体有唯一标识，状态可变
  • • 值对象无标识，不可变，通过值判断相等
  • • 业务逻辑封装在领域对象中
  • • 工厂方法保证对象创建的有效性
