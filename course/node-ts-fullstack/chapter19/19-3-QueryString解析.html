19.3 Query String 解析

处理 URL 查询字符串

  • 理解编码处理

📚 querystring 模块

const querystring = require('querystring'); // 解析查询字符串 const str = 'name=Alice&age=25&active=true'; const obj = querystring.parse(str); console.log(obj); // { name: 'Alice', age: '25', active: 'true' } // 指定分隔符 const obj2 = querystring.parse('name:Alice;age:25', ';', ':'); console.log(obj2); // { name: 'Alice', age: '25' } // 字符串化 const obj3 = { name: 'Alice', age: 25, active: true }; const str2 = querystring.stringify(obj3); console.log(str2); // name=Alice&age=25&active=true // 指定分隔符 const str3 = querystring.stringify(obj3, ';', ':'); console.log(str3); // name:Alice;age:25;active:true // URL 编码 const encoded = querystring.escape('你好世界'); console.log(encoded); // %E4%BD%A0%E5%A5%BD%E4%B8%96%E7%95%8C // URL 解码 const decoded = querystring.unescape('%E4%BD%A0%E5%A5%BD%E4%B8%96%E7%95%8C'); console.log(decoded); // 你好世界

🔧 URLSearchParams

// 推荐：使用 URLSearchParams const params = new URLSearchParams('name=Alice&age=25&tags=a&tags=b'); // 获取值 console.log(params.get('name')); // 'Alice' console.log(params.getAll('tags')); // ['a', 'b'] // 检查存在 console.log(params.has('name')); // true // 设置值 params.set('name', 'Bob'); params.append('tags', 'c'); // 删除 params.delete('age'); // 遍历 for (const [key, value] of params) { console.log(`${key}: ${value}`); } // 转换为对象 const obj = Object.fromEntries(params); console.log(obj); // { name: 'Bob', tags: 'c' } // 注意：Object.fromEntries 只保留最后一个同名参数 // 转换为字符串 console.log(params.toString()); // name=Bob&tags=a&tags=b&tags=c // 从对象创建 const params2 = new URLSearchParams({ name: 'Alice', age: 25 }); console.log(params2.toString()); // name=Alice&age=25

📦 复杂参数处理

// 处理数组参数 function parseArrayParams(str) { const params = new URLSearchParams(str); const result = {}; for (const [key, value] of params) { if (key.endsWith('[]')) { const realKey = key.slice(0, -2); if (!result[realKey]) result[realKey] = []; result[realKey].push(value); } else { result[key] = value; } } return result; } const arrStr = 'tags[]=js&tags[]=node&name=Alice'; console.log(parseArrayParams(arrStr)); // { tags: ['js', 'node'], name: 'Alice' } // 处理嵌套对象参数 function parseNestedParams(str) { const params = new URLSearchParams(str); const result = {}; for (const [key, value] of params) { const parts = key.split('.'); let obj = result; for (let i = 0; i < parts.length - 1; i++) { if (!obj[parts[i]]) obj[parts[i]] = {}; obj = obj[parts[i]]; } obj[parts[parts.length - 1]] = value; } return result; } const nestedStr = 'user.name=Alice&user.age=25&title=Hello'; console.log(parseNestedParams(nestedStr)); // { user: { name: 'Alice', age: '25' }, title: 'Hello' } // 类型转换 function parseWithTypes(str) { const params = new URLSearchParams(str); const result = {}; for (const [key, value] of params) { // 尝试转换为合适类型 if (value === 'true') result[key] = true; else if (value === 'false') result[key] = false; else if (value === 'null') result[key] = null; else if (!isNaN(value) && value !== '') result[key] = Number(value); else result[key] = value; } return result; } console.log(parseWithTypes('active=true&count=10&name=Alice')); // { active: true, count: 10, name: 'Alice' }

⚙️ 实际应用示例

const http = require('http'); const url = require('url'); // HTTP 请求中解析查询参数 const server = http.createServer((req, res) => { const parsedUrl = new URL(req.url, `http://${req.headers.host}`); // 获取查询参数 const page = parsedUrl.searchParams.get('page') || '1'; const limit = parsedUrl.searchParams.get('limit') || '10'; const search = parsedUrl.searchParams.get('search') || ''; console.log('分页:', { page, limit, search }); // 获取所有同名参数 const tags = parsedUrl.searchParams.getAll('tag'); console.log('标签:', tags); // 转换为对象 const query = Object.fromEntries(parsedUrl.searchParams); res.writeHead(200, { 'Content-Type': 'application/json' }); res.end(JSON.stringify({ query, tags })); }); server.listen(3000); // 构建分页 URL function buildPaginationUrl(baseUrl, currentPage, totalPages, params = {}) { const url = new URL(baseUrl); Object.entries(params).forEach(([k, v]) => { url.searchParams.set(k, v); }); return { first: buildPageUrl(url, 1), prev: currentPage > 1 ? buildPageUrl(url, currentPage - 1) : null, current: buildPageUrl(url, currentPage), next: currentPage < totalPages ? buildPageUrl(url, currentPage + 1) : null, last: buildPageUrl(url, totalPages) }; } function buildPageUrl(url, page) { const newUrl = new URL(url.href); newUrl.searchParams.set('page', page); return newUrl.href; } const pagination = buildPaginationUrl('https://api.example.com/users', 2, 10, { limit: 20 }); console.log(pagination);
