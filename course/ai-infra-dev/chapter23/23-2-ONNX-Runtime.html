什么是ONNX Runtime

ONNX Runtime是微软开发的跨平台高性能推理引擎。 它基于ONNX（Open Neural Network Exchange）格式，支持在各种硬件和操作系统上运行深度学习模型， 提供了统一的推理API和优秀的性能优化。

🌐 ONNX简介

ONNX标准格式

  • 标准格式：用于表示深度学习模型的开放标准
  • 跨框架：PyTorch、TensorFlow、Keras等都支持导出ONNX
  • 互操作性：实现不同框架之间的模型共享
  • 版本兼容：支持Opset版本管理

💻 使用示例

import onnx
import onnxruntime as ort
import numpy as np
import torch

# 将PyTorch模型转换为ONNX
def export_to_onnx(model, dummy_input, onnx_path):
    """
    导出ONNX模型
    """
    torch.onnx.export(
        model,
        dummy_input,
        onnx_path,
        export_params=True,
        opset_version=14,  # ONNX opset版本
        do_constant_folding=True,
        input_names=['input'],
        output_names=['output'],
        dynamic_axes={
            'input': {0: 'batch_size'},
            'output': {0: 'batch_size'}
        }
    )

# 创建ONNX Runtime推理会话
def create_inference_session(onnx_path, providers=None):
    """
    创建推理会话
    """
    if providers is None:
        # 自动选择执行提供者
        providers = [
            ('CUDAExecutionProvider', {
                'device_id': 0,
                'arena_extend_strategy': 'kNextPowerOfTwo',
                'gpu_mem_limit': 2 * 1024 * 1024 * 1024,  # 2GB
                'cudnn_conv_algo_search': 'EXHAUSTIVE'
            }),
            'CPUExecutionProvider'
        ]

    # 创建会话
    session = ort.InferenceSession(
        onnx_path,
        providers=providers
    )

    return session

# 使用ONNX Runtime推理
def onnx_infer(session, input_data):
    """
    执行推理
    """
    # 获取输入输出名称
    input_name = session.get_inputs()[0].name
    output_name = session.get_outputs()[0].name

    # 执行推理
    outputs = session.run(
        [output_name],
        {input_name: input_data}
    )

    return outputs[0]

# 量化模型
def quantize_model(onnx_path, quantized_path):
    """
    量化ONNX模型
    """
    from onnxruntime.quantization import quantize_dynamic

    quantize_dynamic(
        onnx_path,
        quantized_path,
        weight_type=ort.QuantType.QUInt8  # 或QInt8
    )

# 完整示例
if __name__ == "__main__":
    # 1. 导出ONNX模型
    model = torch.load("model.pth")
    dummy_input = torch.randn(1, 3, 224, 224)
    export_to_onnx(model, dummy_input, "model.onnx")

    # 2. 量化模型（可选）
    quantize_model("model.onnx", "model_quantized.onnx")

    # 3. 创建推理会话
    session = create_inference_session("model.onnx")

    # 4. 获取模型信息
    print(f"输入: {[input.name for input in session.get_inputs()]}")
    print(f"输出: {[output.name for output in session.get_outputs()]}")

    # 5. 执行推理
    input_data = np.random.randn(1, 3, 224, 224).astype(np.float32)
    output = onnx_infer(session, input_data)

    print(f"推理结果: {output}")
