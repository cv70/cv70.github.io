加密货币风险管理

加密货币市场的高波动性和不成熟性带来了独特的风险挑战。从交易所倒闭到智能合约漏洞，从极端行情到监管冲击，风险管理是加密货币量化交易的核心能力。

⚠️ 加密货币特有风险

市场风险

  • 极端价格波动
  • 流动性枯竭
  • 市场操纵
  • 价格发现失灵

交易所风险

  • 交易所倒闭（如FTX）
  • 提现冻结
  • 系统宕机
  • 数据丢失

技术风险

  • 智能合约漏洞
  • 私钥泄露
  • 网络攻击
  • 51%攻击

监管风险

  • 政策不确定性
  • 交易所封禁
  • 税收政策变化
  • 跨境交易限制

# 加密货币风险管理系统
import numpy as np
import pandas as pd
from typing import Dict, List

class CryptoRiskManager:
    """加密货币风险管理器"""

    def __init__(self, config):
        self.config = config
        self.positions = {}
        self.risk_limits = config['risk_limits']

    def check_position_size(self, symbol, intended_size):
        """检查仓位大小是否合规"""
        max_size = self.risk_limits.get('max_position_size', {}).get(symbol, float('inf'))

        current_size = self.positions.get(symbol, 0)
        new_size = current_size + intended_size

        if abs(new_size) > max_size:
            return False, f"Position size {abs(new_size)} exceeds limit {max_size}"
        return True, "OK"

    def calculate_var(self, returns, confidence=0.95, method='historical'):
        """计算VaR（在险价值）"""
        if method == 'historical':
            # 历史模拟法
            var = np.percentile(returns, (1 - confidence) * 100)
        elif method == 'parametric':
            # 参数法（假设正态分布）
            mean = np.mean(returns)
            std = np.std(returns)
            var = mean + std * np.percentile(np.random.randn(10000), (1 - confidence) * 100)

        return var

    def calculate_max_drawdown(self, equity_curve):
        """计算最大回撤"""
        equity = pd.Series(equity_curve)

        # 计算累积最高点
        cummax = equity.cummax()

        # 计算回撤
        drawdown = (equity - cummax) / cummax

        # 最大回撤
        max_dd = drawdown.min()

        return max_dd

    def check_margin_requirement(self, exchange_account):
        """检查保证金充足性"""
        total_margin = exchange_account['margin_used']
        total_balance = exchange_account['balance']

        margin_ratio = total_margin / total_balance

        if margin_ratio > self.risk_limits['max_margin_ratio']:
            return False, f"Margin ratio {margin_ratio:.2%} exceeds limit"
        return True, "OK"

    def assess_exchange_risk(self, exchange):
        """评估交易所风险"""
        risk_score = 0

        # TVL（总锁定价值）
        if exchange['tvl'] < self.risk_limits['min_tvl']:
            risk_score += 2

        # 24h交易量
        if exchange['volume_24h'] < self.risk_limits['min_volume']:
            risk_score += 2

        # 监管合规性
        if not exchange['regulated']:
            risk_score += 3

        # 历史安全事故
        if exchange['security_incidents'] > 0:
            risk_score += exchange['security_incidents'] * 2

        # 风险等级
        if risk_score == 0:
            return 'low'
        elif risk_score <= 4:
            return 'medium'
        else:
            return 'high'

class PositionSizer:
    """仓位管理器"""

    def __init__(self, risk_per_trade=0.02):
        self.risk_per_trade = risk_per_trade

    def calculate_position_size(self, account_balance, entry_price, stop_loss_price):
        """根据风险计算仓位大小"""
        risk_amount = account_balance * self.risk_per_trade
        risk_per_share = abs(entry_price - stop_loss_price)

        if risk_per_share == 0:
            return 0

        position_size = risk_amount / risk_per_share
        return position_size

    def kelly_criterion_position(self, win_rate, avg_win, avg_loss, account_balance):
        """凯利公式计算仓位"""
        if avg_loss == 0:
            return 0

        win_loss_ratio = avg_win / abs(avg_loss)
        kelly_fraction = (win_rate * win_loss_ratio - (1 - win_rate)) / win_loss_ratio

        # 保守处理，使用半凯利
        position_size = account_balance * max(0, kelly_fraction) * 0.5

        return position_size

class StopLossManager:
    """止损管理器"""

    def __init__(self):
        self.stop_losses = {}

    def set_stop_loss(self, symbol, entry_price, method='percentage', params=None):
        """设置止损"""
        if method == 'percentage':
            stop_price = entry_price * (1 - params.get('percentage', 0.05))
        elif method == 'atr':
            atr = params.get('atr', 0)
            stop_price = entry_price - params.get('multiplier', 2) * atr
        elif method == 'support':
            stop_price = params.get('support_level', entry_price * 0.95)

        self.stop_losses[symbol] = {
            'entry_price': entry_price,
            'stop_price': stop_price,
            'method': method
        }

        return stop_price

    def check_stop_loss(self, symbol, current_price):
        """检查是否触发止损"""
        if symbol not in self.stop_losses:
            return False

        stop_price = self.stop_losses[symbol]['stop_price']

        if current_price <= stop_price:
            return True

        return False

class RiskMonitor:
    """风险监控器"""

    def __init__(self):
        self.alerts = []

    def monitor_risk_metrics(self, portfolio_data):
        """监控风险指标"""
        alerts = []

        # 检查VaR
        var_95 = portfolio_data.get('var_95', 0)
        if var_95 > portfolio_data.get('var_limit', 0):
            alerts.append({
                'type': 'VAR_EXCEEDED',
                'message': f'VaR {var_95:.2f} exceeds limit',
                'severity': 'high'
            })

        # 检查最大回撤
        max_dd = portfolio_data.get('max_drawdown', 0)
        if max_dd < portfolio_data.get('max_drawdown_limit', -0.2):
            alerts.append({
                'type': 'MAX_DRAWDOWN_EXCEEDED',
                'message': f'Max drawdown {max_dd:.2%} exceeded',
                'severity': 'critical'
            })

        # 检查仓位集中度
        concentration = portfolio_data.get('concentration', 0)
        if concentration > portfolio_data.get('concentration_limit', 0.3):
            alerts.append({
                'type': 'CONCENTRATION_HIGH',
                'message': f'Position concentration {concentration:.2%} too high',
                'severity': 'medium'
            })

        self.alerts = alerts
        return alerts
