🛡️ 安全中间件

helmet - 安全 HTTP 头

const helmet = require('helmet'); // 基本使用 - 启用所有安全头 app.use(helmet()); // 自定义配置 app.use(helmet({ contentSecurityPolicy: { directives: { defaultSrc: ["'self'"], styleSrc: ["'self'", "'unsafe-inline'"], scriptSrc: ["'self'", "trusted-cdn.com"] } }, crossOriginEmbedderPolicy: true, crossOriginOpenerPolicy: true, crossOriginResourcePolicy: { policy: "same-origin" }, dnsPrefetchControl: { allow: false }, frameguard: { action: 'deny' }, hidePoweredBy: true, hsts: { maxAge: 31536000, includeSubDomains: true }, ieNoOpen: true, noSniff: true, originAgentCluster: true, permittedCrossDomainPolicies: true, referrerPolicy: { policy: 'strict-origin-when-cross-origin' }, xssFilter: true })); // 单独使用某个功能 app.use(helmet.contentSecurityPolicy({ directives: { defaultSrc: ["'self'"] } }));

cors - 跨域处理

const cors = require('cors'); // 基本使用 - 允许所有来源 app.use(cors()); // 自定义配置 app.use(cors({ origin: 'https://example.com', // 指定来源 methods: ['GET', 'POST', 'PUT', 'DELETE'], allowedHeaders: ['Content-Type', 'Authorization'], credentials: true, // 允许携带凭证 maxAge: 86400, // 预检缓存时间 exposedHeaders: ['X-Custom-Header'] })); // 动态来源 app.use(cors({ origin: (origin, callback) => { const whitelist = ['https://example.com', 'https://api.example.com']; if (whitelist.indexOf(origin) !== -1 || !origin) { callback(null, true); } else { callback(new Error('Not allowed by CORS')); } } })); // 特定路由使用 app.get('/public', cors(), (req, res) => { res.json({ message: 'Public API' }); });

express-rate-limit - 速率限制

const rateLimit = require('express-rate-limit'); // API 限流 const apiLimiter = rateLimit({ windowMs: 15 * 60 * 1000, // 15 分钟 max: 100, // 最多 100 次请求 message: { error: 'Too many requests' }, standardHeaders: true, // 返回 RateLimit-* 头 legacyHeaders: false, // 禁用 X-RateLimit-* 头 handler: (req, res) => { res.status(429).json({ error: 'Too many requests, please try again later' }); } }); app.use('/api', apiLimiter); // 登录严格限流 const loginLimiter = rateLimit({ windowMs: 60 * 60 * 1000, // 1 小时 max: 5, // 最多 5 次尝试 skipSuccessfulRequests: true, // 成功请求不计入 handler: (req, res) => { res.status(429).json({ error: 'Too many login attempts, account temporarily locked' }); } }); app.post('/login', loginLimiter, loginController);

📝 日志中间件

morgan - HTTP 请求日志

const morgan = require('morgan'); const fs = require('fs'); const path = require('path'); // 基本使用 app.use(morgan('dev')); // 开发环境 app.use(morgan('combined')); // 生产环境 // 预定义格式 // combined - Apache 风格 // common - 简化的 Apache 风格 // dev - 带颜色的开发格式 // short - 简短格式 // tiny - 最小格式 // 写入文件 const accessLogStream = fs.createWriteStream( path.join(__dirname, 'access.log'), { flags: 'a' } ); app.use(morgan('combined', { stream: accessLogStream })); // 自定义格式 morgan.token('custom-date', () => { return new Date().toISOString(); }); app.use(morgan(':custom-date :method :url :status :response-time ms')); // 条件使用 if (process.env.NODE_ENV === 'development') { app.use(morgan('dev')); } else { app.use(morgan('combined', { stream: accessLogStream, skip: (req, res) => res.statusCode < 400 // 只记录错误 })); }

⚡ 性能中间件

compression - 响应压缩

const compression = require('compression'); // 基本使用 app.use(compression()); // 自定义配置 app.use(compression({ filter: (req, res) => { // 自定义过滤逻辑 if (req.headers['x-no-compression']) { return false; } return compression.filter(req, res); }, threshold: 1024, // 超过 1KB 才压缩 level: 6, // 压缩级别 0-9 chunkSize: 16384, // 分块大小 memLevel: 8 // 内存级别 })); // 压缩效果 // 原始大小: 100KB // 压缩后: 约 10-30KB（取决于内容类型）

🍪 Cookie 和 Session

cookie-parser - Cookie 解析

const cookieParser = require('cookie-parser'); // 基本使用 app.use(cookieParser()); // 带密钥 app.use(cookieParser('secret-key')); // 读取 Cookie app.get('/', (req, res) => { console.log(req.cookies); // 无签名的 cookies console.log(req.signedCookies); // 签名的 cookies res.send('OK'); }); // 设置 Cookie app.get('/set-cookie', (req, res) => { res.cookie('name', 'value', { maxAge: 900000, // 过期时间（毫秒） httpOnly: true, // 仅 HTTP 访问 secure: true, // 仅 HTTPS signed: true, // 签名 sameSite: 'strict' // CSRF 防护 }); res.send('Cookie set'); }); // 清除 Cookie app.get('/clear-cookie', (req, res) => { res.clearCookie('name'); res.send('Cookie cleared'); });

express-session - Session 管理

const session = require('express-session'); // 基本使用 app.use(session({ secret: 'your-secret-key', resave: false, saveUninitialized: false, cookie: { secure: false, // 生产环境应为 true httpOnly: true, maxAge: 3600000 // 1 小时 } })); // 使用 Session app.post('/login', (req, res) => { req.session.userId = user.id; req.session.role = user.role; res.json({ message: 'Logged in' }); }); app.get('/profile', (req, res) => { if (!req.session.userId) { return res.status(401).json({ error: 'Not authenticated' }); } res.json({ userId: req.session.userId }); }); app.post('/logout', (req, res) => { req.session.destroy(err => { if (err) { return res.status(500).json({ error: 'Logout failed' }); } res.json({ message: 'Logged out' }); }); });

📤 文件上传中间件

multer - 文件上传

const multer = require('multer'); const path = require('path'); // 基本配置 const upload = multer({ dest: 'uploads/' }); // 单文件上传 app.post('/upload/single', upload.single('avatar'), (req, res) => { console.log(req.file); res.json({ file: req.file }); }); // 多文件上传（同名字段） app.post('/upload/multiple', upload.array('photos', 5), (req, res) => { console.log(req.files); res.json({ files: req.files }); }); // 多字段上传 app.post('/upload/fields', upload.fields([ { name: 'avatar', maxCount: 1 }, { name: 'gallery', maxCount: 8 } ]), (req, res) => { console.log(req.files); res.json({ files: req.files }); }); // 自定义存储 const storage = multer.diskStorage({ destination: (req, file, cb) => { cb(null, 'uploads/'); }, filename: (req, file, cb) => { const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1E9); cb(null, file.fieldname + '-' + uniqueSuffix + path.extname(file.originalname)); } }); const uploadCustom = multer({ storage: storage, limits: { fileSize: 5 * 1024 * 1024, // 5MB files: 10 }, fileFilter: (req, file, cb) => { const allowedTypes = /jpeg|jpg|png|gif/; const extname = allowedTypes.test(path.extname(file.originalname).toLowerCase()); const mimetype = allowedTypes.test(file.mimetype); if (extname && mimetype) { cb(null, true); } else { cb(new Error('Only images are allowed')); } } });

📊 完整配置示例

const express = require('express'); const helmet = require('helmet'); const cors = require('cors'); const compression = require('compression'); const morgan = require('morgan'); const rateLimit = require('express-rate-limit'); const cookieParser = require('cookie-parser'); const session = require('express-session'); const app = express(); // ============ 安全中间件 ============ app.use(helmet()); // ============ CORS ============ app.use(cors({ origin: process.env.CORS_ORIGIN, credentials: true })); // ============ 速率限制 ============ const limiter = rateLimit({ windowMs: 15 * 60 * 1000, max: 100 }); app.use('/api', limiter); // ============ 压缩 ============ app.use(compression()); // ============ 日志 ============ if (process.env.NODE_ENV === 'development') { app.use(morgan('dev')); } else { app.use(morgan('combined')); } // ============ 请求解析 ============ app.use(express.json({ limit: '10kb' })); app.use(express.urlencoded({ extended: true })); // ============ Cookie & Session ============ app.use(cookieParser()); app.use(session({ secret: process.env.SESSION_SECRET, resave: false, saveUninitialized: false, cookie: { secure: process.env.NODE_ENV === 'production' } })); // ============ 静态文件 ============ app.use('/static', express.static('public')); // ============ 路由 ============ app.use('/api', require('./routes/api')); module.exports = app;

✅ 小结

📌
本节要点
  • helmet 提供安全 HTTP 头，cors 处理跨域
  • express-rate-limit 防止滥用和 DDoS 攻击
  • morgan 记录请求日志，compression 压缩响应
  • cookie-parser 和 express-session 管理会话
  • multer 处理文件上传，支持多种配置
