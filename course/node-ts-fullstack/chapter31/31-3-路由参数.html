📖 参数类型概述

在 Express 中，获取客户端数据有三种主要方式：

参数类型 位置 获取方式 示例
路由参数 URL 路径中 req.params /users/123 中的 123
查询参数 URL 问号后 req.query ?page=1&size=10
请求体 请求正文中 req.body JSON 表单数据
// 完整示例：展示三种参数 app.post('/users/:id/posts', (req, res) => { // 路由参数 const { id } = req.params; // URL 中的 :id // 查询参数 const { draft, notify } = req.query; // ?draft=true¬ify=yes // 请求体 const { title, content } = req.body; // POST 数据 res.json({ userId: id, draft: draft === 'true', notify: notify === 'yes', post: { title, content } }); }); // 请求示例： // POST /users/123/posts?draft=true¬ify=yes // Body: { "title": "Hello", "content": "World" }

🔢 路由参数详解

基本使用

// 单个参数 app.get('/users/:id', (req, res) => { console.log(req.params); // { id: '123' } res.send(`User ID: ${req.params.id}`); }); // 多个参数 app.get('/posts/:postId/comments/:commentId', (req, res) => { console.log(req.params); // { postId: '1', commentId: '2' } res.json(req.params); }); // 嵌套参数 app.get('/orgs/:orgId/teams/:teamId/members/:memberId', (req, res) => { const { orgId, teamId, memberId } = req.params; res.json({ orgId, teamId, memberId }); });

参数格式限制

// 只匹配数字 ID app.get('/users/:id(\\d+)', (req, res) => { // /users/123 -> 匹配 // /users/abc -> 404 res.send(`Numeric ID: ${req.params.id}`); }); // 只匹配特定长度 app.get('/code/:code([A-Z]{3,6})', (req, res) => { // 只匹配 3-6 位大写字母 res.send(`Code: ${req.params.code}`); }); // 匹配 UUID app.get('/items/:uuid([0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12})', (req, res) => { res.send(`UUID: ${req.params.uuid}`); }); // 匹配 MongoDB ObjectId app.get('/documents/:id([0-9a-fA-F]{24})', (req, res) => { res.send(`ObjectId: ${req.params.id}`); });

❓ 查询参数详解

基本使用

// GET /search?keyword=express&page=1&size=10 app.get('/search', (req, res) => { const { keyword, page = 1, size = 10 } = req.query; res.json({ keyword, page: parseInt(page), size: parseInt(size) }); }); // 多值参数（使用数组） // GET /filter?tags=node&tags=express&tags=typescript app.get('/filter', (req, res) => { const { tags } = req.query; // tags 可能是字符串或数组 const tagList = Array.isArray(tags) ? tags : [tags]; res.json({ tags: tagList }); });

分页查询封装

// 分页中间件 const pagination = (req, res, next) => { const page = parseInt(req.query.page) || 1; const size = parseInt(req.query.size) || 10; const limit = Math.min(size, 100); // 最大 100 条 const offset = (page - 1) * limit; req.pagination = { page, size: limit, offset }; next(); }; // 使用 app.get('/articles', pagination, async (req, res) => { const { page, size, offset } = req.pagination; // 模拟数据库查询 const articles = await Article.findAll({ offset, limit: size }); res.json({ data: articles, page, size, total: await Article.count() }); });

搜索和过滤

// 综合查询示例 // GET /products?category=electronics&minPrice=100&maxPrice=1000&sort=price&order=asc app.get('/products', async (req, res) => { const { category, minPrice, maxPrice, sort = 'createdAt', order = 'desc' } = req.query; // 构建查询条件 const where = {}; if (category) where.category = category; if (minPrice || maxPrice) { where.price = {}; if (minPrice) where.price.gte = parseFloat(minPrice); if (maxPrice) where.price.lte = parseFloat(maxPrice); } // 查询数据库 const products = await Product.findAll({ where, order: [[sort, order]] }); res.json({ data: products }); });

📦 请求体详解

解析中间件配置

const express = require('express'); const app = express(); // 解析 JSON 格式 app.use(express.json()); // 解析 URL 编码格式 app.use(express.urlencoded({ extended: true })); // 解析原始文本 app.use(express.text()); // 解析 Buffer app.use(express.raw()); // 使用 app.post('/api/json', (req, res) => { console.log(req.body); // JSON 对象 res.json(req.body); }); app.post('/api/form', (req, res) => { console.log(req.body); // 表单对象 res.json(req.body); });

🔄 参数预处理

使用中间件预处理

// 参数验证中间件 const validateUserId = (req, res, next) => { const { id } = req.params; // 检查是否为有效数字 if (!/^\d+$/.test(id)) { return res.status(400).json({ success: false, message: 'Invalid user ID format' }); } // 转换为数字并附加到请求对象 req.userId = parseInt(id); next(); }; // 使用中间件 app.get('/users/:id', validateUserId, async (req, res) => { const user = await User.findById(req.userId); res.json(user); });

通用参数处理器

// 解析整数参数 const parseIntParam = (paramName) => { return (req, res, next) => { const value = req.params[paramName]; const parsed = parseInt(value); if (isNaN(parsed)) { return res.status(400).json({ success: false, message: `Invalid ${paramName}: must be a number` }); } req[`${paramName}Int`] = parsed; next(); }; }; // 使用 app.get('/users/:id', parseIntParam('id'), (req, res) => { // req.idInt 已经是解析后的整数 res.send(`User ID: ${req.idInt}`); });

参数转换工厂

// 通用参数转换器 const transformParam = (paramName, transformer) => { return (req, res, next) => { try { req.params[paramName] = transformer(req.params[paramName]); next(); } catch (error) { res.status(400).json({ success: false, message: `Invalid ${paramName}: ${error.message}` }); } }; }; // 使用示例 app.get('/users/:id', transformParam('id', (val) => { const num = parseInt(val); if (isNaN(num)) throw new Error('Must be a number'); return num; }), (req, res) => { res.send(`User ID: ${req.params.id}`); } );

🛡️ 参数验证最佳实践

// 使用 Joi 进行验证 const Joi = require('joi'); // 验证模式 const userSchema = Joi.object({ name: Joi.string().min(2).max(50).required(), email: Joi.string().email().required(), age: Joi.number().integer().min(0).max(150), role: Joi.string().valid('user', 'admin').default('user') }); // 验证中间件 const validateBody = (schema) => { return (req, res, next) => { const { error, value } = schema.validate(req.body, { abortEarly: false, // 返回所有错误 stripUnknown: true // 移除未知字段 }); if (error) { const messages = error.details.map(d => d.message); return res.status(400).json({ success: false, errors: messages }); } req.body = value; // 使用验证后的值 next(); }; }; // 使用 app.post('/users', validateBody(userSchema), async (req, res) => { // req.body 已经过验证和转换 const user = await User.create(req.body); res.status(201).json(user); });

✅ 小结

📌
本节要点
  • 路由参数用 req.params，查询参数用 req.query，请求体用 req.body
  • 路由参数可以用正则限制格式，如 :id(\\d+)
  • 查询参数适合分页、过滤、搜索等场景
  • 使用中间件进行参数预处理和验证
  • 推荐使用 Joi 等库进行复杂的参数验证
