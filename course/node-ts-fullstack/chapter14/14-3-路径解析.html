📚 path.parse 方法

const path = require('path'); // 解析路径为对象 const filePath = '/user/docs/node/app.js'; const parsed = path.parse(filePath); console.log(parsed); // { // root: '/', // 根目录 // dir: '/user/docs/node', // 目录路径 // base: 'app.js', // 文件名（含扩展名） // ext: '.js', // 扩展名 // name: 'app' // 文件名（不含扩展名） // } // Windows 路径解析 const winPath = 'C:\\Users\\Admin\\Documents\\file.txt'; const winParsed = path.win32.parse(winPath); console.log(winParsed); // { // root: 'C:\\', // dir: 'C:\\Users\\Admin\\Documents', // base: 'file.txt', // ext: '.txt', // name: 'file' // } // 相对路径解析 console.log(path.parse('./src/app.js')); // { // root: '', // dir: './src', // base: 'app.js', // ext: '.js', // name: 'app' // } // 只有文件名 console.log(path.parse('app.js')); // { // root: '', // dir: '', // base: 'app.js', // ext: '.js', // name: 'app' // }

🔧 路径对象详解

const path = require('path'); // 路径对象各字段含义 // root - 根目录 // Unix: '/' 或空 // Windows: 'C:\\' 或空 // dir - 完整目录路径（不含文件名） // 包含 root // base - 文件名（含扩展名） // base = name + ext // ext - 扩展名（包含点） // 从最后一个点开始 // name - 文件名（不含扩展名） // 多级扩展名 console.log(path.parse('archive.tar.gz')); // { // root: '', // dir: '', // base: 'archive.tar.gz', // ext: '.gz', // name: 'archive.tar' // } // 无扩展名 console.log(path.parse('README')); // { // root: '', // dir: '', // base: 'README', // ext: '', // name: 'README' // } // 隐藏文件 console.log(path.parse('.gitignore')); // { // root: '', // dir: '', // base: '.gitignore', // ext: '', // name: '.gitignore' // }
💡
扩展名规则

ext 是最后一个点及其后面的内容。以点开头的隐藏文件如果后面没有其他点，则没有扩展名。

📦 path.format 方法

const path = require('path'); // 从对象创建路径字符串 const pathObj = { root: '/', dir: '/user/docs/node', base: 'app.js', ext: '.js', name: 'app' }; console.log(path.format(pathObj)); // '/user/docs/node/app.js' // 优先级规则 // dir 优先于 root // base 优先于 name + ext // 示例 1：使用 dir + base console.log(path.format({ dir: '/user/docs', base: 'file.txt' })); // '/user/docs/file.txt' // 示例 2：使用 root + name + ext console.log(path.format({ root: '/user/', name: 'file', ext: '.txt' })); // '/user/file.txt' // 示例 3：dir 覆盖 root console.log(path.format({ root: '/ignored/', dir: '/user/docs', name: 'file', ext: '.txt' })); // '/user/docs/file.txt' // 示例 4：base 覆盖 name + ext console.log(path.format({ dir: '/user/docs', base: 'config.json', name: 'app', ext: '.js' })); // '/user/docs/config.json'

⚙️ 实际应用

const path = require('path'); const fs = require('fs'); // 1. 修改文件扩展名 function changeExtension(filePath, newExt) { const parsed = path.parse(filePath); parsed.ext = newExt; parsed.base = parsed.name + newExt; return path.format(parsed); } console.log(changeExtension('/data/app.js', '.ts')); // '/data/app.ts' // 2. 获取文件信息 function getFileInfo(filePath) { const parsed = path.parse(filePath); return { directory: parsed.dir, filename: parsed.base, name: parsed.name, extension: parsed.ext, fullPath: path.resolve(filePath) }; } console.log(getFileInfo('./src/components/Button.tsx')); // { // directory: './src/components', // filename: 'Button.tsx', // name: 'Button', // extension: '.tsx', // fullPath: '/home/user/project/src/components/Button.tsx' // } // 3. 批量处理文件 function processFiles(files, targetDir, targetExt) { return files.map(file => { const parsed = path.parse(file); return path.join(targetDir, parsed.name + targetExt); }); } const files = ['a.js', 'b.js', 'c.js']; console.log(processFiles(files, './dist', '.min.js')); // ['./dist/a.min.js', './dist/b.min.js', './dist/c.min.js'] // 4. 根据扩展名分类文件 function classifyFiles(files) { return files.reduce((acc, file) => { const ext = path.extname(file).toLowerCase(); if (!acc[ext]) acc[ext] = []; acc[ext].push(file); return acc; }, {}); }

📝 本节小结

✅
  • • path.parse 将路径解析为对象
  • • 对象包含 root/dir/base/ext/name
  • • path.format 从对象创建路径
  • • dir 优先于 root，base 优先于 name+ext
