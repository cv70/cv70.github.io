37.4 请求与响应

深入理解 Koa 的请求响应处理

🎯 学习目标

  • 掌握获取各种请求参数的方法
  • 学会处理不同类型的请求体
  • 理解响应的设置和发送方式
  • 了解请求头和响应头的操作

📥 获取 URL 参数

// GET /users?page=2&limit=10&sort=name app.use(async ctx => { // 获取完整 URL console.log(ctx.url); // /users?page=2&limit=10&sort=name // 获取路径 console.log(ctx.path); // /users // 获取查询参数对象 console.log(ctx.query); // { page: '2', limit: '10', sort: 'name' } // 获取查询字符串 console.log(ctx.querystring); // page=2&limit=10&sort=name ctx.body = ctx.query; });
💡
参数类型

ctx.query 返回的是对象，所有值都是字符串类型，需要手动转换数字。

📮 获取路由参数

使用 koa-router 获取动态路由参数。

const Router = require('koa-router'); const router = new Router(); // 动态路由参数 router.get('/users/:id', async ctx => { const { id } = ctx.params; ctx.body = { userId: id }; }); // 多个参数 router.get('/posts/:postId/comments/:commentId', async ctx => { const { postId, commentId } = ctx.params; ctx.body = { postId, commentId }; }); app.use(router.routes());

📦 获取请求体

Koa 默认不解析请求体，需要使用中间件。

// 安装 koa-bodyparser npm install koa-bodyparser
const bodyParser = require('koa-bodyparser'); // 注册中间件 app.use(bodyParser()); app.use(async ctx => { // 获取 JSON 或表单数据 const body = ctx.request.body; ctx.body = { received: body }; });

JSON 请求

POST /api/users Content-Type: application/json { "name": "John", "email": "john@example.com" }

表单请求

POST /api/users Content-Type: application/x-www-form-urlencoded name=John&email=john@example.com

📋 获取请求头

app.use(async ctx => { // 获取所有请求头 console.log(ctx.headers); // 获取单个请求头 const contentType = ctx.get('Content-Type'); const authorization = ctx.get('Authorization'); const userAgent = ctx.get('User-Agent'); // 简写方式 console.log(ctx.header['content-type']); ctx.body = { contentType, authorization, userAgent }; });

📤 设置响应

app.use(async ctx => { // 设置状态码 ctx.status = 201; // 设置响应头 ctx.set('X-Custom-Header', 'value'); ctx.set({ 'X-Request-Id': '12345', 'X-Response-Time': '10ms' }); // 设置 Content-Type ctx.type = 'application/json'; // 或者 ctx.type = 'json'; // 设置响应体 - 不同类型 // 字符串 ctx.body = 'Hello World'; // 对象 (自动转 JSON) ctx.body = { message: 'Success', data: [] }; // Buffer ctx.body = Buffer.from('Hello'); // Stream ctx.body = fs.createReadStream('file.pdf'); });

🔄 重定向响应

app.use(async ctx => { // 临时重定向 (302) ctx.redirect('/login'); // 永久重定向 (301) ctx.redirect('/new-url'); // 返回上一页 ctx.redirect('back'); // 自定义重定向状态码 ctx.status = 301; ctx.redirect('/new-location'); });

📁 文件下载响应

const fs = require('fs'); app.use(async ctx => { // 设置为下载文件 ctx.attachment('report.pdf'); // 或者直接发送文件流 ctx.body = fs.createReadStream('./files/report.pdf'); }); // 封装下载函数 async function downloadFile(ctx, filepath, filename) { ctx.attachment(filename); ctx.body = fs.createReadStream(filepath); }

💻 完整示例

const Koa = require('koa'); const Router = require('koa-router'); const bodyParser = require('koa-bodyparser'); const app = new Koa(); const router = new Router(); app.use(bodyParser()); // GET 请求 - 查询列表 router.get('/api/users', async ctx => { const { page = 1, limit = 10 } = ctx.query; ctx.set('X-Total-Count', '100'); ctx.body = { success: true, data: [], pagination: { page: +page, limit: +limit } }; }); // GET 请求 - 获取单个 router.get('/api/users/:id', async ctx => { const { id } = ctx.params; ctx.body = { success: true, data: { id, name: 'User ' + id } }; }); // POST 请求 - 创建 router.post('/api/users', async ctx => { const user = ctx.request.body; ctx.status = 201; ctx.body = { success: true, data: { id: Date.now(), ...user } }; }); app.use(router.routes()); app.listen(3000);
