32.2 路由级中间件

掌握路由级中间件的使用场景和最佳实践

🎯 学习目标

  • 理解路由级中间件与应用级中间件的区别
  • 掌握 router.use() 的使用方法
  • 学会在路由模块中应用中间件
  • 理解中间件的作用域范围

📖 路由级中间件概述

路由级中间件与应用级中间件类似，但它绑定到 express.Router() 实例上。它只对挂载该路由器的请求生效，提供更好的模块化和隔离性。

对比应用级中间件

特性 应用级中间件 路由级中间件
绑定对象 app.use() router.use()
作用范围 整个应用 特定路由模块
生命周期 应用启动时加载 路由加载时生效
适合场景 全局功能 模块特定功能

🔧 router.use() 基本用法

基本示例

const express = require('express'); const router = express.Router(); // 路由级中间件 - 只对此路由模块生效 router.use((req, res, next) => { console.log('Router middleware'); next(); }); // 路由处理 router.get('/users', (req, res) => { res.json({ users: [] }); }); module.exports = router;

路径限定的路由级中间件

const router = express.Router(); // 只对 /admin 路径下的请求生效 router.use('/admin', (req, res, next) => { console.log('Admin section accessed'); next(); }); // 认证中间件 router.use('/admin', (req, res, next) => { if (!req.user || !req.user.isAdmin) { return res.status(403).json({ error: 'Admin access required' }); } next(); }); // 路由 router.get('/admin/dashboard', (req, res) => { res.json({ message: 'Admin Dashboard' }); });

📁 路由模块中的中间件

完整的路由模块示例

// routes/api/users.js const express = require('express'); const router = express.Router(); const userController = require('../../controllers/userController'); const authMiddleware = require('../../middleware/auth'); const validateMiddleware = require('../../middleware/validate'); // ============ 路由级中间件 ============ // 所有用户路由需要认证 router.use(authMiddleware.requireAuth); // 加载用户参数 router.param('id', async (req, res, next, id) => { try { const user = await User.findById(id); if (!user) { return res.status(404).json({ message: 'User not found' }); } req.targetUser = user; next(); } catch (error) { next(error); } }); // ============ 路由定义 ============ // 用户列表 router.get('/', userController.list ); // 创建用户 router.post('/', validateMiddleware.userSchema, userController.create ); // 获取单个用户 router.get('/:id', userController.show ); // 更新用户 router.put('/:id', validateMiddleware.userSchema, userController.update ); // 删除用户 - 需要管理员权限 router.delete('/:id', authMiddleware.requireAdmin, userController.destroy ); module.exports = router;

🔒 认证中间件示例

// middleware/auth.js const jwt = require('jsonwebtoken'); // 需要认证 exports.requireAuth = (req, res, next) => { const token = req.headers.authorization?.replace('Bearer ', ''); if (!token) { return res.status(401).json({ success: false, message: 'Authentication required' }); } try { const decoded = jwt.verify(token, process.env.JWT_SECRET); req.user = decoded; next(); } catch (error) { return res.status(401).json({ success: false, message: 'Invalid or expired token' }); } }; // 需要管理员权限 exports.requireAdmin = (req, res, next) => { if (!req.user || req.user.role !== 'admin') { return res.status(403).json({ success: false, message: 'Admin access required' }); } next(); }; // 可选认证（不强制） exports.optionalAuth = (req, res, next) => { const token = req.headers.authorization?.replace('Bearer ', ''); if (token) { try { const decoded = jwt.verify(token, process.env.JWT_SECRET); req.user = decoded; } catch (error) { // 忽略错误，继续但不设置用户 } } next(); };

🔄 中间件组合模式

1. 中间件工厂

// 创建角色检查中间件 const requireRole = (...roles) => { return (req, res, next) => { if (!req.user) { return res.status(401).json({ message: 'Authentication required' }); } if (!roles.includes(req.user.role)) { return res.status(403).json({ message: 'Insufficient permissions' }); } next(); }; }; // 使用 router.get('/admin', authMiddleware.requireAuth, requireRole('admin'), adminController.dashboard ); router.get('/editor', authMiddleware.requireAuth, requireRole('admin', 'editor'), editorController.dashboard );

2. 条件中间件

// 条件中间件 const conditionalMiddleware = (condition, middleware) => { return (req, res, next) => { if (condition(req)) { return middleware(req, res, next); } next(); }; }; // 使用：只对 POST 请求验证 router.post('/posts', conditionalMiddleware( req => req.method === 'POST', validatePostSchema ), postController.create );

3. 中间件链

// 组合多个中间件 const middlewareChain = (...middlewares) => { return (req, res, next) => { const executeMiddleware = (index) => { if (index >= middlewares.length) { return next(); } middlewares[index](req, res, (err) => { if (err) return next(err); executeMiddleware(index + 1); }); }; executeMiddleware(0); }; }; // 使用 router.get('/protected', middlewareChain( authMiddleware.requireAuth, authMiddleware.requireAdmin, rateLimitMiddleware ), protectedHandler );

📊 实战：API 路由完整示例

// routes/api/index.js const express = require('express'); const router = express.Router(); // API 通用中间件 router.use((req, res, next) => { // 设置响应类型 res.set('Content-Type', 'application/json'); // API 日志 console.log(`[API] ${req.method} ${req.path}`); next(); }); // API 速率限制 const apiLimiter = rateLimit({ windowMs: 15 * 60 * 1000, max: 100 }); router.use(apiLimiter); // 子路由 router.use('/users', require('./users')); router.use('/posts', require('./posts')); router.use('/auth', require('./auth')); // 404 处理 router.use((req, res) => { res.status(404).json({ error: 'API endpoint not found' }); }); module.exports = router; // routes/api/posts.js const express = require('express'); const router = express.Router(); const postController = require('../../controllers/postController'); const auth = require('../../middleware/auth'); // 获取文章列表 - 不需要认证 router.get('/', postController.list); // 获取单篇文章 - 不需要认证 router.get('/:id', postController.show); // 以下路由需要认证 router.use(auth.requireAuth); // 创建文章 router.post('/', validate.postSchema, postController.create ); // 更新文章 - 需要是作者 router.put('/:id', postController.checkOwner, postController.update ); // 删除文章 - 需要是作者或管理员 router.delete('/:id', postController.checkOwnerOrAdmin, postController.destroy ); module.exports = router;

✅ 小结

📌
本节要点
  • 路由级中间件绑定到 Router 实例，只对该路由模块生效
  • 使用 router.use() 添加路由级中间件
  • 适合模块特定的功能：认证、验证、权限控制等
  • 可以通过中间件工厂创建可配置的中间件
  • 中间件可以组合、链式调用，实现复杂逻辑

  • 创建一个路由级中间件，验证用户对资源的所有权
  • 实现一个中间件工厂，根据配置限制访问权限
  • 为 API 路由模块添加完整的认证和权限控制
  • 实现一个条件中间件，只对特定 HTTP 方法进行验证
