68.5 第一个测试用例

第十部分：测试篇 - 动手编写你的第一个测试

1. 项目结构

让我们创建一个简单的项目结构来开始我们的测试之旅。

my-project/ ├── src/ │ ├── utils/ │ │ └── calculator.ts # 要测试的代码 │ └── index.ts ├── tests/ │ └── utils/ │ └── calculator.test.ts # 测试文件 ├── jest.config.ts # Jest 配置 ├── tsconfig.json # TypeScript 配置 └── package.json

package.json

{ "name": "my-project", "version": "1.0.0", "scripts": { "test": "jest", "test:watch": "jest --watch", "test:coverage": "jest --coverage" }, "devDependencies": { "@types/jest": "^29.5.11", "@types/node": "^20.10.0", "jest": "^29.7.0", "ts-jest": "^29.1.1", "typescript": "^5.3.0" } }

2. 创建被测试代码

// src/utils/calculator.ts export class Calculator { /** * 加法运算 */ add(a: number, b: number): number { return a + b; } /** * 减法运算 */ subtract(a: number, b: number): number { return a - b; } /** * 乘法运算 */ multiply(a: number, b: number): number { return a * b; } /** * 除法运算 * @throws Error 当除数为 0 时抛出异常 */ divide(a: number, b: number): number { if (b === 0) { throw new Error('除数不能为 0'); } return a / b; } /** * 计算百分比 */ percentage(value: number, total: number): number { if (total === 0) { return 0; } return (value / total) * 100; } }

3. 编写测试文件

// tests/utils/calculator.test.ts import { Calculator } from '../../src/utils/calculator'; describe('Calculator', () => { let calculator: Calculator; // 在每个测试前创建新的 Calculator 实例 beforeEach(() => { calculator = new Calculator(); }); describe('add', () => { it('should add two positive numbers correctly', () => { const result = calculator.add(2, 3); expect(result).toBe(5); }); it('should add negative numbers correctly', () => { const result = calculator.add(-2, -3); expect(result).toBe(-5); }); it('should add zero correctly', () => { const result = calculator.add(5, 0); expect(result).toBe(5); }); it('should handle decimal numbers', () => { const result = calculator.add(0.1, 0.2); expect(result).toBeCloseTo(0.3); }); }); describe('subtract', () => { it('should subtract two numbers correctly', () => { expect(calculator.subtract(5, 3)).toBe(2); }); it('should handle negative results', () => { expect(calculator.subtract(3, 5)).toBe(-2); }); }); describe('multiply', () => { it('should multiply two numbers correctly', () => { expect(calculator.multiply(3, 4)).toBe(12); }); it('should return 0 when multiplying by 0', () => { expect(calculator.multiply(5, 0)).toBe(0); }); }); describe('divide', () => { it('should divide two numbers correctly', () => { expect(calculator.divide(10, 2)).toBe(5); }); it('should throw error when dividing by 0', () => { expect(() => { calculator.divide(10, 0); }).toThrow('除数不能为 0'); }); it('should handle decimal division', () => { expect(calculator.divide(10, 3)).toBeCloseTo(3.333); }); }); describe('percentage', () => { it('should calculate percentage correctly', () => { expect(calculator.percentage(50, 200)).toBe(25); }); it('should return 0 when total is 0', () => { expect(calculator.percentage(50, 0)).toBe(0); }); }); });

5. 测试结构解析

describe('Calculator', () => { // ↑ 测试套件名称 beforeEach(() => { // ↑ 每个测试前执行的钩子 }); describe('add', () => { // ↑ 嵌套的测试套件 it('should add two numbers', () => { // ↑ 测试用例描述 // 应该清晰描述测试的行为 const result = calculator.add(2, 3); // ↑ Arrange (准备)：创建测试数据 // Act (执行)：调用被测试的方法 // (这里已经在上一步完成) expect(result).toBe(5); // ↑ Assert (断言)：验证结果 }); }); });

AAA 模式

💡
AAA 模式
  • Arrange - 准备：创建测试数据和对象
  • Act - 执行：调用被测试的方法
  • Assert - 断言：验证结果是否符合预期

6. 常见断言示例

describe('常见断言', () => { test('相等性断言', () => { // 严格相等 (===) expect(1 + 1).toBe(2); expect(true).toBe(true); // 深度相等 (递归比较对象) expect({ a: 1 }).toEqual({ a: 1 }); expect([1, 2, 3]).toEqual([1, 2, 3]); }); test('真值断言', () => { expect(true).toBeTruthy(); expect(false).toBeFalsy(); expect(null).toBeNull(); expect(undefined).toBeUndefined(); expect(1).toBeDefined(); }); test('数字断言', () => { expect(5).toBeGreaterThan(3); expect(5).toBeGreaterThanOrEqual(5); expect(3).toBeLessThan(5); expect(0.1 + 0.2).toBeCloseTo(0.3); }); test('字符串断言', () => { expect('hello world').toMatch(/world/); expect('hello world').toContain('hello'); expect('hello world').toHaveLength(11); }); test('数组断言', () => { expect([1, 2, 3]).toContain(2); expect([1, 2, 3]).toHaveLength(3); expect([{ a: 1 }]).toContainEqual({ a: 1 }); }); test('异常断言', () => { expect(() => { throw new Error('error message'); }).toThrow(); expect(() => { throw new Error('error message'); }).toThrow('error message'); expect(() => { throw new Error('error message'); }).toThrow(/message/); }); test('对象断言', () => { const user = { name: 'John', age: 30 }; expect(user).toHaveProperty('name'); expect(user).toHaveProperty('age', 30); expect(user).toMatchObject({ name: 'John' }); }); test('取反断言', () => { expect(1).not.toBe(2); expect(true).not.toBeFalsy(); expect([1, 2, 3]).not.toContain(4); }); });

⚠️
覆盖率目标

一般项目建议覆盖率在 70-80% 以上，核心业务逻辑建议达到 90% 以上。但记住，高覆盖率不等于高质量测试，关键是测试关键路径和边界情况。

8. 本节小结

📚
关键要点
  • 测试文件应与源文件对应，命名为 *.test.ts 或 *.spec.ts
  • 使用 describe 组织测试，it/test 编写测试用例
  • 遵循 AAA 模式：Arrange、Act、Assert
  • 使用 beforeEach/afterEach 进行测试前后的设置和清理
  • 测试要覆盖正常情况、边界情况和异常情况
