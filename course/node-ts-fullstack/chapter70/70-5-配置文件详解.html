70.5 配置文件详解

第十部分：测试篇 - 深入了解 Jest 配置选项

1. 配置文件位置

Jest 按以下顺序查找配置文件：

  1. package.json 中的 "jest" 字段
  2. jest.config.js
  3. jest.config.ts
  4. jest.config.json
  5. 通过 --config 指定的文件
# 使用特定配置文件 jest --config custom-jest.config.ts

2. 核心配置选项

测试文件配置

module.exports = { // 测试环境 testEnvironment: 'node', // 或 'jsdom', '@custom/environment' // 根目录 rootDir: './', // 搜索测试文件的目录 roots: ['<rootDir>/src', '<rootDir>/tests'], // 测试文件匹配模式 testMatch: [ '**/__tests__/**/*.[jt]s?(x)', '**/?(*.)+(spec|test).[jt]s?(x)' ], // 测试文件忽略模式 testPathIgnorePatterns: [ '/node_modules/', '/dist/', '/coverage/' ], // 正则匹配测试文件（与 testMatch 互斥） testRegex: '.*\\.test\\.(ts|js)$', // 测试文件扩展名 moduleFileExtensions: ['ts', 'tsx', 'js', 'jsx', 'json', 'node'] };

3. TypeScript 配置

module.exports = { // 使用预设 preset: 'ts-jest', // 或手动配置 transform: { '^.+\\.tsx?$': [ 'ts-jest', { // 指定 tsconfig tsconfig: 'tsconfig.json', // 隔离模块（性能更好） isolatedModules: true, // 启用 ESM useESM: true, // 自定义编译器选项 compilerOptions: { module: 'CommonJS' } } ] }, // ESM 支持 extensionsToTreatAsEsm: ['.ts'], // 静态文件处理 transformIgnorePatterns: [ 'node_modules/(?!(module-to-transform)/)' ] };

4. 模块解析配置

module.exports = { // 路径别名 moduleNameMapper: { // 路径映射 '^@/(.*)$': '<rootDir>/src/$1', '^@tests/(.*)$': '<rootDir>/tests/$1', '^@services/(.*)$': '<rootDir>/src/services/$1', '^@models/(.*)$': '<rootDir>/src/models/$1', // CSS 模块 Mock '\\.css$': 'identity-obj-proxy', '\\.scss$': 'identity-obj-proxy', // 静态资源 Mock '\\.(jpg|jpeg|png|gif|svg)$': '<rootDir>/__mocks__/fileMock.js', '\\.(woff|woff2|ttf|eot)$': '<rootDir>/__mocks__/fileMock.js' }, // 模块搜索路径 modulePaths: ['<rootDir>/src'], // 忽略的模块路径 modulePathIgnorePatterns: ['<rootDir>/dist/'], // 模块目录 moduleDirectories: ['node_modules', 'src'] };

5. 测试环境配置

module.exports = { // 全局设置 globalSetup: '<rootDir>/tests/setup/global-setup.ts', globalTeardown: '<rootDir>/tests/setup/global-teardown.ts', // 环境设置文件 setupFiles: [ '<rootDir>/tests/setup/env.ts', 'dotenv/config' ], // 测试框架设置 setupFilesAfterEnv: [ '<rootDir>/tests/setup/jest-setup.ts', 'jest-extended/all', '<rootDir>/tests/setup/matchers.ts' ], // 测试环境选项 testEnvironmentOptions: { url: 'http://localhost:3000', userAgent: 'Mozilla/5.0...' }, // 项目引用（monorepo） projects: [ '<rootDir>/packages/*/jest.config.js' ] };

设置文件示例

// tests/setup/global-setup.ts import { setupDatabase } from '../helpers/database'; export default async function globalSetup() { console.log('Setting up test environment...'); await setupDatabase(); process.env.TEST_MODE = 'true'; } // tests/setup/jest-setup.ts import { toBeWithinRange } from './matchers'; expect.extend({ toBeWithinRange }); // 增加超时时间 jest.setTimeout(30000); // 全局 beforeAll/afterAll beforeAll(() => { console.log('Starting test suite'); }); afterAll(() => { console.log('Finished test suite'); });

6. 覆盖率配置

module.exports = { // 收集覆盖率 collectCoverage: false, // 通过 --coverage 启用 // 覆盖率收集范围 collectCoverageFrom: [ 'src/**/*.{ts,tsx}', '!src/**/*.d.ts', '!src/**/index.{ts,tsx}', '!src/main.ts', '!src/**/__tests__/**' ], // 输出目录 coverageDirectory: 'coverage', // 报告格式 coverageReporters: ['text', 'text-summary', 'lcov', 'html'], // 覆盖率提供者 coverageProvider: 'v8', // 阈值配置 coverageThreshold: { global: { branches: 80, functions: 80, lines: 80, statements: 80 }, './src/core/': { branches: 90, functions: 90 } }, // 忽略覆盖率路径 coveragePathIgnorePatterns: [ '/node_modules/', '/tests/' ] };

7. 性能优化配置

module.exports = { // 最大并行工作进程 maxWorkers: '50%', // 每个测试的超时时间 testTimeout: 5000, // 慢测试阈值 slowTestThreshold: 300, // 启用缓存 cache: true, cacheDirectory: '<rootDir>/.jest-cache', // 并发限制 maxConcurrency: 5, // Watch 插件 watchPlugins: [ 'jest-watch-typeahead/filename', 'jest-watch-typeahead/testname' ], // 强制退出 forceExit: true, // 检测未关闭的资源 detectOpenHandles: true, // 错误时立即退出 bail: 1, // 第一个失败后退出 // 或 bail: 3 // 前 3 个失败后退出 };

8. Mock 配置

module.exports = { // 自动 Mock automock: false, // 清除 Mock clearMocks: true, // 重置 Mock resetMocks: false, // 恢复 Mock restoreMocks: false, // Mock 路径 rootDir: './', roots: ['<rootDir>/src'], // 快照序列化器 snapshotSerializers: [ 'jest-serializer-html' ], // 快照格式 snapshotFormat: { escapeString: true, printBasicPrototype: true } };

9. 完整配置示例

// jest.config.ts import type { Config } from 'jest'; const config: Config = { // 基础配置 preset: 'ts-jest', testEnvironment: 'node', rootDir: './', // 文件匹配 roots: ['<rootDir>/src', '<rootDir>/tests'], testMatch: ['**/*.test.ts', '**/*.spec.ts'], testPathIgnorePatterns: ['/node_modules/', '/dist/'], moduleFileExtensions: ['ts', 'tsx', 'js', 'jsx', 'json'], // TypeScript transform: { '^.+\\.tsx?$': ['ts-jest', { tsconfig: 'tsconfig.json', isolatedModules: true }] }, // 模块解析 moduleNameMapper: { '^@/(.*)$': '<rootDir>/src/$1', '^@tests/(.*)$': '<rootDir>/tests/$1' }, // 覆盖率 collectCoverageFrom: [ 'src/**/*.ts', '!src/**/*.d.ts', '!src/main.ts' ], coverageDirectory: 'coverage', coverageReporters: ['text', 'lcov', 'html'], coverageThreshold: { global: { branches: 70, functions: 70, lines: 70, statements: 70 } }, // 环境 setupFilesAfterEnv: ['<rootDir>/tests/setup/jest-setup.ts'], // 性能 maxWorkers: '50%', testTimeout: 10000, cache: true, // 调试 verbose: true, forceExit: true, detectOpenHandles: true, // 随机顺序 seed: 1234, randomize: true }; export default config;

10. 本节小结
