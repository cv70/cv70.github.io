📚 DDD 核心概念

// 领域驱动设计（Domain-Driven Design） // 核心思想：以领域为核心，通过统一语言沟通，解决复杂业务问题 const dddCore = { 领域: '业务问题的范畴，如电商、支付、物流', 子域: '领域的细分，如订单子域、库存子域', 核心域: '业务核心竞争力所在的子域', 支撑域: '支撑核心业务的子域', 通用域: '多个子域共用的通用能力' }; // 战略设计 vs 战术设计 const designLevels = { 战略设计: { 目的: '划分系统边界，定义上下文关系', 要素: ['限界上下文', '上下文映射', '统一语言'], 关注: '宏观架构' }, 战术设计: { 目的: '细化模型设计，指导代码实现', 要素: ['实体', '值对象', '聚合', '领域服务', '仓储', '工厂'], 关注: '微观建模' } }; // 限界上下文（Bounded Context） // 定义模型的边界，同一概念在不同上下文中含义不同 const contexts = { 订单上下文: { 商品: '订单项中的商品快照', 用户: '下单用户信息' }, 商品上下文: { 商品: '完整的商品信息', 用户: '商品管理员' }, 用户上下文: { 商品: '用户收藏的商品', 用户: '完整的用户档案' } };
💡
统一语言

开发团队与业务专家使用相同的术语沟通，确保代码与业务概念一致。

🏗️ 分层架构

// DDD 分层架构 /* ┌─────────────────────────────────────┐ │ 用户界面层 (UI) │ │ Controller, GraphQL Resolver │ ├─────────────────────────────────────┤ │ 应用层 (Application) │ │ Application Service, DTO │ ├─────────────────────────────────────┤ │ 领域层 (Domain) │ │ Entity, Value Object, Aggregate │ │ Domain Service, Domain Event │ ├─────────────────────────────────────┤ │ 基础设施层 (Infrastructure) │ │ Repository, External Service │ └─────────────────────────────────────┘ */ // 各层职责 const layerResponsibilities = { 用户界面层: '处理用户请求，转换数据格式', 应用层: '协调领域对象完成用例，事务控制', 领域层: '核心业务逻辑，领域模型', 基础设施层: '技术实现，持久化，外部服务' }; // NestJS 项目结构示例 const projectStructure = ` src/ ├── modules/ │ ├── order/ # 订单模块（限界上下文） │ │ ├── domain/ # 领域层 │ │ │ ├── entities/ # 实体 │ │ │ ├── value-objects/ # 值对象 │ │ │ ├── aggregates/ # 聚合 │ │ │ ├── services/ # 领域服务 │ │ │ ├── events/ # 领域事件 │ │ │ └── repositories/ # 仓储接口 │ │ ├── application/ # 应用层 │ │ │ ├── services/ # 应用服务 │ │ │ ├── dto/ # 数据传输对象 │ │ │ └── commands/ # 命令 │ │ ├── infrastructure/ # 基础设施层 │ │ │ ├── persistence/ # 持久化 │ │ │ ├── repositories/ # 仓储实现 │ │ │ └── adapters/ # 适配器 │ │ └── interfaces/ # 接口层 │ │ ├── controllers/ # 控制器 │ │ └── resolvers/ # GraphQL解析器 │ ├── product/ # 商品模块 │ └── user/ # 用户模块 ├── shared/ # 共享内核 │ ├── domain/ # 通用领域基础 │ └── infrastructure/ # 通用基础设施 └── main.ts `;
