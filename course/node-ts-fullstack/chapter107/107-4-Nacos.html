🔧 服务注册与发现

// Nacos Node.js SDK import { NacosNamingClient } from 'nacos'; // 创建命名客户端 const namingClient = new NacosNamingClient({ serverList: '127.0.0.1:8848', namespace: 'public', logger: console }); await namingClient.ready(); // ==================== 服务注册 ==================== // 注册服务实例 await namingClient.registerInstance('user-service', { ip: '192.168.1.100', port: 3001, weight: 1, metadata: { version: 'v1', region: 'cn-east' } }); // 注册临时实例（心跳保活） await namingClient.registerInstance('order-service', { ip: '192.168.1.101', port: 3002, weight: 1, ephemeral: true // 临时实例，需要心跳 }, 'DEFAULT_GROUP'); // 注销服务 await namingClient.deregisterInstance('user-service', { ip: '192.168.1.100', port: 3001 }); // ==================== 服务发现 ==================== // 获取服务实例列表 const instances = await namingClient.getAllInstances('user-service'); console.log('Available instances:', instances); // 选择一个健康实例 const healthyInstances = instances.filter(i => i.healthy); const instance = healthyInstances[Math.floor(Math.random() * healthyInstances.length)]; console.log(`Selected: ${instance.ip}:${instance.port}`); // 订阅服务变化 namingClient.subscribe('user-service', (instances) => { console.log('Service instances updated:', instances.length); // 更新本地缓存 updateLocalCache(instances); }); // 取消订阅 namingClient.unsubscribe('user-service', handler);

🔀 配置管理

import { NacosConfigClient } from 'nacos'; // 创建配置客户端 const configClient = new NacosConfigClient({ serverAddr: '127.0.0.1:8848', namespace: 'public' }); // ==================== 配置操作 ==================== // 发布配置 await configClient.publishSingle({ dataId: 'database.yaml', group: 'DEFAULT_GROUP', content: ` host: localhost port: 3306 database: myapp username: root password: secret ` }); // 获取配置 const config = await configClient.getConfig( 'database.yaml', 'DEFAULT_GROUP' ); console.log('Config:', config); // 删除配置 await configClient.remove({ dataId: 'database.yaml', group: 'DEFAULT_GROUP' }); // ==================== 配置监听 ==================== // 监听配置变化 configClient.subscribe({ dataId: 'database.yaml', group: 'DEFAULT_GROUP' }, (content) => { console.log('Config changed:', content); // 解析新配置 const newConfig = yaml.parse(content); // 重新加载配置 reloadDatabaseConfig(newConfig); }); // 取消监听 configClient.removeListener({ dataId: 'database.yaml', group: 'DEFAULT_GROUP' }, listener); // ==================== 配置中心集成 ==================== class NacosConfigManager { private client: NacosConfigClient; private cache: Map = new Map(); constructor(serverAddr: string) { this.client = new NacosConfigClient({ serverAddr }); } // 获取配置（带缓存） async getConfig(dataId: string, parse: (content: string) => T): Promise { const cacheKey = dataId; if (this.cache.has(cacheKey)) { return this.cache.get(cacheKey); } const content = await this.client.getConfig(dataId, 'DEFAULT_GROUP'); const config = parse(content); this.cache.set(cacheKey, config); return config; } // 监听配置变化 async watchConfig( dataId: string, parse: (content: string) => T, onChange: (config: T) => void ) { // 初始加载 const config = await this.getConfig(dataId, parse); onChange(config); // 监听变化 this.client.subscribe({ dataId, group: 'DEFAULT_GROUP' }, (content) => { const newConfig = parse(content); this.cache.set(dataId, newConfig); onChange(newConfig); }); } } // 使用示例 const configManager = new NacosConfigManager('127.0.0.1:8848'); await configManager.watchConfig( 'app-config.yaml', (content) => yaml.parse(content), (config) => { console.log('App config updated:', config); } );
