72.1 测试模块配置

第十部分：测试篇 - NestJS 测试基础

1. NestJS 测试概述

NestJS 内置了强大的测试支持，使用 Jest 作为默认测试框架，提供了 @nestjs/testing 包来简化测试配置。

💡
NestJS 测试优势
  • 内置测试工具包
  • 依赖注入便于 Mock
  • 模块化架构便于隔离测试
  • CLI 自动生成测试模板

2. 安装和配置

# NestJS 项目默认已安装测试依赖 # 如果需要手动安装 npm install --save-dev @nestjs/testing jest ts-jest @types/jest

jest 配置

// jest.config.js module.exports = { moduleFileExtensions: ['js', 'json', 'ts'], rootDir: 'src', testRegex: '.*\\.spec\\.ts$', transform: { '^.+\\.(t|j)s$': 'ts-jest' }, collectCoverageFrom: [ '**/*.(t|j)s' ], coverageDirectory: '../coverage', testEnvironment: 'node' };

3. Test.createTestingModule

import { Test, TestingModule } from '@nestjs/testing'; import { UserService } from './user.service'; import { UserRepository } from './user.repository'; describe('UserService', () => { let service: UserService; let module: TestingModule; beforeEach(async () => { // 创建测试模块 module = await Test.createTestingModule({ providers: [ UserService, { provide: UserRepository, useValue: { findById: jest.fn(), create: jest.fn(), update: jest.fn(), } } ] }).compile(); // 获取服务实例 service = module.get<UserService>(UserService); }); it('should be defined', () => { expect(service).toBeDefined(); }); });

4. Mock Provider

// 方式一：使用 useValue { provide: UserRepository, useValue: { findById: jest.fn().mockResolvedValue({ id: '1', name: 'John' }), create: jest.fn(), } } // 方式二：使用 useClass { provide: UserRepository, useClass: MockUserRepository } // 方式三：使用 useFactory { provide: 'DATABASE_CONNECTION', useFactory: () => ({ query: jest.fn(), execute: jest.fn(), }) } // 方式四：使用 existing { provide: 'ALIAS_NAME', useExisting: UserRepository }

创建 Mock 类

class MockUserRepository { private data = []; async findById(id: string) { return this.data.find(u => u.id === id); } async create(userData: any) { const user = { id: Date.now().toString(), ...userData }; this.data.push(user); return user; } async clear() { this.data = []; } } // 在测试中使用 beforeEach(async () => { const mockRepository = new MockUserRepository(); module = await Test.createTestingModule({ providers: [ UserService, { provide: UserRepository, useValue: mockRepository } ] }).compile(); });

5. 覆盖模块中的 Provider

import { Test, TestingModule } from '@nestjs/testing'; import { AppModule } from './app.module'; import { ConfigService } from '@nestjs/config'; describe('App Module', () => { let module: TestingModule; beforeEach(async () => { module = await Test.createTestingModule({ imports: [AppModule] }) .overrideProvider(ConfigService) .useValue({ get: jest.fn((key: string) => { switch (key) { case 'DATABASE_URL': return 'sqlite::memory:'; default: return null; } }) }) .overrideProvider('DATABASE_CONNECTION') .useFactory({ factory: () => createMockDatabase() }) .compile(); }); });

6. 测试异步 Provider

// 异步 Provider { provide: 'ASYNC_SERVICE', useFactory: async (configService: ConfigService) => { const url = configService.get('API_URL'); const client = await connectToService(url); return client; }, inject: [ConfigService] } // 测试异步 Provider beforeEach(async () => { module = await Test.createTestingModule({ providers: [ { provide: ConfigService, useValue: { get: jest.fn().mockReturnValue('http://test-api.com') } }, { provide: 'ASYNC_SERVICE', useFactory: async (config: ConfigService) => { return { getData: jest.fn().mockResolvedValue({ data: 'mocked' }) }; }, inject: [ConfigService] } ] }).compile(); const service = await module.get('ASYNC_SERVICE'); });

7. 测试守卫和拦截器

import { Test, TestingModule } from '@nestjs/testing'; import { ExecutionContext, CallHandler } from '@nestjs/common'; import { AuthGuard } from './auth.guard'; import { JwtService } from '@nestjs/jwt'; describe('AuthGuard', () => { let guard: AuthGuard; let jwtService: JwtService; beforeEach(async () => { const module = await Test.createTestingModule({ providers: [ AuthGuard, { provide: JwtService, useValue: { verify: jest.fn() } } ] }).compile(); guard = module.get<AuthGuard>(AuthGuard); jwtService = module.get<JwtService>(JwtService); }); it('should allow access with valid token', async () => { const mockContext = { switchToHttp: () => ({ getRequest: () => ({ headers: { authorization: 'Bearer valid-token' } }) }) } as ExecutionContext; jwtService.verify.mockReturnValue({ userId: '123' }); const result = await guard.canActivate(mockContext); expect(result).toBe(true); }); it('should deny access without token', async () => { const mockContext = { switchToHttp: () => ({ getRequest: () => ({ headers: {} }) }) } as ExecutionContext; await expect(guard.canActivate(mockContext)).rejects.toThrow(); }); });

8. 本节小结

📚
关键要点
  • 使用 Test.createTestingModule 创建测试模块
  • 使用 useValue/useClass/useFactory Mock Provider
  • 使用 overrideProvider 覆盖模块中的依赖
  • 通过 module.get() 获取服务实例
  • 测试守卫和拦截器需要模拟 ExecutionContext
