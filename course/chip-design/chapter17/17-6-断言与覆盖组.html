✅ 断言

// 即时断言
always @(posedge clk) begin
    assert (state != IDLE || req == 0)
        else $error("Invalid request in IDLE state");
end

// 并发断言
property req_grant;
    @(posedge clk) req |-> ##[1:10] grant;
endproperty

assert property (req_grant)
    else $error("Grant not received within 10 cycles");
