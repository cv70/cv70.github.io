📚 Plugin 概述

Plugin 用于扩展 Webpack 功能，执行范围更广的任务，从打包优化到资源管理。

# Plugin 与 Loader 的区别 Loader: 转换特定类型的文件 Plugin: 执行更广泛的任务，监听构建生命周期 # 使用方式 const Plugin = require('some-webpack-plugin'); module.exports = { plugins: [ new Plugin({ options }), ], }; # 常用插件分类 1. HTML 生成 2. CSS 提取和优化 3. 代码压缩 4. 环境变量注入 5. 构建分析

🔧 HTML 相关插件

const HtmlWebpackPlugin = require('html-webpack-plugin'); module.exports = { plugins: [ new HtmlWebpackPlugin({ // 模板文件 template: './src/index.html', // 输出文件名 filename: 'index.html', // 注入位置 inject: 'body', // 'head' | 'body' | false // 压缩 HTML minify: { collapseWhitespace: true, removeComments: true, removeRedundantAttributes: true, }, // 指定引入的 chunk chunks: ['main'], // 标题 title: 'My App', // 自定义变量 templateParameters: { ENV: 'production', }, // favicon favicon: './src/favicon.ico', }), // 多页面应用 new HtmlWebpackPlugin({ template: './src/pages/home.html', filename: 'home.html', chunks: ['home'], }), new HtmlWebpackPlugin({ template: './src/pages/about.html', filename: 'about.html', chunks: ['about'], }), ], };

⚙️ CSS 处理插件

const MiniCssExtractPlugin = require('mini-css-extract-plugin'); const CssMinimizerPlugin = require('css-minimizer-webpack-plugin'); module.exports = { module: { rules: [ { test: /\.css$/, use: [ // 开发环境用 style-loader，生产环境用 MiniCssExtractPlugin process.env.NODE_ENV === 'production' ? MiniCssExtractPlugin.loader : 'style-loader', 'css-loader', ], }, ], }, plugins: [ new MiniCssExtractPlugin({ filename: 'css/[name].[contenthash:8].css', chunkFilename: 'css/[name].[contenthash:8].chunk.css', }), ], optimization: { minimizer: [ new CssMinimizerPlugin(), ], }, }; // CopyWebpackPlugin - 复制静态资源 const CopyPlugin = require('copy-webpack-plugin'); module.exports = { plugins: [ new CopyPlugin({ patterns: [ { from: 'public', to: 'public' }, { from: 'src/assets', to: 'assets' }, ], }), ], };
💡
CSS 提取

MiniCssExtractPlugin 将 CSS 提取到独立文件，有利于缓存和并行加载，但开发环境建议使用 style-loader 支持 HMR。

🚀 环境和优化插件

const webpack = require('webpack'); const { BundleAnalyzerPlugin } = require('webpack-bundle-analyzer'); module.exports = { plugins: [ // 定义环境变量 new webpack.DefinePlugin({ 'process.env.NODE_ENV': JSON.stringify(process.env.NODE_ENV), 'process.env.API_URL': JSON.stringify('https://api.example.com'), VERSION: JSON.stringify(require('./package.json').version), }), // 提供全局变量 new webpack.ProvidePlugin({ $: 'jquery', _: 'lodash', React: 'react', }), // 进度显示 new webpack.ProgressPlugin(), // 模块热替换 new webpack.HotModuleReplacementPlugin(), // 构建分析 new BundleAnalyzerPlugin({ analyzerMode: 'static', reportFilename: 'report.html', openAnalyzer: false, }), // IgnorePlugin - 忽略特定模块 new webpack.IgnorePlugin({ resourceRegExp: /^\.\/locale$/, contextRegExp: /moment$/, }), ], }; // 生产环境配置 const TerserPlugin = require('terser-webpack-plugin'); module.exports = { optimization: { minimize: true, minimizer: [ new TerserPlugin({ parallel: true, terserOptions: { compress: { drop_console: true, }, }, }), ], }, };

📊 缓存优化插件

// WorkboxWebpackPlugin - PWA 支持 const WorkboxPlugin = require('workbox-webpack-plugin'); module.exports = { plugins: [ new WorkboxPlugin.GenerateSW({ clientsClaim: true, skipWaiting: true, include: [/\.html$/, /\.js$/, /\.css$/], }), ], }; // CompressionPlugin - Gzip 压缩 const CompressionPlugin = require('compression-webpack-plugin'); module.exports = { plugins: [ new CompressionPlugin({ algorithm: 'gzip', test: /\.(js|css|html|svg)$/, threshold: 10240, // 10KB 以上才压缩 minRatio: 0.8, }), ], }; // HashedModuleIdsPlugin - 稳定的模块 ID module.exports = { plugins: [ new webpack.HashedModuleIdsPlugin(), ], optimization: { runtimeChunk: 'single', splitChunks: { chunks: 'all', cacheGroups: { vendor: { test: /[\\/]node_modules[\\/]/, name: 'vendors', chunks: 'all', }, }, }, }, };
