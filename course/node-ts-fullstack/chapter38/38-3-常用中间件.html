38.3 常用中间件

Koa 生态中必备的中间件

  • 掌握常用 Koa 中间件的安装和配置
  • 学会使用路由、请求体解析等核心中间件
  • 了解跨域、压缩、静态文件等实用中间件
  • 掌握中间件的最佳实践

🔀 koa-router 路由中间件

const Koa = require('koa'); const Router = require('koa-router'); const app = new Koa(); const router = new Router({ prefix: '/api' }); // 定义路由 router.get('/users', async ctx => { ctx.body = { users: [] }; }); router.post('/users', async ctx => { ctx.body = { message: 'Created' }; }); // 注册路由 app.use(router.routes()); app.use(router.allowedMethods()); app.listen(3000);

📥 koa-bodyparser 请求体解析

const bodyParser = require('koa-bodyparser'); app.use(bodyParser({ enableTypes: ['json', 'form'], jsonLimit: '1mb', formLimit: '1mb' })); app.use(async ctx => { const body = ctx.request.body; ctx.body = { received: body }; });
💡
支持的类型

koa-bodyparser 支持 JSON 和表单数据，不支持文件上传。如需文件上传，请使用 koa-body。

📁 koa-body 文件上传

const { koaBody } = require('koa-body'); app.use(koaBody({ multipart: true, // 支持文件上传 formidable: { uploadDir: './uploads', // 上传目录 keepExtensions: true, // 保留文件扩展名 maxFileSize: 10 * 1024 * 1024, // 最大 10MB } })); // 处理文件上传 router.post('/upload', async ctx => { const file = ctx.request.files.file; ctx.body = { message: 'File uploaded', filename: file.newFilename, path: file.filepath }; }); // 处理表单数据 router.post('/submit', async ctx => { const { name, email } = ctx.request.body; ctx.body = { name, email }; });

🌐 @koa/cors 跨域处理

const cors = require('@koa/cors'); // 简单使用 app.use(cors()); // 详细配置 app.use(cors({ origin: 'https://example.com', // 允许的源 allowMethods: ['GET', 'POST', 'PUT', 'DELETE'], allowHeaders: ['Content-Type', 'Authorization'], exposeHeaders: ['X-Total-Count'], credentials: true, // 允许携带 cookie maxAge: 86400 // 预检请求缓存时间 })); // 动态配置 app.use(cors({ origin: ctx => { const allowedOrigins = ['http://localhost:3000', 'https://example.com']; const requestOrigin = ctx.get('Origin'); return allowedOrigins.includes(requestOrigin) ? requestOrigin : ''; } }));

📄 koa-static 静态文件

const serve = require('koa-static'); const path = require('path'); // 提供静态文件服务 app.use(serve(path.join(__dirname, 'public'))); // 多个静态目录 app.use(serve(path.join(__dirname, 'public'))); app.use(serve(path.join(__dirname, 'uploads'))); // 带配置项 app.use(serve(path.join(__dirname, 'public'), { maxage: 86400000, // 缓存时间（毫秒） gzip: true, // 启用 gzip index: 'index.html' // 默认文件 }));

📉 koa-compress 响应压缩

const compress = require('koa-compress'); app.use(compress({ filter: contentType => { return /text|json|javascript|css/i.test(contentType); }, threshold: 1024, // 超过 1KB 才压缩 gzip: { flush: require('zlib').constants.Z_SYNC_FLUSH } }));
✅
压缩效果

对于大型 JSON 响应或文本内容，压缩可以显著减少传输数据量，提高响应速度。

📝 koa-logger 请求日志

const logger = require('koa-logger'); // 使用默认日志 app.use(logger()); // 自定义日志格式 app.use(logger({ transporter: (str, args) => { // 发送到日志服务 console.log(str); } }));

🔐 koa-session Session 管理

const session = require('koa-session'); // 配置 app.keys = ['secret-key', 'another-secret']; const CONFIG = { key: 'koa:sess', // cookie key maxAge: 86400000, // 过期时间（毫秒） autoCommit: true, overwrite: true, httpOnly: true, signed: true, rolling: false, // 每次请求更新过期时间 renew: false, }; app.use(session(CONFIG, app)); // 使用 session router.get('/login', async ctx => { ctx.session.user = { id: 1, name: 'John' }; ctx.body = { message: 'Logged in' }; }); router.get('/profile', async ctx => { const user = ctx.session.user; if (!user) { ctx.status = 401; ctx.body = { error: 'Not logged in' }; return; } ctx.body = { user }; }); router.get('/logout', async ctx => { ctx.session = null; ctx.body = { message: 'Logged out' }; });

💻 完整项目配置示例

const Koa = require('koa'); const Router = require('koa-router'); const bodyParser = require('koa-bodyparser'); const cors = require('@koa/cors'); const compress = require('koa-compress'); const logger = require('koa-logger'); const serve = require('koa-static'); const path = require('path'); const app = new Koa(); const router = new Router({ prefix: '/api' }); // 1. 错误处理中间件（最外层） app.use(async (ctx, next) => { try { await next(); } catch (err) { ctx.status = err.status || 500; ctx.body = { error: err.message }; ctx.app.emit('error', err, ctx); } }); // 2. 日志中间件 app.use(logger()); // 3. 压缩中间件 app.use(compress()); // 4. 跨域中间件 app.use(cors({ origin: '*' })); // 5. 请求体解析中间件 app.use(bodyParser()); // 6. 静态文件服务 app.use(serve(path.join(__dirname, 'public'))); // 7. 路由 router.get('/users', async ctx => { ctx.body = { users: [] }; }); app.use(router.routes()); app.use(router.allowedMethods()); app.listen(3000);
