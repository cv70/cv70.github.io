🛡️ 全局错误处理中间件

// 最外层错误处理中间件 app.use(async (ctx, next) => { try { await next(); } catch (err) { // 设置状态码 ctx.status = err.status || err.statusCode || 500; // 设置响应体 ctx.body = { success: false, error: err.message, code: err.code || 'INTERNAL_ERROR' }; // 开发环境返回堆栈信息 if (process.env.NODE_ENV === 'development') { ctx.body.stack = err.stack; } // 触发应用级错误事件 ctx.app.emit('error', err, ctx); } });
💡
中间件顺序

错误处理中间件必须放在最外层（最先注册），才能捕获所有后续中间件的错误。

⚠️ 自定义错误类

// errors/AppError.js class AppError extends Error { constructor(message, statusCode = 500, code = 'APP_ERROR') { super(message); this.statusCode = statusCode; this.code = code; this.status = statusCode; Error.captureStackTrace(this, this.constructor); } } // 错误类型定义 class ValidationError extends AppError { constructor(message) { super(message, 400, 'VALIDATION_ERROR'); } } class UnauthorizedError extends AppError { constructor(message = 'Unauthorized') { super(message, 401, 'UNAUTHORIZED'); } } class ForbiddenError extends AppError { constructor(message = 'Forbidden') { super(message, 403, 'FORBIDDEN'); } } class NotFoundError extends AppError { constructor(message = 'Resource not found') { super(message, 404, 'NOT_FOUND'); } } class ConflictError extends AppError { constructor(message) { super(message, 409, 'CONFLICT'); } } module.exports = { AppError, ValidationError, UnauthorizedError, ForbiddenError, NotFoundError, ConflictError };

💻 使用自定义错误

const { ValidationError, NotFoundError, UnauthorizedError } = require('./errors'); // 验证错误 router.post('/users', async ctx => { const { name, email } = ctx.request.body; if (!name || !email) { throw new ValidationError('Name and email are required'); } ctx.body = { message: 'User created' }; }); // 资源未找到 router.get('/users/:id', async ctx => { const user = await User.findById(ctx.params.id); if (!user) { throw new NotFoundError('User not found'); } ctx.body = user; }); // 未授权 router.get('/profile', async ctx => { if (!ctx.state.user) { throw new UnauthorizedError('Please login first'); } ctx.body = ctx.state.user; });

📡 全局错误事件监听

// 监听应用级错误事件 app.on('error', (err, ctx) => { // 记录错误日志 console.error('Global error handler:'); console.error('Status:', err.status || 500); console.error('Message:', err.message); console.error('URL:', ctx.url); console.error('Method:', ctx.method); console.error('Stack:', err.stack); // 发送到错误监控服务 // sendToSentry(err, ctx); // sendToLogService(err, ctx); }); // 处理未捕获的 Promise rejection process.on('unhandledRejection', (reason, promise) => { console.error('Unhandled Rejection:', reason); }); // 处理未捕获的异常 process.on('uncaughtException', (err) => { console.error('Uncaught Exception:', err); process.exit(1); });

📝 错误日志记录

// middlewares/errorLogger.js const fs = require('fs'); const path = require('path'); const logStream = fs.createWriteStream( path.join(__dirname, '../logs/error.log'), { flags: 'a' } ); function errorLogger() { return async (ctx, next) => { try { await next(); } catch (err) { // 记录错误日志 const logEntry = { timestamp: new Date().toISOString(), method: ctx.method, url: ctx.url, status: err.status || 500, message: err.message, stack: err.stack, body: ctx.request.body, query: ctx.query, ip: ctx.ip }; logStream.write(JSON.stringify(logEntry) + '\n'); throw err; // 继续抛出给上层处理 } }; } module.exports = errorLogger;

📊 生产环境错误处理

// 完整的生产环境错误处理 const { AppError } = require('./errors'); app.use(async (ctx, next) => { try { await next(); } catch (err) { // 记录错误 console.error(err); // 判断错误类型 const isAppError = err instanceof AppError; const isDev = process.env.NODE_ENV === 'development'; // 设置状态码 ctx.status = err.status || 500; // 构建响应 const response = { success: false, error: isAppError ? err.message : 'Internal Server Error', code: err.code || 'INTERNAL_ERROR' }; // 开发环境额外信息 if (isDev) { response.stack = err.stack; response.detail = err.detail; } // 生产环境隐藏内部错误详情 if (!isAppError && !isDev) { response.error = 'An unexpected error occurred'; } ctx.body = response; // 触发错误事件 ctx.app.emit('error', err, ctx); } }); // 404 处理 app.use(async ctx => { ctx.status = 404; ctx.body = { success: false, error: 'Not Found', code: 'NOT_FOUND' }; });

🔧 完整错误处理方案

// app.js - 完整示例 const Koa = require('koa'); const app = new Koa(); // 1. 全局错误事件监听 app.on('error', (err, ctx) => { // 发送到 Sentry / LogRocket 等 console.error('App error:', err); }); // 2. 错误处理中间件（最外层） app.use(async (ctx, next) => { try { await next(); } catch (err) { ctx.status = err.status || 500; ctx.body = { success: false, error: err.message, code: err.code || 'ERROR' }; ctx.app.emit('error', err, ctx); } }); // 3. 404 处理中间件（最内层） app.use(async (ctx, next) => { await next(); if (ctx.status === 404 && !ctx.body) { ctx.status = 404; ctx.body = { success: false, error: 'Not Found', code: 'NOT_FOUND' }; } }); // 4. 业务路由 app.use(router.routes()); app.listen(3000);
