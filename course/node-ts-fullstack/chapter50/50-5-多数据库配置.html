50.5 多数据库配置

配置和管理多个数据库连接

🎯 学习目标

  • 理解多数据库配置场景
  • 掌握 TypeORM 多数据库配置
  • 掌握 Prisma 多数据库配置
  • 了解读写分离配置

📚 多数据库应用场景

  • 读写分离：主库写，从库读
  • 微服务数据隔离：每个服务独立数据库
  • 多租户系统：每个租户独立数据库
  • 数据分片：按业务分库
  • 遗留系统集成：连接不同类型的数据库

🔧 TypeORM 多数据库配置

// app.module.ts import { Module } from '@nestjs/common'; import { TypeOrmModule } from '@nestjs/typeorm'; @Module({ imports: [ // 主数据库 TypeOrmModule.forRoot({ name: 'primary', type: 'postgres', host: 'localhost', port: 5432, username: 'postgres', password: 'password', database: 'primary_db', entities: [__dirname + '/**/*.entity{.ts,.js}'], }), // 从数据库 TypeOrmModule.forRoot({ name: 'secondary', type: 'mysql', host: 'localhost', port: 3306, username: 'root', password: 'password', database: 'secondary_db', entities: [__dirname + '/**/*.secondary.entity{.ts,.js}'], }), ], }) export class AppModule {} // 模块中注册 Repository @Module({ imports: [ // 指定使用哪个数据库连接 TypeOrmModule.forFeature([User], 'primary'), TypeOrmModule.forFeature([Log], 'secondary'), ], }) export class UsersModule {} // 服务中注入 @Injectable() export class UsersService { constructor( @InjectRepository(User, 'primary') private usersRepository: Repository, @InjectRepository(Log, 'secondary') private logsRepository: Repository, ) {} async findUser(id: number) { return this.usersRepository.findOneBy({ id }); } async createLog(data: any) { return this.logsRepository.save(data); } }
💡
连接名称

使用 name 属性区分不同的数据库连接。InjectRepository 的第二个参数指定使用的连接名称。

📦 TypeORM 读写分离

// app.module.ts TypeOrmModule.forRoot({ type: 'mysql', replication: { master: { host: 'master.db.com', port: 3306, username: 'root', password: 'password', database: 'mydb', }, slaves: [ { host: 'slave1.db.com', port: 3306, username: 'root', password: 'password', database: 'mydb', }, { host: 'slave2.db.com', port: 3306, username: 'root', password: 'password', database: 'mydb', }, ], }, entities: [__dirname + '/**/*.entity{.ts,.js}'], }) // 写操作自动使用主库 await repository.save(user); // master // 读操作自动使用从库 await repository.find(); // slave // 强制使用主库 await repository.find({ cache: false, }); // 负载均衡到 slave

🔄 Prisma 多数据库

// 创建多个 Prisma Service // prisma/prisma-primary.service.ts @Injectable() export class PrismaPrimaryService extends PrismaClient implements OnModuleInit { constructor() { super({ datasources: { db: { url: process.env.DATABASE_URL_PRIMARY, }, }, }); } async onModuleInit() { await this.$connect(); } } // prisma/prisma-secondary.service.ts @Injectable() export class PrismaSecondaryService extends PrismaClient implements OnModuleInit { constructor() { super({ datasources: { db: { url: process.env.DATABASE_URL_SECONDARY, }, }, }); } async onModuleInit() { await this.$connect(); } } // prisma.module.ts @Global() @Module({ providers: [PrismaPrimaryService, PrismaSecondaryService], exports: [PrismaPrimaryService, PrismaSecondaryService], }) export class PrismaModule {} // 服务中使用 @Injectable() export class UsersService { constructor( private primaryDb: PrismaPrimaryService, private secondaryDb: PrismaSecondaryService, ) {} async findUser(id: number) { return this.primaryDb.user.findUnique({ where: { id } }); } async findLog(id: number) { return this.secondaryDb.log.findUnique({ where: { id } }); } }

🌐 动态数据库切换

// 多租户数据库切换 @Injectable() export class TenantService { private tenantConnections: Map = new Map(); async getTenantConnection(tenantId: string): Promise { if (this.tenantConnections.has(tenantId)) { return this.tenantConnections.get(tenantId)!; } const dataSource = new DataSource({ type: 'postgres', host: `${tenantId}.db.example.com`, port: 5432, username: 'tenant_user', password: 'tenant_password', database: `tenant_${tenantId}`, entities: [User, Order], }); await dataSource.initialize(); this.tenantConnections.set(tenantId, dataSource); return dataSource; } async closeAllConnections() { for (const [_, ds] of this.tenantConnections) { await ds.destroy(); } this.tenantConnections.clear(); } } // 使用中间件提取租户信息 @Injectable() export class TenantMiddleware implements NestMiddleware { constructor(private tenantService: TenantService) {} async use(req: Request, res: Response, next: NextFunction) { const tenantId = req.headers['x-tenant-id'] as string; req['tenantConnection'] = await this.tenantService.getTenantConnection(tenantId); next(); } }
⚠️
连接池管理

多数据库配置会增加连接数。合理配置连接池大小，避免资源浪费。动态数据库连接需要做好连接管理和清理。
