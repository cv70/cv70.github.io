13.7 模块实例化与层次引用

构建层次化设计

🎯 学习目标

  • 掌握模块实例化的基本方法
  • 学会使用位置关联和命名关联
  • 理解层次化设计的概念
  • 能够引用其他模块的信号

📦 模块实例化概述

模块实例化是将一个模块作为组件在另一个模块中使用的过程。通过实例化，可以构建层次化的设计结构。

// 实例化基本语法
模块名 实例名 (
    .端口名(连接信号),
    .端口名(连接信号)
);

🔌 命名关联（推荐）

.port(signal)

命名关联明确指定端口和信号的对应关系，代码可读性高，不易出错。

// D触发器模块
module dff(
    input  clk,
    input  rst_n,
    input  d,
    output q
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            q <= 1'b0;
        else
            q <= d;
    end
endmodule

// 实例化D触发器
module shift_reg(
    input  clk,
    input  rst_n,
    input  data_in,
    output data_out
);
    wire q0, q1, q2;

    // 使用命名关联实例化
    dff u0 (
        .clk(clk),       // 端口clk连接到信号clk
        .rst_n(rst_n),   // 端口rst_n连接到信号rst_n
        .d(data_in),     // 端口d连接到信号data_in
        .q(q0)           // 端口q连接到信号q0
    );

    dff u1 (
        .clk(clk),
        .rst_n(rst_n),
        .d(q0),
        .q(q1)
    );

    dff u2 (
        .clk(clk),
        .rst_n(rst_n),
        .d(q1),
        .q(data_out)
    );
endmodule

📍 位置关联

(signal1, signal2, ...)

位置关联按照端口声明的顺序连接信号，简洁但易错，不推荐用于复杂设计。

// 使用位置关联实例化
dff u0 (clk, rst_n, data_in, q0);  // 按顺序：clk, rst_n, d, q
dff u1 (clk, rst_n, q0, q1);
dff u2 (clk, rst_n, q1, data_out);

// ⚠️ 注意：位置关联容易出错，特别是端口较多时

🔗 层次引用

层次路径

层次引用允许访问设计树中任何模块的信号，常用于调试和仿真。

// 层次路径格式
顶层模块名.子模块名.信号名

// 示例层次结构
top
  ├── cpu
  │   ├── alu
  │   └── reg_file
  └── memory
      └── cache

// 层次引用示例
top.cpu.alu.result        // 访问CPU中ALU的result信号
top.memory.cache.valid    // 访问cache的valid信号

// 在测试平台中使用层次引用
module tb;
    reg clk;
    wire [31:0] data;

    top u_top (
        .clk(clk),
        .data(data)
    );

    initial begin
        // 监控层次信号
        $monitor("Time=%0t, ALU Result=%h",
                 $time, u_top.cpu.alu.result);
    end
endmodule

🏗️ 完整示例：4位加法器

// 1位全加器
module full_adder(
    input  a,
    input  b,
    input  cin,
    output sum,
    output cout
);
    assign sum  = a ^ b ^ cin;
    assign cout = (a & b) | (b & cin) | (a & cin);
endmodule

// 4位行波进位加法器
module adder_4bit(
    input  [3:0] a,
    input  [3:0] b,
    input        cin,
    output [3:0] sum,
    output       cout
);
    wire c0, c1, c2;

    // 实例化4个全加器
    full_adder fa0 (
        .a(a[0]),
        .b(b[0]),
        .cin(cin),
        .sum(sum[0]),
        .cout(c0)
    );

    full_adder fa1 (
        .a(a[1]),
        .b(b[1]),
        .cin(c0),
        .sum(sum[1]),
        .cout(c1)
    );

    full_adder fa2 (
        .a(a[2]),
        .b(b[2]),
        .cin(c1),
        .sum(sum[2]),
        .cout(c2)
    );

    full_adder fa3 (
        .a(a[3]),
        .b(b[3]),
        .cin(c2),
        .sum(sum[3]),
        .cout(cout)
    );
endmodule

📊 参数传递

// 带参数的模块
module counter #(parameter WIDTH = 8)(
    input             clk,
    input             rst_n,
    input             enable,
    output reg [WIDTH-1:0] count
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            count <= 0;
        else if (enable)
            count <= count + 1;
    end
endmodule

// 实例化时传递参数
module top;
    reg clk, rst_n, enable;
    wire [7:0]  count8;
    wire [15:0] count16;

    // 使用默认参数
    counter #(.WIDTH(8)) u_counter8 (
        .clk(clk),
        .rst_n(rst_n),
        .enable(enable),
        .count(count8)
    );

    // 传递新参数
    counter #(.WIDTH(16)) u_counter16 (
        .clk(clk),
        .rst_n(rst_n),
        .enable(enable),
        .count(count16)
    );
endmodule
💡
最佳实践
  • • 优先使用命名关联而非位置关联
  • • 实例名使用有意义的名称（如u_alu而非u1）
  • • 保持层次结构清晰，避免过深嵌套
  • • 使用参数化设计提高复用性

✅

  • • 命名关联比位置关联更安全

  • • 参数传递提高模块复用性
