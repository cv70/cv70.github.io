40.4 路由类型

类型安全的路由定义

🎯 学习目标

  • 掌握 koa-router 的类型定义
  • 学会定义类型安全的路由处理器
  • 理解 RouterContext 类型参数
  • 学会路由模块化组织

📚 Router 基本类型

import Router, { RouterContext, Middleware } from '@koa/router'; const router = new Router(); // RouterContext 泛型参数 // RouterContext // 基本路由 router.get('/', async (ctx: RouterContext) => { ctx.body = { message: 'Hello' }; }); // 带参数的路由 router.get('/users/:id', async (ctx: RouterContext<{ id: string }>) => { const { id } = ctx.params; // 类型安全 ctx.body = { userId: id }; }); // 多参数路由 router.get( '/posts/:postId/comments/:commentId', async (ctx: RouterContext<{ postId: string; commentId: string }>) => { const { postId, commentId } = ctx.params; ctx.body = { postId, commentId }; } );

🔧 定义路由类型

// src/types/routes.ts import { RouterContext } from '@koa/router'; // 路由参数类型 export interface UserRoutes { ':id': string; } export interface PostRoutes { ':postId': string; ':postId/comments/:commentId': { postId: string; commentId: string }; } // 查询参数类型 export interface PaginationQuery { page?: string; limit?: string; sort?: 'asc' | 'desc'; } // 请求体类型 export interface CreateUserBody { username: string; email: string; password: string; } // 完整的路由上下文类型 export type UserRouteContext = RouterContext< Params, any, Query > & { request: { body: Body; }; };

👤 用户路由示例

// src/routes/userRoutes.ts import Router, { RouterContext } from '@koa/router'; import { User, CreateUserDTO, UpdateUserDTO } from '@/types/user'; import { ApiResponse, PaginatedResponse } from '@/types/response'; const router = new Router({ prefix: '/api/users' }); // GET /api/users - 获取列表 router.get('/', async (ctx: RouterContext) => { const page = parseInt(ctx.query.page || '1', 10); const limit = parseInt(ctx.query.limit || '10', 10); const result = await UserService.findAll({ page, limit }); const response: PaginatedResponse = { success: true, ...result }; ctx.body = response; }); // GET /api/users/:id - 获取单个 router.get('/:id', async (ctx: RouterContext<{ id: string }>) => { const user = await UserService.findById(ctx.params.id); if (!user) { ctx.status = 404; ctx.body = { success: false, error: 'User not found' }; return; } ctx.body = { success: true, data: user }; }); // POST /api/users - 创建 router.post('/', async (ctx: RouterContext) => { const data = ctx.request.body; const user = await UserService.create(data); ctx.status = 201; ctx.body = { success: true, data: user }; }); // PUT /api/users/:id - 更新 router.put('/:id', async (ctx: RouterContext<{ id: string }, any, any, UpdateUserDTO>) => { const { id } = ctx.params; const data = ctx.request.body; const user = await UserService.update(id, data); ctx.body = { success: true, data: user }; }); // DELETE /api/users/:id - 删除 router.delete('/:id', async (ctx: RouterContext<{ id: string }>) => { await UserService.delete(ctx.params.id); ctx.body = { success: true, message: 'User deleted' }; }); export default router;

🧩 控制器模式

// src/controllers/userController.ts import { RouterContext, Next } from '@koa/router'; import { User, CreateUserDTO, UpdateUserDTO } from '@/types/user'; // 控制器类型 type ControllerAction = ( ctx: RouterContext & { request: { body: Body } }, next: Next ) => Promise; class UserController { // 获取用户列表 index: ControllerAction = async (ctx) => { const page = parseInt(ctx.query.page || '1', 10); const limit = parseInt(ctx.query.limit || '10', 10); const result = await UserService.findAll({ page, limit }); ctx.body = { success: true, ...result }; }; // 获取单个用户 show: ControllerAction<{ id: string }> = async (ctx) => { const user = await UserService.findById(ctx.params.id); if (!user) { ctx.status = 404; ctx.body = { success: false, error: 'User not found' }; return; } ctx.body = { success: true, data: user }; }; // 创建用户 store: ControllerAction = async (ctx) => { const data = ctx.request.body; const user = await UserService.create(data); ctx.status = 201; ctx.body = { success: true, data: user }; }; // 更新用户 update: ControllerAction<{ id: string }, any, UpdateUserDTO> = async (ctx) => { const { id } = ctx.params; const data = ctx.request.body; const user = await UserService.update(id, data); ctx.body = { success: true, data: user }; }; // 删除用户 destroy: ControllerAction<{ id: string }> = async (ctx) => { await UserService.delete(ctx.params.id); ctx.body = { success: true, message: 'User deleted' }; }; } export default new UserController();
// src/routes/userRoutes.ts import Router from '@koa/router'; import UserController from '@/controllers/userController'; const router = new Router({ prefix: '/api/users' }); router.get('/', UserController.index); router.get('/:id', UserController.show); router.post('/', UserController.store); router.put('/:id', UserController.update); router.delete('/:id', UserController.destroy); export default router;

🔗 路由模块化

// src/routes/index.ts import Router from '@koa/router'; import userRoutes from './userRoutes'; import authRoutes from './authRoutes'; import postRoutes from './postRoutes'; const router = new Router(); // 注册子路由 router.use(userRoutes.routes()); router.use(authRoutes.routes()); router.use(postRoutes.routes()); // 健康检查 router.get('/health', async (ctx) => { ctx.body = { status: 'ok', timestamp: new Date().toISOString() }; }); export default router;
// src/app.ts import Koa from 'koa'; import bodyParser from 'koa-bodyparser'; import routes from './routes'; const app = new Koa(); app.use(bodyParser()); app.use(routes.routes()); app.use(routes.allowedMethods()); export default app;

🔐 带中间件的路由

// src/routes/adminRoutes.ts import Router, { RouterContext } from '@koa/router'; import { authMiddleware } from '@/middlewares/auth'; import { adminOnly } from '@/middlewares/admin'; const router = new Router({ prefix: '/api/admin' }); // 所有路由都需要认证和管理员权限 router.use(authMiddleware({ secret: process.env.JWT_SECRET! })); router.use(adminOnly()); // 管理员专用路由 router.get('/users', async (ctx: RouterContext) => { const users = await UserService.findAllAdmin(); ctx.body = { success: true, data: users }; }); router.delete('/users/:id', async (ctx: RouterContext<{ id: string }>) => { await UserService.adminDelete(ctx.params.id); ctx.body = { success: true }; }); export default router;

⚡ 路由装饰器（高级）

// src/decorators/route.ts import Router, { RouterContext, Next } from '@koa/router'; const router = new Router(); // 路由装饰器工厂 function Route(method: 'get' | 'post' | 'put' | 'delete', path: string) { return function ( target: any, propertyKey: string, descriptor: PropertyDescriptor ) { const originalMethod = descriptor.value; router[method](path, async (ctx: RouterContext, next: Next) => { await originalMethod.call(target, ctx, next); }); }; } // 快捷装饰器 const Get = (path: string) => Route('get', path); const Post = (path: string) => Route('post', path); const Put = (path: string) => Route('put', path); const Delete = (path: string) => Route('delete', path); // 使用装饰器 class PostController { @Get('/posts') async index(ctx: RouterContext) { ctx.body = await PostService.findAll(); } @Get('/posts/:id') async show(ctx: RouterContext<{ id: string }>) { ctx.body = await PostService.findById(ctx.params.id); } @Post('/posts') async create(ctx: RouterContext) { const post = await PostService.create(ctx.request.body); ctx.status = 201; ctx.body = post; } } export { router, Get, Post, Put, Delete };
