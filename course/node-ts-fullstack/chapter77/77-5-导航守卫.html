📚 基本认证路由

import { Navigate, useLocation } from 'react-router-dom'; // 认证守卫组件 function ProtectedRoute({ children }: { children: React.ReactNode }) { const { isAuthenticated, isLoading } = useAuth(); const location = useLocation(); // 加载中显示加载状态 if (isLoading) { return <div>Loading...</div>; } // 未认证则重定向到登录页 if (!isAuthenticated) { // 保存当前路径，登录后跳转回来 return <Navigate to="/login" state={{ from: location }} replace />; } return <>{children}</>; } // 路由配置 const router = createBrowserRouter([ { path: '/', element: <Layout />, children: [ { index: true, element: <Home /> }, { path: 'login', element: <Login /> }, { path: 'dashboard', element: ( <ProtectedRoute> <Dashboard /> </ProtectedRoute> ), }, { path: 'admin', element: ( <ProtectedRoute> <AdminPanel /> </ProtectedRoute> ), }, ], }, ]); // 登录后跳转 function Login() { const navigate = useNavigate(); const location = useLocation(); const { login } = useAuth(); const from = location.state?.from?.pathname || '/'; const handleSubmit = async (credentials) => { const success = await login(credentials); if (success) { navigate(from, { replace: true }); } }; return <LoginForm onSubmit={handleSubmit} />; }

🔧 角色权限控制

// 角色权限守卫 interface ProtectedRouteProps { children: React.ReactNode; roles?: string[]; // 允许的角色 } function ProtectedRoute({ children, roles }: ProtectedRouteProps) { const { user, isAuthenticated, isLoading } = useAuth(); const location = useLocation(); if (isLoading) { return <LoadingSpinner />; } if (!isAuthenticated) { return <Navigate to="/login" state={{ from: location }} replace />; } // 检查角色权限 if (roles && !roles.some(role => user.roles.includes(role))) { return <Navigate to="/unauthorized" replace />; } return <>{children}</>; } // 路由配置 const router = createBrowserRouter([ { path: '/', element: <Layout />, children: [ { index: true, element: <Home /> }, { path: 'dashboard', element: ( <ProtectedRoute> <Dashboard /> </ProtectedRoute> ), }, { path: 'admin', element: ( <ProtectedRoute roles={['admin']}> <AdminPanel /> </ProtectedRoute> ), }, { path: 'manager', element: ( <ProtectedRoute roles={['admin', 'manager']}> <ManagerPanel /> </ProtectedRoute> ), }, { path: 'unauthorized', element: <Unauthorized /> }, ], }, ]); // 权限 Hook function usePermission(permission: string) { const { user } = useAuth(); return user?.permissions.includes(permission); } function DeleteButton() { const canDelete = usePermission('delete:posts'); if (!canDelete) { return null; } return <button onClick={handleDelete}>Delete</button>; }

📊 路由加载器

import { useLoaderData, LoaderFunction } from 'react-router-dom'; // 定义加载器 export const userLoader: LoaderFunction = async ({ params }) => { const user = await fetchUser(params.userId!); if (!user) { throw new Response('User not found', { status: 404 }); } return user; }; // 路由配置 const router = createBrowserRouter([ { path: '/users/:userId', element: <UserDetail />, loader: userLoader, errorElement: <UserError />, }, ]); // 组件中使用加载器数据 function UserDetail() { const user = useLoaderData() as User; return ( <div> <h1>{user.name}</h1> <p>{user.email}</p> </div> ); } // 使用 useRouteLoaderData 获取其他路由的数据 function UserHeader() { const user = useRouteLoaderData('user-detail') as User; return <h1>{user.name}</h1>; } // 需要在路由配置中添加 id const router = createBrowserRouter([ { id: 'user-detail', path: '/users/:userId', element: <UserDetail />, loader: userLoader, }, ]); // 权限检查加载器 export const protectedLoader: LoaderFunction = async () => { const user = await checkAuth(); if (!user) { return redirect('/login'); } return user; };
