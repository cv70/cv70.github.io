📖 Router 类概述

express.Router() 类是一个完整的中间件和路由系统，常被称为"迷你应用"。它与主应用有相似的功能，但有一些关键区别。

Router 与 App 的区别

特性 express() express.Router()
顶级对象 是（主应用） 否（路由模块）
设置 app.set()/app.get() 不支持
监听端口 app.listen() 不支持
路由方法 支持 支持
中间件 支持 支持
挂载路径 自动挂载 需通过 app.use() 挂载

🔧 Router 完整 API

创建 Router

// 基本创建 const router = express.Router(); // 带选项创建 const router = express.Router({ caseSensitive: false, // 路由区分大小写（默认 false） mergeParams: false, // 合并父路由参数（默认 false） strict: false // 严格匹配（默认 false） });

路由方法

// 所有 HTTP 方法 router.get(path, [callbacks]) router.post(path, [callbacks]) router.put(path, [callbacks]) router.patch(path, [callbacks]) router.delete(path, [callbacks]) router.head(path, [callbacks]) router.options(path, [callbacks]) // 特殊方法 router.all(path, [callbacks]) // 匹配所有方法 router.use([path], callbacks) // 中间件

路由链式定义

// router.route() - 链式定义同一路径 router.route('/users/:id') .all(authMiddleware) // 所有方法先经过认证 .get(userController.show) // GET 详情 .put(userController.update) // PUT 更新 .delete(userController.delete); // DELETE 删除 // 等价于： router.all('/users/:id', authMiddleware); router.get('/users/:id', userController.show); router.put('/users/:id', userController.update); router.delete('/users/:id', userController.delete);

⚙️ Router 选项详解

caseSensitive - 大小写敏感

// 大小写不敏感（默认） const router = express.Router({ caseSensitive: false }); // /Users 和 /users 都匹配 // 大小写敏感 const router = express.Router({ caseSensitive: true }); // /Users 和 /users 是不同的路由

mergeParams - 合并参数

// 默认不合并 const router = express.Router({ mergeParams: false }); app.use('/users/:userId', router); // router 中无法访问 userId // 合并父路由参数 const router = express.Router({ mergeParams: true }); app.use('/users/:userId', router); // router 中可以访问 req.params.userId // 示例 router.get('/posts/:postId', (req, res) => { const { userId, postId } = req.params; res.json({ userId, postId }); }); // GET /users/123/posts/456 -> { userId: '123', postId: '456' }

strict - 严格模式

// 非严格模式（默认） const router = express.Router({ strict: false }); // /foo 和 /foo/ 被视为相同 // 严格模式 const router = express.Router({ strict: true }); // /foo 和 /foo/ 是不同的路由

🔄 Router 中间件

模块级中间件

// routes/admin.js const express = require('express'); const router = express.Router(); // 此中间件只影响 admin 路由模块 router.use((req, res, next) => { console.log('Admin route accessed'); next(); }); // 管理员认证 router.use((req, res, next) => { if (!req.user || req.user.role !== 'admin') { return res.status(403).json({ message: 'Admin access required' }); } next(); }); router.get('/dashboard', (req, res) => { res.json({ message: 'Admin Dashboard' }); }); module.exports = router;

特定路径的中间件

// 只对特定路径应用中间件 router.use('/settings', (req, res, next) => { // 验证设置修改权限 next(); }); router.get('/settings', getSettings); router.put('/settings', updateSettings); // 其他路由不受影响 router.get('/users', getUsers);

路由级中间件

// 特定路由的中间件 const validatePost = (req, res, next) => { const { title, content } = req.body; if (!title || !content) { return res.status(400).json({ message: 'Title and content required' }); } next(); }; router.post('/posts', validatePost, createPost); router.put('/posts/:id', validatePost, updatePost);

🏗️ Router 高级用法

动态加载路由

// 延迟加载路由模块 let adminRouter = null; app.use('/admin', (req, res, next) => { if (!adminRouter) { adminRouter = require('./routes/admin'); } adminRouter(req, res, next); }); // 或者条件加载 if (process.env.ENABLE_ADMIN === 'true') { app.use('/admin', require('./routes/admin')); }

路由参数预处理

// 参数预处理 router.param('userId', (req, res, next, id) => { // 验证 ID 格式 if (!/^\d+$/.test(id)) { return res.status(400).json({ message: 'Invalid user ID' }); } // 查找用户 User.findById(parseInt(id)) .then(user => { if (!user) { return res.status(404).json({ message: 'User not found' }); } req.user = user; next(); }) .catch(next); }); // 后续路由可直接使用 req.user router.get('/users/:userId', (req, res) => { res.json(req.user); }); router.put('/users/:userId', (req, res) => { req.user.update(req.body); res.json(req.user); });

Router 组合

// 组合多个 Router const express = require('express'); const app = express(); // 用户相关路由 const userRoutes = express.Router(); userRoutes.get('/', listUsers); userRoutes.post('/', createUser); const userPostRoutes = express.Router({ mergeParams: true }); userPostRoutes.get('/', listUserPosts); userPostRoutes.post('/', createUserPost); // 组合 userRoutes.use('/:userId/posts', userPostRoutes); // 公开 API const publicApiRoutes = express.Router(); publicApiRoutes.use('/users', userRoutes); // 私有 API const privateApiRoutes = express.Router(); privateApiRoutes.use(authMiddleware); privateApiRoutes.use('/admin', adminRoutes); // 挂载 app.use('/api/v1', publicApiRoutes); app.use('/api/internal', privateApiRoutes);

📊 实战示例：API 路由架构

// routes/api/index.js const express = require('express'); const router = express.Router(); // API 版本路由 const v1Routes = require('./v1'); const v2Routes = require('./v2'); // API 通用中间件 router.use((req, res, next) => { res.set({ 'Content-Type': 'application/json', 'X-API-Version': req.headers['x-api-version'] || '1' }); next(); }); // API 文档 router.get('/docs', (req, res) => { res.json({ message: 'API Documentation' }); }); // 版本路由 router.use('/v1', v1Routes); router.use('/v2', v2Routes); // 默认使用最新版本 router.use('/', v2Routes); module.exports = router; // routes/api/v1/index.js const express = require('express'); const router = express.Router(); const userRoutes = require('./users'); const postRoutes = require('./posts'); router.use('/users', userRoutes); router.use('/posts', postRoutes); module.exports = router; // routes/api/v1/users.js const express = require('express'); const router = express.Router(); const userController = require('../../../controllers/api/v1/userController'); router.param('id', userController.load); router.route('/') .get(userController.index) .post(userController.create); router.route('/:id') .get(userController.show) .put(userController.update) .patch(userController.patch) .delete(userController.destroy); module.exports = router;

✅ 小结

📌
本节要点
  • Router 是"迷你应用"，支持中间件和路由，但不支持设置和监听
  • Router 选项：caseSensitive、mergeParams、strict
  • mergeParams: true 可访问父路由的参数
  • router.param() 可以预处理路由参数
  • Router 可以嵌套组合，构建复杂的路由结构
