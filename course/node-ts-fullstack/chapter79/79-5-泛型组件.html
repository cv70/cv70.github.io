📚 泛型组件基础

// 问题：不同数据类型需要不同组件 function StringList({ items }: { items: string[] }) { return ( <ul> {items.map((item, i) => <li key={i}>{item}</li>)} </ul> ); } function NumberList({ items }: { items: number[] }) { return ( <ul> {items.map((item, i) => <li key={i}>{item}</li>)} </ul> ); } // 解决：泛型组件 // 使用函数声明 function List<T>({ items }: { items: T[] }) { return ( <ul> {items.map((item, i) => <li key={i}>{String(item)}</li>)} </ul> ); } // 使用箭头函数 const List = <T,>({ items }: { items: T[] }) => ( <ul> {items.map((item, i) => <li key={i}>{String(item)}</li>)} </ul> ); // 使用：类型推断 <List items={['a', 'b', 'c']} /> // T = string <List items={[1, 2, 3]} /> // T = number // 显式指定类型（有时需要） <List<User> items={users} />

🔧 泛型组件示例

// 带渲染函数的列表组件 interface ListProps<T> { items: T[]; renderItem: (item: T, index: number) => ReactNode; keyExtractor: (item: T) => string | number; } function List<T>({ items, renderItem, keyExtractor }: ListProps<T>) { return ( <ul> {items.map((item, index) => ( <li key={keyExtractor(item)}> {renderItem(item, index)} </li> ))} </ul> ); } // 使用 interface User { id: string; name: string; email: string; } function UserList() { const users: User[] = [ { id: '1', name: 'Alice', email: 'alice@example.com' }, { id: '2', name: 'Bob', email: 'bob@example.com' }, ]; return ( <List items={users} keyExtractor={(user) => user.id} renderItem={(user) => ( <div> <strong>{user.name}</strong> <span>{user.email}</span> </div> )} /> ); } // 选择器组件 interface SelectProps<T> { options: T[]; value: T | null; onChange: (value: T) => void; getOptionLabel: (option: T) => string; getOptionValue: (option: T) => string; } function Select<T>({ options, value, onChange, getOptionLabel, getOptionValue, }: SelectProps<T>) { return ( <select value={value ? getOptionValue(value) : ''} onChange={(e) => { const selected = options.find( (o) => getOptionValue(o) === e.target.value ); if (selected) onChange(selected); }} > {options.map((option) => ( <option key={getOptionValue(option)} value={getOptionValue(option)}> {getOptionLabel(option)} </option> ))} </select> ); } // 使用 interface Product { id: number; name: string; price: number; } function ProductSelect() { const [selected, setSelected] = useState<Product | null>(null); const products: Product[] = [...]; return ( <Select options={products} value={selected} onChange={setSelected} getOptionLabel={(p) => `${p.name} - $${p.price}`} getOptionValue={(p) => String(p.id)} /> ); }

📊 泛型约束

// 约束 T 必须有 id 属性 interface WithId { id: string | number; } interface DataTableProps<T extends WithId> { data: T[]; columns: Column<T>[]; } interface Column<T> { key: keyof T; header: string; render?: (value: T[keyof T], row: T) => ReactNode; } function DataTable<T extends WithId>({ data, columns }: DataTableProps<T>) { return ( <table> <thead> <tr> {columns.map((col) => ( <th key={String(col.key)}>{col.header}</th> ))} </tr> </thead> <tbody> {data.map((row) => ( <tr key={row.id}> {columns.map((col) => ( <td key={String(col.key)}> {col.render ? col.render(row[col.key], row) : String(row[col.key])} </td> ))} </tr> ))} </tbody> </table> ); } // 使用 interface User { id: string; name: string; email: string; role: string; } function UserTable() { const users: User[] = [...]; const columns: Column<User>[] = [ { key: 'name', header: 'Name' }, { key: 'email', header: 'Email' }, { key: 'role', header: 'Role', render: (role) => <span className={`badge-${role}`}>{role}</span>, }, ]; return <DataTable data={users} columns={columns} />; } // 多个泛型参数 interface PairProps<K, V> { data: [K, V][]; renderKey: (key: K) => ReactNode; renderValue: (value: V) => ReactNode; } function PairList<K, V>({ data, renderKey, renderValue }: PairProps<K, V>) { return ( <dl> {data.map(([key, value], i) => ( <React.Fragment key={i}> <dt>{renderKey(key)}</dt> <dd>{renderValue(value)}</dd> </React.Fragment> ))} </dl> ); }
