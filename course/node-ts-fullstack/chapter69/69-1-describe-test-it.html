69.1 describe、test、it

第十部分：测试篇 - 掌握 Jest 测试组织的核心函数

1. describe - 测试套件

describe 函数用于创建一个测试套件，将相关的测试用例组织在一起。

describe(name, fn) // name: 测试套件的名称 // fn: 包含测试用例的函数

基本用法

// 创建一个测试套件 describe('Calculator', () => { // 所有的测试用例都在这里 test('should add numbers', () => { // ... }); test('should subtract numbers', () => { // ... }); });

嵌套 describe

describe('UserService', () => { describe('create', () => { test('should create user with valid data', () => {}); test('should throw error with invalid email', () => {}); }); describe('update', () => { test('should update user name', () => {}); test('should not update non-existent user', () => {}); }); describe('delete', () => { test('should delete existing user', () => {}); test('should throw error for non-existent user', () => {}); }); });
💡
命名建议

describe 名称通常使用被测试的类名或模块名，嵌套 describe 使用方法名或功能名称。

2. test 和 it - 测试用例

test 和 it 是完全等价的，都用于定义一个测试用例。it 是行为驱动开发（BDD）风格的写法。

test(name, fn, timeout?) it(name, fn, timeout?) // name: 测试用例描述 // fn: 测试函数 // timeout: 可选的超时时间（毫秒）

两种写法对比

// 使用 test test('should add two numbers', () => { expect(add(1, 2)).toBe(3); }); // 使用 it（推荐） it('should add two numbers', () => { expect(add(1, 2)).toBe(3); }); // it 读起来更自然：it should add two numbers

测试描述最佳实践

describe('Calculator', () => { describe('add', () => { // ✅ 好的描述：清晰说明预期行为 it('should return sum of two positive numbers', () => {}); it('should handle negative numbers', () => {}); it('should return 0 when adding zero', () => {}); // ❌ 不好的描述：模糊或不明确 // it('works', () => {}); // it('test add', () => {}); }); });

3. 跳过测试

describe.skip(name, fn) // 跳过整个测试套件 describe.only(name, fn) // 只运行这个测试套件 test.skip(name, fn) // 跳过测试 it.skip(name, fn) // 同上 test.only(name, fn) // 只运行这个测试 it.only(name, fn) // 同上

使用场景

describe('UserService', () => { // 跳过整个测试套件 describe.skip('deprecated methods', () => { it('should work with old API', () => {}); }); // 只运行这个测试套件（调试时有用） describe.only('create', () => { it('should create user', () => {}); }); describe('update', () => { it('should update user', () => {}); // 跳过单个测试 it.skip('should handle edge case', () => {}); // 只运行这个测试 it.only('should validate email', () => {}); }); });
⚠️
注意事项

使用 skip 和 only 是临时的调试手段，不要将它们提交到代码仓库中。应该在 CI 中检测并拒绝包含 skip 的代码。

4. 每个测试的测试

describe.each(table)(name, fn, timeout) test.each(table)(name, fn, timeout) it.each(table)(name, fn, timeout)

参数化测试示例

// 使用数组 describe.each([ [1, 1, 2], [1, 2, 3], [2, 2, 4], ])('.add(%i, %i)', (a, b, expected) => { it(`should return ${expected}`, () => { expect(add(a, b)).toBe(expected); }); }); // 使用对象 describe.each` a | b | expected ${1} | ${1} | ${2} ${1} | ${2} | ${3} ${2} | ${2} | ${4} `('add($a, $b)', ({ a, b, expected }) => { it(`should return ${expected}`, () => { expect(add(a, b)).toBe(expected); }); }); // test.each 示例 test.each([ [0, 100, 0], [50, 100, 50], [100, 100, 100], ])('percentage(%i, %i) should be %i', (value, total, expected) => { expect(percentage(value, total)).toBe(expected); });

5. 测试结构最佳实践

层次化组织

// 推荐的测试结构 describe('UserService', () => { // 顶层：按类或模块组织 describe('create', () => { // 第二层：按方法组织 describe('validation', () => { // 第三层：按场景组织 it('should validate email format', () => {}); it('should validate password strength', () => {}); }); describe('database operations', () => { it('should insert user to database', () => {}); it('should handle duplicate email', () => {}); }); }); describe('update', () => { // ... }); });

命名约定

层级 命名风格 示例
顶层 describe 类名/模块名 UserService, Calculator
中层 describe 方法名/功能名 create, add, validation
底层 describe 场景名 when user exists, edge cases
it/test should + 行为描述 should return user by id

6. 完整示例

// tests/services/user.service.test.ts import { UserService } from '@/services/user.service'; import { UserRepository } from '@/repositories/user.repository'; // Mock 依赖 jest.mock('@/repositories/user.repository'); describe('UserService', () => { let service: UserService; let mockRepository: jest.Mocked<UserRepository>; beforeEach(() => { // 创建 Mock 实例 mockRepository = { findById: jest.fn(), create: jest.fn(), update: jest.fn(), delete: jest.fn(), } as any; service = new UserService(mockRepository); }); afterEach(() => { jest.clearAllMocks(); }); describe('findById', () => { it('should return user when found', async () => { const mockUser = { id: '1', name: 'John' }; mockRepository.findById.mockResolvedValue(mockUser); const result = await service.findById('1'); expect(result).toEqual(mockUser); expect(mockRepository.findById).toHaveBeenCalledWith('1'); }); it('should return null when user not found', async () => { mockRepository.findById.mockResolvedValue(null); const result = await service.findById('999'); expect(result).toBeNull(); }); it('should throw error when id is invalid', async () => { await expect(service.findById('')).rejects.toThrow('Invalid id'); }); }); describe('create', () => { describe('validation', () => { it('should throw error when email is missing', async () => { await expect(service.create({ name: 'John' })) .rejects.toThrow('Email is required'); }); it('should throw error when email is invalid', async () => { await expect(service.create({ email: 'invalid', name: 'John' })) .rejects.toThrow('Invalid email format'); }); }); describe('success cases', () => { it('should create user with valid data', async () => { const userData = { email: 'john@example.com', name: 'John' }; const createdUser = { id: '1', ...userData }; mockRepository.create.mockResolvedValue(createdUser); const result = await service.create(userData); expect(result).toEqual(createdUser); }); }); }); });

7. 本节小结

📚
关键要点
  • describe 用于组织测试套件，可以嵌套使用
  • test 和 it 完全等价，it 更符合 BDD 风格
  • 使用 skip 和 only 进行调试，但不要提交
  • 使用 each 进行参数化测试，减少重复代码
  • 遵循层次化的测试组织结构
