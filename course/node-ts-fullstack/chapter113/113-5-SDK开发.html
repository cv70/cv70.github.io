🔧 Node.js SDK实现

// index.ts - SDK入口 export { ApiClient } from './client'; export { UserApi, OrderApi, PaymentApi } from './api'; export * from './errors'; // client.ts - 核心客户端 import axios, { AxiosInstance, AxiosRequestConfig } from 'axios'; import crypto from 'crypto'; import { v4 as uuidv4 } from 'uuid'; export interface ClientConfig { appKey: string; appSecret: string; baseUrl?: string; timeout?: number; } export class ApiClient { private config: Required; private httpClient: AxiosInstance; constructor(config: ClientConfig) { this.config = { baseUrl: 'https://api.example.com', timeout: 30000, ...config, }; this.httpClient = axios.create({ baseURL: this.config.baseUrl, timeout: this.config.timeout, }); } // 通用请求方法 async request(method: string, path: string, data?: any): Promise { const timestamp = Date.now().toString(); const nonce = uuidv4(); // 构造签名 const stringToSign = this.buildSignString(method, path, timestamp, nonce, data); const signature = crypto .createHmac('sha256', this.config.appSecret) .update(stringToSign) .digest('hex'); const config: AxiosRequestConfig = { method, url: path, headers: { 'X-App-Key': this.config.appKey, 'X-Timestamp': timestamp, 'X-Nonce': nonce, 'X-Signature': signature, 'Content-Type': 'application/json', }, }; if (method === 'GET') { config.params = data; } else { config.data = data; } try { const response = await this.httpClient.request>(config); if (response.data.code !== 0) { throw new ApiError(response.data.code, response.data.message); } return response.data.data; } catch (error) { if (error instanceof ApiError) { throw error; } throw new NetworkError(error.message); } } private buildSignString( method: string, path: string, timestamp: string, nonce: string, data: any ): string { const params = method === 'GET' ? this.sortParams(data || {}) : JSON.stringify(data || {}); return `${method.toUpperCase()}\n${path}\n${timestamp}\n${nonce}\n${params}`; } private sortParams(params: Record): string { return Object.keys(params) .sort() .map(key => `${key}=${params[key]}`) .join('&'); } } // api/user.ts - 用户API export class UserApi { constructor(private client: ApiClient) {} async getUser(userId: string): Promise { return this.client.request('GET', `/api/v1/users/${userId}`); } async listUsers(params: { page?: number; pageSize?: number }): Promise> { return this.client.request>('GET', '/api/v1/users', params); } async createUser(data: CreateUserInput): Promise { return this.client.request('POST', '/api/v1/users', data); } } // errors.ts - 错误定义 export class ApiError extends Error { constructor(public code: number, message: string) { super(message); this.name = 'ApiError'; } } export class NetworkError extends Error { constructor(message: string) { super(message); this.name = 'NetworkError'; } } // 使用示例 import { ApiClient, UserApi } from '@example/api-sdk'; const client = new ApiClient({ appKey: 'your-app-key', appSecret: 'your-app-secret', }); const userApi = new UserApi(client); // 获取用户 const user = await userApi.getUser('user-123'); console.log(user.name); // 创建用户 const newUser = await userApi.createUser({ name: 'John', email: 'john@example.com', });
