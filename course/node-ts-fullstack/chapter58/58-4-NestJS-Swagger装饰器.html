58.4 NestJS Swagger 装饰器

使用装饰器丰富 API 文档内容

📖 装饰器概述

@nestjs/swagger 提供了丰富的装饰器，通过在控制器和 DTO 上添加装饰器，自动生成详细的 API 文档。

💡
装饰器优势

装饰器与代码紧密结合，文档即代码，易于维护和同步更新。

🏷️ 控制器装饰器

@ApiTags - 接口分组

import { ApiTags } from '@nestjs/swagger'; @ApiTags('用户管理') @Controller('users') export class UserController { // 所有方法都属于 "用户管理" 分组 }

@ApiBearerAuth - Bearer 认证

import { ApiBearerAuth } from '@nestjs/swagger'; @ApiBearerAuth('JWT-auth') // 对应 DocumentBuilder 中的名称 @Controller('users') export class UserController { // 所有接口需要认证 }

@ApiSecurity - 自定义认证

@ApiSecurity('api-key') @Controller('admin') export class AdminController {}

📝 方法装饰器

@ApiOperation - 接口描述

import { ApiOperation } from '@nestjs/swagger'; @Get() @ApiOperation({ summary: '获取用户列表', description: '获取所有用户的分页列表，支持筛选和排序', }) findAll() { return this.userService.findAll(); }

@ApiResponse - 响应说明

import { ApiResponse, ApiOkResponse, ApiCreatedResponse, ApiNotFoundResponse } from '@nestjs/swagger'; @Get(':id') @ApiOkResponse({ description: '获取成功', type: User }) @ApiNotFoundResponse({ description: '用户不存在' }) @ApiUnauthorizedResponse({ description: '未授权' }) findOne(@Param('id') id: string) { return this.userService.findOne(+id); } @Post() @ApiCreatedResponse({ description: '创建成功', type: User, // 或使用数组类型 type: [User], }) create(@Body() createUserDto: CreateUserDto) { return this.userService.create(createUserDto); } // 通用响应装饰器 @ApiResponse({ status: 200, description: '成功响应', type: User, headers: { 'X-Total-Count': { description: '总数', schema: { type: 'integer' }, }, }, })

响应装饰器汇总

装饰器 状态码 使用场景
@ApiOkResponse 200 成功响应
@ApiCreatedResponse 201 创建成功
@ApiNoContentResponse 204 无内容响应
@ApiBadRequestResponse 400 参数错误
@ApiUnauthorizedResponse 401 未认证
@ApiForbiddenResponse 403 禁止访问
@ApiNotFoundResponse 404 资源不存在
@ApiConflictResponse 409 冲突
@ApiTooManyRequestsResponse 429 请求过多
@ApiInternalServerErrorResponse 500 服务器错误

📥 参数装饰器

@ApiParam - 路径参数

import { ApiParam } from '@nestjs/swagger'; @Get(':id') @ApiParam({ name: 'id', type: 'integer', description: '用户 ID', example: 123, required: true, }) findOne(@Param('id') id: string) {} // 多个参数 @Get(':userId/posts/:postId') @ApiParam({ name: 'userId', description: '用户 ID' }) @ApiParam({ name: 'postId', description: '文章 ID' }) getUserPost(@Param() params: { userId: string; postId: string }) {}

@ApiQuery - 查询参数

import { ApiQuery } from '@nestjs/swagger'; @Get() @ApiQuery({ name: 'page', required: false, type: Number, description: '页码', example: 1, }) @ApiQuery({ name: 'pageSize', required: false, type: Number, description: '每页数量', example: 10, }) @ApiQuery({ name: 'status', required: false, enum: ['active', 'inactive'], description: '用户状态', }) @ApiQuery({ name: 'sort', required: false, type: String, description: '排序字段', example: '-createdAt', }) findAll( @Query('page') page = 1, @Query('pageSize') pageSize = 10, ) {}

@ApiBody - 请求体

import { ApiBody } from '@nestjs/swagger'; @Post() @ApiBody({ type: CreateUserDto, description: '用户创建数据', examples: { example1: { summary: '基本示例', value: { name: '张三', email: 'zhangsan@example.com', password: 'Password123!', }, }, }, }) create(@Body() createUserDto: CreateUserDto) {}

📦 DTO 装饰器

@ApiProperty - 属性定义

import { ApiProperty, ApiPropertyOptional, OmitType, PartialType } from '@nestjs/swagger'; export class UserDto { @ApiProperty({ description: '用户 ID', example: 1, type: Number, }) id: number; @ApiProperty({ description: '用户名', example: '张三', minLength: 2, maxLength: 50, }) name: string; @ApiProperty({ description: '邮箱', example: 'zhangsan@example.com', format: 'email', }) email: string; @ApiPropertyOptional({ description: '年龄', example: 25, minimum: 0, maximum: 150, }) age?: number; @ApiPropertyOptional({ description: '用户状态', enum: UserStatus, enumName: 'UserStatus', default: UserStatus.ACTIVE, }) status?: UserStatus; @ApiPropertyOptional({ description: '角色列表', type: [String], example: ['user', 'admin'], }) roles?: string[]; @ApiProperty({ description: '创建时间', format: 'date-time', example: '2024-01-15T10:30:00Z', }) createdAt: Date; } // 可选属性装饰器 export class UpdateUserDto extends PartialType(UserDto) { // 所有属性变为可选 } // 排除某些属性 export class CreateUserDto extends OmitType(UserDto, ['id', 'createdAt']) { @ApiProperty({ description: '密码' }) password: string; }

DTO 工具类型

工具类型 说明 使用场景
PartialType 所有属性变为可选 更新 DTO
OmitType 排除指定属性 创建 DTO（排除 id）
PickType 选取指定属性 部分字段更新
IntersectionType 合并多个类型 组合 DTO

🎨 其他实用装饰器

@ApiExcludeEndpoint - 隐藏接口

@Get('internal') @ApiExcludeEndpoint() // 不在文档中显示 internalMethod() {}

@ApiHeader - 请求头

@ApiHeader({ name: 'X-Custom-Header', description: '自定义请求头', required: true, }) @Get() findAll() {}

@ApiProduces/@ApiConsumes - 内容类型

@ApiProduces('application/json') @ApiConsumes('application/json', 'multipart/form-data') @Post('upload') uploadFile() {}

@ApiExtension - 自定义扩展

@ApiExtension('x-my-extension', { custom: 'value' }) @Get() findAll() {}

💻 完整示例

import { Controller, Get, Post, Put, Delete, Param, Body, Query } from '@nestjs/common'; import { ApiTags, ApiOperation, ApiParam, ApiQuery, ApiBody, ApiOkResponse, ApiCreatedResponse, ApiNotFoundResponse, ApiBearerAuth, } from '@nestjs/swagger'; @ApiTags('用户管理') @ApiBearerAuth() @Controller('users') export class UserController { constructor(private readonly userService: UserService) {} @Get() @ApiOperation({ summary: '获取用户列表' }) @ApiQuery({ name: 'page', required: false, type: Number }) @ApiQuery({ name: 'pageSize', required: false, type: Number }) @ApiOkResponse({ description: '成功获取列表', type: [UserDto], }) findAll( @Query('page') page = 1, @Query('pageSize') pageSize = 10, ) { return this.userService.findAll({ page, pageSize }); } @Get(':id') @ApiOperation({ summary: '获取用户详情' }) @ApiParam({ name: 'id', description: '用户 ID' }) @ApiOkResponse({ description: '获取成功', type: UserDto }) @ApiNotFoundResponse({ description: '用户不存在' }) findOne(@Param('id') id: string) { return this.userService.findOne(+id); } @Post() @ApiOperation({ summary: '创建用户' }) @ApiBody({ type: CreateUserDto }) @ApiCreatedResponse({ description: '创建成功', type: UserDto }) create(@Body() createUserDto: CreateUserDto) { return this.userService.create(createUserDto); } @Put(':id') @ApiOperation({ summary: '更新用户' }) @ApiParam({ name: 'id' }) @ApiBody({ type: UpdateUserDto }) @ApiOkResponse({ type: UserDto }) update( @Param('id') id: string, @Body() updateUserDto: UpdateUserDto, ) { return this.userService.update(+id, updateUserDto); } @Delete(':id') @ApiOperation({ summary: '删除用户' }) @ApiParam({ name: 'id' }) @ApiOkResponse({ description: '删除成功' }) remove(@Param('id') id: string) { return this.userService.remove(+id); } }

📝 小结

  • @ApiTags 用于接口分组
  • @ApiOperation 描述接口功能
  • @ApiParam/@ApiQuery/@ApiBody 描述参数
  • @ApiResponse 系列装饰器描述响应
  • @ApiProperty/@ApiPropertyOptional 定义 DTO 属性
  • 使用 PartialType、OmitType 等工具类型复用 DTO
