📊 成本分析方法

# 成本分析脚本
import pandas as pd

def analyze_training_costs():
    """分析训练成本"""

    # 成本数据
    gpu_hour_cost = 2.0  # 美元/GPU小时
    total_gpu_hours = 10000
    storage_cost_tb_month = 20.0
    storage_tb = 100
    training_months = 2

    # 计算成本
    gpu_cost = total_gpu_hours * gpu_hour_cost
    storage_cost = storage_tb_month * storage_tb * training_months
    total_cost = gpu_cost + storage_cost

    # 成本分解
    cost_breakdown = {
        'GPU计算': gpu_cost,
        '存储': storage_cost,
        '总计': total_cost
    }

    # 生成报告
    report = pd.DataFrame({
        '类别': list(cost_breakdown.keys()),
        '成本(美元)': list(cost_breakdown.values())
    })

    print("\n训练成本分析报告")
    print("=" * 40)
    print(report)
    print(f"\n每1000次迭代成本: ${total_cost / (total_gpu_hours * 3600 * 0.1):.2f}")
    print(f"每美元训练tokens: {1e12 / total_cost:.0f}")

    # 优化建议
    if gpu_cost / total_cost > 0.8:
        print("\n优化建议: GPU成本占比过高，考虑使用Spot实例")

analyze_training_costs()

📈 成本监控

# 成本监控脚本
import requests
import json

class CostMonitor:
    def __init__(self, billing_api_key):
        self.api_key = billing_api_key
        self.base_url = "https://billing.api.cloud.com"

    def get_current_costs(self):
        """获取当前成本"""
        response = requests.get(
            f"{self.base_url}/costs/current",
            headers={"Authorization": f"Bearer {self.api_key}"}
        )
        return response.json()

    def get_cost_forecast(self):
        """获取成本预测"""
        response = requests.get(
            f"{self.base_url}/costs/forecast",
            headers={"Authorization": f"Bearer {self.api_key}"}
        )
        return response.json()

    def check_budget(self, budget_limit):
        """检查预算"""
        current_cost = self.get_current_costs()['total']
        forecast = self.get_cost_forecast()['total']

        alert_needed = False
        if current_cost > budget_limit * 0.8:
            print(f"警告: 已使用80%预算 (${current_cost:.2f}/${budget_limit:.2f})")
            alert_needed = True

        if forecast > budget_limit:
            print(f"警告: 预计超出预算 (${forecast:.2f}/${budget_limit:.2f})")
            alert_needed = True

        return alert_needed
