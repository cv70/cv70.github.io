📦 共享类型定义

// shared/types.ts // 用户 export interface User { id: string; username: string; avatar?: string; status: 'online' | 'offline' | 'typing'; joinedAt: Date; } // 消息 export interface Message { id: string; content: string; type: 'text' | 'image' | 'system'; sender: User; roomId: string; timestamp: Date; read: boolean; } // 房间 export interface Room { id: string; name: string; description?: string; creator: string; users: Map<string, User>; messages: Message[]; createdAt: Date; isPrivate: boolean; } // 服务端到客户端事件 export interface ServerToClientEvents { 'room:joined': (room: Room) => void; 'room:left': (roomId: string) => void; 'room:user-joined': (user: User) => void; 'room:user-left': (user: User) => void; 'message:new': (message: Message) => void; 'user:typing': (userId: string, isTyping: boolean) => void; 'user:updated': (user: User) => void; error: (error: { code: string; message: string }) => void; } // 客户端到服务端事件 export interface ClientToServerEvents { 'room:join': (roomId: string, callback: (room: Room | null, error?: string) => void) => void; 'room:leave': (roomId: string, callback: (success: boolean) => void) => void; 'room:create': (data: { name: string; isPrivate?: boolean }, callback: (room: Room) => void) => void; 'message:send': (data: { roomId: string; content: string }, callback: (message: Message) => void) => void; 'user:typing': (roomId: string, isTyping: boolean) => void; 'message:history': (roomId: string, callback: (messages: Message[]) => void) => void; }

🖥️ 服务端实现

Socket.IO 服务端配置

// server/src/socket.ts import { Server } from 'socket.io'; import type { ServerToClientEvents, ClientToServerEvents, User, Room, Message } from '../../shared/types'; const rooms = new Map<string, Room>(); const userSockets = new Map<string, string>(); // userId -> socketId export function setupSocket(io: Server) { // 认证中间件 io.use((socket, next) => { const userId = socket.handshake.auth.userId; const username = socket.handshake.auth.username; if (!userId || !username) { return next(new Error('认证失败')); } socket.data.userId = userId; socket.data.username = username; next(); }); io.on('connection', (socket) => { const userId = socket.data.userId; const username = socket.data.username; console.log(`用户连接: ${username} (${userId})`); // 创建用户对象 const user: User = { id: userId, username, status: 'online', joinedAt: new Date() }; userSockets.set(userId, socket.id); // ==================== 房间管理 ==================== // 创建房间 socket.on('room:create', (data, callback) => { const roomId = generateRoomId(); const room: Room = { id: roomId, name: data.name, creator: userId, users: new Map([[userId, user]]), messages: [], createdAt: new Date(), isPrivate: data.isPrivate || false }; rooms.set(roomId, room); socket.join(roomId); callback(room); io.emit('room:created', room); // 通知所有人有新房间 }); // 加入房间 socket.on('room:join', (roomId, callback) => { const room = rooms.get(roomId); if (!room) { return callback(null, '房间不存在'); } // 加入房间 socket.join(roomId); room.users.set(userId, user); // 通知房间内其他人 socket.to(roomId).emit('room:user-joined', user); // 返回房间信息 callback({ ...room, users: Array.from(room.users.values()) } as any); }); // 离开房间 socket.on('room:leave', (roomId, callback) => { const room = rooms.get(roomId); if (room) { room.users.delete(userId); socket.leave(roomId); // 通知其他人 socket.to(roomId).emit('room:user-left', user); // 如果房间为空，删除房间 if (room.users.size === 0) { rooms.delete(roomId); } } callback(true); }); // ==================== 消息处理 ==================== // 发送消息 socket.on('message:send', (data, callback) => { const room = rooms.get(data.roomId); if (!room) { return; } const message: Message = { id: generateMessageId(), content: data.content, type: 'text', sender: user, roomId: data.roomId, timestamp: new Date(), read: false }; // 保存消息 room.messages.push(message); // 广播给房间所有人 io.to(data.roomId).emit('message:new', message); callback(message); }); // 获取历史消息 socket.on('message:history', (roomId, callback) => { const room = rooms.get(roomId); callback(room?.messages || []); }); // ==================== 用户状态 ==================== // 输入状态 socket.on('user:typing', (roomId, isTyping) => { socket.to(roomId).emit('user:typing', userId, isTyping); }); // 断开连接 socket.on('disconnect', () => { console.log(`用户断开: ${username}`); // 从所有房间中移除用户 rooms.forEach((room) => { if (room.users.has(userId)) { room.users.delete(userId); socket.to(room.id).emit('room:user-left', user); } }); userSockets.delete(userId); }); }); } // 辅助函数 function generateRoomId(): string { return Math.random().toString(36).substring(2, 10); } function generateMessageId(): string { return Date.now().toString(36) + Math.random().toString(36).substring(2); }

⚛️ React 客户端实现

Socket Hook

// client/src/hooks/useSocket.ts import { useEffect, useState, useRef, useCallback } from 'react'; import { io, Socket } from 'socket.io-client'; import type { ServerToClientEvents, ClientToServerEvents, Message, User, Room } from '../../../shared/types'; export function useChatSocket(url: string) { const socketRef = useRef<Socket<ServerToClientEvents, ClientToServerEvents>>(null); const [isConnected, setIsConnected] = useState(false); const [currentRoom, setCurrentRoom] = useState<Room | null>(null); const [messages, setMessages] = useState<Message[]>([]); const [users, setUsers] = useState<User[]>([]); const [typingUsers, setTypingUsers] = useState<Set<string>>(new Set()); useEffect(() => { const socket = io(url, { auth: { userId: localStorage.getItem('userId'), username: localStorage.getItem('username') } }); socketRef.current = socket; socket.on('connect', () => setIsConnected(true)); socket.on('disconnect', () => setIsConnected(false)); // 房间事件 socket.on('room:joined', (room) => { setCurrentRoom(room); setUsers(Array.from((room as any).users || [])); }); socket.on('room:user-joined', (user) => { setUsers(prev => [...prev, user]); // 添加系统消息 setMessages(prev => [...prev, { id: Date.now().toString(), content: `${user.username} 加入了房间`, type: 'system', sender: user, roomId: currentRoom?.id || '', timestamp: new Date(), read: true }]); }); socket.on('room:user-left', (user) => { setUsers(prev => prev.filter(u => u.id !== user.id)); setMessages(prev => [...prev, { id: Date.now().toString(), content: `${user.username} 离开了房间`, type: 'system', sender: user, roomId: currentRoom?.id || '', timestamp: new Date(), read: true }]); }); // 消息事件 socket.on('message:new', (message) => { setMessages(prev => [...prev, message]); }); // 输入状态 socket.on('user:typing', (userId, isTyping) => { setTypingUsers(prev => { const newSet = new Set(prev); if (isTyping) { newSet.add(userId); } else { newSet.delete(userId); } return newSet; }); }); return () => { socket.disconnect(); }; }, [url]); // 加入房间 const joinRoom = useCallback((roomId: string) => { return new Promise<Room>((resolve, reject) => { socketRef.current?.emit('room:join', roomId, (room, error) => { if (error) reject(error); else resolve(room); }); }); }, []); // 发送消息 const sendMessage = useCallback((content: string) => { if (!currentRoom) return; socketRef.current?.emit('message:send', { roomId: currentRoom.id, content }, (message) => { // 消息已发送 }); }, [currentRoom]); // 设置输入状态 const setTyping = useCallback((isTyping: boolean) => { if (!currentRoom) return; socketRef.current?.emit('user:typing', currentRoom.id, isTyping); }, [currentRoom]); return { isConnected, currentRoom, messages, users, typingUsers, joinRoom, sendMessage, setTyping }; }

聊天室组件

// client/src/components/ChatRoom.tsx import React, { useState, useEffect } from 'react'; import { useChatSocket } from '../hooks/useSocket'; export function ChatRoom() { const { isConnected, messages, users, typingUsers, joinRoom, sendMessage, setTyping } = useChatSocket('http://localhost:3000'); const [input, setInput] = useState(''); const [roomId, setRoomId] = useState(''); const handleSend = () => { if (input.trim()) { sendMessage(input.trim()); setInput(''); setTyping(false); } }; const handleInput = (e: React.ChangeEvent<HTMLInputElement>) => { setInput(e.target.value); setTyping(true); // 停止输入后 1 秒清除状态 clearTimeout(handleInput.timeout); handleInput.timeout = setTimeout(() => setTyping(false), 1000); }; return ( <div className="chat-container"> {/* 侧边栏 - 用户列表 */} <div className="sidebar"> <h3>在线用户 ({users.length})</h3> <ul> {users.map(user => ( <li key={user.id}> {user.username} {typingUsers.has(user.id) && <span>正在输入...</span>} </li> ))} </ul> </div> {/* 主聊天区域 */} <div className="main-chat"> {/* 消息列表 */} <div className="messages"> {messages.map(msg => ( <div key={msg.id} className={`message ${msg.type}"`}> {msg.type === 'system' ? ( <span className="system">{msg.content}</span> ) : ( <> <span className="sender">{msg.sender.username}:</span> <span className="content">{msg.content}</span> </> )} </div> ))} </div> {/* 输入框 */} <div className="input-area"> <input type="text" value={input} onChange={handleInput} onKeyPress={(e) => e.key === 'Enter' && handleSend()} placeholder="输入消息..." /> <button onClick={handleSend}>发送</button> </div> </div> </div> ); }
