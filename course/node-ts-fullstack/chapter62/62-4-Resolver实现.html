62.4 Resolver 实现

NestJS GraphQL Resolver 完整实现

📖 Resolver 结构

在 NestJS 中，Resolver 是处理 GraphQL 查询和变更的类，类似于 REST 中的 Controller。

import { Resolver, Query, Mutation, Args, ResolveField, Parent } from '@nestjs/graphql'; import { UseGuards, Context } from '@nestjs/common'; import { GqlAuthGuard } from '../auth/guards/gql-auth.guard'; import { User } from './entities/user.entity'; import { UserService } from './user.service'; import { CreateUserInput } from './dto/create-user.input'; @Resolver(() => User) export class UserResolver { constructor(private readonly userService: UserService) {} // Query 定义 @Query(() => [User], { name: 'users' }) async findAll(): Promise { return this.userService.findAll(); } // Mutation 定义 @Mutation(() => User) async createUser( @Args('input') input: CreateUserInput, ): Promise { return this.userService.create(input); } // 字段 Resolver @ResolveField(() => [Post]) async posts(@Parent user: User): Promise { return this.userService.getUserPosts(user.id); } }

🔐 认证守卫

// GQL 认证守卫 import { CanActivate, ExecutionContext, Injectable } from '@nestjs/common'; import { GqlExecutionContext } from '@nestjs/graphql'; @Injectable() export class GqlAuthGuard implements CanActivate { canActivate(context: ExecutionContext): boolean { const ctx = GqlExecutionContext.create(context); const request = ctx.getContext().req; return request.user != null; } } // 使用守卫 @Resolver(() => User) @UseGuards(GqlAuthGuard) export class UserResolver { @Query(() => User, { name: 'me' }) async getCurrentUser(@Context() context) { return context.req.user; } }

📝 小结

  • Resolver 类似 REST 的 Controller
  • 使用 @Query 和 @Mutation 定义操作
  • 使用 @ResolveField 定义关联字段
  • 支持 NestJS 守卫和拦截器
  • 通过 @Context 获取请求上下文
