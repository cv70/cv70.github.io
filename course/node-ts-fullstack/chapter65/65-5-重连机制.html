🔄 默认重连机制

默认配置

Socket.IO 默认启用自动重连，配置如下：

const socket = io('http://localhost:3000', { reconnection: true, // 启用自动重连（默认 true） reconnectionAttempts: Infinity, // 无限次重连（默认） reconnectionDelay: 1000, // 初始延迟 1 秒（默认） reconnectionDelayMax: 5000, // 最大延迟 5 秒（默认） randomizationFactor: 0.5, // 随机化因子（默认 0.5） timeout: 20000 // 连接超时 20 秒（默认） });

指数退避算法

重连延迟采用指数退避算法：

// 计算公式 delay = reconnectionDelay * Math.pow(2, attempt - 1) delay = delay * (1 + randomizationFactor * Math.random()) delay = Math.min(delay, reconnectionDelayMax) // 示例（randomizationFactor = 0.5） // 第 1 次: ~1000ms // 第 2 次: ~1500ms (1000 * 1.5) // 第 3 次: ~3000ms (1500 * 2) // 第 4 次: ~5000ms (达到最大值) // 后续: ~5000ms

⚙️ 重连配置详解

限制重连次数

const socket = io('http://localhost:3000', { reconnectionAttempts: 10, // 最多重连 10 次 reconnectionDelay: 1000, reconnectionDelayMax: 5000 }); // 监听重连失败 socket.on('reconnect_failed', () => { console.log('重连失败，已达到最大次数'); showError('无法连接到服务器，请刷新页面重试'); }); // 监听重连尝试 socket.on('reconnect_attempt', (attempt) => { console.log(`第 ${attempt} 次重连尝试`); updateReconnectStatus(attempt); });

调整重连延迟

// 快速重连（适合对实时性要求高的场景） const fastReconnect = io('http://localhost:3000', { reconnectionDelay: 100, reconnectionDelayMax: 1000, randomizationFactor: 0.2 }); // 慢速重连（适合节省资源） const slowReconnect = io('http://localhost:3000', { reconnectionDelay: 5000, reconnectionDelayMax: 60000, randomizationFactor: 0.3 });

禁用自动重连

// 完全禁用自动重连 const socket = io('http://localhost:3000', { reconnection: false }); // 断开后需手动重连 socket.on('disconnect', () => { // 显示重连按钮 showReconnectButton(); }); // 用户点击重连 reconnectButton.onclick = () => { socket.connect(); };

🔧 自定义重连策略

带 UI 提示的重连

class SocketReconnect { private socket: Socket; private attempts = 0; private maxAttempts = 10; constructor(url: string) { this.socket = io(url, { reconnection: false // 禁用默认，自定义实现 }); this.setupEvents(); } private setupEvents() { this.socket.on('connect', () => { this.attempts = 0; this.hideReconnectUI(); }); this.socket.on('disconnect', (reason) => { if (reason === 'io server disconnect') { // 服务端主动断开，不重连 this.showError('您已被服务器断开'); } else { // 其他原因，尝试重连 this.attemptReconnect(); } }); } private attemptReconnect() { if (this.attempts >= this.maxAttempts) { this.showReconnectFailed(); return; } this.attempts++; const delay = this.calculateDelay(); this.showReconnecting(this.attempts, delay); setTimeout(() => { this.socket.connect(); }, delay); } private calculateDelay(): number { // 指数退避 const base = 1000; const max = 30000; const delay = Math.min(base * Math.pow(2, this.attempts - 1), max); // 添加随机抖动 return delay * (1 + 0.5 * Math.random()); } private showReconnecting(attempt: number, delay: number) { const element = document.getElementById('reconnect-status'); if (element) { element.innerHTML = ` 正在重连... 第 ${attempt} 次
将在 ${(delay / 1000).toFixed(1)} 秒后重试 `; element.style.display = 'block'; } } private showReconnectFailed() { const element = document.getElementById('reconnect-status'); if (element) { element.innerHTML = ` 重连失败 `; } } private hideReconnectUI() { const element = document.getElementById('reconnect-status'); if (element) { element.style.display = 'none'; } } retry() { this.attempts = 0; this.attemptReconnect(); } }

网络状态感知重连

class NetworkAwareSocket { private socket: Socket; private isOnline = navigator.onLine; private pendingConnect = false; constructor(url: string) { this.socket = io(url, { autoConnect: false }); this.setupNetworkListeners(); this.setupSocketListeners(); // 初始连接 if (this.isOnline) { this.socket.connect(); } } private setupNetworkListeners() { window.addEventListener('online', () => { console.log('网络已恢复'); this.isOnline = true; if (this.pendingConnect) { this.socket.connect(); this.pendingConnect = false; } }); window.addEventListener('offline', () => { console.log('网络已断开'); this.isOnline = false; // Socket 会自动触发 disconnect 事件 }); } private setupSocketListeners() { this.socket.on('disconnect', (reason) => { if (!this.isOnline) { // 网络断开导致，等待网络恢复 console.log('等待网络恢复...'); this.pendingConnect = true; this.showOfflineMessage(); } else { // 网络正常但断开，尝试重连 this.scheduleReconnect(); } }); this.socket.on('connect', () => { this.hideOfflineMessage(); }); } private scheduleReconnect() { // 检查网络状态后再重连 if (!this.isOnline) { this.pendingConnect = true; return; } setTimeout(() => { this.socket.connect(); }, 1000); } private showOfflineMessage() { document.getElementById('offline-banner')!.style.display = 'block'; } private hideOfflineMessage() { document.getElementById('offline-banner')!.style.display = 'none'; } }

📊 重连事件

事件 参数 说明
reconnect_attempt attempt: number 重连尝试前触发
reconnecting attempt: number 正在重连时触发
reconnect_error error: Error 重连失败时触发
reconnect_failed 无 达到最大重连次数时触发
reconnect attempt: number 重连成功时触发
// 完整的重连事件监听 socket.io.on('reconnect_attempt', (attempt) => { console.log(`准备第 ${attempt} 次重连`); }); socket.io.on('reconnect', (attempt) => { console.log(`重连成功，第 ${attempt} 次尝试`); }); socket.io.on('reconnect_error', (error) => { console.log('重连错误:', error.message); }); socket.io.on('reconnect_failed', () => { console.log('重连失败'); });

⚡ 重连优化技巧

状态恢复

// 重连后恢复之前的状态 let currentRoom: string | null = null; socket.on('connect', () => { // 重连后重新加入房间 if (currentRoom) { socket.emit('join-room', currentRoom); } }); socket.on('disconnect', () => { // 保持状态，等待重连 console.log('断开连接，等待重连'); });

消息缓存

// 断开时缓存消息 const messageQueue: any[] = []; function sendSafe(event: string, data: any) { if (socket.connected) { socket.emit(event, data); } else { messageQueue.push({ event, data }); } } socket.on('connect', () => { // 发送缓存的消息 while (messageQueue.length > 0) { const { event, data } = messageQueue.shift()!; socket.emit(event, data); } });
