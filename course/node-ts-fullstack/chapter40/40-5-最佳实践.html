🏗️ 分层架构

┌─────────────────────────────────────────┐ │ Routes (路由层) │ │ 定义 API 端点和 HTTP 方法 │ └─────────────────────────────────────────┘ │ ▼ ┌─────────────────────────────────────────┐ │ Controllers (控制层) │ │ 处理请求、调用服务、返回响应 │ └─────────────────────────────────────────┘ │ ▼ ┌─────────────────────────────────────────┐ │ Services (服务层) │ │ 业务逻辑、事务管理、服务编排 │ └─────────────────────────────────────────┘ │ ▼ ┌─────────────────────────────────────────┐ │ Repositories (数据访问层) │ │ 数据库操作、缓存操作 │ └─────────────────────────────────────────┘
// src/controllers/userController.ts export class UserController { constructor(private userService: UserService) {} async index(ctx: RouterContext) { const users = await this.userService.findAll(); ctx.body = { success: true, data: users }; } } // src/services/userService.ts export class UserService { constructor(private userRepo: UserRepository) {} async findAll() { return this.userRepo.findAll(); } } // src/repositories/userRepository.ts export class UserRepository { async findAll() { return User.findAll(); } }

📝 日志管理

# 安装 winston npm install winston
// src/utils/logger.ts import winston from 'winston'; const logger = winston.createLogger({ level: process.env.LOG_LEVEL || 'info', format: winston.format.combine( winston.format.timestamp(), winston.format.json() ), transports: [ // 错误日志 new winston.transports.File({ filename: 'logs/error.log', level: 'error' }), // 所有日志 new winston.transports.File({ filename: 'logs/combined.log' }) ] }); // 开发环境输出到控制台 if (process.env.NODE_ENV !== 'production') { logger.add(new winston.transports.Console({ format: winston.format.simple() })); } export default logger;
// 日志中间件 import logger from '@/utils/logger'; app.use(async (ctx, next) => { const start = Date.now(); await next(); const ms = Date.now() - start; logger.info({ method: ctx.method, url: ctx.url, status: ctx.status, duration: ms, ip: ctx.ip }); }); // 错误日志 app.on('error', (err, ctx) => { logger.error({ error: err.message, stack: err.stack, url: ctx.url, method: ctx.method }); });

💉 依赖注入

# 安装 tsyringe npm install tsyringe reflect-metadata
// tsconfig.json { "compilerOptions": { "experimentalDecorators": true, "emitDecoratorMetadata": true } }
// src/index.ts import 'reflect-metadata'; import { container } from 'tsyringe'; // 注册依赖 container.registerSingleton(UserRepository); container.registerSingleton(UserService); container.registerSingleton(UserController); // src/repositories/userRepository.ts import { injectable } from 'tsyringe'; @injectable() export class UserRepository { async findAll() { return User.findAll(); } } // src/services/userService.ts import { injectable, inject } from 'tsyringe'; @injectable() export class UserService { constructor( @inject(UserRepository) private userRepo: UserRepository ) {} async findAll() { return this.userRepo.findAll(); } } // src/controllers/userController.ts import { injectable, inject } from 'tsyringe'; @injectable() export class UserController { constructor( @inject(UserService) private userService: UserService ) {} async index(ctx: RouterContext) { const users = await this.userService.findAll(); ctx.body = { success: true, data: users }; } }

🧪 单元测试

# 安装测试工具 npm install -D jest ts-jest @types/jest supertest @types/supertest
// jest.config.js module.exports = { preset: 'ts-jest', testEnvironment: 'node', roots: ['/src'], testMatch: ['**/__tests__/**/*.ts', '**/*.spec.ts'], moduleNameMapper: { '^@/(.*)$': '/src/$1' } };
// src/__tests__/user.test.ts import request from 'supertest'; import app from '@/app'; describe('User API', () => { describe('GET /api/users', () => { it('should return users list', async () => { const response = await request(app.callback()) .get('/api/users') .expect(200); expect(response.body.success).toBe(true); expect(Array.isArray(response.body.data)).toBe(true); }); }); describe('POST /api/users', () => { it('should create a new user', async () => { const response = await request(app.callback()) .post('/api/users') .send({ username: 'testuser', email: 'test@example.com', password: 'password123' }) .expect(201); expect(response.body.success).toBe(true); expect(response.body.data.username).toBe('testuser'); }); it('should validate input', async () => { const response = await request(app.callback()) .post('/api/users') .send({ username: '' }) .expect(400); expect(response.body.success).toBe(false); }); }); });
