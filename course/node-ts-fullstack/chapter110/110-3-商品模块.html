📚 SPU/SKU模型

// SPU (Standard Product Unit) - 标准产品单位 // SKU (Stock Keeping Unit) - 库存量单位 // 示例：iPhone 15 const example = { SPU: { id: 'spu-001', name: 'iPhone 15', brand: 'Apple', category: '手机', description: 'iPhone 15 全新设计...', attributes: { brand: 'Apple', model: 'iPhone 15', releaseYear: 2023 } }, SKUs: [ { id: 'sku-001', spuId: 'spu-001', name: 'iPhone 15 黑色 128GB', price: 5999, stock: 100, specs: { color: '黑色', storage: '128GB' } }, { id: 'sku-002', spuId: 'spu-001', name: 'iPhone 15 黑色 256GB', price: 6999, stock: 50, specs: { color: '黑色', storage: '256GB' } }, { id: 'sku-003', spuId: 'spu-001', name: 'iPhone 15 白色 128GB', price: 5999, stock: 80, specs: { color: '白色', storage: '128GB' } } ] }; // 规格属性定义 interface SpecDefinition { name: string; values: string[]; } const specDefinitions: SpecDefinition[] = [ { name: '颜色', values: ['黑色', '白色', '蓝色', '粉色'] }, { name: '存储', values: ['128GB', '256GB', '512GB'] } ];
💡
SPU与SKU关系

一个SPU对应多个SKU，SKU是具体的库存单位，包含规格、价格、库存信息。

🔧 实体设计

// spu.entity.ts @Entity('spu') export class SPU { @PrimaryGeneratedColumn('uuid') id: string; @Column() name: string; @Column() brandId: string; @ManyToOne(() => Category) category: Category; @Column() categoryId: string; @Column({ type: 'text' }) description: string; @Column({ type: 'json' }) attributes: Record; @Column({ type: 'json' }) specDefinitions: SpecDefinition[]; @Column({ default: true }) active: boolean; @OneToMany(() => SKU, sku => sku.spu) skus: SKU[]; @CreateDateColumn() createdAt: Date; @UpdateDateColumn() updatedAt: Date; } // sku.entity.ts @Entity('sku') export class SKU { @PrimaryGeneratedColumn('uuid') id: string; @Column() spuId: string; @ManyToOne(() => SPU, spu => spu.skus) spu: SPU; @Column() name: string; @Column({ type: 'decimal', precision: 10, scale: 2 }) price: number; @Column({ type: 'decimal', precision: 10, scale: 2, nullable: true }) originalPrice: number; @Column({ type: 'json' }) specs: Record; @Column({ default: 0 }) stock: number; @Column({ default: 0 }) frozenStock: number; // 冻结库存（已下单未支付） @Column({ nullable: true }) image: string; @Column({ default: true }) active: boolean; // 可用库存 get availableStock(): number { return this.stock - this.frozenStock; } } // 商品服务 @Injectable() export class ProductService { constructor( @InjectRepository(SPU) private spuRepo: Repository, @InjectRepository(SKU) private skuRepo: Repository, ) {} // 创建商品（包含SKU） async createProduct(dto: CreateProductDto): Promise { const spu = this.spuRepo.create({ name: dto.name, description: dto.description, brandId: dto.brandId, categoryId: dto.categoryId, attributes: dto.attributes, specDefinitions: dto.specDefinitions, }); // 生成所有SKU组合 const skuCombinations = this.generateSkuCombinations( dto.specDefinitions, dto.skuData ); spu.skus = skuCombinations.map(skuData => this.skuRepo.create({ name: `${spu.name} ${this.formatSpecs(skuData.specs)}`, price: skuData.price, specs: skuData.specs, stock: skuData.stock, }) ); return this.spuRepo.save(spu); } // 搜索商品 async searchProducts(query: SearchQuery) { const { keyword, categoryId, minPrice, maxPrice, page, limit } = query; const queryBuilder = this.spuRepo.createQueryBuilder('spu') .leftJoinAndSelect('spu.skus', 'sku') .where('spu.active = true'); if (keyword) { queryBuilder.andWhere( 'spu.name LIKE :keyword OR spu.description LIKE :keyword', { keyword: `%${keyword}%` } ); } if (categoryId) { queryBuilder.andWhere('spu.categoryId = :categoryId', { categoryId }); } return queryBuilder .skip((page - 1) * limit) .take(limit) .getManyAndCount(); } }
