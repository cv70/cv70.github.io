📚 Zustand 基础

// 安装 // npm install zustand // Zustand 特点 // - 极简 API，学习成本低 // - 无需 Provider 包裹 // - 支持 TypeScript // - 支持中间件 // - 细粒度订阅，性能优秀 import { create } from 'zustand'; // 创建 store interface CounterState { count: number; increment: () => void; decrement: () => void; incrementByAmount: (amount: number) => void; reset: () => void; } const useCounterStore = create<CounterState>((set) => ({ count: 0, increment: () => set((state) => ({ count: state.count + 1 })), decrement: () => set((state) => ({ count: state.count - 1 })), incrementByAmount: (amount) => set((state) => ({ count: state.count + amount })), reset: () => set({ count: 0 }), })); // 使用 store function Counter() { const count = useCounterStore((state) => state.count); const increment = useCounterStore((state) => state.increment); const decrement = useCounterStore((state) => state.decrement); return ( <div> <p>Count: {count}</p> <button onClick={increment}>+</button> <button onClick={decrement}>-</button> </div> ); } // 选择器优化性能 function CountDisplay() { // 只订阅 count，其他状态变化不会触发重渲染 const count = useCounterStore((state) => state.count); return <div>{count}</div>; } function IncrementButton() { // 只订阅 increment 方法，count 变化不会触发重渲染 const increment = useCounterStore((state) => state.increment); return <button onClick={increment}>+</button>; } // 获取整个 store（不推荐，可能导致不必要的渲染） // const { count, increment } = useCounterStore();

🔧 完整示例：Todo 应用

import { create } from 'zustand'; import { persist, devtools } from 'zustand/middleware'; interface Todo { id: string; text: string; completed: boolean; } interface TodoState { todos: Todo[]; filter: 'all' | 'active' | 'completed'; // Actions addTodo: (text: string) => void; toggleTodo: (id: string) => void; deleteTodo: (id: string) => void; setFilter: (filter: 'all' | 'active' | 'completed') => void; clearCompleted: () => void; // Computed getFilteredTodos: () => Todo[]; } const useTodoStore = create<TodoState>()( devtools( persist( (set, get) => ({ todos: [], filter: 'all', addTodo: (text) => set((state) => ({ todos: [ ...state.todos, { id: crypto.randomUUID(), text, completed: false }, ], })), toggleTodo: (id) => set((state) => ({ todos: state.todos.map((todo) => todo.id === id ? { ...todo, completed: !todo.completed } : todo ), })), deleteTodo: (id) => set((state) => ({ todos: state.todos.filter((todo) => todo.id !== id), })), setFilter: (filter) => set({ filter }), clearCompleted: () => set((state) => ({ todos: state.todos.filter((todo) => !todo.completed), })), getFilteredTodos: () => { const { todos, filter } = get(); if (filter === 'active') return todos.filter((t) => !t.completed); if (filter === 'completed') return todos.filter((t) => t.completed); return todos; }, }), { name: 'todo-storage' } // localStorage key ) ) ); // 组件中使用 function TodoList() { const todos = useTodoStore((state) => state.getFilteredTodos()); const toggleTodo = useTodoStore((state) => state.toggleTodo); const deleteTodo = useTodoStore((state) => state.deleteTodo); return ( <ul> {todos.map((todo) => ( <li key={todo.id}> <input type="checkbox" checked={todo.completed} onChange={() => toggleTodo(todo.id)} /> {todo.text} <button onClick={() => deleteTodo(todo.id)}>Delete</button> </li> ))} </ul> ); } function AddTodo() { const addTodo = useTodoStore((state) => state.addTodo); const [text, setText] = useState(''); const handleSubmit = (e) => { e.preventDefault(); if (text.trim()) { addTodo(text); setText(''); } }; return ( <form onSubmit={handleSubmit}> <input value={text} onChange={(e) => setText(e.target.value)} /> <button type="submit">Add</button> </form> ); }

📊 Zustand 中间件

import { create } from 'zustand'; import { persist, devtools, immer, subscribeWithSelector } from 'zustand/middleware'; // 多个中间件组合 const useStore = create<State>()( devtools( persist( immer((set) => ({ // 使用 immer 可以直接修改 state items: [], addItem: (item) => set((state) => { state.items.push(item); // 直接 push，无需返回新对象 }), updateItem: (id, updates) => set((state) => { const item = state.items.find(i => i.id === id); if (item) { Object.assign(item, updates); // 直接修改 } }), })), { name: 'my-store' } ), { name: 'MyStore' } ) ); // 中间件说明 // - persist：持久化到 localStorage/sessionStorage // - devtools：Redux DevTools 集成 // - immer：支持直接修改 state // - subscribeWithSelector：高级订阅功能 // 在 React 组件外使用 // 直接调用 store 方法 useTodoStore.getState().addTodo('New todo'); // 订阅变化 const unsubscribe = useTodoStore.subscribe((state) => { console.log('Todos changed:', state.todos); }); // 取消订阅 unsubscribe();
