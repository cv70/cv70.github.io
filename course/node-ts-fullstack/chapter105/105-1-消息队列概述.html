📚 什么是消息队列？

// 消息队列（Message Queue）是一种异步通信机制 // 生产者发送消息到队列，消费者从队列获取消息 // 基本概念 const messageQueue = { producer: '生产者 - 发送消息的应用', consumer: '消费者 - 接收消息的应用', queue: '队列 - 存储消息的容器', message: '消息 - 传递的数据单元', broker: '代理 - 消息队列服务器' }; // 消息流转过程 const messageFlow = ` Producer → [Queue/Broker] → Consumer 1. Producer 创建消息 2. Producer 将消息发送到 Queue 3. Queue 存储消息 4. Consumer 从 Queue 获取消息 5. Consumer 处理消息 6. Consumer 确认消息处理完成 `; // 简单示例 class SimpleMessageQueue { private queue: Message[] = []; // 生产者发送消息 async produce(message: Message) { this.queue.push(message); console.log(`Message produced: ${message.id}`); } // 消费者获取消息 async consume(): Promise { const message = this.queue.shift(); if (message) { console.log(`Message consumed: ${message.id}`); } return message || null; } }
💡
核心作用

解耦：生产者和消费者不需要直接交互；异步：提高响应速度；削峰：平滑流量高峰。

🔧 消息队列模式

// 点对点模式（Point-to-Point） // 一条消息只能被一个消费者消费 class PointToPointQueue { private queue: Message[] = []; async send(message: Message) { this.queue.push(message); } async receive(): Promise { return this.queue.shift() || null; } } // 发布订阅模式（Pub/Sub） // 一条消息可以被多个消费者消费 class PubSubQueue { private topics: Map = new Map(); private subscribers: Map = new Map(); async publish(topic: string, message: Message) { const consumers = this.subscribers.get(topic) || []; consumers.forEach(consumer => consumer.onMessage(message)); } subscribe(topic: string, consumer: Consumer) { if (!this.subscribers.has(topic)) { this.subscribers.set(topic, []); } this.subscribers.get(topic)!.push(consumer); } } // 使用示例 const pubsub = new PubSubQueue(); // 订阅者 pubsub.subscribe('order.created', { onMessage: (msg) => console.log('库存服务收到:', msg) }); pubsub.subscribe('order.created', { onMessage: (msg) => console.log('通知服务收到:', msg) }); // 发布消息 pubsub.publish('order.created', { orderId: '123', amount: 100 }); // 库存服务收到: { orderId: '123', amount: 100 } // 通知服务收到: { orderId: '123', amount: 100 }
