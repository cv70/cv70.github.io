62.5 与 TypeORM 集成

GraphQL 与 TypeORM 完美配合

📖 Entity 定义

import { Entity, PrimaryGeneratedColumn, Column, OneToMany } from 'typeorm'; import { ObjectType, Field, ID } from '@nestjs/graphql'; @Entity() @ObjectType() export class User { @PrimaryGeneratedColumn() @Field(() => ID) id: number; @Column() @Field() name: string; @Column({ unique: true }) @Field() email: string; @Column() password: string; // 不添加 @Field，不在 GraphQL 中暴露 @OneToMany(() => Post, post => post.author) @Field(() => [Post]) posts: Post[]; }

🔧 Service 实现

@Injectable() export class UserService { constructor( @InjectRepository(User) private userRepository: Repository, ) {} async findAll(): Promise { return this.userRepository.find({ relations: ['posts'], }); } async findOne(id: number): Promisenull> { return this.userRepository.findOne({ where: { id }, relations: ['posts'], }); } async create(input: CreateUserInput): Promise { const user = this.userRepository.create(input); return this.userRepository.save(user); } async update(id: number, input: UpdateUserInput): Promisenull> { await this.userRepository.update(id, input); return this.findOne(id); } async remove(id: number): Promise<void> { await this.userRepository.delete(id); } }

📝 小结

  • Entity 可以同时作为 TypeORM 实体和 GraphQL 类型
  • 敏感字段不添加 @Field 装饰器
  • 使用 relations 加载关联数据
  • Service 处理业务逻辑和数据访问
  • Resolver 调用 Service 处理请求
