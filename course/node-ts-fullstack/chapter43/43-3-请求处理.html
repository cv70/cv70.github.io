43.3 请求处理

掌握获取请求数据的各种方式

🎯 学习目标

  • 掌握请求对象的使用
  • 学会获取请求体数据
  • 了解请求头和Cookie的处理
  • 掌握文件上传处理

📚 请求装饰器

装饰器 说明 示例
@Req() 请求对象 @Req() req: Request
@Body() 请求体 @Body() body: CreateUserDto
@Param() 路径参数 @Param('id') id: string
@Query() 查询参数 @Query('page') page: number
@Headers() 请求头 @Headers('authorization') token
@Ip() 客户端IP @Ip() ip: string

🔧 请求体处理

import { Controller, Post, Body } from '@nestjs/common'; class CreateUserDto { name: string; email: string; password: string; } @Controller('users') export class UsersController { @Post() create(@Body() createUserDto: CreateUserDto) { console.log(createUserDto); // { name: 'John', email: 'john@example.com', password: 'xxx' } return { message: 'User created', data: createUserDto }; } // 获取单个字段 @Post('login') login( @Body('email') email: string, @Body('password') password: string ) { return { email }; } }
💡
DTO 模式

使用 DTO (Data Transfer Object) 定义请求体结构，可以配合 ValidationPipe 进行自动验证。

🌐 Headers 和 Cookies

import { Controller, Get, Headers, Req, Res } from '@nestjs/common'; import { Request, Response } from 'express'; @Controller('auth') export class AuthController { @Get('me') getCurrentUser( @Headers('authorization') authorization: string, @Headers() headers: Record ) { console.log(authorization); // Bearer xxx console.log(headers); // { host: '...', ... } return { token: authorization }; } @Get('token') getToken(@Req() req: Request) { // Express Request 对象 console.log(req.cookies); // Cookie 对象 console.log(req.signedCookies); // 签名 Cookie return { cookies: req.cookies }; } @Get('set-cookie') setCookie(@Res() res: Response) { res.cookie('token', 'abc123', { httpOnly: true, maxAge: 3600000, }); return res.json({ message: 'Cookie set' }); } }

📎 文件上传

import { Controller, Post, UseInterceptors, UploadedFile, UploadedFiles } from '@nestjs/common'; import { FileInterceptor, FilesInterceptor } from '@nestjs/platform-express'; @Controller('upload') export class UploadController { // 单文件上传 @Post('single') @UseInterceptors(FileInterceptor('file')) uploadSingle(@UploadedFile() file: Express.Multer.File) { console.log(file); // { // fieldname: 'file', // originalname: 'image.jpg', // size: 12345, // mimetype: 'image/jpeg', // buffer: // } return { filename: file.originalname, size: file.size }; } // 多文件上传 @Post('multiple') @UseInterceptors(FilesInterceptor('files', 10)) // 最多10个 uploadMultiple(@UploadedFiles() files: Express.Multer.File[]) { return { count: files.length }; } }
⚠️
文件上传配置

文件上传默认存储在内存中。可以通过配置 Multer 选项指定存储位置、文件大小限制、文件类型限制等。
