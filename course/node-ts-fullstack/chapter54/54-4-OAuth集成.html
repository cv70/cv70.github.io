📚 OAuth 2.0 流程

// OAuth 2.0 授权码流程 ┌─────────┐ ┌─────────────┐ │ User │ │ Client │ │ (Browser)│ │ (Your App) │ └────┬────┘ └──────┬──────┘ │ │ │ 1. 点击"使用 Google 登录" │ │──────────────────────────────────────────────>│ │ │ │ 2. 重定向到 Google 授权页面 │ │<──────────────────────────────────────────────│ │ │ │ 3. 用户同意授权 │ │───────────────────────────────────────>┌─────┴─────┐ │ │ Google │ │ 4. 重定向回应用（带授权码） │ OAuth │ │<────────────────────────────────────────│ Server │ │ └───────────┘ │ │ │ 5. 用授权码换取 Token │ │ │────────>│ │ │ │ │ 6. 返回 Access Token │<────────│ │ │ │ │ 7. 获取用户信息 │ │ │ │────────>│ │ │ │ │ 8. 返回用户信息 │<────────│ │ │ │ │ 9. 创建会话/JWT │ │ │<──────────────────────────────────────────────│ │ // 关键概念： // - client_id: 应用标识 // - client_secret: 应用密钥 // - redirect_uri: 回调地址 // - scope: 请求的权限范围 // - code: 授权码（一次性） // - access_token: 访问令牌

🔧 Google OAuth 配置

// src/config/passport.ts import passport from 'passport'; import { Strategy as GoogleStrategy, Profile } from 'passport-google-oauth20'; import { User } from '../models/User'; // Google OAuth 策略 passport.use( new GoogleStrategy( { clientID: process.env.GOOGLE_CLIENT_ID!, clientSecret: process.env.GOOGLE_CLIENT_SECRET!, callbackURL: process.env.GOOGLE_CALLBACK_URL || '/api/auth/google/callback', scope: ['profile', 'email'], }, async ( accessToken: string, refreshToken: string, profile: Profile, done: Function ) => { try { // 查找是否已绑定 Google 账号 let user = await User.findOne({ 'oauth.google.id': profile.id, }); if (user) { return done(null, user); } // 检查邮箱是否已注册 const email = profile.emails?.[0]?.value; if (email) { user = await User.findOne({ email }); if (user) { // 绑定 Google 账号到现有用户 user.oauth = { ...user.oauth, google: { id: profile.id, accessToken, refreshToken, }, }; await user.save(); return done(null, user); } } // 创建新用户 const newUser = await User.create({ email: email, username: profile.displayName?.replace(/\s+/g, '_').toLowerCase() || `google_${profile.id}`, password: Math.random().toString(36).slice(-16), // 随机密码 avatar: profile.photos?.[0]?.value, oauth: { google: { id: profile.id, accessToken, refreshToken, }, }, isEmailVerified: true, // Google 邮箱已验证 }); done(null, newUser); } catch (error) { done(error, false); } } ) ); export default passport;

🐙 GitHub OAuth 配置

// src/config/passport.ts import { Strategy as GitHubStrategy, Profile } from 'passport-github2'; // GitHub OAuth 策略 passport.use( new GitHubStrategy( { clientID: process.env.GITHUB_CLIENT_ID!, clientSecret: process.env.GITHUB_CLIENT_SECRET!, callbackURL: process.env.GITHUB_CALLBACK_URL || '/api/auth/github/callback', scope: ['user:email'], }, async ( accessToken: string, refreshToken: string, profile: Profile, done: Function ) => { try { // 查找是否已绑定 GitHub 账号 let user = await User.findOne({ 'oauth.github.id': profile.id, }); if (user) { return done(null, user); } // 检查邮箱是否已注册 const email = profile.emails?.[0]?.value; if (email) { user = await User.findOne({ email }); if (user) { // 绑定 GitHub 账号到现有用户 user.oauth = { ...user.oauth, github: { id: profile.id, accessToken, refreshToken, }, }; await user.save(); return done(null, user); } } // 创建新用户 const newUser = await User.create({ email: email || `${profile.id}@github.placeholder`, username: profile.username || `github_${profile.id}`, password: Math.random().toString(36).slice(-16), avatar: profile.photos?.[0]?.value, oauth: { github: { id: profile.id, accessToken, refreshToken, }, }, }); done(null, newUser); } catch (error) { done(error, false); } } ) );

🛣️ OAuth 路由

// src/routes/auth.routes.ts import { Router } from 'express'; import passport from '../config/passport'; import { AuthController } from '../controllers/auth.controller'; import { authenticate } from '../middleware/auth.middleware'; const router = Router(); const authController = new AuthController(); // Google OAuth router.get( '/google', passport.authenticate('google', { scope: ['profile', 'email'] }) ); router.get( '/google/callback', passport.authenticate('google', { failureRedirect: '/login?error=google_auth_failed', session: false, }), authController.oauthCallback ); // GitHub OAuth router.get( '/github', passport.authenticate('github', { scope: ['user:email'] }) ); router.get( '/github/callback', passport.authenticate('github', { failureRedirect: '/login?error=github_auth_failed', session: false, }), authController.oauthCallback ); // 绑定第三方账号（需要登录） router.post( '/bind/google', authenticate, (req, res, next) => { passport.authorize('google', { scope: ['profile', 'email'] })(req, res, next); } ); router.get( '/bind/google/callback', authenticate, passport.authorize('google', { session: false }), authController.bindOAuthCallback ); export default router;

🔐 OAuth 控制器

// src/controllers/auth.controller.ts import { Request, Response } from 'express'; import { JwtService } from '../services/jwt.service'; export class AuthController { // OAuth 回调处理 async oauthCallback(req: Request, res: Response) { const user = req.user as any; if (!user) { return res.redirect(`${process.env.FRONTEND_URL}/login?error=auth_failed`); } // 生成 JWT const payload = { userId: user._id.toString(), username: user.username, role: user.role, }; const token = JwtService.sign(payload); // 重定向到前端，携带 Token res.redirect( `${process.env.FRONTEND_URL}/auth/callback?token=${token}` ); } // 绑定 OAuth 账号回调 async bindOAuthCallback(req: Request, res: Response) { const user = req.user as any; const account = req.account as any; // OAuth 账号信息 if (!account) { return res.redirect(`${process.env.FRONTEND_URL}/settings?error=bind_failed`); } res.redirect( `${process.env.FRONTEND_URL}/settings?success=account_bound` ); } }

⚙️ 环境变量配置

⚠️
安全提示

回调 URL 必须与 OAuth 提供商后台配置的完全一致，包括协议（http/https）、域名、端口和路径。
